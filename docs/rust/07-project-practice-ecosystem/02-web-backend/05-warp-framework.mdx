# Warp 框架

> **版本**：v1.0  
> **最后更新**：2026-04-22

## 📚 概念介绍

Warp 是一个基于**组合式过滤器（Filter）**设计的 Web 框架。每个路由都是一个 Filter，可以通过 `and`、`or`、`map` 等操作符像拼积木一样组合出复杂的路由逻辑。这种函数式编程风格使得路由定义非常灵活且可复用。

## 🎯 基础路由

### Hello World

```rust
use warp::Filter;

#[tokio::main]
async fn main() {
    let hello = warp::path::end()
        .map(|| "Hello, Warp!");

    let health = warp::path("health")
        .map(|| warp::reply::json(&serde_json::json!({"status": "ok"})));

    let routes = hello.or(health);

    println!("服务器启动于 http://localhost:3030");
    warp::serve(routes).run(([0, 0, 0, 0], 3030)).await;
}
```

### Filter 组合

```rust
use warp::Filter;
use serde::{Deserialize, Serialize};

#[derive(Serialize)]
struct User {
    id: u64,
    name: String,
}

#[derive(Deserialize)]
struct ListQuery {
    page: Option<u32>,
    limit: Option<u32>,
}

#[tokio::main]
async fn main() {
    // 路径参数
    let get_user = warp::path!("users" / u64)
        .and(warp::get())
        .map(|user_id: u64| {
            warp::reply::json(&User {
                id: user_id,
                name: format!("User_{}", user_id),
            })
        });

    // 查询参数
    let list_users = warp::path("users")
        .and(warp::get())
        .and(warp::query::<ListQuery>())
        .map(|query: ListQuery| {
            let page = query.page.unwrap_or(1);
            let limit = query.limit.unwrap_or(10);
            warp::reply::json(&serde_json::json!({
                "page": page,
                "limit": limit,
                "users": []
            }))
        });

    // 嵌套路径
    let user_posts = warp::path!("users" / u64 / "posts" / u64)
        .and(warp::get())
        .map(|user_id: u64, post_id: u64| {
            warp::reply::json(&serde_json::json!({
                "user_id": user_id,
                "post_id": post_id
            }))
        });

    let routes = get_user.or(list_users).or(user_posts);
    warp::serve(routes).run(([0, 0, 0, 0], 3030)).await;
}
```

## 📦 JSON 请求与响应

```rust
use warp::Filter;
use serde::{Deserialize, Serialize};

#[derive(Deserialize)]
struct CreateItemRequest {
    name: String,
    price: f64,
}

#[derive(Serialize)]
struct Item {
    id: u64,
    name: String,
    price: f64,
}

#[derive(Serialize)]
struct ApiResponse<T: Serialize> {
    success: bool,
    data: T,
}

fn json_body() -> impl Filter<Extract = (CreateItemRequest,), Error = warp::Rejection> + Clone {
    warp::body::content_length_limit(1024 * 16)
        .and(warp::body::json())
}

#[tokio::main]
async fn main() {
    let create_item = warp::path("items")
        .and(warp::post())
        .and(json_body())
        .map(|body: CreateItemRequest| {
            let item = Item {
                id: 1,
                name: body.name,
                price: body.price,
            };
            warp::reply::with_status(
                warp::reply::json(&ApiResponse {
                    success: true,
                    data: item,
                }),
                warp::http::StatusCode::CREATED,
            )
        });

    let list_items = warp::path("items")
        .and(warp::get())
        .map(|| {
            let items = vec![
                Item { id: 1, name: "Rust Book".to_string(), price: 29.99 },
                Item { id: 2, name: "Tokio Guide".to_string(), price: 19.99 },
            ];
            warp::reply::json(&ApiResponse {
                success: true,
                data: items,
            })
        });

    let routes = create_item.or(list_items);
    warp::serve(routes).run(([0, 0, 0, 0], 3030)).await;
}
```

## 🗄️ 共享状态与可复用 Filter

```rust
use std::collections::HashMap;
use std::sync::Arc;
use tokio::sync::RwLock;
use warp::Filter;
use serde::{Deserialize, Serialize};

type KeyValueStore = Arc<RwLock<HashMap<String, String>>>;

fn with_store(
    store: KeyValueStore,
) -> impl Filter<Extract = (KeyValueStore,), Error = std::convert::Infallible> + Clone {
    warp::any().map(move || store.clone())
}

#[derive(Deserialize)]
struct SetValueRequest {
    value: String,
}

#[derive(Serialize)]
struct StoreEntry {
    key: String,
    value: String,
}

#[tokio::main]
async fn main() {
    let store: KeyValueStore = Arc::new(RwLock::new(HashMap::new()));

    let get_value = warp::path!("kv" / String)
        .and(warp::get())
        .and(with_store(store.clone()))
        .and_then(|key: String, store: KeyValueStore| async move {
            let data = store.read().await;
            match data.get(&key) {
                Some(value) => Ok(warp::reply::json(&StoreEntry {
                    key,
                    value: value.clone(),
                })),
                None => Err(warp::reject::not_found()),
            }
        });

    let set_value = warp::path!("kv" / String)
        .and(warp::put())
        .and(warp::body::json::<SetValueRequest>())
        .and(with_store(store.clone()))
        .and_then(|key: String, body: SetValueRequest, store: KeyValueStore| async move {
            store.write().await.insert(key.clone(), body.value.clone());
            Ok::<_, warp::Rejection>(warp::reply::json(&StoreEntry {
                key,
                value: body.value,
            }))
        });

    let delete_value = warp::path!("kv" / String)
        .and(warp::delete())
        .and(with_store(store.clone()))
        .and_then(|key: String, store: KeyValueStore| async move {
            let removed = store.write().await.remove(&key);
            match removed {
                Some(_) => Ok(warp::reply::with_status(
                    warp::reply::json(&serde_json::json!({"deleted": key})),
                    warp::http::StatusCode::OK,
                )),
                None => Err(warp::reject::not_found()),
            }
        });

    let list_keys = warp::path("kv")
        .and(warp::get())
        .and(with_store(store.clone()))
        .and_then(|store: KeyValueStore| async move {
            let data = store.read().await;
            let keys: Vec<&String> = data.keys().collect();
            Ok::<_, warp::Rejection>(warp::reply::json(&keys))
        });

    let routes = list_keys.or(get_value).or(set_value).or(delete_value);

    println!("KV Store 启动于 http://localhost:3030");
    warp::serve(routes).run(([0, 0, 0, 0], 3030)).await;
}
```

## ⚠️ 错误处理

```rust
use warp::Filter;
use serde::Serialize;
use std::convert::Infallible;

#[derive(Serialize)]
struct ErrorResponse {
    code: u16,
    message: String,
}

async fn handle_rejection(err: warp::Rejection) -> Result<impl warp::Reply, Infallible> {
    let (code, message) = if err.is_not_found() {
        (warp::http::StatusCode::NOT_FOUND, "资源未找到".to_string())
    } else if err.find::<warp::reject::MethodNotAllowed>().is_some() {
        (warp::http::StatusCode::METHOD_NOT_ALLOWED, "方法不允许".to_string())
    } else if err.find::<warp::reject::PayloadTooLarge>().is_some() {
        (warp::http::StatusCode::PAYLOAD_TOO_LARGE, "请求体过大".to_string())
    } else {
        (warp::http::StatusCode::INTERNAL_SERVER_ERROR, "服务器内部错误".to_string())
    };

    let json = warp::reply::json(&ErrorResponse {
        code: code.as_u16(),
        message,
    });

    Ok(warp::reply::with_status(json, code))
}

#[tokio::main]
async fn main() {
    let api = warp::path("api")
        .and(warp::path("items"))
        .and(warp::get())
        .map(|| warp::reply::json(&serde_json::json!({"items": []})));

    let routes = api.recover(handle_rejection);
    warp::serve(routes).run(([0, 0, 0, 0], 3030)).await;
}
```
