## 📚 概念介绍

并发处理是高并发系统的核心能力，决定了系统能够同时处理多少请求。本节将深入讲解线程池设计、异步非阻塞模型、背压控制和无锁数据结构等关键技术。

## 🎯 线程池设计

### 线程池核心参数

线程池的核心在于合理配置参数，避免资源浪费和性能瓶颈。

```js
// 线程池参数配置示例
class ThreadPoolConfig {
  constructor() {
    // 核心线程数：常驻线程数，即使空闲也不会被回收
    this.corePoolSize = 8
    
    // 最大线程数：线程池允许的最大线程数
    this.maximumPoolSize = 16
    
    // 线程空闲时间：超过核心线程数的空闲线程存活时间（毫秒）
    this.keepAliveTime = 60000
    
    // 任务队列：用于存储待执行的任务
    this.workQueue = new ArrayBlockingQueue(1000)
    
    // 拒绝策略：当队列满且线程数达到最大时的处理方式
    this.rejectedExecutionHandler = new CallerRunsPolicy()
  }
}

// 拒绝策略实现
class CallerRunsPolicy {
  rejectedExecution(task, executor) {
    // 由调用线程执行该任务，降低提交速度
    task.run()
  }
}

// 线程池使用示例
class ThreadPoolExecutor {
  constructor(config) {
    this.corePoolSize = config.corePoolSize
    this.maximumPoolSize = config.maximumPoolSize
    this.workQueue = config.workQueue
    this.rejectedHandler = config.rejectedExecutionHandler
    this.workers = []
    this.activeCount = 0
  }

  execute(task) {
    if (this.activeCount < this.corePoolSize) {
      // 核心线程数未满，创建新线程
      this.addWorker(task, true)
    } else if (this.workQueue.offer(task)) {
      // 任务队列未满，任务入队
      console.log('Task queued')
    } else if (this.activeCount < this.maximumPoolSize) {
      // 队列已满但线程数未达最大，创建非核心线程
      this.addWorker(task, false)
    } else {
      // 拒绝任务
      this.rejectedHandler.rejectedExecution(task, this)
    }
  }

  addWorker(task, core) {
    const worker = {
      task,
      core,
      run: () => {
        this.activeCount++
        try {
          task.run()
        } finally {
          this.activeCount--
          this.processQueue()
        }
      }
    }
    this.workers.push(worker)
    worker.run()
  }

  processQueue() {
    while (this.activeCount < this.corePoolSize) {
      const task = this.workQueue.poll()
      if (task) {
        this.addWorker(task, true)
      } else {
        break
      }
    }
  }
}

// 使用示例
const config = new ThreadPoolConfig()
const executor = new ThreadPoolExecutor(config)

for (let i = 0; i < 20; i++) {
  const taskId = i
  executor.execute({
    run: () => {
      console.log(`Task ${taskId} executed by thread`)
    }
  })
}
```

### 线程池参数调优公式

```js
// CPU 密集型任务：线程数 = CPU 核心数 + 1
function cpuIntensiveThreads() {
  const cpuCores = 8
  return cpuCores + 1  // 9 个线程
}

// IO 密集型任务：线程数 = CPU 核心数 / (1 - 阻塞系数)
function ioIntensiveThreads(blockingCoefficient = 0.8) {
  const cpuCores = 8
  return Math.floor(cpuCores / (1 - blockingCoefficient))  // 40 个线程
}

// 混合型任务：根据实际监控数据调整
function mixedThreads(cpuRatio, ioRatio, cpuCores = 8) {
  const cpuThreads = Math.floor(cpuCores * cpuRatio)
  const ioThreads = Math.floor(cpuCores * ioRatio / 0.2)
  return cpuThreads + ioThreads
}

console.log('CPU 密集型线程数:', cpuIntensiveThreads())
console.log('IO 密集型线程数:', ioIntensiveThreads())
console.log('混合型线程数:', mixedThreads(0.3, 0.7))
```

## 🎯 异步非阻塞模型

### Reactor 模式

Reactor 模式通过事件循环和回调实现高并发处理。

```js
// Reactor 模式实现
class Reactor {
  constructor() {
    this.eventLoop = []
    this.running = false
  }

  // 注册事件处理器
  register(eventType, handler) {
    this.eventLoop.push({ eventType, handler })
  }

  // 触发事件
  emit(eventType, data) {
    this.eventLoop
      .filter(item => item.eventType === eventType)
      .forEach(item => item.handler(data))
  }

  // 启动事件循环
  start() {
    this.running = true
    console.log('Reactor event loop started')
  }

  stop() {
    this.running = false
    console.log('Reactor event loop stopped')
  }
}

// 使用示例
const reactor = new Reactor()

// 注册请求处理
reactor.register('request', (req) => {
  console.log(`Processing request: ${req.id}`)
  
  // 模拟异步处理
  setTimeout(() => {
    reactor.emit('response', { id: req.id, result: 'success' })
  }, 100)
})

// 注册响应处理
reactor.register('response', (res) => {
  console.log(`Response received: ${res.result}`)
})

reactor.start()

// 模拟请求
for (let i = 0; i < 5; i++) {
  reactor.emit('request', { id: i })
}
```

### Promise/Async-Await 模式

```js
// 异步请求处理
class AsyncRequestHandler {
  constructor() {
    this.requestQueue = []
    this.maxConcurrent = 10
    this.activeRequests = 0
  }

  async processRequest(request) {
    return new Promise((resolve, reject) => {
      if (this.activeRequests >= this.maxConcurrent) {
        this.requestQueue.push({ request, resolve, reject })
        console.log(`Request ${request.id} queued`)
      } else {
        this.executeRequest(request, resolve, reject)
      }
    })
  }

  async executeRequest(request, resolve, reject) {
    this.activeRequests++
    console.log(`Processing request ${request.id}, active: ${this.activeRequests}`)

    try {
      // 模拟异步处理
      await new Promise(r => setTimeout(r, 100))
      resolve({ id: request.id, status: 'success' })
    } catch (error) {
      reject(error)
    } finally {
      this.activeRequests--
      this.processNext()
    }
  }

  processNext() {
    if (this.requestQueue.length > 0) {
      const next = this.requestQueue.shift()
      this.executeRequest(next.request, next.resolve, next.reject)
    }
  }
}

// 使用示例
async function testAsyncHandler() {
  const handler = new AsyncRequestHandler()
  const requests = Array.from({ length: 20 }, (_, i) => ({ id: i }))

  const results = await Promise.all(
    requests.map(req => handler.processRequest(req))
  )

  console.log(`Completed ${results.length} requests`)
}

testAsyncHandler()
```

## 🎯 背压控制（Backpressure）

### 背压控制实现

背压控制防止生产者过快压垮消费者。

```js
// 背压控制器
class BackpressureController {
  constructor(consumerCapacity = 100) {
    this.consumerCapacity = consumerCapacity
    this.buffer = []
    this.bufferSize = consumerCapacity * 2
    this.paused = false
  }

  // 生产者推送数据
  async produce(data) {
    while (this.buffer.length >= this.bufferSize) {
      // 缓冲区满，等待消费者处理
      await this.wait(10)
    }

    this.buffer.push(data)
    console.log(`Produced: ${data}, buffer size: ${this.buffer.length}`)

    // 检查是否需要暂停生产
    if (this.buffer.length >= this.bufferSize * 0.8 && !this.paused) {
      this.paused = true
      console.log('Backpressure: Pausing production')
    }
  }

  // 消费者消费数据
  consume() {
    if (this.buffer.length === 0) {
      return null
    }

    const data = this.buffer.shift()
    console.log(`Consumed: ${data}, buffer size: ${this.buffer.length}`)

    // 恢复生产
    if (this.paused && this.buffer.length < this.bufferSize * 0.3) {
      this.paused = false
      console.log('Backpressure: Resuming production')
    }

    return data
  }

  wait(ms) {
    return new Promise(resolve => setTimeout(resolve, ms))
  }
}

// 使用示例
async function testBackpressure() {
  const controller = new BackpressureController(10)

  // 生产者
  const producer = async () => {
    for (let i = 0; i < 50; i++) {
      await controller.produce(i)
      await controller.wait(5)
    }
  }

  // 消费者
  const consumer = async () => {
    let count = 0
    while (count < 50) {
      const data = controller.consume()
      if (data !== null) {
        count++
      }
      await controller.wait(20)
    }
  }

  await Promise.all([producer(), consumer()])
}

testBackpressure()
```

## 🎯 无锁数据结构

### 无锁队列实现

使用 CAS（Compare-And-Swap）实现无锁队列。

```js
// 无锁队列节点
class Node {
  constructor(value) {
    this.value = value
    this.next = null
  }
}

// 无锁队列
class LockFreeQueue {
  constructor() {
    this.head = new Node(null)
    this.tail = this.head
  }

  // 入队（无锁）
  enqueue(value) {
    const newNode = new Node(value)
    let oldTail = null

    while (true) {
      oldTail = this.tail
      const next = oldTail.next

      if (oldTail === this.tail) {
        if (next === null) {
          // 尝试设置 next 指针
          if (this.cas(oldTail.next, null, newNode)) {
            break
          }
        } else {
          // 帮助推进 tail
          this.cas(this.tail, oldTail, next)
        }
      }
    }

    // 尝试推进 tail
    this.cas(this.tail, oldTail, newNode)
  }

  // 出队（无锁）
  dequeue() {
    while (true) {
      const oldHead = this.head
      const oldTail = this.tail
      const next = oldHead.next

      if (oldHead === this.head) {
        if (oldHead === oldTail) {
          if (next === null) {
            // 队列为空
            return null
          }
          // 帮助推进 tail
          this.cas(this.tail, oldTail, next)
        } else {
          const value = next.value
          // 尝试推进 head
          if (this.cas(this.head, oldHead, next)) {
            return value
          }
        }
      }
    }
  }

  // CAS 操作（模拟）
  cas(target, expected, newValue) {
    // 在实际实现中，这是原子操作
    if (target === expected) {
      target = newValue
      return true
    }
    return false
  }

  size() {
    let count = 0
    let current = this.head.next
    while (current) {
      count++
      current = current.next
    }
    return count
  }
}

// 使用示例
const queue = new LockFreeQueue()

// 并发入队
for (let i = 0; i < 10; i++) {
  queue.enqueue(i)
}

console.log('Queue size:', queue.size())

// 并发出队
while (queue.size() > 0) {
  const value = queue.dequeue()
  console.log('Dequeued:', value)
}
```

## 📝 实际应用示例

### 综合并发处理系统

```js
// 综合并发处理系统
class ConcurrentProcessingSystem {
  constructor() {
    this.threadPool = new ThreadPoolExecutor(new ThreadPoolConfig())
    this.reactor = new Reactor()
    this.backpressure = new BackpressureController(100)
  }

  // 处理请求
  async processRequest(request) {
    // 1. 背压检查
    await this.backpressure.produce(request)

    // 2. 提交到线程池
    return new Promise((resolve, reject) => {
      this.threadPool.execute({
        run: async () => {
          try {
            // 3. 异步处理
            const result = await this.handleAsync(request)
            
            // 4. 发送响应事件
            this.reactor.emit('response', { requestId: request.id, result })
            
            // 5. 从背压缓冲区移除
            this.backpressure.consume()
            
            resolve(result)
          } catch (error) {
            reject(error)
          }
        }
      })
    })
  }

  async handleAsync(request) {
    // 模拟异步处理
    await new Promise(r => setTimeout(r, Math.random() * 100))
    return { id: request.id, processed: true }
  }

  // 注册响应处理器
  onResponse(handler) {
    this.reactor.register('response', handler)
  }

  start() {
    this.reactor.start()
  }
}

// 使用示例
async function testSystem() {
  const system = new ConcurrentProcessingSystem()
  system.start()

  // 注册响应处理
  system.onResponse((res) => {
    console.log(`Response: ${JSON.stringify(res)}`)
  })

  // 模拟并发请求
  const requests = Array.from({ length: 50 }, (_, i) => ({ id: i }))
  await Promise.all(requests.map(req => system.processRequest(req)))
}

testSystem()
```

## 🔗 相关概念

- **线程池**：复用线程资源，减少线程创建销毁开销
- **Reactor 模式**：基于事件循环的异步非阻塞 I/O 模型
- **背压控制**：防止生产者过快压垮消费者的流量控制机制
- **无锁数据结构**：使用 CAS 等原子操作实现的线程安全数据结构
- **协程**：轻量级的用户态线程，适合高并发场景
