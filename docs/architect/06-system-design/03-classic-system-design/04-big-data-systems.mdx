```js
// 实时计算平台
class StreamProcessingPlatform {
  constructor(config) {
    this.sources = new Map()
    this.processors = new Map()
    this.sinks = new Map()
    this.topologies = new Map()
  }

  // 注册数据源
  registerSource(name, source) {
    this.sources.set(name, source)
  }

  // 注册处理器
  registerProcessor(name, processor) {
    this.processors.set(name, processor)
  }

  // 注册数据汇
  registerSink(name, sink) {
    this.sinks.set(name, sink)
  }

  // 创建拓扑
  createTopology(name, config) {
    const topology = {
      name,
      sources: config.sources || [],
      processors: config.processors || [],
      sinks: config.sinks || [],
      running: false
    }
    this.topologies.set(name, topology)
    return topology
  }

  // 启动拓扑
  async startTopology(name) {
    const topology = this.topologies.get(name)
    if (!topology) {
      throw new Error(`Topology ${name} not found`)
    }

    console.log(`[StreamPlatform] Starting topology: ${name}`)
    topology.running = true

    // 启动数据源
    for (const sourceName of topology.sources) {
      const source = this.sources.get(sourceName)
      if (source) {
        await source.start()
      }
    }

    // 启动处理器
    for (const processorConfig of topology.processors) {
      const processor = this.processors.get(processorConfig.name)
      if (processor) {
        await processor.start(processorConfig)
      }
    }

    console.log(`[StreamPlatform] Topology ${name} started`)
  }

  // 停止拓扑
  async stopTopology(name) {
    const topology = this.topologies.get(name)
    if (!topology) {
      throw new Error(`Topology ${name} not found`)
    }

    console.log(`[StreamPlatform] Stopping topology: ${name}`)
    topology.running = false

    // 停止处理器
    for (const processorConfig of topology.processors) {
      const processor = this.processors.get(processorConfig.name)
      if (processor) {
        await processor.stop()
      }
    }

    // 停止数据源
    for (const sourceName of topology.sources) {
      const source = this.sources.get(sourceName)
      if (source) {
        await source.stop()
      }
    }

    console.log(`[StreamPlatform] Topology ${name} stopped`)
  }
}

// 流处理器
class StreamProcessor {
  constructor(config) {
    this.name = config.name
    this.running = false
    this.windowSize = config.windowSize || 5000  // 窗口大小（毫秒）
    this.slidingInterval = config.slidingInterval || 1000  // 滑动间隔（毫秒）
    this.state = new Map()
  }

  async start(config) {
    this.running = true
    this.source = config.source
    this.sink = config.sink
    console.log(`[StreamProcessor] Started: ${this.name}`)
  }

  async stop() {
    this.running = false
    console.log(`[StreamProcessor] Stopped: ${this.name}`)
  }

  // 处理消息
  async process(message) {
    // 应用业务逻辑
    const result = this.applyLogic(message)
    
    // 更新状态
    this.updateState(message, result)
    
    return result
  }

  // 应用业务逻辑
  applyLogic(message) {
    // 窗口聚合
    const windowKey = this.getWindowKey(message.timestamp)
    
    if (!this.state.has(windowKey)) {
      this.state.set(windowKey, {
        count: 0,
        sum: 0,
        messages: []
      })
    }
    
    const window = this.state.get(windowKey)
    window.count++
    window.sum += message.value || 1
    window.messages.push(message)
    
    return {
      windowKey,
      count: window.count,
      sum: window.sum,
      average: window.sum / window.count
    }
  }

  // 获取窗口键
  getWindowKey(timestamp) {
    return Math.floor(timestamp / this.windowSize)
  }

  // 更新状态
  updateState(message, result) {
    // 清理过期窗口
    const currentWindow = this.getWindowKey(Date.now())
    
    for (const [windowKey] of this.state) {
      if (currentWindow - windowKey > 10) {
        this.state.delete(windowKey)
      }
    }
  }

  // 获取聚合结果
  getAggregations() {
    const results = []
    
    for (const [windowKey, data] of this.state) {
      results.push({
        windowKey,
        count: data.count,
        sum: data.sum,
        average: data.sum / data.count
      })
    }
    
    return results
  }
}

  async batchWrite(records) {

      await this.write(record)
    }
  }

  query(filter = {}) {
    return this.data.filter(record => {
      return Object.keys(filter).every(key => record[key] === filter[key])
    })
  }
}
```

```js
// 日志聚合器（模拟 Logstash）
class LogAggregator {
  constructor(config) {
    this.name = config.name
    this.running = false
    this.filters = config.filters || []
  }

  async start(config) {
    this.running = true
    this.source = config.source
    this.storage = config.storage
    console.log(`[LogAggregator] Started: ${this.name}`)
  }

  async stop() {
    this.running = false
    console.log(`[LogAggregator] Stopped: ${this.name}`)
  }

  // 聚合日志
  async aggregate(logs) {
    const processedLogs = []

    for (const log of logs) {
      // 应用过滤器
      let processedLog = log

      for (const filter of this.filters) {
        processedLog = filter.apply(processedLog)
      }

      processedLogs.push(processedLog)
    }

    return processedLogs
  }

  // 处理日志
  async process() {
    if (!this.running) return

    // 从收集器获取日志
    const collector = this.source
    const logs = collector.getBuffer()

    if (logs.length > 0) {
      // 聚合日志
      const processedLogs = await this.aggregate(logs)

      // 存储日志
      if (this.storage) {
        await this.storage.index(processedLogs)
      }
    }
  }
}

// 日志存储（模拟 Elasticsearch）
class LogStorage {
  constructor(config) {
    this.index = config.index
    this.documents = new Map()
  }

  // 索引日志
  async index(logs) {
    for (const log of logs) {
      const id = this.generateId()
      this.documents.set(id, {
        ...log,
        _id: id,
        _index: this.index
      })
    }
    console.log(`[LogStorage] Indexed ${logs.length} logs to ${this.index}`)
  }

  // 搜索日志
  async search(query) {
    const results = []

    for (const doc of this.documents.values()) {
      if (this.matchesQuery(doc, query)) {
        results.push(doc)
      }
    }

    return results
  }

  // 匹配查询
  matchesQuery(doc, query) {
    return Object.keys(query).every(key => {
      const value = query[key]
      const docValue = doc[key]

      if (typeof value === 'string' && typeof docValue === 'string') {
        return docValue.toLowerCase().includes(value.toLowerCase())
      }

      return docValue === value
    })
  }

  generateId() {
    return `${Date.now()}_${Math.random().toString(36).substr(2, 9)}`
  }
}

// 日志过滤器
class LogFilter {
  constructor(type, config) {
    this.type = type
    this.config = config
  }

  apply(log) {
    switch (this.type) {
      case 'grok':
        return this.applyGrok(log)
      case 'date':
        return this.applyDate(log)
      case 'mutate':
        return this.applyMutate(log)
      default:
        return log
    }
  }

  applyGrok(log) {
    // 简化的 Grok 解析
    const pattern = this.config.pattern
    const match = log.message.match(pattern)
    
    if (match) {
      return {
        ...log,
        ...match.groups
      }
    }
    
    return log
  }

  applyDate(log) {
    const field = this.config.field
    const format = this.config.format || 'timestamp'
    
    if (log[field]) {
      return {
        ...log,
        [`${field}_parsed`]: new Date(log[field]).toISOString()
      }
    }
    
    return log
  }

  applyMutate(log) {
    const result = { ...log }
    
    // 添加字段
    if (this.config.add) {
      Object.assign(result, this.config.add)
    }
    
    // 删除字段
    if (this.config.remove) {
      for (const field of this.config.remove) {
        delete result[field]
      }
    }
    
    // 重命名字段
    if (this.config.rename) {
      for (const [oldName, newName] of Object.entries(this.config.rename)) {
        result[newName] = result[oldName]
        delete result[oldName]
      }
    }
    
    return result
  }
}
```

## 🎯 数据同步平台

### CDC 数据同步

CDC（Change Data Capture）实时捕获数据库变更并同步到目标系统。

```js
// 数据同步平台
class DataSyncPlatform {
  constructor(config) {
    this.sources = new Map()
    this.destinations = new Map()
    this.tasks = new Map()
  }

  // 注册数据源
  registerSource(name, source) {
    this.sources.set(name, source)
  }

  // 注册目标系统
  registerDestination(name, destination) {
    this.destinations.set(name, destination)
  }

  // 创建同步任务
  createTask(name, config) {
    const task = {
      name,
      source: config.source,
      destination: config.destination,
      tables: config.tables || [],
      mode: config.mode || 'full',  // full | incremental | cdc
      running: false,
      statistics: {
        recordsProcessed: 0,
        errors: 0,
        lastSyncTime: null
      }
    }
    this.tasks.set(name, task)
    return task
  }

  // 启动任务
  async startTask(name) {
    const task = this.tasks.get(name)
    if (!task) {
      throw new Error(`Task ${name} not found`)
    }

    console.log(`[DataSync] Starting task: ${name}`)
    task.running = true

    const source = this.sources.get(task.source)
    const destination = this.destinations.get(task.destination)

    if (task.mode === 'cdc') {
      await this.startCDC(task, source, destination)
    } else {
      await this.startBatchSync(task, source, destination)
    }

    console.log(`[DataSync] Task ${name} started`)
  }

  // 启动 CDC 同步
  async startCDC(task, source, destination) {
    // 订阅数据源变更
    source.subscribe(async (change) => {
      if (!task.running) return

      try {
        // 同步变更到目标
        await destination.applyChange(change)
        task.statistics.recordsProcessed++
        task.statistics.lastSyncTime = Date.now()
      } catch (error) {
        task.statistics.errors++
        console.error(`[DataSync] Error processing change:`, error.message)
      }
    })
  }

  // 启动批量同步
  async startBatchSync(task, source, destination) {
    for (const table of task.tables) {
      console.log(`[DataSync] Syncing table: ${table}`)
      
      // 从源读取数据
      const data = await source.readTable(table)
      
      // 写入目标
      await destination.writeTable(table, data)
      
      task.statistics.recordsProcessed += data.length
      task.statistics.lastSyncTime = Date.now()
    }
  }

  // 停止任务
  async stopTask(name) {
    const task = this.tasks.get(name)
    if (!task) {
      throw new Error(`Task ${name} not found`)
    }

    console.log(`[DataSync] Stopping task: ${name}`)
    task.running = false

    const source = this.sources.get(task.source)
    if (source && source.unsubscribe) {
      source.unsubscribe()
    }

    console.log(`[DataSync] Task ${name} stopped`)
  }

  // 获取任务统计
  getTaskStatistics(name) {
    const task = this.tasks.get(name)
    if (!task) {
      throw new Error(`Task ${name} not found`)
    }
    return task.statistics
  }
}

// 数据源（模拟 MySQL）
class MySQLSource {
  constructor(config) {
    this.host = config.host
    this.database = config.database
    this.tables = new Map()
    this.subscribers = []
    this.binlogPosition = 0
  }

  // 读取表数据
  async readTable(tableName) {
    const table = this.tables.get(tableName)
    if (!table) {
      throw new Error(`Table ${tableName} not found`)
    }
    return table.data
  }

  // 订阅变更
  subscribe(callback) {
    this.subscribers.push(callback)
  }

  // 取消订阅
  unsubscribe() {
    this.subscribers = []
  }

  // 模拟数据变更
  emitChange(change) {
    this.binlogPosition++
    for (const subscriber of this.subscribers) {
      subscriber({
        ...change,
        binlogPosition: this.binlogPosition
      })
    }
  }

  // 初始化表数据
  initializeTable(tableName, data) {
    this.tables.set(tableName, { data })
  }
}

// 目标系统（模拟 Kafka）
class KafkaDestination {
  constructor(config) {
    this.topic = config.topic
    this.producer = new Map()
  }

  // 应用变更
  async applyChange(change) {
    const key = `${change.table}_${change.id}`
    this.producer.set(key, {
      topic: this.topic,
      key,
      value: JSON.stringify(change),
      timestamp: Date.now()
    })
    console.log(`[KafkaDestination] Produced to ${this.topic}:`, change.operation, change.table, change.id)
  }

  // 写入表
  async writeTable(tableName, data) {
    console.log(`[KafkaDestination] Writing ${data.length} records from ${tableName}`)
    
    for (const record of data) {
      await this.applyChange({
        operation: 'insert',
        table: tableName,
        data: record,
        id: record.id
      })
    }
  }
}
```

```js
// 指标分析系统
class MetricsAnalysisSystem {
  constructor(config) {
    this.collectors = new Map()
    this.storage = new MetricsStorage()
    this.alerts = new Map()
    this.dashboard = new MetricsDashboard()
  }

  // 注册指标收集器
  registerCollector(name, collector) {
    this.collectors.set(name, collector)
  }

  // 注册告警规则
  registerAlert(name, rule) {
    this.alerts.set(name, rule)
  }

  // 收集指标
  async collectMetrics() {
    const allMetrics = []

    for (const [name, collector] of this.collectors) {
      const metrics = await collector.collect()
      
      for (const metric of metrics) {
        allMetrics.push({
          ...metric,
          source: name
        })
      }
    }

    // 存储指标
    await this.storage.write(allMetrics)

    // 检查告警
    await this.checkAlerts(allMetrics)

    return allMetrics
  }

  // 检查告警
  async checkAlerts(metrics) {
    for (const [name, rule] of this.alerts) {
      const matchedMetrics = metrics.filter(metric => {
        return rule.query(metric)
      })

      if (matchedMetrics.length > 0) {
        await rule.trigger(matchedMetrics)
      }
    }
  }

  // 查询指标
  async queryMetrics(query) {
    return await this.storage.query(query)
  }

  // 更新仪表板
  async updateDashboard() {
    const metrics = await this.queryMetrics({})
    this.dashboard.update(metrics)
  }
}

// 指标存储（模拟 Prometheus）
class MetricsStorage {
  constructor() {
    this.timeseries = new Map()
  }

  // 写入指标
  async write(metrics) {
    const timestamp = Date.now()

    for (const metric of metrics) {
      const key = this.getMetricKey(metric)
      
      if (!this.timeseries.has(key)) {
        this.timeseries.set(key, {
          name: metric.name,
          labels: metric.labels || {},
          samples: []
        })
      }

      const timeseries = this.timeseries.get(key)
      timeseries.samples.push({
        value: metric.value,
        timestamp
      })

      // 只保留最近 1000 个样本
      if (timeseries.samples.length > 1000) {
        timeseries.samples.shift()
      }
    }
  }

  // 查询指标
  async query(query) {
    const results = []

    for (const [key, timeseries] of this.timeseries) {
      if (this.matchesQuery(timeseries, query)) {
        results.push({
          metric: {
            name: timeseries.name,
            labels: timeseries.labels
          },
          values: timeseries.samples
        })
      }
    }

    return results
  }

  // 获取指标键
  getMetricKey(metric) {
    const labels = metric.labels || {}
    const labelString = Object.entries(labels)
      .sort()
      .map(([k, v]) => `${k}="${v}"`)
      .join(',')
    
    return `${metric.name}{${labelString}}`
  }

  // 匹配查询
  matchesQuery(timeseries, query) {
    if (query.name && timeseries.name !== query.name) {
      return false
    }

    if (query.labels) {
      for (const [key, value] of Object.entries(query.labels)) {
        if (timeseries.labels[key] !== value) {
          return false
        }
      }
    }

    return true
  }
}

// 指标收集器
class MetricsCollector {
  constructor(config) {
    this.name = config.name
    this.scrapeInterval = config.scrapeInterval || 10000
    this.targets = config.targets || []
  }

  // 收集指标
  async collect() {
    const metrics = []

    for (const target of this.targets) {
      const targetMetrics = await this.scrapeTarget(target)
      metrics.push(...targetMetrics)
    }

    return metrics
  }

  // 抓取目标
  async scrapeTarget(target) {
    // 模拟 HTTP 请求抓取指标
    const metrics = []

      labels: {
        instance: target.instance,
        job: target.job
      },
      value: Math.random() * 100
    })

    // 内存使用率
    metrics.push({
      name: 'memory_usage',
      labels: {
        instance: target.instance,
        job: target.job
      },
      value: Math.random() * 100
    })

    // 请求速率
    metrics.push({
      name: 'request_rate',
      labels: {
        instance: target.instance,
        job: target.job,
        endpoint: target.endpoint
      },
      value: Math.random() * 1000
    })

    return metrics
  }
}

class MetricsDashboard {
  constructor() {
    this.panels = []
  }

  // 添加面板
  addPanel(panel) {
    this.panels.push(panel)
  }

  // 更新仪表板
  update(metrics) {
    console.log('[Dashboard] Updating panels...')
    
    for (const panel of this.panels) {
      const panelMetrics = this.filterMetrics(metrics, panel.query)
      console.log(`[Dashboard] Panel "${panel.title}":`, panelMetrics.length, 'metrics')
    }
  }

  // 过滤指标
  filterMetrics(metrics, query) {
    return metrics.filter(metric => {
      if (query.name && metric.metric.name !== query.name) {
        return false
      }
      return true
    })
  }
}

// 告警规则
class AlertRule {
  constructor(config) {
    this.name = config.name
    this.query = config.query
    this.threshold = config.threshold
    this.duration = config.duration || 60000
    this.handler = config.handler
  }

  // 触发告警
  async trigger(metrics) {
    console.log(`[Alert] ${this.name} triggered by ${metrics.length} metrics`)
    
    if (this.handler) {
      await this.handler(metrics)
    }
  }
}
```
