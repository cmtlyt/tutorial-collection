# 日志管理

> **版本**：v1.0
> **最后更新**：2026-04-24

## 📚 概念介绍

日志是可观测性的三大支柱之一，记录了系统运行过程中发生的事件。本节将讲解如何构建企业级日志管理方案，实现日志的收集、存储、分析和可视化。

### 核心概念

- **结构化日志**：使用 JSON 等格式，便于机器解析和分析
- **日志级别**：DEBUG、INFO、WARN、ERROR、FATAL
- **日志聚合**：将分散的日志集中收集到统一的存储系统
- **日志分析**：通过搜索、过滤、聚合等操作提取有价值的信息

## 🎯 结构化日志

### 日志格式设计

**JSON 结构化日志示例：**
```javascript
const winston = require('winston');

const logger = winston.createLogger({
  level: 'info',
  format: winston.format.combine(
    winston.format.timestamp(),
    winston.format.errors({ stack: true }),
    winston.format.splat(),
    winston.format.json()
  ),
  defaultMeta: {
    service: 'my-app',
    environment: process.env.NODE_ENV || 'development',
    version: process.env.APP_VERSION || '1.0.0'
  },
  transports: [
    new winston.transports.Console({
      format: winston.format.simple()
    }),
    new winston.transports.File({ 
      filename: 'error.log', 
      level: 'error' 
    }),
    new winston.transports.File({ 
      filename: 'combined.log' 
    })
  ]
});

// 使用示例
logger.info('User logged in', {
  userId: '12345',
  username: 'john.doe',
  ip: '192.168.1.1',
  userAgent: 'Mozilla/5.0...',
  duration: 45
});

logger.error('Database connection failed', {
  error: 'Connection timeout',
  database: 'postgres',
  host: 'localhost',
  port: 5432,
  stack: new Error().stack
});

logger.warn('High memory usage', {
  memoryUsage: process.memoryUsage(),
  threshold: 500 * 1024 * 1024
});
```

### 日志级别规范

```javascript
// 日志级别使用规范
const LogLevel = {
  DEBUG: 'debug',    // 调试信息，仅在开发环境使用
  INFO: 'info',      // 一般信息，记录正常的业务流程
  WARN: 'warn',      // 警告信息，不影响业务但需要关注
  ERROR: 'error',    // 错误信息，影响业务但可以恢复
  FATAL: 'fatal'     // 致命错误，导致服务不可用
};

// 日志记录最佳实践
function processPayment(paymentData) {
  logger.info('Payment processing started', {
    paymentId: paymentData.id,
    amount: paymentData.amount,
    currency: paymentData.currency
  });

  try {
    // 业务逻辑
    const result = await paymentService.process(paymentData);
    
    logger.info('Payment processed successfully', {
      paymentId: paymentData.id,
      transactionId: result.transactionId,
      duration: result.duration
    });
    
    return result;
  } catch (error) {
    logger.error('Payment processing failed', {
      paymentId: paymentData.id,
      error: error.message,
      code: error.code,
      stack: error.stack,
      retryable: error.retryable
    });
    
    throw error;
  }
}
```

### Logstash 配置

**logstash.conf：**
```conf
input {
  # 从 Beats 接收日志
  beats {
    port => 5044
  }
  
  # 从文件读取日志
  file {
    path => "/var/log/*.log"
    start_position => "beginning"
    sincedb_path => "/dev/null"
  }
}

filter {
  # 解析 JSON 日志
  if [message] =~ /^\{.*\}$/ {
    json {
      source => "message"
    }
  }
  
  # 添加时间戳
  date {
    match => ["timestamp", "ISO8601"]
    target => "@timestamp"
  }
  
  # 添加主机信息
  mutate {
    add_field => {
      "[host][name]" => "${HOSTNAME}"
      "[host][ip]" => "${HOST_IP}"
    }
  }
  
  # 根据日志级别添加标签
  if [level] == "error" or [level] == "fatal" {
    mutate {
      add_tag => ["error"]
    }
  }
  
  # 移除不需要的字段
  mutate {
    remove_field => ["message"]
  }
  
  # 用户代理解析
  if [userAgent] {
    useragent {
      source => "userAgent"
      target => "ua"
    }
  }
  
  # IP 地址地理位置解析
  if [ip] {
    geoip {
      source => "ip"
      target => "geo"
    }
  }
}

output {
  # 输出到 Elasticsearch
  elasticsearch {
    hosts => ["elasticsearch:9200"]
    index => "logs-%{+YYYY.MM.dd}"
    user => "elastic"
    password => "${ES_PASSWORD}"
  }
  
  # 输出到标准输出（调试用）
  stdout {
    codec => rubydebug
  }
}
```

## 🔧 日志采集

### Filebeat 配置

**filebeat.yml：**
```yaml
filebeat.inputs:
- type: log
  enabled: true
  paths:
    - /var/log/my-app/*.log
  fields:
    service: my-app
    environment: production
  fields_under_root: true
  multiline:
    pattern: '^\['
    negate: true
    match: after

- type: container
  enabled: true
  paths:
    - /var/log/containers/*.log
  processors:
  - add_kubernetes_metadata:
      host: ${NODE_NAME}
      matchers:
      - logs_path:
          logs_path: "/var/log/containers/"

# 输出到 Logstash
output.logstash:
  hosts: ["logstash:5044"]
  loadbalance: true

# 输出到 Elasticsearch（备选）
# output.elasticsearch:
#   hosts: ["elasticsearch:9200"]
#   index: "filebeat-%{+yyyy.MM.dd}"

# 处理器配置
processors:
  - add_host_metadata:
      when.not.contains.tags: forwarded
  - add_cloud_metadata: ~
  - add_docker_metadata: ~
  - add_kubernetes_metadata: ~

# 日志级别配置
logging.level: info
logging.to_files: true
logging.files:
  path: /var/log/filebeat
  name: filebeat
  keepfiles: 7
  permissions: 0644
```

## 📝 日志分析

### Elasticsearch 查询

```javascript
// 基础查询
GET /logs-*/_search
{
  "query": {
    "match_all": {}
  }
}

// 按日志级别查询
GET /logs-*/_search
{
  "query": {
    "term": {
      "level": "error"
    }
  }
}

// 按时间范围查询
GET /logs-*/_search
{
  "query": {
    "range": {
      "@timestamp": {
        "gte": "now-1h",
        "lte": "now"
      }
    }
  }
}

// 组合查询
GET /logs-*/_search
{
  "query": {
    "bool": {
      "must": [
        { "term": { "level": "error" } },
        { "term": { "service": "my-app" } }
      ],
      "filter": [
        { "range": { "@timestamp": { "gte": "now-1h" } } }
      ]
    }
  }
}

// 聚合统计
GET /logs-*/_search
{
  "size": 0,
  "aggs": {
    "by_level": {
      "terms": {
        "field": "level"
      }
    },
    "by_service": {
      "terms": {
        "field": "service"
      }
    },
    "errors_over_time": {
      "date_histogram": {
        "field": "@timestamp",
        "calendar_interval": "1h"
      },
      "aggs": {
        "top_errors": {
          "top_hits": {
            "size": 10,
            "sort": [
              { "@timestamp": { "order": "desc" } }
            ]
          }
        }
      }
    }
  }
}
```
