# 主流云平台与服务

> **版本**：v1.0
> **最后更新**：2026-04-24

## 📚 概念介绍

云平台提供了弹性的计算、存储、网络等基础设施服务，帮助企业快速构建和部署应用。本节将讲解主流云平台的核心服务和 Serverless 平台。

### 云服务模型

```
SaaS（软件即服务）
    ↑
PaaS（平台即服务）
    ↑
IaaS（基础设施即服务）
    ↑
物理硬件
```

### 主流云平台对比

| 平台 | 优势 | 核心服务 | 适用场景 |
|------|------|---------|---------|
| **阿里云** | 国内市场领先、生态完善 | ECS、RDS、OSS、SLB | 国内业务、企业应用 |
| **AWS** | 全球覆盖、服务最全 | EC2、RDS、S3、Lambda | 全球业务、初创公司 |
| **Azure** | 企业级、微软生态 | VM、SQL Database、Blob | 企业应用、混合云 |
| **GCP** | 数据分析、AI 能力强 | Compute Engine、Cloud SQL、BigQuery | 数据分析、AI 应用 |

## 💾 存储服务

### 对象存储

**阿里云 OSS：**
```javascript
const OSS = require('ali-oss');

const client = new OSS({
  region: 'oss-cn-hangzhou',
  accessKeyId: process.env.OSS_ACCESS_KEY_ID,
  accessKeySecret: process.env.OSS_ACCESS_KEY_SECRET,
  bucket: 'my-bucket'
});

// 上传文件
async function uploadFile(filePath, objectName) {
  try {
    const result = await client.put(objectName, filePath);
    console.log('Upload success:', result.url);
    return result;
  } catch (error) {
    console.error('Upload failed:', error);
    throw error;
  }
}

// 下载文件
async function downloadFile(objectName, localPath) {
  try {
    await client.get(objectName, localPath);
    console.log('Download success');
  } catch (error) {
    console.error('Download failed:', error);
    throw error;
  }
}

// 删除文件
async function deleteFile(objectName) {
  try {
    await client.delete(objectName);
    console.log('Delete success');
  } catch (error) {
    console.error('Delete failed:', error);
    throw error;
  }
}

// 列出文件
async function listFiles(prefix) {
  try {
    const result = await client.list({
      prefix: prefix,
      'max-keys': 100
    });
    return result.objects;
  } catch (error) {
    console.error('List failed:', error);
    throw error;
  }
}
```

**AWS S3：**
```javascript
const { S3Client, PutObjectCommand, GetObjectCommand, DeleteObjectCommand, ListObjectsV2Command } = require('@aws-sdk/client-s3');

const client = new S3Client({
  region: 'us-west-2',
  credentials: {
    accessKeyId: process.env.AWS_ACCESS_KEY_ID,
    secretAccessKey: process.env.AWS_SECRET_ACCESS_KEY
  }
});

// 上传文件
async function uploadFile(bucketName, key, body) {
  try {
    const command = new PutObjectCommand({
      Bucket: bucketName,
      Key: key,
      Body: body
    });
    const result = await client.send(command);
    console.log('Upload success:', result.ETag);
    return result;
  } catch (error) {
    console.error('Upload failed:', error);
    throw error;
  }
}

// 下载文件
async function downloadFile(bucketName, key) {
  try {
    const command = new GetObjectCommand({
      Bucket: bucketName,
      Key: key
    });
    const result = await client.send(command);
    return result.Body;
  } catch (error) {
    console.error('Download failed:', error);
    throw error;
  }
}

// 删除文件
async function deleteFile(bucketName, key) {
  try {
    const command = new DeleteObjectCommand({
      Bucket: bucketName,
      Key: key
    });
    await client.send(command);
    console.log('Delete success');
  } catch (error) {
    console.error('Delete failed:', error);
    throw error;
  }
}

// 列出文件
async function listFiles(bucketName, prefix) {
  try {
    const command = new ListObjectsV2Command({
      Bucket: bucketName,
      Prefix: prefix,
      MaxKeys: 100
    });
    const result = await client.send(command);
    return result.Contents;
  } catch (error) {
    console.error('List failed:', error);
    throw error;
  }
}
```

## ⚡ Serverless 平台

### 函数计算

**阿里云函数计算：**
```javascript
// 函数代码
exports.handler = async (event, context) => {
  console.log('Received event:', JSON.stringify(event, null, 2));
  
  try {
    // 解析事件
    const body = JSON.parse(event.body || '{}');
    
    // 业务逻辑
    const result = {
      message: 'Hello from Function Compute!',
      input: body,
      timestamp: new Date().toISOString()
    };
    
    // 返回响应
    return {
      statusCode: 200,
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(result)
    };
  } catch (error) {
    console.error('Error:', error);
    
    return {
      statusCode: 500,
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        error: error.message
      })
    };
  }
};
```

### Serverless 最佳实践

**1. 冷启动优化：**
```javascript
// 预热连接
let dbConnection = null;

async function getDBConnection() {
  if (!dbConnection) {
    dbConnection = await createDBConnection();
  }
  return dbConnection;
}

exports.handler = async (event) => {
  const db = await getDBConnection();
  // 使用数据库连接
};
```

**2. 资源限制：**
```javascript
// 设置合理的内存和超时时间
const AWS = require('aws-sdk');
const lambda = new AWS.Lambda();

lambda.updateFunctionConfiguration({
  FunctionName: 'my-function',
  MemorySize: 256,
  Timeout: 30
}).promise();
```

**3. 错误处理：**
```javascript
exports.handler = async (event) => {
  try {
    // 业务逻辑
    return await processEvent(event);
  } catch (error) {
    // 记录错误
    console.error('Error:', error);
    
    // 发送到错误追踪服务
    await sendToErrorTracking(error);
    
    // 抛出错误以触发重试
    throw error;
  }
};
```
