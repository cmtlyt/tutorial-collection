## 📚 ER 建模

### 实体关系模型

```js
// ER 模型核心概念
const erModel = {
  entity: {
    description: "实体",
    example: "用户、订单、商品",
    notation: "矩形"
  },
  attribute: {
    description: "属性",
    example: "姓名、年龄、价格",
    notation: "椭圆"
  },
  relationship: {
    description: "关系",
    example: "购买、关注、雇佣",
    notation: "菱形"
  },
  cardinality: {
    description: "基数",
    types: [
      "一对一（1:1）",
      "一对多（1:N）",
      "多对多（M:N）"
    ]
  }
}
```

### 数据库规范化

```js
// 规范化范式
const normalizationForms = {
  firstNF: {
    name: "第一范式（1NF）",
    requirement: "每个属性都是原子值，不可再分",
    example: `
-- 违反 1NF
用户: "Alice, Bob" (多个值在一个字段)

-- 符合 1NF
用户1: "Alice"
用户2: "Bob"
    `
  },
  secondNF: {
    name: "第二范式（2NF）",
    requirement: "满足 1NF，且非主属性完全依赖于主键",
    example: `
-- 违反 2NF
订单表: (订单ID, 商品ID, 商品名称, 数量)
-- 商品名称只依赖于商品ID，不依赖于(订单ID, 商品ID)

-- 符合 2NF
订单表: (订单ID, 商品ID, 数量)
商品表: (商品ID, 商品名称)
    `
  },
  thirdNF: {
    name: "第三范式（3NF）",
    requirement: "满足 2NF，且非主属性不传递依赖于主键",
    example: `
-- 违反 3NF
用户表: (用户ID, 姓名, 部门ID, 部门名称)
-- 部门名称传递依赖于用户ID

-- 符合 3NF
用户表: (用户ID, 姓名, 部门ID)
部门表: (部门ID, 部门名称)
    `
  }
}
```

## 📚 维度建模

### 星型模型

```js
// 星型模型结构
const starSchema = {
  description: "一个事实表连接多个维度表",
  structure: {
    factTable: {
      description: "事实表",
      features: [
        "存储度量值（数值）",
        "包含外键引用维度表",
        "数据量大，增长快"
      ],
      example: "销售事实表：订单ID、用户ID、商品ID、销售额、数量"
    },
    dimensionTable: {
      description: "维度表",
      features: [
        "存储描述性属性",
        "数据量相对较小",
        "变化缓慢"
      ],
      examples: [
        "用户维度：用户ID、姓名、年龄、性别",
        "商品维度：商品ID、名称、类别、品牌",
        "时间维度：日期ID、年、月、日、星期"
      ]
    }
  },
  advantages: [
    "查询性能好（JOIN 少）",
    "结构简单，易于理解",
    "适合快速分析"
  ],
  disadvantages: [
    "数据冗余",
    "存储空间占用大"
  ]
}
```

### 雪花模型

```js
// 雪花模型结构
const snowflakeSchema = {
  description: "维度表进一步规范化，形成层次结构",
  structure: {
    factTable: "与星型模型相同",
    normalizedDimensions: {
      description: "规范化的维度表",
      example: `
商品维度表: 商品ID, 名称, 类别ID, 品牌ID
类别维度表: 类别ID, 类别名称, 父类别ID
品牌维度表: 品牌ID, 品牌名称, 国家ID
国家维度表: 国家ID, 国家名称, 大洲
      `
    }
  },
  advantages: [
    "数据冗余少",
    "存储空间占用小",
    "维护方便"
  ],
  disadvantages: [
    "查询性能差（JOIN 多）",
    "结构复杂"
  ]
}
```

### 星型 vs 雪花

| 特性 | 星型模型 | 雪花模型 |
|------|---------|---------|
| 查询性能 | 高（JOIN 少） | 中（JOIN 多） |
| 存储空间 | 大（数据冗余） | 小（规范化） |
| 结构复杂度 | 简单 | 复杂 |
| 维护性 | 中 | 高 |
| 适用场景 | 快速分析、数据集市 | 复杂分析、企业级数仓 |

## 📚 数据仓库分层设计

### 分层架构

```js
// 数据仓库分层模型
const dataWarehouseLayers = {
  ODS: {
    name: "操作数据层（Operational Data Store）",
    description: "原始数据存储层",
    features: [
      "与源系统数据结构一致",
      "不做清洗和转换",
      "保留历史数据"
    ],
    example: "原始订单表、原始用户表"
  },
  DWD: {
    name: "数据仓库明细层（Data Warehouse Detail）",
    description: "明细数据层",
    features: [
      "数据清洗和规范化",
      "统一维度和度量",
      "保持明细粒度"
    ],
    example: "订单明细表、用户明细表"
  },
  DWS: {
    name: "数据仓库汇总层（Data Warehouse Summary）",
    description: "轻度汇总层",
    features: [
      "按主题域汇总",
      "预计算常用指标",
      "支持多维分析"
    ],
    example: "日销售汇总、月用户活跃度"
  },
  ADS: {
    name: "应用数据层（Application Data Store）",
    description: "应用数据层",
    features: [
      "面向业务应用",
      "高度汇总",
      "直接用于报表和 BI"
    ],
    example: "销售报表、用户增长报表"
  }
}
```

### 分层示例

```sql
-- ODS 层：原始订单数据
CREATE TABLE ods_order_raw (
    order_id BIGINT,
    user_id BIGINT,
    product_id BIGINT,
    amount DECIMAL(10,2),
    status VARCHAR(20),
    created_at TIMESTAMP,
    updated_at TIMESTAMP
) PARTITIONED BY (dt STRING);

-- DWD 层：清洗后的订单明细
CREATE TABLE dwd_order_detail (
    order_id BIGINT,
    user_id BIGINT,
    product_id BIGINT,
    amount DECIMAL(10,2),
    status VARCHAR(20),
    order_date DATE,
    created_at TIMESTAMP,
    updated_at TIMESTAMP
) PARTITIONED BY (dt STRING);

-- DWS 层：日销售汇总
CREATE TABLE dws_sales_daily (
    product_id BIGINT,
    sales_date DATE,
    order_count BIGINT,
    total_amount DECIMAL(15,2),
    avg_amount DECIMAL(10,2)
) PARTITIONED BY (dt STRING);

-- ADS 层：销售报表
CREATE TABLE ads_sales_report (
    product_id BIGINT,
    product_name STRING,
    period STRING,
    total_sales DECIMAL(15,2),
    growth_rate DECIMAL(5,2)
);
```

## 📚 数据湖与湖仓一体

### 数据湖架构

```js
// 数据湖特点
const dataLake = {
  description: "存储原始数据的集中式存储库",
  features: [
    "存储所有格式的数据（结构化、半结构化、非结构化）",
    "Schema-on-read（读取时定义模式）",
    "低成本存储",
    "支持大规模数据处理"
  ],
  advantages: [
    "数据存储成本低",
    "支持多种数据类型",
    "灵活的数据处理"
  ],
  disadvantages: [
    "缺乏数据治理",
    "查询性能相对较差",
    "数据质量难以保证"
  ]
}
```

### 湖仓一体架构

```js
// 湖仓一体（Lakehouse）
const lakehouse = {
  description: "结合数据湖的灵活性和数据仓库的性能",
  features: [
    "数据湖的低成本存储",
    "数据仓库的 ACID 事务",
    "Schema enforcement（模式强制）",
    "支持 BI 和机器学习",
    "统一的数据治理"
  ],
  technologies: [
    "Delta Lake",
    "Apache Iceberg",
    "Apache Hudi"
  ],
  advantages: [
    "存储成本低",
    "查询性能高",
    "支持事务",
    "统一的数据管理"
  ]
}
```

### 数据架构演进

```
单体数据库
    ↓
数据仓库（结构化数据）
    ↓
数据湖（多源异构数据）
    ↓
湖仓一体（Lakehouse）
    ↓
实时数据平台（流批一体）
```

## 📝 实际应用示例

### 电商数据仓库设计

```sql
-- 1. ODS 层：原始数据
CREATE TABLE ods_orders (
    order_id BIGINT,
    user_id BIGINT,
    order_items ARRAY<STRUCT<product_id BIGINT, quantity INT, price DECIMAL(10,2)>>,
    total_amount DECIMAL(10,2),
    status VARCHAR(20),
    created_at TIMESTAMP,
    updated_at TIMESTAMP
) PARTITIONED BY (dt STRING);

-- 2. DWD 层：订单明细（展开商品）
CREATE TABLE dwd_order_items (
    order_id BIGINT,
    user_id BIGINT,
    product_id BIGINT,
    quantity INT,
    price DECIMAL(10,2),
    amount DECIMAL(10,2),
    status VARCHAR(20),
    order_date DATE,
    created_at TIMESTAMP
) PARTITIONED BY (dt STRING);

-- 3. DWS 层：商品日销售汇总
CREATE TABLE dws_product_sales_daily (
    product_id BIGINT,
    sales_date DATE,
    order_count BIGINT,
    total_quantity BIGINT,
    total_amount DECIMAL(15,2),
    avg_price DECIMAL(10,2),
    distinct_users BIGINT
) PARTITIONED BY (dt STRING);

-- 4. ADS 层：商品销售趋势
CREATE TABLE ads_product_trend (
    product_id BIGINT,
    product_name STRING,
    period STRING,
    total_sales DECIMAL(15,2),
    order_count BIGINT,
    mom_growth DECIMAL(5,2),
    yoy_growth DECIMAL(5,2)
);
```
