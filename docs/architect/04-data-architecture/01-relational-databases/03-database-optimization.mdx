## 📚 查询优化

### 执行计划分析

```sql
-- 使用 EXPLAIN 分析查询执行计划
EXPLAIN SELECT 
    o.id,
    o.order_no,
    u.username,
    p.product_name,
    oi.quantity,
    oi.price
FROM orders o
INNER JOIN users u ON o.user_id = u.id
INNER JOIN order_items oi ON o.id = oi.order_id
INNER JOIN products p ON oi.product_id = p.id
WHERE o.status = 'pending'
AND o.created_at >= '2025-01-01'
ORDER BY o.created_at DESC
LIMIT 100;

-- 执行计划关键字段说明
const explainColumns = {
  id: "查询序列号，表示执行顺序",
  select_type: "查询类型（SIMPLE/PRIMARY/SUBQUERY/DERIVED）",
  type: "访问类型（性能：const > eq_ref > ref > range > index > ALL）",
  possible_keys: "可能使用的索引",
  key: "实际使用的索引",
  key_len: "使用的索引长度",
  ref: "索引引用的列",
  rows: "预估扫描行数",
  filtered: "过滤后的行数百分比",
  Extra: "额外信息（Using index/Using filesort/Using temporary）"
```

### SQL 优化技巧

```sql
-- 1. 避免 SELECT *
-- 错误示例
SELECT * FROM orders WHERE user_id = 123;

-- 正确示例：只查询需要的字段
SELECT id, order_no, status, total_amount 
FROM orders 
WHERE user_id = 123;

-- 2. 避免在索引列上使用函数
-- 错误示例
SELECT * FROM users WHERE YEAR(created_at) = 2025;

-- 正确示例：使用范围查询
SELECT * FROM users 
WHERE created_at >= '2025-01-01' 
AND created_at < '2026-01-01';

-- 3. 使用 UNION ALL 替代 UNION（如果不需要去重）
-- UNION 会进行去重操作，性能较差
SELECT user_id FROM orders WHERE status = 'pending'
UNION
SELECT user_id FROM orders WHERE status = 'processing';

-- UNION ALL 不会去重，性能更好
SELECT user_id FROM orders WHERE status = 'pending'
UNION ALL
SELECT user_id FROM orders WHERE status = 'processing';

-- 4. 使用 LIMIT 限制结果集
SELECT * FROM orders 
WHERE status = 'pending' 
ORDER BY created_at DESC 
LIMIT 20;
```

## 📚 分库分表

### 垂直分库 vs 水平分表

```js
// 分库分表策略对比
const shardingStrategies = {
  vertical: {
    description: "按业务模块拆分",
    example: "用户库、订单库、商品库",
    advantages: [
      "业务边界清晰",
      "便于团队协作",
      "降低单库压力"
    ],
    disadvantages: [
      "跨库 JOIN 困难",
      "分布式事务复杂"
    ]
  },
  horizontal: {
    description: "按数据量拆分",
    example: "订单表按用户 ID 取模分 16 张表",
    advantages: [
      "解决单表数据量过大",
      "提升查询性能"
    ],
    disadvantages: [
      "跨分片查询复杂",
      "数据迁移困难"
    ]
  }
}
```

### 分片键选择

```js
// 分片键设计原则
const shardingKeyDesign = {
  principles: [
    "查询频率高的字段",
    "数据分布均匀",
    "避免热点数据",
    "考虑业务扩展性"
  ],
  commonShardingKeys: {
    user_id: "用户相关数据，按用户 ID 取模",
    order_id: "订单数据，按订单 ID 哈希",
    date: "按时间范围分片（适合报表查询）",
    region: "按地域分片（就近访问）"
  }
}

// 分片算法示例
const shardAlgorithms = {
  modulo: {
    name: "取模分片",
    formula: "shard = id % shard_count",
    example: "user_id % 16",
    advantages: "数据分布均匀",
    disadvantages: "扩容困难（需要数据迁移）"
  },
  range: {
    name: "范围分片",
    formula: "shard = (id - min_id) / range_size",
    example: "user_id / 1000000",
    advantages: "范围查询高效",
    disadvantages: "数据分布不均"
  },
  hash: {
    name: "哈希分片",
    formula: "shard = hash(id) % shard_count",
    example: "MD5(user_id) % 16",
    advantages: "数据分布均匀",
    disadvantages: "不支持范围查询"
  }
}
```

## 📝 实际应用示例

### 综合优化方案

```sql
-- 场景：优化订单查询性能

-- 1. 创建合适的索引
CREATE INDEX idx_order_user_status_created ON orders(user_id, status, created_at);
CREATE INDEX idx_order_status_created ON orders(status, created_at);

-- 2. 优化查询语句
SELECT 
    o.id,
    o.order_no,
    o.status,
    o.total_amount,
    o.created_at
FROM orders o
WHERE o.user_id = ?
AND o.status IN ('pending', 'processing')
AND o.created_at >= ?
ORDER BY o.created_at DESC
LIMIT 20;

-- 3. 分析执行计划
EXPLAIN SELECT ...;

-- 4. 监控慢查询
-- 开启慢查询日志
SET GLOBAL slow_query_log = 'ON';
```
