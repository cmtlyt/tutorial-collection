## 📚 Neo4j 与 Cypher

### Cypher 查询语言

```cypher
-- Cypher 基本语法示例

-- 1. 创建节点
CREATE (p:Person {name: 'Alice', age: 30})
CREATE (c:Company {name: 'TechCorp', founded: 2010})

-- 2. 创建关系
MATCH (p:Person {name: 'Alice'})
MATCH (c:Company {name: 'TechCorp'})
CREATE (p)-[:WORKS_FOR {since: 2020, position: 'Engineer'}]->(c)

-- 3. 查询节点
MATCH (p:Person)
WHERE p.age > 25
RETURN p.name, p.age
ORDER BY p.age DESC
LIMIT 10

-- 4. 查询关系和路径
MATCH path = (p1:Person)-[:FRIENDS]->(p2:Person)
WHERE p1.name = 'Alice'
RETURN path

-- 5. 多跳查询（朋友的朋友）
MATCH (p:Person {name: 'Alice'})-[:FRIENDS]->(friend)-[:FRIENDS]->(fof)
RETURN fof.name

-- 6. 最短路径
MATCH (start:Person {name: 'Alice'}), (end:Person {name: 'Bob'})
MATCH path = shortestPath((start)-[*]-(end))
RETURN path

-- 7. 聚合查询
MATCH (p:Person)-[:WORKS_FOR]->(c:Company)
RETURN c.name, count(p) as employee_count
ORDER BY employee_count DESC
```

### 图遍历模式

```js
// 常见的图遍历模式
const graphTraversalPatterns = {
  oneHop: {
    description: "一跳查询",
    cypher: "MATCH (a)-[r]->(b) WHERE a.name = 'Alice' RETURN b",
    use: "查询直接关联的节点"
  },
  multiHop: {
    description: "多跳查询",
    cypher: "MATCH (a)-[*2..3]->(b) WHERE a.name = 'Alice' RETURN b",
    use: "查询多度关系"
  },
  bidirectional: {
    description: "双向查询",
    cypher: "MATCH (a)-[r]-(b) WHERE a.name = 'Alice' RETURN b",
    use: "查询无向关系"
  },
  shortestPath: {
    description: "最短路径",
    cypher: "MATCH p = shortestPath((a)-[*]-(b)) RETURN p",
    use: "查找两个节点间的最短路径"
  },
  allPaths: {
    description: "所有路径",
    cypher: "MATCH p = (a)-[*]-(b) RETURN p",
    use: "查找所有可能的路径"
  }
}
```

## 📚 图建模最佳实践

### 社交网络建模

```cypher
-- 社交网络数据模型

-- 创建用户节点
CREATE (alice:User {id: 'user1', name: 'Alice', age: 30})
CREATE (bob:User {id: 'user2', name: 'Bob', age: 25})
CREATE (charlie:User {id: 'user3', name: 'Charlie', age: 35})

-- 创建关注关系
CREATE (alice)-[:FOLLOWS {since: datetime('2025-01-01')}]->(bob)
CREATE (bob)-[:FOLLOWS {since: datetime('2025-02-01')}]->(alice)
CREATE (alice)-[:FOLLOWS {since: datetime('2025-03-01')}]->(charlie)

-- 创建帖子节点
CREATE (post1:Post {id: 'post1', content: 'Hello World!', createdAt: datetime('2025-04-20')})
CREATE (post2:Post {id: 'post2', content: 'Graph databases are awesome!', createdAt: datetime('2025-04-21')})

-- 创建发布关系
CREATE (alice)-[:POSTED]->(post1)
CREATE (bob)-[:POSTED]->(post2)

-- 创建点赞关系
CREATE (bob)-[:LIKED {at: datetime('2025-04-20')}]->(post1)
CREATE (charlie)-[:LIKED {at: datetime('2025-04-20')}]->(post1)
CREATE (alice)-[:LIKED {at: datetime('2025-04-21')}]->(post2)
```

### 推荐系统建模

```cypher
-- 推荐系统：基于协同过滤

-- 查询共同好友
MATCH (u1:User {name: 'Alice'})-[:FRIENDS]->(common)-[:FRIENDS]->(u2:User)
WHERE u1 <> u2
RETURN u2.name, count(common) as mutual_friends
ORDER BY mutual_friends DESC
LIMIT 10

-- 查询推荐关注（好友的好友）
MATCH (u:User {name: 'Alice'})-[:FRIENDS]->(friend)-[:FRIENDS]->(recommendation)
WHERE NOT (u)-[:FRIENDS]->(recommendation) AND u <> recommendation
RETURN recommendation.name, count(friend) as common_friends
ORDER BY common_friends DESC
LIMIT 10

-- 商品推荐（基于购买行为）
MATCH (u:User {name: 'Alice'})-[:PURCHASED]->(:Product)-[:PURCHASED]->(other:Product)
WHERE NOT (u)-[:PURCHASED]->(other)
RETURN other.name, count(*) as purchase_count
ORDER BY purchase_count DESC
LIMIT 10
```

### 知识图谱建模

```cypher
-- 知识图谱示例

-- 创建实体
CREATE (person1:Person {name: 'Elon Musk'})
CREATE (company1:Company {name: 'Tesla'})
CREATE (company2:Company {name: 'SpaceX'})
CREATE (company3:Company {name: 'Twitter'})

-- 创建关系
CREATE (person1)-[:FOUNDED {year: 2003}]->(company1)
CREATE (person1)-[:FOUNDED {year: 2002}]->(company2)
CREATE (person1)-[:ACQUIRED {year: 2022}]->(company3)

-- 查询某人的所有公司
MATCH (p:Person {name: 'Elon Musk'})-[r]->(c:Company)
RETURN c.name, type(r) as relationship, r.year

-- 查询公司间的关联
MATCH (c1:Company)<-[:FOUNDED]-(p:Person)-[:FOUNDED]->(c2:Company)
WHERE c1.name = 'Tesla'
RETURN c2.name
```

## 📝 实际应用示例

### 欺诈检测

```cypher
-- 欺诈检测场景：检测异常交易模式

-- 1. 检测循环转账（洗钱模式）
MATCH (a:Account)-[t1:TRANSFERRED]->(b:Account)-[t2:TRANSFERRED]->(c:Account)-[t3:TRANSFERRED]->(a)
WHERE t1.amount = t2.amount AND t2.amount = t3.amount
RETURN a, b, c, t1.amount as suspicious_amount

-- 2. 检测短时间内的多笔大额交易
MATCH (a:Account)-[t:TRANSFERRED]->(:Account)
WHERE t.amount > 10000
WITH a, count(t) as large_transactions, sum(t.amount) as total_amount
WHERE large_transactions > 5
RETURN a.id, large_transactions, total_amount

-- 3. 检测共享账户（多个账户使用同一设备）
MATCH (d:Device)<-[:USED]-(a1:Account)
MATCH (d)<-[:USED]-(a2:Account)
WHERE a1 <> a2
RETURN d.id, collect(DISTINCT a1.id) as accounts
WHERE size(accounts) > 5
```

### 路径规划

```cypher
-- 路径规划场景：查找最短路线

-- 创建地点和道路
CREATE (a:Location {name: 'A'})
CREATE (b:Location {name: 'B'})
CREATE (c:Location {name: 'C'})
CREATE (d:Location {name: 'D'})

CREATE (a)-[:ROAD {distance: 10}]->(b)
CREATE (b)-[:ROAD {distance: 15}]->(c)
CREATE (c)-[:ROAD {distance: 20}]->(d)
CREATE (a)-[:ROAD {distance: 30}]->(d)

-- 查找最短路径
MATCH (start:Location {name: 'A'}), (end:Location {name: 'D'})
MATCH path = shortestPath((start)-[*]-(end))
RETURN path, reduce(total = 0, r IN relationships(path) | total + r.distance) as total_distance

-- 查找所有路径并按距离排序
MATCH (start:Location {name: 'A'}), (end:Location {name: 'D'})
MATCH path = (start)-[*]-(end)
WITH path, reduce(total = 0, r IN relationships(path) | total + r.distance) as total_distance
RETURN path, total_distance
ORDER BY total_distance
LIMIT 5
```
