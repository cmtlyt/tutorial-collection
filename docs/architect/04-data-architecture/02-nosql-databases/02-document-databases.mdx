## 📚 文档建模策略

### 嵌入 vs 引用

```js
// 嵌入模式（Embedding）
const embeddedOrder = {
  _id: ObjectId("64f8a1b2c3d4e5f6a7b8c9d0"),
  orderNo: "ORD-2025-001",
  userId: ObjectId("64f8a1b2c3d4e5f6a7b8c9d1"),
  status: "pending",
  items: [
    {
      productId: "PROD-001",
      productName: "Laptop",
      quantity: 1,
      price: 999.99,
      specs: {
        cpu: "Intel i7",
        ram: "16GB",
        storage: "512GB SSD"
      }
    },
    {
      productId: "PROD-002",
      productName: "Mouse",
      quantity: 2,
      price: 29.99
    }
  ],
  shipping: {
    address: "123 Main St, New York",
    method: "express",
    trackingNumber: "TRK123456789"
  },
  totalAmount: 1059.97,
  createdAt: ISODate("2025-04-20T10:00:00Z")
}

// 引用模式（Referencing）
const referencedOrder = {
  _id: ObjectId("64f8a1b2c3d4e5f6a7b8c9d0"),
  orderNo: "ORD-2025-001",
  userId: ObjectId("64f8a1b2c3d4e5f6a7b8c9d1"),
  itemIds: [
    ObjectId("64f8a1b2c3d4e5f6a7b8c9d2"),
    ObjectId("64f8a1b2c3d4e5f6a7b8c9d3")
  ],
  shippingId: ObjectId("64f8a1b2c3d4e5f6a7b8c9d4"),
  totalAmount: 1059.97,
  createdAt: ISODate("2025-04-20T10:00:00Z")
}

// 选择策略
const modelingStrategy = {
  embed: {
    useWhen: [
      "数据经常一起访问",
      "子文档数据量小且稳定",
      "数据一致性要求高",
      "一对多关系且'多'端数量有限"
    ],
    advantages: ["查询性能好", "原子性操作", "减少连接"],
    disadvantages: ["文档大小限制（16MB）", "数据冗余", "更新复杂"]
  },
  reference: {
    useWhen: [
      "子文档数据量大",
      "数据频繁变化",
      "多对多关系",
      "需要独立访问子文档"
    ],
    advantages: ["文档大小可控", "数据不冗余", "更新简单"],
    disadvantages: ["需要多次查询", "应用层连接", "一致性复杂"]
  }
}
```

## 📚 聚合管道

### 基本聚合操作

```js
// 聚合管道示例
const aggregationPipeline = [
  // 1. 匹配阶段（过滤文档）
  {
    $match: {
      status: "completed",
      createdAt: {
        $gte: new Date("2025-01-01"),
        $lt: new Date("2025-02-01")
      }
    }
  },
  // 2. 分组阶段
  {
    $group: {
      _id: "$userId",
      totalOrders: { $sum: 1 },
      totalAmount: { $sum: "$totalAmount" },
      avgOrderValue: { $avg: "$totalAmount" }
    }
  },
  // 3. 排序阶段
  {
    $sort: {
      totalAmount: -1
    }
  },
  // 4. 限制结果
  {
    $limit: 10
  },
  // 5. 查找关联数据
  {
    $lookup: {
      from: "users",
      localField: "_id",
      foreignField: "_id",
      as: "userInfo"
    }
  },
  // 6. 展开数组
  {
    $unwind: "$userInfo"
  },
  // 7. 投影字段
  {
    $project: {
      username: "$userInfo.username",
      email: "$userInfo.email",
      totalOrders: 1,
      totalAmount: 1,
      avgOrderValue: 1
    }
  }
]

// 聚合最佳实践
const aggregationBestPractices = {
  matchEarly: "尽早使用 $match 过滤数据，减少处理量",
  limitEarly: "尽早使用 $limit 限制结果集",
  useIndexes: "确保 $match 和 $sort 使用索引",
  avoidUnnecessaryStages: "移除不必要的聚合阶段",
  monitorPerformance: "使用 explain() 分析聚合性能"
}
```
