## 📚 列式存储 vs 行式存储

### 存储架构对比

```js
// 行式存储示例
const rowStorage = {
  description: "同一行的数据存储在一起",
  structure: `
Row 1: [id:1, name:Alice, age:25, city:NYC]
Row 2: [id:2, name:Bob, age:30, city:LA]
Row 3: [id:3, name:Charlie, age:35, city:Chicago]
  `,
  advantages: [
    "适合事务处理（OLTP）",
    "单行查询高效",
    "插入和更新性能好"
  ],
  disadvantages: [
    "分析查询需要读取整列",
    "压缩率较低"
  ]
}

// 列式存储示例
const columnStorage = {
  description: "同一列的数据存储在一起",
  structure: `
id:   [1, 2, 3]
name: [Alice, Bob, Charlie]
age:  [25, 30, 35]
city: [NYC, LA, Chicago]
  `,
  advantages: [
    "适合分析查询（OLAP）",
    "只读取需要的列",
    "压缩率高（同列数据相似）",
    "聚合查询高效"
  ],
  disadvantages: [
    "单行查询性能差",
    "插入和更新开销大"
  ]
}
```

### 适用场景对比

| 特性 | 行式存储 | 列式存储 |
|------|---------|---------|
| 典型场景 | OLTP（事务处理） | OLAP（数据分析） |
| 查询类型 | 单行查询、小范围查询 | 聚合分析、报表查询 |
| 写入性能 | 高 | 中 |
| 读取性能 | 单行快、聚合慢 | 单行慢、聚合快 |
| 压缩率 | 低 | 高（可达 70%+） |
| 代表产品 | MySQL、PostgreSQL | HBase、Cassandra、ClickHouse |

```js

const timeSeriesDataFeatures = {
```

## 📝 实际应用示例

### IoT 监控系统设计

```js
// 场景：物联网设备监控

// InfluxDB 数据模型设计
const iotMonitoring = {
  measurement: "sensor_data",
  tags: {
    device_id: "设备 ID",
    device_type: "设备类型",
    location: "安装位置",
    region: "区域"
  },
  fields: {
    temperature: "温度值",
    humidity: "湿度值",
    pressure: "压力值",
    status: "设备状态"
  },
  retentionPolicies: {
    raw: "RP_RAW: 7 天原始数据",
    hourly: "RP_HOURLY: 90 天小时级数据",
    daily: "RP_DAILY: 365 天天级数据"
  }
}

// 数据写入示例（伪代码）
const writeSensorData = async (data) => {
  const point = {
    measurement: 'sensor_data',
    tags: {
      device_id: data.deviceId,
      device_type: data.deviceType,
      location: data.location,
      region: data.region
    },
    fields: {
      temperature: data.temperature,
      humidity: data.humidity,
      pressure: data.pressure,
      status: data.status
    },
    timestamp: data.timestamp
  }
  
  await influxdb.write(point)
}

// 连续查询：数据降采样
const continuousQuery = `
CREATE CONTINUOUS QUERY cq_downsample_hourly 
ON sensor_db 
BEGIN
  SELECT mean(*) INTO "RP_HOURLY".sensor_data 
  FROM "RP_RAW".sensor_data 
  GROUP BY time(1h), device_id, device_type
END
`
```

### HBase RowKey 设计

```js
// HBase RowKey 设计原则
const rowkeyDesign = {
  principles: [
    "长度原则：越短越好（建议 16 字节以内）",
    "散列原则：避免热点（使用哈希前缀）",
    "唯一性：必须唯一",
    "排序性：利用字典序特性"
  ],
  patterns: {
    reverseTimestamp: {
      description: "反转时间戳",
      example: "Long.MAX_VALUE - timestamp",
      use: "获取最新数据"
    },
    hashPrefix: {
      description: "哈希前缀",
      example: "MD5(userId).substring(0, 4) + userId",
      use: "避免热点"
    },
    composite: {
      description: "组合键",
      example: "hashPrefix + userId + timestamp",
      use: "支持范围查询"
    }
  }
}

// 示例：订单表 RowKey 设计
const orderRowKey = {
  bad: "user_id + timestamp",
  reason: "同一用户的数据会集中到同一 Region",
  good: "MD5(user_id).substring(0, 4) + user_id + timestamp",
  reason: "数据均匀分布，支持按用户查询"
}
```
