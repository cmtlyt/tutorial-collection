## 🔢 主要图像生成模型

### 1. Stable Diffusion

#### 核心思想

Stable Diffusion 是基于潜在扩散模型（Latent Diffusion Models）的开源图像生成模型，通过在潜在空间中进行扩散过程，高效地生成高质量图像。

#### 技术特点

1. **潜在空间扩散**
   - 在压缩的潜在空间中工作
   - 大幅降低计算成本
   - 保持图像质量

2. **开源生态**
   - 模型权重完全开源
   - 活跃的社区贡献
   - 丰富的插件生态

3. **灵活部署**
   - 本地部署
   - 云端部署
   - 移动端部署

```javascript
import { StableDiffusionPipeline } from '@xenova/transformers';

class StableDiffusionGenerator {
  constructor() {
    this.pipeline = null;
  }

  /**
   * 加载 Stable Diffusion 模型
   */
  async loadModel() {
    // 使用 Hugging Face 的 Stable Diffusion 模型
    this.pipeline = await StableDiffusionPipeline.from_pretrained(
      'runwayml/stable-diffusion-v1-5',
      {
        revision: 'fp16',
        dtype: 'fp16'
      }
    );
  }

  /**
   * 文本到图像生成
   * 
   * @param {string} prompt - 文本提示
   * @param {Object} options - 生成选项
   * @returns {Promise<Buffer>} - 生成的图像
   */
  async generateImage(prompt, options = {}) {
    if (!this.pipeline) {
      await this.loadModel();
    }

    const {
      negativePrompt = '',
      width = 512,
      height = 512,
      steps = 50,
      guidanceScale = 7.5,
      seed = null
    } = options;

    // 生成图像
    const result = await this.pipeline(prompt, {
      negative_prompt: negativePrompt,
      width: width,
      height: height,
      num_inference_steps: steps,
      guidance_scale: guidanceScale,
      generator: seed ? { seed: seed } : undefined
    });

    return result.images[0];
  }

  /**
   * 图像到图像生成
   * 
   * @param {Buffer} inputImage - 输入图像
   * @param {string} prompt - 文本提示
   * @param {number} strength - 变化强度 (0-1)
   * @returns {Promise<Buffer>} - 生成的图像
   */
  async img2img(inputImage, prompt, strength = 0.5) {
    if (!this.pipeline) {
      await this.loadModel();
    }

    const result = await this.pipeline(prompt, {
      image: inputImage,
      strength: strength
    });

    return result.images[0];
  }

  /**
   * 图像修复
   * 
   * @param {Buffer} inputImage - 输入图像
   * @param {Buffer} mask - 修复掩码
   * @param {string} prompt - 文本提示
   * @returns {Promise<Buffer>} - 修复后的图像
   */
  async inpaint(inputImage, mask, prompt) {
    if (!this.pipeline) {
      await this.loadModel();
    }

    const result = await this.pipeline(prompt, {
      image: inputImage,
      mask: mask
    });

    return result.images[0];
  }
}

// 使用示例
async function example() {
  const sd = new StableDiffusionGenerator();
  
  // 文本到图像
  const image1 = await sd.generateImage(
    'A beautiful sunset over the ocean, photorealistic, 4K',
    {
      negativePrompt: 'blurry, low quality, distorted',
      width: 1024,
      height: 1024,
      steps: 50,
      guidanceScale: 7.5
    }
  );
  
  // 图像到图像
  const image2 = await sd.img2img(
    image1,
    'Convert to oil painting style',
    0.6
  );
}
```

```javascript
class DALLEGenerator {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.baseUrl = 'https://api.openai.com/v1';
  }

  /**
   * 生成图像
   * 
   * @param {string} prompt - 文本提示
   * @param {Object} options - 生成选项
   * @returns {Promise<Object>} - 生成结果
   */
  async generateImage(prompt, options = {}) {
    const {
      model = 'dall-e-3',
      size = '1024x1024',
      quality = 'standard',
      style = 'vivid',
      n = 1
    } = options;

    const response = await fetch(`${this.baseUrl}/images/generations`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: JSON.stringify({
        model: model,
        prompt: prompt,
        size: size,
        quality: quality,
        style: style,
        n: n
      })
    });

    const data = await response.json();
    return data.data[0];
  }

  /**
   * 编辑图像
   * 
   * @param {Buffer} image - 原始图像
   * @param {Buffer} mask - 编辑掩码
   * @param {string} prompt - 编辑提示
   * @returns {Promise<Object>} - 编辑结果
   */
  async editImage(image, mask, prompt) {
    const response = await fetch(`${this.baseUrl}/images/edits`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: this.createFormData({
        image: image,
        mask: mask,
        prompt: prompt,
        n: 1,
        size: '1024x1024'
      })
    });

    const data = await response.json();
    return data.data[0];
  }

  /**
   * 图像变体
   * 
   * @param {Buffer} image - 原始图像
   * @returns {Promise<Object>} - 变体结果
   */
  async createVariation(image) {
    const response = await fetch(`${this.baseUrl}/images/variations`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: this.createFormData({
        image: image,
        n: 1,
        size: '1024x1024'
      })
    });

    const data = await response.json();
    return data.data[0];
  }

  /**
   * 创建 FormData
   * 
   * @param {Object} data - 数据对象
   * @returns {FormData} - FormData 对象
   */
  createFormData(data) {
    const formData = new FormData();
    
    for (const [key, value] of Object.entries(data)) {
      if (value instanceof Buffer) {
        formData.append(key, new Blob([value]), 'image.png');
      } else {
        formData.append(key, value);
      }
    }
    
    return formData;
  }
}
```

```javascript
class MidjourneyGenerator {
  constructor(discordToken, channelId) {
    this.discordToken = discordToken;
    this.channelId = channelId;
    this.baseUrl = 'https://discord.com/api/v9';
  }

  /**
   * 生成图像
   * 
   * @param {string} prompt - 文本提示
   * @param {Object} options - 生成选项
   * @returns {Promise<Object>} - 生成结果
   */
  async generateImage(prompt, options = {}) {
    const {
      aspectRatio = '1:1',
      quality = 1,
      stylize = 250,
      chaos = 0
    } = options;

    // 构建命令
    const command = this.buildCommand(prompt, {
      ar: aspectRatio,
      q: quality,
      s: stylize,
      c: chaos
    });

    // 发送到 Discord
    const response = await this.sendToDiscord(command);
    
    // 等待生成完成
    return await this.waitForCompletion(response.id);
  }

  /**
   * 构建命令
   * 
   * @param {string} prompt - 文本提示
   * @param {Object} params - 参数
   * @returns {string} - 命令字符串
   */
  buildCommand(prompt, params) {
    let command = `/imagine prompt: ${prompt}`;
    
    if (params.ar) command += ` --ar ${params.ar}`;
    if (params.q) command += ` --q ${params.q}`;
    if (params.s) command += ` --s ${params.s}`;
    if (params.c) command += ` --c ${params.c}`;
    
    return command;
  }

  /**
   * 发送到 Discord
   * 
   * @param {string} command - 命令
   * @returns {Promise<Object>} - 响应
   */
  async sendToDiscord(command) {
    const response = await fetch(
      `${this.baseUrl}/channels/${this.channelId}/messages`,
      {
        method: 'POST',
        headers: {
          'Authorization': `Bot ${this.discordToken}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          content: command
        })
      }
    );

    return await response.json();
  }

  /**
   * 等待生成完成
   * 
   * @param {string} messageId - 消息 ID
   * @returns {Promise<Object>} - 生成结果
   */
  async waitForCompletion(messageId) {
    // 轮询检查生成状态
    const maxAttempts = 60;
    const interval = 5000;

    for (let i = 0; i < maxAttempts; i++) {
      await new Promise(resolve => setTimeout(resolve, interval));

      const response = await this.checkMessage(messageId);
      
      if (response.status === 'completed') {
        return response;
      }
    }

    throw new Error('生成超时');
  }

  /**
   * 检查消息状态
   * 
   * @param {string} messageId - 消息 ID
   * @returns {Promise<Object>} - 消息状态
   */
  async checkMessage(messageId) {
    const response = await fetch(
      `${this.baseUrl}/channels/${this.channelId}/messages/${messageId}`,
      {
        headers: {
          'Authorization': `Bot ${this.discordToken}`
        }
      }
    );

    const data = await response.json();
    
    // 解析消息内容，提取图像 URL
    const imageUrl = this.extractImageUrl(data.content);
    
    if (imageUrl) {
      return {
        status: 'completed',
        imageUrl: imageUrl
      };
    }

    return {
      status: 'processing'
    };
  }

  /**
   * 提取图像 URL
   * 
   * @param {string} content - 消息内容
   * @returns {string|null} - 图像 URL
   */
  extractImageUrl(content) {
    // 从 Discord 消息中提取图像 URL
    const match = content.match(/https?:\/\/[^\s]+\.(png|jpg|jpeg|webp)/);
    return match ? match[0] : null;
  }
}

// 使用示例
async function example() {
  const midjourney = new MidjourneyGenerator(
    'your-discord-token',
    'your-channel-id'
  );
  
  // 生成图像
  const result = await midjourney.generateImage(
    'A mystical forest with glowing mushrooms, ethereal lighting, fantasy art style',
    {
      aspectRatio: '16:9',
      quality: 2,
      stylize: 750,
      chaos: 0
    }
  );
  
  console.log('生成结果:', result);
}
```

### 4. ControlNet

#### 核心思想

ControlNet 是为 Stable Diffusion 添加条件控制的扩展，允许用户通过额外的条件（如边缘图、深度图、姿态图等）精确控制生成结果。

#### 技术特点

1. **精确控制**
   - 边缘控制
   - 深度控制
   - 姿态控制
   - 多种条件输入

2. **保持一致性**
   - 保持原始结构
   - 可控的风格变化
   - 精确的细节控制

3. **灵活性**
   - 多种预训练模型
   - 可组合使用
   - 易于集成

### 5. LoRA 微调

#### 核心思想

LoRA（Low-Rank Adaptation）是一种高效的模型微调方法，通过添加低秩矩阵来适配特定风格或主题，而不需要修改原始模型参数。

#### 技术特点

1. **高效训练**
   - 参数量少
   - 训练速度快
   - 显存占用低

2. **灵活组合**
   - 多个 LoRA 组合
   - 权重调整
   - 风格混合

3. **易于分享**
   - 文件体积小
   - 易于分发
   - 社区共享

#### 代码实现

```javascript
class LoRATrainer {
  constructor(baseModel) {
    this.baseModel = baseModel;
    this.loraLayers = {};
  }

  /**
   * 准备训练数据
   * 
   * @param {Array} images - 训练图像列表
   * @param {Array} prompts - 对应的提示列表
   * @returns {Object} - 训练数据集
   */
  prepareDataset(images, prompts) {
    return images.map((image, index) => ({
      image: image,
      prompt: prompts[index],
      id: index
    }));
  }

  /**
   * 创建 LoRA 层
   * 
   * @param {number} rank - 低秩矩阵的秩
   * @param {number} alpha - 缩放因子
   * @returns {Object} - LoRA 层配置
   */
  createLoRALayer(rank, alpha) {
    return {
      rank: rank,
      alpha: alpha,
      W_down: null,  // 下投影矩阵
      W_up: null,    // 上投影矩阵
      trained: false
    };
  }

  /**
   * 训练 LoRA
   * 
   * @param {Object} dataset - 训练数据集
   * @param {Object} config - 训练配置
   * @returns {Promise<Object>} - 训练结果
   */
  async trainLoRA(dataset, config = {}) {
    const {
      rank = 4,
      alpha = 1,
      epochs = 10,
      batchSize = 4,
      learningRate = 1e-4
    } = config;

    // 创建 LoRA 层
    this.loraLayers = this.createLoRALayer(rank, alpha);

    // 训练循环
    for (let epoch = 0; epoch < epochs; epoch++) {
      const totalLoss = await this.trainEpoch(
        dataset,
        batchSize,
        learningRate
      );

      console.log(`Epoch ${epoch + 1}/${epochs}, Loss: ${totalLoss}`);
    }

    this.loraLayers.trained = true;
    return this.loraLayers;
  }

  /**
   * 训练一个 epoch
   * 
   * @param {Object} dataset - 训练数据集
   * @param {number} batchSize - 批次大小
   * @param {number} learningRate - 学习率
   * @returns {Promise<number>} - 总损失
   */
  async trainEpoch(dataset, batchSize, learningRate) {
    let totalLoss = 0;
    const numBatches = Math.ceil(dataset.length / batchSize);

    for (let batch = 0; batch < numBatches; batch++) {
      const batchData = dataset.slice(
        batch * batchSize,
        (batch + 1) * batchSize
      );

      const batchLoss = await this.trainBatch(batchData, learningRate);
      totalLoss += batchLoss;
    }

    return totalLoss / numBatches;
  }

  /**
   * 训练一个批次
   *
```
