# 语音合成（TTS）

> **版本**：v1.0  
> **最后更新**：2026-04-15

## 📖 概述

文本转语音（Text-to-Speech，TTS）是将文本转换为自然语音的技术。随着深度学习的发展，TTS 技术已经从简单的拼接合成发展到现在的神经语音合成，生成的语音质量和自然度都得到了显著提升。本章节将详细介绍主流的 TTS 模型及其应用。

## 🎯 为什么需要语音合成？

### 从文本到语音

1. **无障碍访问**
   - 视觉障碍用户辅助
   - 阅读困难用户支持
   - 信息无障碍化

2. **广泛应用**
   - 智能助手
   - 导航系统
   - 有声读物
   - 游戏配音

3. **技术演进**
   - 从参数合成到波形拼接
   - 从统计模型到神经网络
   - 从单一音色到多音色

## 🔢 主要 TTS 模型

### 1. VITS

#### 核心思想

VITS（Conditional Variational Autoencoder with Adversarial Learning for End-to-End Text-to-Speech）是一种基于条件变分自编码器和对抗学习的端到端 TTS 模型。

#### 技术特点

1. **端到端架构**
   - 统一的训练和推理流程
   - 无需复杂的流水线
   - 高效的推理速度

2. **变分自编码器**
   - 潜在空间建模
   - 语音风格控制
   - 多说话人支持

3. **对抗学习**
   - 提升语音自然度
   - 增强音频质量
   - 减少人工痕迹

```javascript
import { pipeline } from '@xenova/transformers';

class VITSModel {
  constructor() {
    this.model = null;
  }

  /**
   * 加载 VITS 模型
   */
  async loadModel() {
    // 使用 Hugging Face 的 VITS 模型
    this.model = await pipeline(
      'text-to-speech',
      'Xenova/vits-vctk'
    );
  }

  /**
   * 文本转语音
   * 
   * @param {string} text - 输入文本
   * @param {Object} options - 合成选项
   * @returns {Promise<ArrayBuffer>} - 音频数据
   */
  async synthesize(text, options = {}) {
    if (!this.model) {
      await this.loadModel();
    }

    const {
      speakerId = 0,
      speed = 1.0,
      noiseScale = 0.667,
      noiseScaleW = 0.8
    } = options;

    // 合成语音
    const result = await this.model(text, {
      speaker_embeddings: this.getSpeakerEmbedding(speakerId),
      speed: speed,
      noise_scale: noiseScale,
      noise_scale_w: noiseScaleW
    });

    return result.audio;
  }

  /**
   * 获取说话人嵌入
   * 
   * @param {number} speakerId - 说话人 ID
   * @returns {Array} - 说话人嵌入向量
   */
  getSpeakerEmbedding(speakerId) {
    // 基于说话人 ID 生成嵌入向量
    // 实际实现中应该从预训练的说话人嵌入矩阵中获取
    const speakerCount = 3;
    const embeddingDim = 256;
    
    // 使用伪随机数生成器确保相同 ID 生成相同嵌入
    const seed = speakerId * 12345;
    const random = this.seededRandom(seed);
    
    const embedding = [];
    for (let i = 0; i < embeddingDim; i++) {
      embedding.push(random() * 2 - 1); // 归一化到 [-1, 1]
    }
    
    return embedding;
  }

  /**
   * 种子随机数生成器
   * 
   * @param {number} seed - 种子值
   * @returns {Function} - 随机数生成函数
   */
  seededRandom(seed) {
    let value = seed;
    
    return function() {
      value = (value * 9301 + 49297) % 233280;
      return value / 233280;
    };
  }

  /**
   * 批量合成
   * 
   * @param {Array<string>} texts - 文本列表
   * @param {Object} options - 合成选项
   * @returns {Promise<Array<ArrayBuffer>>} - 音频数据列表
   */
  async batchSynthesize(texts, options = {}) {
    const results = [];
    
    for (const text of texts) {
      try {
        const audio = await this.synthesize(text, options);
        results.push({
          text: text,
          success: true,
          audio: audio
        });
      } catch (error) {
        results.push({
          text: text,
          success: false,
          error: error.message
        });
      }
    }
    
    return results;
  }

  /**
   * 实时合成
   * 
   * @param {string} text - 输入文本
   * @param {Function} onAudioChunk - 音频块回调
   * @param {Object} options - 合成选项
   * @returns {Promise<void>}
   */
  async synthesizeRealtime(text, onAudioChunk, options = {}) {
    const {
      chunkSize = 100,
      delay = 100
    } = options;

    // 分块处理文本
    const chunks = this.splitText(text, chunkSize);
    
    for (const chunk of chunks) {
      const audio = await this.synthesize(chunk, options);
      onAudioChunk(audio);
      
      // 添加延迟以模拟实时播放
      // 实际实现中应该根据音频长度计算延迟
      const audioDuration = this.calculateAudioDuration(audio);
      await new Promise(resolve => setTimeout(resolve, audioDuration * 1000));
    }
  }

  /**
   * 计算音频时长
   * 
   * @param {ArrayBuffer} audio - 音频数据
   * @returns {number} - 音频时长（秒）
   */
  calculateAudioDuration(audio) {
    // 简化的音频时长计算
    // 实际实现中应该解析音频头部信息
    const bytesPerSample = 2; // 16-bit
    const sampleRate = 24000; // VITS 采样率
    const channels = 1;
    
    const totalSamples = audio.byteLength / (bytesPerSample * channels);
    return totalSamples / sampleRate;
  }

  /**
   * 分割文本
   * 
   * @param {string} text - 输入文本
   * @param {number} chunkSize - 块大小
   * @returns {Array<string>} - 文本块列表
   */
  splitText(text, chunkSize) {
    const chunks = [];
    let currentChunk = '';
    
    for (let i = 0; i < text.length; i++) {
      currentChunk += text[i];
      
      if (currentChunk.length >= chunkSize && 
          (text[i] === ' ' || text[i] === '，' || text[i] === '。')) {
        chunks.push(currentChunk.trim());
        currentChunk = '';
      }
    }
    
    if (currentChunk.trim()) {
      chunks.push(currentChunk.trim());
    }
    
    return chunks;
  }

  /**
   * 获取支持的说话人列表
   * 
   * @returns {Array<Object>} - 说话人列表
   */
  getSpeakers() {
    return [
      { id: 0, name: 'Speaker 1', gender: 'female' },
      { id: 1, name: 'Speaker 2', gender: 'male' },
      { id: 2, name: 'Speaker 3', gender: 'female' }
    ];
  }

  /**
   * 调整语音参数
   * 
   * @param {ArrayBuffer} audio - 原始音频
   * @param {Object} params - 调整参数
   * @returns {Promise<ArrayBuffer>} - 调整后的音频
   */
  async adjustAudio(audio, params = {}) {
    const {
      speed = 1.0,
      pitch = 1.0,
      volume = 1.0
    } = params;

    // 使用 Web Audio API 调整音频
    const audioContext = new AudioContext();
    const audioBuffer = await audioContext.decodeAudioData(audio);
    
    // 创建离线上下文进行处理
    const offlineContext = new OfflineAudioContext(
      audioBuffer.numberOfChannels,
      audioBuffer.length * speed,
      audioBuffer.sampleRate
    );
    
    const source = offlineContext.createBufferSource();
    source.buffer = audioBuffer;
    source.playbackRate.value = speed;
    
    // 创建增益节点调整音量
    const gainNode = offlineContext.createGain();
    gainNode.gain.value = volume;
    
    source.connect(gainNode);
    gainNode.connect(offlineContext.destination);
    
    source.start();
    
    const renderedBuffer = await offlineContext.startRendering();
    
    // 转换为 ArrayBuffer
    const wavBuffer = this.audioBufferToWav(renderedBuffer);
    
    return wavBuffer;
  }

  /**
   * 将 AudioBuffer 转换为 WAV 格式
   * 
   * @param {AudioBuffer} audioBuffer - AudioBuffer 对象
   * @returns {ArrayBuffer} - WAV 格式数据
   */
  audioBufferToWav(audioBuffer) {
    const numChannels = audioBuffer.numberOfChannels;
    const sampleRate = audioBuffer.sampleRate;
    const format = 1; // PCM
    const bitDepth = 16;
    
    const bytesPerSample = bitDepth / 8;
    const blockAlign = numChannels * bytesPerSample;
    
    const dataLength = audioBuffer.length * blockAlign;
    const bufferLength = 44 + dataLength;
    
    const arrayBuffer = new ArrayBuffer(bufferLength);
    const view = new DataView(arrayBuffer);
    
    // WAV 文件头
    this.writeString(view, 0, 'RIFF');
    view.setUint32(4, 36 + dataLength, true);
    this.writeString(view, 8, 'WAVE');
    this.writeString(view, 12, 'fmt ');
    view.setUint32(16, 16, true);
    view.setUint16(20, format, true);
    view.setUint16(22, numChannels, true);
    view.setUint32(24, sampleRate, true);
    view.setUint32(28, sampleRate * blockAlign, true);
    view.setUint16(32, blockAlign, true);
    view.setUint16(34, bitDepth, true);
    this.writeString(view, 36, 'data');
    view.setUint32(40, dataLength, true);
    
    // 写入音频数据
    const offset = 44;
    for (let i = 0; i < audioBuffer.length; i++) {
      for (let channel = 0; channel < numChannels; channel++) {
        const sample = audioBuffer.getChannelData(channel)[i];
        const intSample = sample < 0 ? sample * 0x8000 : sample * 0x7FFF;
        view.setInt16(offset + (i * blockAlign) + (channel * bytesPerSample), intSample, true);
      }
    }
    
    return arrayBuffer;
  }

  /**
   * 写入字符串
   * 
   * @param {DataView} view - DataView 对象
   * @param {number} offset - 偏移量
   * @param {string} string - 字符串
   */
  writeString(view, offset, string) {
    for (let i = 0; i < string.length; i++) {
      view.setUint8(offset + i, string.charCodeAt(i));
    }
  }
}

  const vits = new VITSModel();
  
  // 文本转语音

    '第一段文本',
    '第二段文本',

  ];
  
  const results = await vits.batchSynthesize(texts);
```

```javascript
class ElevenLabsTTS {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.baseUrl = 'https://api.elevenlabs.io/v1';
  }

  /**
   * 文本转语音
   * 
   * @param {string} text - 输入文本
   * @param {Object} options - 合成选项
   * @returns {Promise<ArrayBuffer>} - 音频数据
   */
  async synthesize(text, options = {}) {
    const {
      voiceId = '21m00Tcm4TlvDq8ikWAM',
      modelId = 'eleven_monolingual_v1',
      stability = 0.5,
      similarityBoost = 0.75,
      style = 0.0,
      speakerBoost = false
    } = options;

    const response = await fetch(
      `${this.baseUrl}/text-to-speech/${voiceId}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'xi-api-key': this.apiKey,
          'Accept': 'audio/mpeg'
        },
        body: JSON.stringify({
          text: text,
          model_id: modelId,
          voice_settings: {
            stability: stability,
            similarity_boost: similarityBoost,
            style: style,
            use_speaker_boost: speakerBoost
          }
        })
      }
    );

    if (!response.ok) {
      throw new Error(`ElevenLabs API 错误: ${response.statusText}`);
    }

    return await response.arrayBuffer();
  }

  /**
   * 流式合成
   * 
   * @param {string} text - 输入文本
   * @param {Function} onChunk - 音频块回调
   * @param {Object} options - 合成选项
   * @returns {Promise<void>}
   */
  async synthesizeStream(text, onChunk, options = {}) {
    const {
      voiceId = '21m00Tcm4TlvDq8ikWAM',
      modelId = 'eleven_monolingual_v1',
      stability = 0.5,
      similarityBoost = 0.75
    } = options;

    const response = await fetch(
      `${this.baseUrl}/text-to-speech/${voiceId}/stream`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'xi-api-key': this.apiKey,
          'Accept': 'audio/mpeg'
        },
        body: JSON.stringify({
          text: text,
          model_id: modelId,
          voice_settings: {
            stability: stability,
            similarity_boost: similarityBoost
          }
        })
      }
    );

    if (!response.ok) {
      throw new Error(`ElevenLabs API 错误: ${response.statusText}`);
    }

    const reader = response.body.getReader();
    const decoder = new TextDecoder();

    while (true) {
      const { done, value } = await reader.read();
      
      if (done) {
        break;
      }

      onChunk(value);
    }
  }

  /**
   * 获取可用音色列表
   * 
   * @returns {Promise<Array<Object>>} - 音色列表
   */
  async getVoices() {
    const response = await fetch(
      `${this.baseUrl}/voices`,
      {
        method: 'GET',
        headers: {
          'xi-api-key': this.apiKey
        }
      }
    );

    if (!response.ok) {
      throw new Error(`ElevenLabs API 错误: ${response.statusText}`);
    }

    const data = await response.json();
    return data.voices;
  }

  /**
   * 克隆音色
   * 
   * @param {string} name - 音色名称
   * @param {ArrayBuffer} audio - 音频样本
   * @param {string} description - 描述
   * @returns {Promise<Object>} - 克隆结果
   */
  async cloneVoice(name, audio, description = '') {
    const formData = new FormData();
    formData.append('name', name);
    formData.append('files', new Blob([audio]), 'audio.mp3');
    formData.append('description', description);

    const response = await fetch(
      `${this.baseUrl}/voices/add`,
      {
        method: 'POST',
        headers: {
          'xi-api-key': this.apiKey
        },
        body: formData
      }
    );

    if (!response.ok) {
      throw new Error(`ElevenLabs API 错误: ${response.statusText}`);
    }

    return await response.json();
  }

  /**
```

### 4. Azure TTS

#### 核心思想

Azure Text-to-Speech 是微软提供的云语音服务，支持多种语言和音色，提供高质量的语音合成能力。

#### 技术特点

1. **多语言支持**
   - 支持 90+ 语言
   - 300+ 音色
   - 神经语音引擎

2. **高质量输出**
   - 自然流畅
   - 情感丰富
   - 风格多样

3. **企业级服务**
   - 高可用性
   - 低延迟
   - 可扩展

```javascript
class AzureTTS {
  constructor(subscriptionKey, region) {
    this.subscriptionKey = subscriptionKey;
    this.region = region;
    this.baseUrl = `https://${region}.tts.speech.microsoft.com/cognitiveservices/v1`;
  }

  /**
   * 文本转语音
   * 
   * @param {string} text - 输入文本
   * @param {Object} options - 合成选项
   * @returns {Promise<ArrayBuffer>} - 音频数据
   */
  async synthesize(text, options = {}) {
    const {
      voice = 'zh-CN-XiaoxiaoNeural',
      rate = 'medium',
      pitch = 'medium',
      volume = 'default',
      outputFormat = 'audio-24khz-48kbitrate-mono-mp3'
    } = options;

    // 构建 SSML
    const ssml = this.buildSSML(text, {
      voice: voice,
      rate: rate,
      pitch: pitch,
      volume: volume
    });

    const response = await fetch(
      `${this.baseUrl}?outputFormat=${outputFormat}`,
      {
        method: 'POST',
        headers: {
          'Ocp-Apim-Subscription-Key': this.subscriptionKey,
          'Content-Type': 'application/ssml+xml',
          'X-Microsoft-OutputFormat': outputFormat
        },
        body: ssml
      }
    );

    if (!response.ok) {
      throw new Error(`Azure TTS API 错误: ${response.statusText}`);
    }

    return await response.arrayBuffer();
  }

  /**
   * 构建 SSML
   * 
   * @param {string} text - 输入文本
   * @param {Object} options - SSML 选项
   * @returns {string} - SSML 字符串
   */
  buildSSML(text, options) {
    const { voice, rate, pitch, volume } = options;

    return `
      <speak version='1.0' xml:lang='zh-CN'>
        <voice name='${voice}'>
          <prosody rate='${rate}' pitch='${pitch}' volume='${volume}'>
            ${text}
          </prosody>
        </voice>
      </speak>
    `.trim();
  }

  /**
   * 获取可用音色列表
   * 
   * @param {string} locale - 语言区域
   * @returns {Promise<Array<Object>>} - 音色列表
   */
  async getVoices(locale = 'zh-CN') {
    const response = await fetch(
      `https://${this.region}.tts.speech.microsoft.com/cognitiveservices/voices/list`,
      {
        method: 'GET',
        headers: {
          'Ocp-Apim-Subscription-Key': this.subscriptionKey
        }
      }
    );

    if (!response.ok) {
      throw new Error(`Azure TTS API 错误: ${response.statusText}`);
    }

    const voices = await response.json();
    
    // 过滤指定语言的音色
    return voices.filter(voice => voice.Locale.startsWith(locale));
  }

  /**
   * 带情感的语音合成
   * 
   * @param {string} text - 输入文本
   * @param {string} emotion - 情感类型
   * @param {Object} options - 合成选项
   * @returns {Promise<ArrayBuffer>} - 音频数据
   */
  async synthesizeWithEmotion(text, emotion, options = {}) {
    const emotionStyles = {
      calm: 'calm',
      cheerful: 'cheerful',
      sad: 'sad',
      angry: 'angry',
      fearful: 'fearful',
      disgusted: 'disgruntled',
      shouting: 'shouting'
    };

    const style = emotionStyles[emotion] || 'general';
    
    // 构建带情感的 SSML
    const ssml = this.buildEmotionalSSML(text, style, options);
    
    const response = await fetch(
      `${this.baseUrl}?outputFormat=audio-24khz-48kbitrate-mono-mp3`,
      {
        method: 'POST',
        headers: {
          'Ocp-Apim-Subscription-Key': this.subscriptionKey,
          'Content-Type': 'application/ssml+xml'
        },
        body: ssml
      }
    );

    if (!response.ok) {
      throw new Error(`Azure TTS API 错误: ${response.statusText}`);
    }

    return await response.arrayBuffer();
  }

  /**
   * 构建带情感的 SSML
   * 
   * @param {string} text - 输入文本
   * @param {string} style - 情感风格
   * @param {Object} options - 合成选项
   * @returns {string} - SSML 字符串
   */
  buildEmotionalSSML(text, style, options) {
    const { voice = 'zh-CN-XiaoxiaoNeural' } = options;

    return `
      <speak version='1.0' xml:lang='zh-CN' xmlns='http://www.w3.org/2001/10/synthesis' xmlns:mstts='http://www.w3.org/2001/mstts'>
        <voice name='${voice}'>
          <mstts:express-as style='${style}'>
            ${text}
          </mstts:express-as>
        </voice>
      </speak>
    `.trim();
  }

  /**
   * 长文本合成
   * 
   * @param {string} text - 输入文本
   * @param {Function} onProgress - 进度回调
   * @param {Object} options - 合成选项
   * @returns {Promise<ArrayBuffer>} - 音频数据
   */
  async synthesizeLongText(text, onProgress, options = {}) {
    // 分割文本
    const chunks = this.splitText(text, 1000);
    
    const audioChunks = [];
    
    for (let i = 0; i < chunks.length; i++) {
      const audio = await this.synthesize(chunks[i], options);
      audioChunks.push(audio);
      
      if (onProgress) {
        onProgress({
          current: i + 1,
          total: chunks.length,
          progress: ((i + 1) / chunks.length) * 100
        });
      }
    }
    
    // 合并音频块
    return this.mergeAudioChunks(audioChunks);
  }

  /**
   * 分割文本
   * 
   * @param {string} text - 输入文本
   * @param {number} maxLength - 最大长度
   * @returns {Array<string>} - 文本块列表
   */
  splitText(text, maxLength) {
    const chunks = [];
    let currentChunk = '';
    
    // 按句子分割
    const sentences = text.split(/([。！？.!?])/);
    
    for (let i = 0; i < sentences.length; i += 2) {
      const sentence = sentences[i] + (sentences[i + 1] || '');
      
      if (currentChunk.length + sentence.length > maxLength && currentChunk.length > 0) {
        chunks.push(currentChunk.trim());
        currentChunk = sentence;
      } else {
        currentChunk += sentence;
      }
    }
    
    if (currentChunk.trim()) {
      chunks.push(currentChunk.trim());
    }
    
    return chunks;
  }

  /**
   * 合并音频块
   * 
   * @param {Array<ArrayBuffer>} chunks - 音频块列表
   * @returns {Promise<ArrayBuffer>} - 合并后的音频
   */
  async mergeAudioChunks(chunks) {
    if (chunks.length === 0) {
      return new ArrayBuffer(0);
    }

    if (chunks.length === 1) {
      return chunks[0];
    }

    // 解码所有音频块
    const audioContext = new AudioContext({ sampleRate: 24000 });
    const decodedBuffers = [];
    
    for (const chunk of chunks) {
      try {
        const audioBuffer = await audioContext.decodeAudioData(chunk);
        decodedBuffers.push(audioBuffer);
      } catch (error) {
        console.error('音频解码错误:', error);
      }
    }

    if (decodedBuffers.length === 0) {
      return new ArrayBuffer(0);
    }

    // 计算总长度
    const totalLength = decodedBuffers.reduce(
      (sum, buffer) => sum + buffer.length,
      0
    );

    // 创建合并后的音频缓冲区
    const numberOfChannels = decodedBuffers[0].numberOfChannels;
    const sampleRate = decodedBuffers[0].sampleRate;
    
    const mergedBuffer = audioContext.createBuffer(
      numberOfChannels,
      totalLength,
      sampleRate
    );

    // 合并音频数据
    let offset = 0;
    for (const buffer of decodedBuffers) {
      for (let channel = 0; channel < numberOfChannels; channel++) {
        const channelData = buffer.getChannelData(channel);
        const mergedChannelData = mergedBuffer.getChannelData(channel);
        
        for (let i = 0; i < channelData.length; i++) {
          mergedChannelData[offset + i] = channelData[i];
        }
      }
      
      offset += buffer.length;
    }

    // 转换为 WAV 格式
    return this.audioBufferToWav(mergedBuffer);
  }

  /**
   * 获取语音风格列表
   * 
   * @param {string} voice - 音色名称
   * @returns {Promise<Array<string>>} - 风格列表
   */
  async getVoiceStyles(voice) {
    // Azure TTS 支持的情感风格
    const styles = [
      'general',
      'calm',
      'cheerful',
      'sad',
      'angry',
      'fearful',
      'disgruntled',
      'shouting',
      'whispering',
      'empathetic'
    ];
    
    return styles;
  }
}

  const azureTTS = new AzureTTS('your-subscription-key', 'eastasia');
  
  // 文本转语音
  const audio1 = await azureTTS.synthesize(
```
