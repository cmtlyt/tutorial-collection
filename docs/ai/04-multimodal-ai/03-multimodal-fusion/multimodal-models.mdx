# 多模态模型

> **版本**：v1.0  
> **最后更新**：2026-04-15

## 📖 概述

多模态模型是能够同时处理和理解多种模态数据（如文本、图像、音频、视频等）的 AI 模型。通过融合不同模态的信息，多模态模型能够实现更强大的理解和生成能力，接近人类的感知和认知水平。本章节将详细介绍主流的多模态模型及其应用。

## 🎯 为什么需要多模态模型？

### 从单一模态到多模态融合

1. **更全面的理解**
   - 整合多种信息源
   - 提供更丰富的上下文
   - 增强理解准确性

2. **更自然的交互**
   - 接近人类感知方式
   - 支持多模态输入
   - 提供多模态输出

3. **更广泛的应用**
   - 跨模态检索
   - 多模态内容生成
   - 智能辅助决策

1. **原生多模态架构**
   - 统一的 Transformer 架构

3. **实时交互**

#### 代码实现

```javascript
class GPT4Omni {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.baseUrl = 'https://api.openai.com/v1';
  }

  /**
   * 多模态理解
   * 
   * @param {Object} input - 多模态输入
   * @param {Object} options - 选项
   * @returns {Promise<Object>} - 理解结果
   */
  async understand(input, options = {}) {
    const { text, image, audio } = input;
    const { model = 'gpt-4o' } = options;

    const messages = [];
    
    // 构建消息
    if (text) {
      messages.push({
        role: 'user',
        content: [
          { type: 'text', text: text }
        ]
      });
    }

    if (image) {
      if (messages.length === 0) {
        messages.push({ role: 'user', content: [] });
      }
      
      messages[0].content.push({
        type: 'image_url',
        image_url: { url: image }
      });
    }

    if (audio) {
      if (messages.length === 0) {
        messages.push({ role: 'user', content: [] });
      }
      
      messages[0].content.push({
        type: 'audio_url',
        audio_url: { url: audio }
      });
    }

    const response = await fetch(`${this.baseUrl}/chat/completions`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: JSON.stringify({
        model: model,
        messages: messages,
        max_tokens: 2000
      })
    });

    const data = await response.json();
    return {
      text: data.choices[0].message.content,
      model: model,
      usage: data.usage
    };
  }

  /**
   * 图像理解
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {string} prompt - 提示文本
   * @returns {Promise<string>} - 理解结果
   */
  async understandImage(imageUrl, prompt) {
    return await this.understand(
      { image: imageUrl, text: prompt },
      { model: 'gpt-4o' }
    );
  }

  /**
   * 音频理解
   * 
   * @param {string} audioUrl - 音频 URL
   * @param {string} prompt - 提示文本
   * @returns {Promise<string>} - 理解结果
   */
  async understandAudio(audioUrl, prompt) {
    return await this.understand(
      { audio: audioUrl, text: prompt },
      { model: 'gpt-4o' }
    );
  }

  /**
   * 多模态对话
   * 
   * @param {Array} messages - 对话历史
   * @returns {Promise<string>} - 回复
   */
  async chat(messages) {
    const response = await fetch(`${this.baseUrl}/chat/completions`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: JSON.stringify({
        model: 'gpt-4o',
        messages: messages,
        max_tokens: 2000
      })
    });

    const data = await response.json();
    return data.choices[0].message.content;
  }

  /**
   * 实时语音对话
   * 
   * @param {MediaStream} stream - 音频流
   * @param {Function} onResponse - 响应回调
   * @returns {Promise<void>}
   */
  async realtimeConversation(stream, onResponse) {
    const audioContext = new AudioContext({ sampleRate: 24000 });
    const source = audioContext.createMediaStreamSource(stream);
    const processor = audioContext.createScriptProcessor(4096, 1, 1);

    let audioBuffer = [];

    processor.onaudioprocess = async (event) => {
      const inputData = event.inputBuffer.getChannelData(0);
      audioBuffer.push(...inputData);

      // 每 2 秒处理一次
      if (audioBuffer.length >= 48000) {
        const audioData = new Float32Array(audioBuffer);
        
        try {
          const response = await this.transcribeAndRespond(audioData);
          onResponse(response);
        } catch (error) {
          console.error('处理错误:', error);
        }

        audioBuffer = [];
      }
    };

    source.connect(processor);
    processor.connect(audioContext.destination);
  }

  /**
   * 转录并响应
   * 
   * @param {Float32Array} audioData - 音频数据
   * @returns {Promise<Object>} - 响应结果
   */
  async transcribeAndRespond(audioData) {
    // 转录音频
    const transcription = await this.transcribeAudio(audioData);
    
    // 生成回复
    const response = await this.chat([
      { role: 'user', content: transcription }
    ]);
    
    // 合成语音
    const audio = await this.synthesizeSpeech(response);
    
    return {
      transcription: transcription,
      response: response,
      audio: audio
    };
  }

  /**
   * 转录音频
   * 
   * @param {Float32Array} audioData - 音频数据
   * @returns {Promise<string>} - 转录文本
   */
  async transcribeAudio(audioData) {
    // 使用 Whisper 转录
    const response = await fetch(`${this.baseUrl}/audio/transcriptions`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: this.createFormData({
        file: new Blob([audioData], { type: 'audio/wav' }),
        model: 'whisper-1'
      })
    });

    const data = await response.json();
    return data.text;
  }

  /**
   * 合成语音
   * 
   * @param {string} text - 文本
   * @returns {Promise<ArrayBuffer>} - 音频数据
   */
  async synthesizeSpeech(text) {
    const response = await fetch(`${this.baseUrl}/audio/speech`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: JSON.stringify({
        model: 'tts-1',
        input: text,
        voice: 'alloy'
      })
    });

    return await response.arrayBuffer();
  }

  /**
   * 创建 FormData
   * 
   * @param {Object} data - 数据对象
   * @returns {FormData} - FormData 对象
   */
  createFormData(data) {
    const formData = new FormData();
    
    for (const [key, value] of Object.entries(data)) {
      if (value instanceof Blob) {
        formData.append(key, value, 'audio.wav');
      } else {
        formData.append(key, value);
      }
    }
    
    return formData;
  }
}

// 使用示例
async function example() {
  const gpt4o = new GPT4Omni('your-api-key');
  
  // 图像理解
  const imageResult = await gpt4o.understandImage(
    'https://example.com/image.jpg',
    '请详细描述这张图片的内容'
  );
  
  console.log('图像理解结果:', imageResult);
  
  // 音频理解
  const audioResult = await gpt4o.understandAudio(
    'https://example.com/audio.mp3',
    '请总结这段音频的主要内容'
  );
  
  console.log('音频理解结果:', audioResult);
  
  // 多模态对话
  const messages = [
    {
      role: 'user',
      content: [
        { type: 'text', text: '这张图片里有什么？' },
        { type: 'image_url', image_url: { url: 'https://example.com/image.jpg' } }
      ]
    }
  ];
  
  const chatResponse = await gpt4o.chat(messages);
  console.log('对话回复:', chatResponse);
}
```

#### 代码实现

```javascript
class GeminiVision {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.baseUrl = 'https://generativelanguage.googleapis.com/v1beta';
  }

  /**
   * 图像理解
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {string} prompt - 提示文本
   * @param {Object} options - 选项
   * @returns {Promise<Object>} - 理解结果
   */
  async understandImage(imageUrl, prompt, options = {}) {
    const {
      model = 'gemini-pro-vision',
      temperature = 0.7,
      maxOutputTokens = 2048
    } = options;

    const response = await fetch(
      `${this.baseUrl}/models/${model}:generateContent?key=${this.apiKey}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          contents: [
            {
              parts: [
                { text: prompt },
                {
                  inline_data: {
                    mime_type: 'image/jpeg',
                    data: await this.imageToBase64(imageUrl)
                  }
                }
              ]
            }
          ],
          generationConfig: {
            temperature: temperature,
            maxOutputTokens: maxOutputTokens
          }
        })
      }
    );

    const data = await response.json();
    return {
      text: data.candidates[0].content.parts[0].text,
      model: model,
      usage: data.usageMetadata
    };
  }

  /**
   * 视频理解
   * 
   * @param {string} videoUrl - 视频 URL
   * @param {string} prompt - 提示文本
   * @returns {Promise<Object>} - 理解结果
   */
  async understandVideo(videoUrl, prompt) {
    const response = await fetch(
      `${this.baseUrl}/models/gemini-pro-vision:generateContent?key=${this.apiKey}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          contents: [
            {
              parts: [
                { text: prompt },
                {
                  file_data: {
                    mime_type: 'video/mp4',
                    file_uri: videoUrl
                  }
                }
              ]
            }
          ]
        })
      }
    );

    const data = await response.json();
    return {
      text: data.candidates[0].content.parts[0].text,
      model: 'gemini-pro-vision'
    };
  }

  /**
   * 多图像理解
   * 
   * @param {Array<string>} imageUrls - 图像 URL 列表
   * @param {string} prompt - 提示文本
   * @returns {Promise<Object>} - 理解结果
   */
  async understandMultipleImages(imageUrls, prompt) {
    const parts = [{ text: prompt }];
    
    for (const imageUrl of imageUrls) {
      parts.push({
        inline_data: {
          mime_type: 'image/jpeg',
          data: await this.imageToBase64(imageUrl)
        }
      });
    }

    const response = await fetch(
      `${this.baseUrl}/models/gemini-pro-vision:generateContent?key=${this.apiKey}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          contents: [
            {
              parts: parts
            }
          ]
        })
      }
    );

    const data = await response.json();
    return {
      text: data.candidates[0].content.parts[0].text,
      model: 'gemini-pro-vision'
    };
  }

  /**
   * 图像描述生成
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {Object} options - 选项
   * @returns {Promise<string>} - 描述文本
   */
  async describeImage(imageUrl, options = {}) {
    const { detail = 'detailed' } = options;
    
    const prompts = {
      brief: '请简要描述这张图片',
      detailed: '请详细描述这张图片的内容、颜色、构图等',
      creative: '请用富有创意的语言描述这张图片'
    };

    const result = await this.understandImage(
      imageUrl,
      prompts[detail] || prompts.detailed
    );

    return result.text;
  }

  /**
   * 图像问答
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {string} question - 问题
   * @returns {Promise<string>} - 答案
   */
  async askAboutImage(imageUrl, question) {
    const result = await this.understandImage(imageUrl, question);
    return result.text;
  }

  /**
   * 图像对比
   * 
   * @param {string} imageUrl1 - 第一张图像 URL
   * @param {string} imageUrl2 - 第二张图像 URL
   * @returns {Promise<string>} - 对比结果
   */
  async compareImages(imageUrl1, imageUrl2) {
    const prompt = '请对比这两张图片的异同点';
    const result = await this.understandMultipleImages(
      [imageUrl1, imageUrl2],
      prompt
    );

    return result.text;
  }

  /**
   * 图像转 Base64
   * 
   * @param {string} imageUrl - 图像 URL
   * @returns {Promise<string>} - Base64 编码
   */
  async imageToBase64(imageUrl) {
    const response = await fetch(imageUrl);
    const blob = await response.blob();
    
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onloadend = () => {
        const base64 = reader.result.split(',')[1];
        resolve(base64);
      };
      reader.onerror = reject;
      reader.readAsDataURL(blob);
    });
  }

  /**
   * 流式响应
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {string} prompt - 提示文本
   * @param {Function} onChunk - 块回调
   * @returns {Promise<void>}
   */
  async streamResponse(imageUrl, prompt, onChunk) {
    const response = await fetch(
      `${this.baseUrl}/models/gemini-pro-vision:streamGenerateContent?key=${this.apiKey}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          contents: [
            {
              parts: [
                { text: prompt },
                {
                  inline_data: {
                    mime_type: 'image/jpeg',
                    data: await this.imageToBase64(imageUrl)
                  }
                }
              ]
            }
          ]
        })
      }
    );

    const reader = response.body.getReader();
    const decoder = new TextDecoder();

    while (true) {
      const { done, value } = await reader.read();
      
      if (done) {
        break;
      }

      const chunk = decoder.decode(value);
      const lines = chunk.split('\n');
      
      for (const line of lines) {
        if (line.startsWith('data: ')) {
          const data = JSON.parse(line.slice(6));
          if (data.candidates && data.candidates[0]) {
            const text = data.candidates[0].content?.parts?.[0]?.text;
            if (text) {
              onChunk(text);
            }
          }
        }
      }
    }
  }
}

  const gemini = new GeminiVision('your-api-key');
  
  // 图像理解
  const result1 = await gemini.understandImage(
    'https://example.com/image.jpg',
    '请详细描述这张图片'
  );
  
  console.log('图像理解结果:', result1.text);

    { detail: 'detailed' }
  );
  
  console.log('图像描述:', description);
  
  // 图像对比
  const comparison = await gemini.compareImages(

    'https://example.com/image2.jpg'
  );
  
  console.log('图像对比:', comparison);
}
```

### 3. LLaVA

#### 核心思想

LLaVA（Large Language-and-Vision Assistant）是一个开源的多模态对话模型，通过连接视觉编码器和大语言模型，实现强大的视觉理解和对话能力。

#### 技术特点

1. **开源可部署**
   - 模型权重开源
   - 支持本地部署
   - 社区活跃

2. **高效架构**
   - 视觉编码器 + LLM
   - 轻量级适配器
   - 高效推理

3. **对话能力**
   - 多轮对话
   - 上下文理解
   - 自然交互

#### 代码实现

```javascript
import { pipeline } from '@xenova/transformers';

class LLaVAModel {
  constructor() {
    this.model = null;
  }

  /**
   * 加载 LLaVA 模型
   */
  async loadModel() {
    // 使用 Hugging Face 的 LLaVA 模型
    this.model = await pipeline(
      'visual-question-answering',
      'llava-hf/llava-1.5-7b-hf'
    );
  }

  /**
   * 图像理解
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {string} prompt - 提示文本
   * @returns {Promise<string>} - 理解结果
   */
  async understandImage(imageUrl, prompt) {
    if (!this.model) {
      await this.loadModel();
    }

    const result = await this.model({
      image: imageUrl,
      question: prompt
    });

    return result.answer;
  }

  /**
   * 多轮对话
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {Array} messages - 对话历史
   * @returns {Promise<string>} - 回复
   */
  async chat(imageUrl, messages) {
    if (!this.model) {
      await this.loadModel();
    }

    // 构建对话上下文
    const context = this.buildContext(messages);
    
    const result = await this.model({
      image: imageUrl,
      question: context
    });

    return result.answer;
  }

  /**
   * 构建对话上下文
   * 
   * @param {Array} messages - 对话历史
   * @returns {string} - 上下文文本
   */
  buildContext(messages) {
    let context = '';
    
    for (const message of messages) {
      if (message.role === 'user') {
        context += `User: ${message.content}\n`;
      } else if (message.role === 'assistant') {
        context += `Assistant: ${message.content}\n`;
      }
    }
    
    return context;
  }

  /**
   * 图像描述生成
   * 
   * @param {string} imageUrl - 图像 URL
   * @returns {Promise<string>} - 描述文本
   */
  async describeImage(imageUrl) {
    return await this.understandImage(
      imageUrl,
      'Please describe this image in detail.'
    );
  }

  /**
   * 视觉问答
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {string} question - 问题
   * @returns {Promise<string>} - 答案
   */
  async askQuestion(imageUrl, question) {
    return await this.understandImage(imageUrl, question);
  }

  /**
   * 批量处理
   * 
   * @param {Array<Object>} items - 处理项列表
   * @returns {Promise<Array>} - 处理结果
   */
  async batchProcess(items) {
    const results = [];
    
    for (const item of items) {
      try {
        const result = await this.understandImage(
          item.imageUrl,
          item.prompt
        );
        
        results.push({
          imageUrl: item.imageUrl,
          prompt: item.prompt,
          result: result,
          success: true
        });
      } catch (error) {
        results.push({
          imageUrl: item.imageUrl,
          prompt: item.prompt,
          error: error.message,
          success: false
        });
      }
    }
    
    return results;
  }

  /**
   * 实时对话
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {Function} onMessage - 消息回调
   * @returns {Promise<void>}
   */
  async realtimeChat(imageUrl, onMessage) {
    let messages = [];
    
    const respond = async (userMessage) => {
      messages.push({ role: 'user', content: userMessage });
      
      const response = await this.chat(imageUrl, messages);
      
      messages.push({ role: 'assistant', content: response });
      
      onMessage({
        role: 'assistant',
        content: response
      });
    };
    
    return respond;
  }
}

// 使用示例
async function example() {
  const llava = new LLaVAModel();
  
  // 图像理解
  const result1 = await llava.understandImage(
    'https://example.com/image.jpg',
    'What do you see in this image?'
  );
  
  console.log('图像理解结果:', result1);
  
  // 多轮对话
  const messages = [
    { role: 'user', content: 'What is in this image?' },
    { role: 'assistant', content: 'This image shows a cat sitting on a table.' },
    { role: 'user', content: 'What color is the cat?' }
  ];
  
  const chatResponse = await llava.chat(
    'https://example.com/image.jpg',
    messages
  );
  
  console.log('对话回复:', chatResponse);
  
  // 批量处理
  const items = [
    { imageUrl: 'https://example.com/image1.jpg', prompt: 'Describe this image' },
    { imageUrl: 'https://example.com/image2.jpg', prompt: 'What objects are in this image?' }
```

### 4. Flamingo

#### 核心思想

Flamingo 是 DeepMind 推出的视觉语言模型，通过在视觉和语言数据上进行大规模预训练，实现少样本的多模态理解能力。

#### 技术特点

1. **少样本学习**
   - 快速适应新任务
   - 少样本示例学习
   - 强泛化能力

2. **视觉语言对齐**
   - 深度跨模态对齐
   - 统一表示空间
   - 高效信息融合

3. **多任务能力**
   - 图像描述
   - 视觉问答
   - 图像分类

#### 代码实现

```javascript
class FlamingoModel {
  constructor() {
    this.model = null;
  }

  /**
   * 加载 Flamingo 模型
   */
  async loadModel() {
    // 使用 Hugging Face 的 Flamingo 模型
    this.model = await pipeline(
      'image-to-text',
      'deepmind/flamingo'
    );
  }

  /**
   * 少样本图像描述
   * 
   * @param {string} imageUrl - 目标图像 URL
   * @param {Array<Object>} examples - 示例列表
   * @returns {Promise<string>} - 描述文本
   */
  async fewShotDescribe(imageUrl, examples) {
    if (!this.model) {
      await this.loadModel();
    }

    // 构建少样本提示
    const prompt = this.buildFewShotPrompt(examples);
    
    const result = await this.model({
      image: imageUrl,
      prompt: prompt
    });

    return result[0].generated_text;
  }

  /**
   * 构建少样本提示
   * 
   * @param {Array<Object>} examples - 示例列表
   * @returns {string} - 提示文本
   */
  buildFewShotPrompt(examples) {
    let prompt = '';
    
    for (const example of examples) {
      prompt += `Image: ${example.imageUrl}\n`;
      prompt += `Description: ${example.description}\n\n`;
    }
    
    prompt += 'Image: <TARGET_IMAGE>\n';
    prompt += 'Description:';
    
    return prompt;
  }

  /**
   * 少样本视觉问答
   * 
   * @param {string} imageUrl - 目标图像 URL
   * @param {string} question - 问题
   * @param {Array<Object>} examples - 示例列表
   * @returns {Promise<string>} - 答案
   */
  async fewShotVQA(imageUrl, question, examples) {
    if (!this.model) {
      await this.loadModel();
    }

    // 构建少样本提示
    const prompt = this.buildVQAPrompt(question, examples);
    
    const result = await this.model({
      image: imageUrl,
      prompt: prompt
    });

    return result[0].generated_text;
  }

  /**
   * 构建 VQA 提示
   * 
   * @param {string} question - 问题
   * @param {Array<Object>} examples - 示例列表
   * @returns {string} - 提示文本
   */
  buildVQAPrompt(question, examples) {
    let prompt = '';
    
    for (const example of examples) {
      prompt += `Image: ${example.imageUrl}\n`;
      prompt += `Question: ${example.question}\n`;
      prompt += `Answer: ${example.answer}\n\n`;
    }
    
    prompt += `Image: <TARGET_IMAGE>\n`;
    prompt += `Question: ${question}\n`;
    prompt += `Answer:`;
    
    return prompt;
  }

  /**
   * 零样本图像分类
   * 
   * @param {string} imageUrl - 图像 URL
   * @param {Array<string>} classes - 类别列表
   * @returns {Promise<Object>} - 分类结果
   */
  async zeroShotClassify(imageUrl, classes) {
    if (!this.model) {
      await this.loadModel();
    }

    const prompt = `What is in this image? Choose from: ${classes.join(', ')}`;
    
    const result = await this.model({
      image: imageUrl,
      prompt: prompt
    });

    const answer = result[0].generated_text;
    
    // 匹配最可能的类别
    const matchedClass = this.matchClass(answer, classes);
    
    return {
      class: matchedClass,
      confidence: this.calculateConfidence(answer, matchedClass)
    };
  }

  /**
   * 匹配类别
   * 
   * @param {string} answer - 模型回答
   * @param {Array<string>} classes - 类别列表
   * @returns {string} - 匹配的类别
   */
  matchClass(answer, classes) {
    const lowerAnswer = answer.toLowerCase();
    
    for (const className of classes) {
      if (lowerAnswer.includes(className.toLowerCase())) {
        return className;
      }
    }
    
    // 如果没有匹配，返回最相似的类别
    return classes[0];
  }

  /**
   * 计算置信度
   * 
   * @param {string} answer - 模型回答
   * @param {string} matchedClass - 匹配的类别
   * @returns {number} - 置信度
   */
  calculateConfidence(answer, matchedClass) {
    const lowerAnswer = answer.toLowerCase();
    const lowerClass = matchedClass.toLowerCase();
    
    if (lowerAnswer.includes(lowerClass)) {
      return 0.9;
    } else {
      return 0.5;
    }
  }

  /**
   * 图像检索
   * 
   * @param {string} query - 查询文本
   * @param {Array<string>} imageUrls - 图像 URL 列表
   * @returns {Promise<Array>} - 检索结果
   */
  async retrieveImages(query, imageUrls) {
    if (!this.model) {
      await this.loadModel();
    }

    const results = [];
    
    for (const imageUrl of imageUrls) {
      const prompt = `Does this image match the query: "${query}"? Answer yes or no.`;
      
      const result = await this.model({
        image: imageUrl,
        prompt: prompt
      });

      const answer = result[0].generated_text.toLowerCase();
      const matches = answer.includes('yes');
      
      results.push({
        imageUrl: imageUrl,
        matches: matches,
        confidence: matches ? 0.8 : 0.2
      });
    }
    
    // 按置信度排序
    results.sort((a, b) => b.confidence - a.confidence);
    
    return results;
  }
}

async function example() {
  const flamingo = new FlamingoModel();
  
  // 少样本图像描述
  const examples = [
    {
      imageUrl: 'https://example.com/example1.jpg',
      description: 'A cat sitting on a windowsill'
    },
    {
      imageUrl: 'https://example.com/example2.jpg',
      description: 'A dog playing in the park'
    }
  ];
  
  const description = await flamingo.fewShotDescribe(
    'https://example.com/target.jpg',
    examples
  );
  
  console.log('图像描述:', description);
  
  // 零样本图像分类
  const classes = ['cat', 'dog', 'bird', 'car'];
  const classification = await flamingo.zeroShotClassify(
    'https://example.com/image.jpg',
    classes
  );
  
  console.log('分类结果:', classification);

  const imageUrls = [
    'https://example.com/image1.jpg',
    'https://example.com/image2.jpg',
    'https://example.com/image3.jpg'
  ];
  
  const retrievalResults = await flamingo.retrieveImages(query, imageUrls);
  console.log('检索结果:', retrievalResults);
}
```
