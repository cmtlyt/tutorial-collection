## 📖 概述

神经符号 AI（Neuro-Symbolic AI）是结合神经网络和符号推理的混合智能系统。它融合了深度学习的感知能力和符号推理的逻辑能力，旨在构建既具备强大感知能力又具有可解释性和推理能力的智能系统。

## 🧠 什么是神经符号 AI

神经符号 AI 的核心思想是：**结合神经网络的学习能力和符号推理的逻辑能力，构建更强大、更可靠的智能系统**。

### 神经网络 vs 符号系统

| 特性 | 神经网络 | 符号系统 |
|------|---------|---------|
| 优势 | 感知能力强、学习能力强、适应性强 | 推理能力强、可解释性强、逻辑严密 |
| 劣势 | 可解释性差、推理能力弱、需要大量数据 | 感知能力弱、学习困难、难以处理不确定性 |
| 适用场景 | 图像识别、语音识别、自然语言处理 | 逻辑推理、规划、知识表示 |
| 数据需求 | 大量标注数据 | 领域知识、规则 |

### 神经符号 AI 的优势

- **可解释性**：结合符号推理，提供清晰的决策过程
- **推理能力**：支持复杂的逻辑推理和规划
- **学习能力**：利用神经网络从数据中学习
- **泛化能力**：结合归纳推理和演绎推理
- **可靠性**：符号系统提供逻辑约束，减少错误

## 🔢 符号推理

符号推理是基于符号和规则的推理方法，使用逻辑和知识库进行推理。

### 符号推理基础

#### 逻辑表示

```python
# Python 示例：符号推理基础
from typing import List, Dict, Set, Optional
from dataclasses import dataclass
from enum import Enum

class LogicalOperator(Enum):
    """逻辑运算符"""
    AND = "and"
    OR = "or"
    NOT = "not"
    IMPLIES = "implies"
    IFF = "iff"

@dataclass
class Atom:
    """原子命题"""
    name: str
    negated: bool = False
    
    def __str__(self):
        prefix = "¬" if self.negated else ""
        return f"{prefix}{self.name}"
    
    def evaluate(self, assignment: Dict[str, bool]) -> bool:
        """评估原子命题"""
        value = assignment.get(self.name, False)
        return not value if self.negated else value

@dataclass
class Formula:
    """逻辑公式"""
    left: 'Formula' | Atom
    operator: Optional[LogicalOperator]
    right: Optional['Formula' | Atom] = None
    
    def evaluate(self, assignment: Dict[str, bool]) -> bool:
        """评估公式"""
        left_val = self.left.evaluate(assignment) if isinstance(self.left, Formula) else self.left.evaluate(assignment)
        
        if self.operator == LogicalOperator.NOT:
            return not left_val
        
        right_val = self.right.evaluate(assignment) if isinstance(self.right, Formula) else self.right.evaluate(assignment)
        
        if self.operator == LogicalOperator.AND:
            return left_val and right_val
        elif self.operator == LogicalOperator.OR:
            return left_val or right_val
        elif self.operator == LogicalOperator.IMPLIES:
            return (not left_val) or right_val
        elif self.operator == LogicalOperator.IFF:
            return left_val == right_val
        
        return left_val
    
    def __str__(self):
        if self.operator == LogicalOperator.NOT:
            return f"¬{self.left}"
        return f"({self.left} {self.operator.value} {self.right})"

class KnowledgeBase:
    """知识库"""
    
    def __init__(self):
        self.facts: Dict[str, bool] = {}
        self.rules: List[Formula] = []
    
    def add_fact(self, name: str, value: bool = True):
        """添加事实"""
        self.facts[name] = value
    
    def add_rule(self, rule: Formula):
        """添加规则"""
        self.rules.append(rule)
    
    def query(self, query: Formula) -> bool:
        """查询"""
        return query.evaluate(self.facts)
    
    def forward_chaining(self) -> Dict[str, bool]:
        """前向链推理"""
        changed = True
        while changed:
            changed = False
            for rule in self.rules:
                if rule.operator == LogicalOperator.IMPLIES:
                    # 规则形式: A -> B
                    if isinstance(rule.left, Atom) and isinstance(rule.right, Atom):
                        antecedent = rule.left.evaluate(self.facts)
                        consequent = rule.right.name
                        
                        if antecedent and consequent not in self.facts:
                            self.facts[consequent] = True
                            changed = True
        
        return self.facts
    
    def backward_chaining(self, goal: Atom) -> bool:
        """后向链推理"""
        if goal.name in self.facts:
            return self.facts[goal.name]
        
        for rule in self.rules:
            if rule.operator == LogicalOperator.IMPLIES:
                if isinstance(rule.right, Atom) and rule.right.name == goal.name:
                    if isinstance(rule.left, Atom):
                        return self.backward_chaining(rule.left)
                    elif isinstance(rule.left, Formula):
                        # 处理复合前提
                        return self._evaluate_formula(rule.left)
        
        return False
    
    def _evaluate_formula(self, formula: Formula) -> bool:
        """评估复合公式"""
        if isinstance(formula, Atom):
            return self.backward_chaining(formula)
        
        if formula.operator == LogicalOperator.AND:
            return (self._evaluate_formula(formula.left) and 
                    self._evaluate_formula(formula.right))
        elif formula.operator == LogicalOperator.OR:
            return (self._evaluate_formula(formula.left) or 
                    self._evaluate_formula(formula.right))
        
        return False
```

#### 知识图谱推理

```python
# Python 示例：知识图谱推理
from typing import List, Tuple, Dict, Set
import networkx as nx

class KnowledgeGraph:
    """知识图谱"""
    
    def __init__(self):
        self.graph = nx.DiGraph()
        self.entity_types: Dict[str, str] = {}
        self.relation_types: Set[str] = set()
    
    def add_entity(self, entity: str, entity_type: str):
        """添加实体"""
        self.graph.add_node(entity)
        self.entity_types[entity] = entity_type
    
    def add_relation(self, subject: str, relation: str, object: str):
        """添加关系"""
        self.graph.add_edge(subject, object, relation=relation)
        self.relation_types.add(relation)
    
    def get_relations(self, entity: str, relation: str = None) -> List[Tuple[str, str, str]]:
        """获取关系"""
        relations = []
        
        if relation:
            # 获取特定关系
            for _, obj, data in self.graph.out_edges(entity, data=True):
                if data.get('relation') == relation:
                    relations.append((entity, relation, obj))
        else:
            # 获取所有关系
            for _, obj, data in self.graph.out_edges(entity, data=True):
                relations.append((entity, data.get('relation'), obj))
        
        return relations
    
    def find_path(self, start: str, end: str) -> List[str]:
        """查找路径"""
        try:
            return nx.shortest_path(self.graph, start, end)
        except nx.NetworkXNoPath:
            return []
    
    def infer_relations(self, entity: str, max_depth: int = 3) -> Dict[str, List[str]]:
        """推断关系"""
        inferred = {}
        
        for relation in self.relation_types:
            inferred_relations = []
            visited = set()
            
            # 使用广度优先搜索
            queue = [(entity, 0)]
            
            while queue:
                current, depth = queue.pop(0)
                
                if depth >= max_depth or current in visited:
                    continue
                
                visited.add(current)
                
                for _, obj, data in self.graph.out_edges(current, data=True):
                    if data.get('relation') == relation:
                        inferred_relations.append(obj)
                        queue.append((obj, depth + 1))
            
            if inferred_relations:
                inferred[relation] = inferred_relations
        
        return inferred
    
    def transitive_closure(self, relation: str) -> Set[Tuple[str, str]]:
        """计算传递闭包"""
        edges = set()
        
        for subject in self.graph.nodes():
            # 使用深度优先搜索
            stack = [(subject, [subject])]
            
            while stack:
                current, path = stack.pop()
                
                for _, obj, data in self.graph.out_edges(current, data=True):
                    if data.get('relation') == relation and obj not in path:
                        edges.add((subject, obj))
                        stack.append((obj, path + [obj]))
        
        return edges

class SymbolicReasoner:
    """符号推理器"""
    
    def __init__(self, knowledge_graph: KnowledgeGraph):
        self.kg = knowledge_graph
        self.rules: List[Dict] = []
    
    def add_rule(self, rule: Dict):
        """
        添加推理规则
        
        规则格式:
        {
            'premises': [(subject, relation, object), ...],
            'conclusion': (subject, relation, object),
            'confidence': float
        }
        """
        self.rules.append(rule)
    
    def apply_rules(self, entity: str) -> List[Dict]:
        """应用推理规则"""
        conclusions = []
        
        for rule in self.rules:
            premises = rule['premises']
            conclusion = rule['conclusion']
            
            # 检查所有前提是否满足
            all_premises_true = True
            for subject, relation, object in premises:
                relations = self.kg.get_relations(subject, relation)
                if not any(obj == object for _, _, obj in relations):
                    all_premises_true = False
                    break
            
            # 如果所有前提都满足，则得出结论
            if all_premises_true:
                conclusions.append({
                    'conclusion': conclusion,
                    'confidence': rule.get('confidence', 1.0),
                    'rule': rule
                })
        
        return conclusions
    
    def explain(self, conclusion: Dict) -> str:
        """解释推理过程"""
        rule = conclusion['rule']
        premises = rule['premises']
        
        explanation = "推理过程:\n"
        for i, (subject, relation, object) in enumerate(premises, 1):
            explanation += f"{i}. {subject} {relation} {object}\n"
        
        conclusion_str = f"{conclusion['conclusion'][0]} {conclusion['conclusion'][1]} {conclusion['conclusion'][2]}"
        explanation += f"\n因此: {conclusion_str}"
        
        return explanation
```

```python
# Python 示例：神经符号集成
import torch
import torch.nn as nn
from typing import List, Dict, Optional

class NeuralSymbolicIntegrator:
    """神经符号集成器"""
    
    def __init__(
        self,
        neural_network: nn.Module,
        symbolic_reasoner: SymbolicReasoner
    ):
        self.neural_network = neural_network
        self.symbolic_reasoner = symbolic_reasoner
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    
    def perceive(self, input_data: torch.Tensor) -> Dict:
        """感知阶段：使用神经网络提取特征"""
        with torch.no_grad():
            features = self.neural_network(input_data.to(self.device))
        
        # 将神经网络输出转换为符号表示
        symbols = self._features_to_symbols(features)
        
        return symbols
    
    def reason(self, symbols: Dict) -> List[Dict]:
        """推理阶段：使用符号推理"""
        conclusions = []
        
        for entity, relations in symbols.items():
            # 应用推理规则
            entity_conclusions = self.symbolic_reasoner.apply_rules(entity)
            conclusions.extend(entity_conclusions)
        
        return conclusions
    
    def act(self, conclusions: List[Dict]) -> Dict:
        """行动阶段：根据推理结果生成行动"""
        # 选择置信度最高的结论
        if conclusions:
            best_conclusion = max(conclusions, key=lambda x: x['confidence'])
            return {
                'action': best_conclusion['conclusion'],
                'confidence': best_conclusion['confidence']
            }
        
        return {'action': None, 'confidence': 0.0}
    
    def _features_to_symbols(self, features: torch.Tensor) -> Dict:
        """将特征转换为符号"""
        # 简化的实现：使用阈值将连续特征转换为离散符号
        symbols = {}
        
        # 假设特征的维度对应不同的实体
        for i, feature in enumerate(features):
            entity_name = f"entity_{i}"
            
            # 将特征向量转换为关系
            relations = []
            for j, value in enumerate(feature):
                if value > 0.5:  # 阈值
                    relation_name = f"has_property_{j}"
                    relations.append((entity_name, relation_name, "true"))
            
            if relations:
                symbols[entity_name] = relations
        
        return symbols
    
    def forward(self, input_data: torch.Tensor) -> Dict:
        """完整的前向传播"""
        # 感知
        symbols = self.perceive(input_data)
        
        # 推理
        conclusions = self.reason(symbols)
        
        # 行动
        action = self.act(conclusions)
        
        return action

class NeuroSymbolicAgent:
    """神经符号智能体"""
    
    def __init__(
        self,
        neural_network: nn.Module,
        knowledge_graph: KnowledgeGraph
    ):
        self.neural_network = neural_network
        self.knowledge_graph = knowledge_graph
        
        # 创建符号推理器
        self.symbolic_reasoner = SymbolicReasoner(knowledge_graph)
        
        # 创建集成器
        self.integrator = NeuralSymbolicIntegrator(
            neural_network,
            symbolic_reasoner
        )
        
        # 添加推理规则
        self._add_default_rules()
    
    def _add_default_rules(self):
        """添加默认推理规则"""
        # 传递性规则
        self.symbolic_reasoner.add_rule({
            'premises': [
                ('A', 'related_to', 'B'),
                ('B', 'related_to', 'C')
            ],
            'conclusion': ('A', 'related_to', 'C'),
            'confidence': 0.8
        })
        
        # 对称性规则
        self.symbolic_reasoner.add_rule({
            'premises': [('A', 'related_to', 'B')],
            'conclusion': ('B', 'related_to', 'A'),
            'confidence': 0.9
        })
    
    def process(self, input_data: torch.Tensor) -> Dict:
        """处理输入"""
        return self.integrator.forward(input_data)
    
    def explain(self, input_data: torch.Tensor) -> str:
        """解释决策过程"""
        # 获取感知结果
        symbols = self.integrator.perceive(input_data)
        
        # 获取推理结果
        conclusions = self.integrator.reason(symbols)
        
        # 生成解释
        explanation = "决策过程:\n\n"
        
        explanation += "1. 感知阶段:\n"
        explanation += f"   从输入中提取了 {len(symbols)} 个实体\n\n"
        
        explanation += "2. 推理阶段:\n"
        for i, conclusion in enumerate(conclusions, 1):
            rule_explanation = self.symbolic_reasoner.explain(conclusion)
            explanation += f"   {i}. {rule_explanation}\n"
        
        explanation += "\n3. 行动阶段:\n"
        action = self.integrator.act(conclusions)
        explanation += f"   执行行动: {action}\n"
        
        return explanation
```

```python
# Python 示例：神经增强符号推理
import torch
import torch.nn as nn

class NeuralEnhancedReasoner:
    """神经增强推理器"""
    
    def __init__(
        self,
        base_reasoner: SymbolicReasoner,
        confidence_network: nn.Module
    ):
        self.base_reasoner = base_reasoner
        self.confidence_network = confidence_network
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.confidence_network.to(self.device)
    
    def reason_with_confidence(
        self,
        entity: str,
        context: torch.Tensor
    ) -> List[Dict]:
        """带置信度的推理"""
        # 使用基础推理器进行推理
        conclusions = self.base_reasoner.apply_rules(entity)
        
        # 使用神经网络调整置信度
        enhanced_conclusions = []
        for conclusion in conclusions:
            # 将结论和上下文输入神经网络
            confidence_input = self._prepare_confidence_input(
                conclusion,
                context
            )
            
            with torch.no_grad():
                adjusted_confidence = self.confidence_network(
                    confidence_input.to(self.device)
                )
            
            enhanced_conclusion = conclusion.copy()
            enhanced_conclusion['confidence'] = adjusted_confidence.item()
            enhanced_conclusions.append(enhanced_conclusion)
        
        return enhanced_conclusions
    
    def _prepare_confidence_input(
        self,
        conclusion: Dict,
        context: torch.Tensor
    ) -> torch.Tensor:
        """准备置信度网络的输入"""
        # 简化的实现：将结论编码为向量
        conclusion_vec = self._encode_conclusion(conclusion)
        
        # 与上下文拼接
        combined = torch.cat([context, conclusion_vec], dim=-1)
        
        return combined
    
    def _encode_conclusion(self, conclusion: Dict) -> torch.Tensor:
        """编码结论"""
        # 简化的实现：使用哈希编码
        subject, relation, object = conclusion['conclusion']
        
        # 将字符串转换为简单的数值编码
        subject_hash = hash(subject) % 100
        relation_hash = hash(relation) % 100
        object_hash = hash(object) % 100
        
        return torch.tensor([
            subject_hash / 100.0,
            relation_hash / 100.0,
            object_hash / 100.0,
            conclusion.get('confidence', 0.0)
        ]).unsqueeze(0)

class ConfidenceNetwork(nn.Module):
    """置信度网络"""
    
    def __init__(self, input_dim: int = 8, hidden_dim: int = 64):
        super().__init__()
        
        self.network = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(hidden_dim // 2, 1),
            nn.Sigmoid()
        )
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        return self.network(x)
```

```python
# Python 示例：符号引导神经网络
import torch
import torch.nn as nn

class SymbolGuidedNetwork(nn.Module):
    """符号引导网络"""
    
    def __init__(
        self,
        base_network: nn.Module,
        symbolic_constraints: List[Dict]
    ):
        super().__init__()
        
        self.base_network = base_network
        self.symbolic_constraints = symbolic_constraints
        
        # 添加约束层
        self.constraint_layer = ConstraintLayer(symbolic_constraints)
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        # 基础网络前向传播
        features = self.base_network(x)
        
        # 应用符号约束
        constrained_output = self.constraint_layer(features)
        
        return constrained_output
    
    def apply_constraints(self, output: torch.Tensor) -> torch.Tensor:
        """应用符号约束"""
        return self.constraint_layer(output)

class ConstraintLayer(nn.Module):
    """约束层"""
    
    def __init__(self, constraints: List[Dict]):
        super().__init__()
        
        self.constraints = constraints
        
        # 为每个约束学习权重
        self.constraint_weights = nn.ParameterList([
            nn.Parameter(torch.randn(1)) for _ in constraints
        ])
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """应用约束"""
        output = x.clone()
        
        for i, constraint in enumerate(self.constraints):
            weight = self.constraint_weights[i]
            output = self._apply_single_constraint(output, constraint, weight)
        
        return output
    
    def _apply_single_constraint(
        self,
        x: torch.Tensor,
        constraint: Dict,
        weight: torch.Tensor
    ) -> torch.Tensor:
        """应用单个约束"""
        constraint_type = constraint['type']
        
        if constraint_type == 'monotonic':
            # 单调性约束
            return self._apply_monotonic_constraint(x, constraint, weight)
        elif constraint_type == 'range':
            # 范围约束
            return self._apply_range_constraint(x, constraint, weight)
        elif constraint_type == 'logical':
            # 逻辑约束
            return self._apply_logical_constraint(x, constraint, weight)
        
        return x
    
    def _apply_monotonic_constraint(
        self,
        x: torch.Tensor,
        constraint: Dict,
        weight: torch.Tensor
    ) -> torch.Tensor:
        """应用单调性约束"""
        feature_idx = constraint['feature_index']
        direction = constraint.get('direction', 'increasing')
        
        # 确保特征的单调性
        if direction == 'increasing':
            # 确保特征值递增
            sorted_values, _ = torch.sort(x[:, feature_idx])
            x[:, feature_idx] = sorted_values
        else:
            # 确保特征值递减
            sorted_values, _ = torch.sort(x[:, feature_idx], descending=True)
            x[:, feature_idx] = sorted_values
        
        return x
    
    def _apply_range_constraint(
        self,
        x: torch.Tensor,
        constraint: Dict,
        weight: torch.Tensor
    ) -> torch.Tensor:
        """应用范围约束"""
        feature_idx = constraint['feature_index']
        min_val = constraint.get('min', 0.0)
        max_val = constraint.get('max', 1.0)
        
        # 将特征值限制在指定范围内
        x[:, feature_idx] = torch.clamp(x[:, feature_idx], min_val, max_val)
        
        return x
    
    def _apply_logical_constraint(
        self,
        x: torch.Tensor,
        constraint: Dict,
        weight: torch.Tensor
    ) -> torch.Tensor:
        """应用逻辑约束"""
        # 简化的实现：确保某些特征之间的关系
        features = constraint['features']
        relation = constraint.get('relation', 'and')
        
        if relation == 'and':
            # 所有特征都应该为正
            for feature_idx in features:
                x[:, feature_idx] = torch.relu(x[:, feature_idx])
        elif relation == 'or':
            # 至少一个特征应该为正
            mask = (x[:, features] > 0).any(dim=1)
            x[~mask, features] = torch.abs(x[~mask, features])
        
        return x
```

```python
# Python 示例：解释生成
from typing import List, Dict, Optional

class ExplanationGenerator:
    """解释生成器"""
    
    def __init__(self, neuro_symbolic_agent: NeuroSymbolicAgent):
        self.agent = neuro_symbolic_agent
    
    def generate_explanation(
        self,
        input_data: torch.Tensor,
        decision: Dict
    ) -> str:
        """生成解释"""
        explanation_parts = []
        
        # 1. 感知阶段解释
        perception_explanation = self._explain_perception(input_data)
        explanation_parts.append(perception_explanation)
        
        # 2. 推理阶段解释
        reasoning_explanation = self._explain_reasoning(input_data, decision)
        explanation_parts.append(reasoning_explanation)
        
        # 3. 决策阶段解释
        decision_explanation = self._explain_decision(decision)
        explanation_parts.append(decision_explanation)
        
        # 组合所有部分
        full_explanation = "\n\n".join(explanation_parts)
        
        return full_explanation
    
    def _explain_perception(self, input_data: torch.Tensor) -> str:
        """解释感知阶段"""
        symbols = self.agent.integrator.perceive(input_data)
        
        explanation = "📊 感知阶段分析:\n"
        explanation += f"从输入数据中识别出 {len(symbols)} 个实体:\n"
        
        for entity, relations in symbols.items():
            explanation += f"\n  • {entity}:\n"
            for subject, relation, object in relations:
                explanation += f"    - {subject} {relation} {object}\n"
        
        return explanation
    
    def _explain_reasoning(
        self,
        input_data: torch.Tensor,
        decision: Dict
    ) -> str:
        """解释推理阶段"""
        symbols = self.agent.integrator.perceive(input_data)
        conclusions = self.agent.integrator.reason(symbols)
        
        explanation = "🧠 推理阶段分析:\n"
        explanation += f"应用了 {len(self.agent.symbolic_reasoner.rules)} 条推理规则\n\n"
        
        for i, conclusion in enumerate(conclusions, 1):
            rule_explanation = self.agent.symbolic_reasoner.explain(conclusion)
            explanation += f"推理 {i} (置信度: {conclusion['confidence']:.2f}):\n"
            explanation += f"{rule_explanation}\n\n"
        
        return explanation
    
    def _explain_decision(self, decision: Dict) -> str:
        """解释决策阶段"""
        explanation = "🎯 决策阶段分析:\n"
        
        if decision.get('action'):
            action = decision['action']
            confidence = decision.get('confidence', 0.0)
            
            explanation += f"最终决策: {action[0]} {action[1]} {action[2]}\n"
            explanation += f"决策置信度: {confidence:.2f}\n"
            
            if confidence > 0.8:
                explanation += "决策可靠性: 高"
            elif confidence > 0.5:
                explanation += "决策可靠性: 中"
            else:
                explanation += "决策可靠性: 低，建议人工复核"
        else:
            explanation += "未生成有效决策"
        
        return explanation
    
    def generate_contrastive_explanation(
        self,
        input_data: torch.Tensor,
        alternative_input: torch.Tensor
    ) -> str:
        """生成对比解释"""
        decision1 = self.agent.process(input_data)
        decision2 = self.agent.process(alternative_input)
        
        explanation = "🔄 对比分析:\n\n"
        
        explanation += "场景 1:\n"
        explanation += self.generate_explanation(input_data, decision1)
        
        explanation += "\n\n" + "="*50 + "\n\n"
        
        explanation += "场景 2:\n"
        explanation += self.generate_explanation(alternative_input, decision2)
        
        explanation += "\n\n" + "="*50 + "\n\n"
        
        explanation += "差异分析:\n"
        explanation += self._analyze_differences(decision1, decision2)
        
        return explanation
    
    def _analyze_differences(
        self,
        decision1: Dict,
        decision2: Dict
    ) -> str:
        """分析决策差异"""
        analysis = ""
        
        action1 = decision1.get('action')
        action2 = decision2.get('action')
        
        if action1 and action2:
            if action1 != action2:
                analysis += f"决策不同:\n"
                analysis += f"  场景1: {action1[0]} {action1[1]} {action1[2]}\n"
                analysis += f"  场景2: {action2[0]} {action2[1]} {action2[2]}\n"
            else:
                analysis += f"决策相同: {action1[0]} {action1[1]} {action1[2]}\n"
                
                confidence1 = decision1.get('confidence', 0.0)
                confidence2 = decision2.get('confidence', 0.0)
                
                if abs(confidence1 - confidence2) > 0.1:
                    analysis += f"但置信度不同:\n"
                    analysis += f"  场景1: {confidence1:.2f}\n"
                    analysis += f"  场景2: {confidence2:.2f}\n"
        
        return analysis
```
