## 🧠 Mixture of Experts (MoE)

Mixture of Experts (MoE) 是一种通过激活模型的不同部分来处理不同输入的架构，能够在保持模型能力的同时显著提高计算效率。

### 专家路由

专家路由机制决定输入应该由哪些专家处理。

```javascript
// architectures/moe/router.js
class MoERouter {
  constructor(config) {
    /**
     * MoE 路由器
     * 
     * 参数:
     *   config: 配置对象
     */
    this.config = {
      numExperts: config.numExperts || 8,
      topK: config.topK || 2, // 激活的专家数量
      capacityFactor: config.capacityFactor || 1.5, // 容量因子
      noiseEnabled: config.noiseEnabled || true,
      ...config
    };

    this.gateWeights = this._initializeGateWeights();
    this.expertCapacity = Math.floor(
      this.config.capacityFactor * this.config.topK
    );
  }

  _initializeGateWeights() {
    /**
     * 初始化门控权重
     * 
     * 返回:
     *   权重矩阵
     */
    const inputDim = 512; // 假设输入维度
    const weights = [];

    for (let i = 0; i < this.config.numExperts; i++) {
      const expertWeights = [];
      for (let j = 0; j < inputDim; j++) {
        expertWeights.push((Math.random() - 0.5) * 0.1);
      }
      weights.push(expertWeights);
    }

    return weights;
  }

  route(input) {
    /**
     * 路由输入到专家
     * 
     * 参数:
     *   input: 输入向量
     * 
     * 返回:
     *   路由结果
     */
    // 计算门控分数
    const gateScores = this._computeGateScores(input);

    // 选择 top-k 专家
    const topExperts = this._selectTopKExperts(gateScores);

    // 计算路由权重
    const weights = this._computeRoutingWeights(topExperts);

    return {
      expertIndices: topExperts.map(e => e.index),
      expertWeights: weights,
      gateScores: gateScores
    };
  }

  _computeGateScores(input) {
    /**
     * 计算门控分数
     * 
     * 参数:
     *   input: 输入向量
     * 
     * 返回:
     *   分数数组
     */
    const scores = [];

    for (let i = 0; i < this.config.numExperts; i++) {
      // 计算点积
      let score = 0;
      for (let j = 0; j < input.length && j < this.gateWeights[i].length; j++) {
        score += input[j] * this.gateWeights[i][j];
      }

      // 添加噪声（训练时）
      if (this.config.noiseEnabled) {
        score += this._generateNoise();
      }

      scores.push(score);
    }

    return scores;
  }

  _generateNoise() {
    /**
     * 生成噪声
     * 
     * 返回:
     *   噪声值
     */
    const mean = 0;
    const std = 0.1;
    const u1 = Math.random();
    const u2 = Math.random();
    const z0 = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
    return mean + std * z0;
  }

  _selectTopKExperts(scores) {
    /**
     * 选择 top-k 专家
     * 
     * 参数:
     *   scores: 分数数组
     * 
     * 返回:
     *   专家数组
     */
    const indexedScores = scores.map((score, index) => ({
      index,
      score
    }));

    indexedScores.sort((a, b) => b.score - a.score);

    return indexedScores.slice(0, this.config.topK);
  }

  _computeRoutingWeights(topExperts) {
    /**
     * 计算路由权重
     * 
     * 参数:
     *   topExperts: top-k 专家
     * 
     * 返回:
     *   权重数组
     */
    const scores = topExperts.map(e => e.score);
    const maxScore = Math.max(...scores);

    // 计算指数
    const expScores = scores.map(s => Math.exp(s - maxScore));

    // 归一化
    const sumExp = expScores.reduce((sum, e) => sum + e, 0);
    const weights = expScores.map(e => e / sumExp);

    return weights;
  }

  batchRoute(inputs) {
    /**
     * 批量路由
     * 
     * 参数:
     *   inputs: 输入数组
     * 
     * 返回:
     *   批量路由结果
     */
    return inputs.map(input => this.route(input));
  }
}

// 使用示例
function moERouterExample() {
  const router = new MoERouter({
    numExperts: 8,
    topK: 2,
    capacityFactor: 1.5,
    noiseEnabled: true
  });

  // 单个输入路由
  const input = Array.from({ length: 512 }, () => Math.random());
  const routing = router.route(input);

  console.log('选中的专家:', routing.expertIndices);
  console.log('专家权重:', routing.expertWeights);
  console.log('门控分数:', routing.gateScores);

  // 批量路由
  const inputs = Array.from({ length: 4 }, () => 
    Array.from({ length: 512 }, () => Math.random())
  );

  const batchRouting = router.batchRoute(inputs);
  console.log('批量路由结果:', batchRouting);
}
```

### 负载均衡

负载均衡确保专家之间的负载分布均匀。

```javascript
// architectures/moe/load-balancer.js
class MoELoadBalancer {
  constructor(config) {
    /**
     * MoE 负载均衡器
     * 
     * 参数:
     *   config: 配置对象
     */
    this.config = {
      numExperts: config.numExperts || 8,
      auxLossCoefficient: config.auxLossCoefficient || 0.01,
      ...config
    };

    this.expertLoads = new Array(this.config.numExperts).fill(0);
    this.expertCapacities = new Array(this.config.numExperts).fill(0);
  }

  updateExpertLoads(expertIndices) {
    /**
     * 更新专家负载
     * 
     * 参数:
     *   expertIndices: 专家索引数组
     */
    for (const index of expertIndices) {
      this.expertLoads[index]++;
    }
  }

  getExpertLoads() {
    /**
     * 获取专家负载
     * 
     * 返回:
     *   负载数组
     */
    return this.expertLoads;
  }

  calculateAuxiliaryLoss(gateScores, expertIndices) {
    /**
     * 计算辅助损失
     * 
     * 参数:
     *   gateScores: 门控分数
     *   expertIndices: 专家索引
     * 
     * 返回:
     *   辅助损失
     */
    // 计算每个专家的平均分数
    const expertScores = new Array(this.config.numExperts).fill(0);
    const expertCounts = new Array(this.config.numExperts).fill(0);

    for (let i = 0; i < expertIndices.length; i++) {
      const expertIndex = expertIndices[i];
      expertScores[expertIndex] += gateScores[i];
      expertCounts[expertIndex]++;
    }

    // 计算平均分数
    const meanScores = expertScores.map((sum, i) => 
      expertCounts[i] > 0 ? sum / expertCounts[i] : 0
    );

    // 计算所有分数的平均值
    const overallMean = meanScores.reduce((sum, s) => sum + s, 0) / meanScores.length;

    // 计算辅助损失（鼓励专家负载均衡）
    const auxLoss = meanScores.reduce((sum, s) => 
      sum + Math.pow(s - overallMean, 2), 0
    ) / meanScores.length;

    return auxLoss * this.config.auxLossCoefficient;
  }

  applyCapacityConstraints(routingResults) {
    /**
     * 应用容量约束
     * 
     * 参数:
     *   routingResults: 路由结果数组
     * 
     * 返回:
     *   过滤后的结果
     */
    const filteredResults = [];

    for (const result of routingResults) {
      const validExperts = [];
      const validWeights = [];

      for (let i = 0; i < result.expertIndices.length; i++) {
        const expertIndex = result.expertIndices[i];

        // 检查专家容量
        if (this.expertLoads[expertIndex] < this.expertCapacities[expertIndex]) {
          validExperts.push(expertIndex);
          validWeights.push(result.expertWeights[i]);
        }
      }

      // 如果没有有效专家，选择负载最低的专家
      if (validExperts.length === 0) {
        const minLoadIndex = this.expertLoads.indexOf(Math.min(...this.expertLoads));
        validExperts.push(minLoadIndex);
        validWeights.push(1.0);
      }

      // 重新归一化权重
      const sumWeights = validWeights.reduce((sum, w) => sum + w, 0);
      const normalizedWeights = validWeights.map(w => w / sumWeights);

      filteredResults.push({
        ...result,
        expertIndices: validExperts,
        expertWeights: normalizedWeights
      });
    }

    return filteredResults;
  }

  resetLoads() {
    /**
     * 重置负载
     */
    this.expertLoads.fill(0);
  }

  getLoadStatistics() {
    /**
     * 获取负载统计
     * 
     * 返回:
     *   统计对象
     */
    const loads = this.expertLoads;
    const mean = loads.reduce((sum, l) => sum + l, 0) / loads.length;
    const variance = loads.reduce((sum, l) => sum + Math.pow(l - mean, 2), 0) / loads.length;
    const std = Math.sqrt(variance);

    return {
      mean,
      std,
      min: Math.min(...loads),
      max: Math.max(...loads),
      loads
    };
  }
}

// 使用示例
function moELoadBalancerExample() {
  const balancer = new MoELoadBalancer({
    numExperts: 8,
    auxLossCoefficient: 0.01
  });

  // 模拟路由结果
  const routingResults = [
    { expertIndices: [0, 2], expertWeights: [0.7, 0.3], gateScores: [1.2, 0.8] },
    { expertIndices: [1, 3], expertWeights: [0.6, 0.4], gateScores: [1.0, 0.9] },
    { expertIndices: [0, 4], expertWeights: [0.8, 0.2], gateScores: [1.5, 0.7] }
  ];

  // 更新负载
  for (const result of routingResults) {
    balancer.updateExpertLoads(result.expertIndices);
  }

  // 获取负载统计
  const stats = balancer.getLoadStatistics();
  console.log('负载统计:', stats);

  // 计算辅助损失
  const auxLoss = balancer.calculateAuxiliaryLoss(
    [1.2, 0.8, 1.0, 0.9, 1.5, 0.7],
    [0, 2, 1, 3, 0, 4]
  );

  console.log('辅助损失:', auxLoss);

  // 应用容量约束
  const filtered = balancer.applyCapacityConstraints(routingResults);
  console.log('过滤后的结果:', filtered);
}
```

### 训练策略

MoE 模型的训练需要特殊的策略来确保专家的多样性和负载均衡。

```javascript
// architectures/moe/trainer.js
class MoETrainer {
  constructor(config) {
    /**
     * MoE 训练器
     * 
     * 参数:
     *   config: 配置对象
     */
    this.config = {
      numExperts: config.numExperts || 8,
      learningRate: config.learningRate || 0.001,
      auxLossWeight: config.auxLossWeight || 0.01,
      zLossWeight: config.zLossWeight || 0.0001,
      ...config
    };

    this.expertWeights = this._initializeExpertWeights();
    this.gateWeights = this._initializeGateWeights();
    this.optimizer = new SGD(this.config.learningRate);
  }

  _initializeExpertWeights() {
    /**
     * 初始化专家权重
     * 
     * 返回:
     *   专家权重数组
     */
    const experts = [];
    const inputDim = 512;
    const outputDim = 512;

    for (let i = 0; i < this.config.numExperts; i++) {
      const expert = {
        weights: Array.from({ length: inputDim * outputDim }, () => 
          (Math.random() - 0.5) * 0.1
        ),
        bias: Array.from({ length: outputDim }, () => 0)
      };
      experts.push(expert);
    }

    return experts;
  }

  _initializeGateWeights() {
    /**
     * 初始化门控权重
     * 
     * 返回:
     *   门控权重矩阵
     */
    const inputDim = 512;
    const weights = [];

    for (let i = 0; i < this.config.numExperts; i++) {
      const expertWeights = Array.from({ length: inputDim }, () => 
        (Math.random() - 0.5) * 0.1
      );
      weights.push(expertWeights);
    }

    return weights;
  }

  forward(input) {
    /**
     * 前向传播
     * 
     * 参数:
     *   input: 输入向量
     * 
     * 返回:
     *   输出和中间结果
     */
    // 计算门控分数
    const gateScores = this._computeGateScores(input);

    // 选择 top-k 专家
    const topExperts = this._selectTopKExperts(gateScores);

    // 计算路由权重
    const weights = this._computeRoutingWeights(topExperts);

    // 专家计算
    const expertOutputs = [];
    for (const expert of topExperts) {
      const output = this._expertForward(input, expert.index);
      expertOutputs.push({
        index: expert.index,
        output
      });
    }

    // 组合输出
    const combinedOutput = this._combineOutputs(expertOutputs, weights);

    return {
      output: combinedOutput,
      gateScores,
      topExperts,
      weights,
      expertOutputs
    };
  }

  _computeGateScores(input) {
    /**
     * 计算门控分数
     * 
     * 参数:
     *   input: 输入向量
     * 
     * 返回:
     *   分数数组
     */
    const scores = [];

    for (let i = 0; i < this.config.numExperts; i++) {
      let score = 0;
      for (let j = 0; j < input.length && j < this.gateWeights[i].length; j++) {
        score += input[j] * this.gateWeights[i][j];
      }
      scores.push(score);
    }

    return scores;
  }

  _selectTopKExperts(scores) {
    /**
     * 选择 top-k 专家
     * 
     * 参数:
     *   scores: 分数数组
     * 
     * 返回:
     *   专家数组
     */
    const indexedScores = scores.map((score, index) => ({
      index,
      score
    }));

    indexedScores.sort((a, b) => b.score - a.score);
    return indexedScores.slice(0, 2); // top-2
  }

  _computeRoutingWeights(topExperts) {
    /**
     * 计算路由权重
     * 
     * 参数:
     *   topExperts: top-k 专家
     * 
     * 返回:
     *   权重数组
     */
    const scores = topExperts.map(e => e.score);
    const maxScore = Math.max(...scores);
    const expScores = scores.map(s => Math.exp(s - maxScore));
    const sumExp = expScores.reduce((sum, e) => sum + e, 0);
    return expScores.map(e => e / sumExp);
  }

  _expertForward(input, expertIndex) {
    /**
     * 专家前向传播
     * 
     * 参数:
     *   input: 输入向量
     *   expertIndex: 专家索引
     * 
     * 返回:
     *   输出向量
     */
    const expert = this.expertWeights[expertIndex];
    const inputDim = 512;
    const outputDim = 512;

    const output = [];

    for (let i = 0; i < outputDim; i++) {
      let sum = expert.bias[i];

      for (let j = 0; j < inputDim; j++) {
        sum += input[j] * expert.weights[i * inputDim + j];
      }

      output.push(sum);
    }

    return output;
  }

  _combineOutputs(expertOutputs, weights) {
    /**
     * 组合专家输出
     * 
     * 参数:
     *   expertOutputs: 专家输出数组
     *   weights: 权重数组
     * 
     * 返回:
     *   组合输出
     */
    const outputDim = 512;
    const combined = Array.from({ length: outputDim }, () => 0);

    for (let i = 0; i < expertOutputs.length; i++) {
      const weight = weights[i];
      const expertOutput = expertOutputs[i].output;

      for (let j = 0; j < outputDim; j++) {
        combined[j] += weight * expertOutput[j];
      }
    }

    return combined;
  }

  backward(input, forwardResult, gradients) {
    /**
     * 反向传播
     * 
     * 参数:
     *   input: 输入向量
     *   forwardResult: 前向传播结果
     *   gradients: 输出梯度
     * 
     * 返回:
     *   损失和辅助损失
     */
    // 计算辅助损失（负载均衡）
    const auxLoss = this._computeAuxiliaryLoss(forwardResult.gateScores);

    // 计算 Z 损失（门控分数稳定性）
    const zLoss = this._computeZLoss(forwardResult.gateScores);

    // 总损失
    const totalLoss = auxLoss * this.config.auxLossWeight + 
                     zLoss * this.config.zLossWeight;

    // 计算梯度
    const expertGradients = this._computeExpertGradients(
      input,
      forwardResult,
      gradients
    );

    const gateGradients = this._computeGateGradients(
      input,
      forwardResult,
      gradients
    );

    // 更新权重
    this.optimizer.step(expertGradients, gateGradients);

    return {
      loss: totalLoss,
      auxLoss,
      zLoss
    };
  }

  _computeAuxiliaryLoss(gateScores) {
    /**
     * 计算辅助损失
     * 
     * 参数:
     *   gateScores: 门控分数
     * 
     * 返回:
     *   辅助损失
     */
    const mean = gateScores.reduce((sum, s) => sum + s, 0) / gateScores.length;
    const variance = gateScores.reduce((sum, s) => sum + Math.pow(s - mean, 2), 0) / gateScores.length;
    return variance;
  }

  _computeZLoss(gateScores) {
    /**
     * 计算 Z 损失
     * 
     * 参数:
     *   gateScores: 门控分数
     * 
     * 返回:
     *   Z 损失
     */
    const sumSquares = gateScores.reduce((sum, s) => sum + Math.pow(s, 2), 0);
    return Math.pow(Math.sqrt(sumSquares) - 1, 2);
  }

  _computeExpertGradients(input, forwardResult, gradients) {
    /**
     * 计算专家梯度
     * 
     * 参数:
     *   input: 输入向量
     *   forwardResult: 前向传播结果
     *   gradients: 输出梯度
     * 
     * 返回:
     *   专家梯度
     */
    const expertGradients = [];

    for (const expertOutput of forwardResult.expertOutputs) {
      const weight = forwardResult.weights[forwardResult.expertOutputs.indexOf(expertOutput)];
      const gradient = gradients.map(g => g * weight);

      expertGradients.push({
        expertIndex: expertOutput.index,
        gradient
      });
    }

    return expertGradients;
  }

  _computeGateGradients(input, forwardResult, gradients) {
    /**
     * 计算门控梯度
     * 
     * 参数:
     *   input: 输入向量
     *   forwardResult: 前向传播结果
     *   gradients: 输出梯度
     * 
     * 返回:
     *   门控梯度
     */
    // 简化实现
    return this.gateWeights.map(weights => 
      weights.map(() => (Math.random() - 0.5) * 0.01)
    );
  }

  train(inputs, targets, epochs = 10) {
    /**
     * 训练模型
     * 
     * 参数:
     *   inputs: 输入数组
     *   targets: 目标数组
     *   epochs: 训练轮数
     * 
     * 返回:
     *   训练历史
     */
    const history = [];

    for (let epoch = 0; epoch < epochs; epoch++) {
      let totalLoss = 0;
      let totalAuxLoss = 0;
      let totalZLoss = 0;

      for (let i = 0; i < inputs.length; i++) {
        // 前向传播
        const forwardResult = this.forward(inputs[i]);

        // 计算损失
        const loss = this._computeLoss(forwardResult.output, targets[i]);

        // 反向传播
        const backwardResult = this.backward(
          inputs[i],
          forwardResult,
          this._computeGradients(forwardResult.output, targets[i])
        );

        totalLoss += loss;
        totalAuxLoss += backwardResult.auxLoss;
        totalZLoss += backwardResult.zLoss;
      }

      history.push({
        epoch: epoch + 1,
        loss: totalLoss / inputs.length,
        auxLoss: totalAuxLoss / inputs.length,
        zLoss: totalZLoss / inputs.length
      });

      console.log(`Epoch ${epoch + 1}: Loss = ${totalLoss / inputs.length}`);
    }

    return history;
  }

  _computeLoss(output, target) {
    /**
     * 计算损失
     * 
     * 参数:
     *   output: 输出向量
     *   target: 目标向量
     * 
     * 返回:
     *   损失值
     */
    let loss = 0;
    for (let i = 0; i < output.length && i < target.length; i++) {
      loss += Math.pow(output[i] - target[i], 2);
    }
    return loss / output.length;
  }

  _computeGradients(output, target) {
    /**
     * 计算梯度
     * 
     * 参数:
     *   output: 输出向量
     *   target: 目标向量
     * 
     * 返回:
     *   梯度向量
     */
    return output.map((o, i) => 2 * (o - target[i]) / output.length);
  }
}
```

## 🌊 状态空间模型

状态空间模型（SSM）是一种新型的序列建模架构，通过状态空间理论处理序列数据，具有线性复杂度和高效的长上下文处理能力。

### Mamba

Mamba 是一种基于选择性状态空间模型的高效序列建模架构。

```javascript
// architectures/ssm/mamba.js
class MambaBlock {
  constructor(config) {
    /**
     * Mamba 块
     * 
     * 参数:
     *   config: 配置对象
     */
    this.config = {
      dModel: config.dModel || 512,
      dState: config.dState || 16,
      dConv: config.dConv || 4,
      expand: config.expand || 2,
      ...config
    };

    this.dInner = this.config.dModel * this.config.expand;
    this.dtProj = this._initializeProjection(this.dInner);
    this.ALog = this._initializeALog();
    this.D = this._initializeD();
    this.xProj = this._initializeProjection(this.dInner);
    this.dtProj = this._initializeProjection(this.dInner);
    this.outProj = this._initializeProjection(this.config.dModel);
  }

  _initializeProjection(d) {
    /**
     * 初始化投影权重
     * 
     * 参数:
     *   d: 维度
     * 
     * 返回:
     *   权重矩阵
     */
    return Array.from({ length: d * d }, () => (Math.random() - 0.5) * 0.1);
  }

  _initializeALog() {
    /**
     * 初始化 A log 参数
     * 
     * 返回:
     *   A log 参数
     */
    return Array.from({ length: this.config.dState }, (_, i) => 
      Math.log(Math.exp(-5 + 10 * i / (this.config.dState - 1)))
    );
  }

  _initializeD() {
    /**
     * 初始化 D 参数
     * 
     * 返回:
     *   D 参数
     */
    return Array.from({ length: this.config.dModel }, () => 
      (Math.random() - 0.5) * 0.1
    );
  }

  forward(x) {
    /**
     * 前向传播
     * 
     * 参数:
     *   x: 输入序列 [batch, seqLen, dModel]
     * 
     * 返回:
     *   输出序列
     */
    const [batchSize, seqLen] = [x.length, x[0].length];

    // 投影到内部维度
    const xProj = this._project(x, this.xProj, this.dInner);

    // 计算时间步长
    const dt = this._computeDt(xProj);

    // 计算状态空间参数
    const A = this._computeA();
    const B = this._computeB(xProj);
    const C = this._computeC(xProj);

    // 选择性扫描
    const y = this._selectiveScan(xProj, dt, A, B, C);

    // 输出投影
    const output = this._project(y, this.outProj, this.config.dModel);

    return output;
  }

  _project(x, weights, dOut) {
    /**
     * 投影
     * 
     * 参数:
     *   x: 输入
     *   weights: 权重
     *   dOut: 输出维度
     * 
     * 返回:
     *   投影结果
     */
    const batchSize = x.length;
    const seqLen = x[0].length;
    const dIn = x[0][0].length;

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      const batchOutput = [];

      for (let t = 0; t < seqLen; t++) {
        const timestepOutput = [];

        for (let o = 0; o < dOut; o++) {
          let sum = 0;

          for (let i = 0; i < dIn; i++) {
            sum += x[b][t][i] * weights[o * dIn + i];
          }

          timestepOutput.push(sum);
        }

        batchOutput.push(timestepOutput);
      }

      output.push(batchOutput);
    }

    return output;
  }

  _computeDt(xProj) {
    /**
     * 计算时间步长
     * 
     * 参数:
     *   xProj: 投影后的输入
     * 
     * 返回:
     *   时间步长
     */
    const batchSize = xProj.length;
    const seqLen = xProj[0].length;
    const dt = [];

    for (let b = 0; b < batchSize; b++) {
      const batchDt = [];

      for (let t = 0; t < seqLen; t++) {
        // 简化实现：使用 sigmoid 激活
        const dtValue = this._sigmoid(xProj[b][t][0]);
        batchDt.push(dtValue);
      }

      dt.push(batchDt);
    }

    return dt;
  }

  _sigmoid(x) {
    /**
     * Sigmoid 激活函数
     * 
     * 参数:
     *   x: 输入值
     * 
     * 返回:
     *   激活值
     */
    return 1 / (1 + Math.exp(-x));
  }

  _computeA() {
    /**
     * 计算 A 矩阵
     * 
     * 返回:
     *   A 矩阵
     */
    const A = [];

    for (let i = 0; i < this.config.dState; i++) {
      const row = [];
      for (let j = 0; j < this.config.dState; j++) {
        if (i === j) {
          row.push(Math.exp(this.ALog[i]));
        } else {
          row.push(0);
        }
      }
      A.push(row);
    }

    return A;
  }

  _computeB(xProj) {
    /**
     * 计算 B 矩阵
     * 
     * 参数:
     *   xProj: 投影后的输入
     * 
     * 返回:
     *   B 矩阵
     */
    const batchSize = xProj.length;
    const seqLen = xProj[0].length;
    const B = [];

    for (let b = 0; b < batchSize; b++) {
      const batchB = [];

      for (let t = 0; t < seqLen; t++) {
        const timestepB = [];

        for (let s = 0; s < this.config.dState; s++) {
          // 简化实现：使用 tanh 激活
          const bValue = Math.tanh(xProj[b][t][s + 1]);
          timestepB.push(bValue);
        }

        batchB.push(timestepB);
      }

      B.push(batchB);
    }

    return B;
  }

  _computeC(xProj) {
    /**
     * 计算 C 矩阵
     * 
     * 参数:
     *   xProj: 投影后的输入
     * 
     * 返回:
     *   C 矩阵
     */
    const batchSize = xProj.length;
    const seqLen = xProj[0].length;
    const C = [];

    for (let b = 0; b < batchSize; b++) {
      const batchC = [];

      for (let t = 0; t < seqLen; t++) {
        const timestepC = [];

        for (let s = 0; s < this.config.dState; s++) {
          // 简化实现：使用 sigmoid 激活
          const cValue = this._sigmoid(xProj[b][t][s + 1 + this.config.dState]);
          timestepC.push(cValue);
        }

        batchC.push(timestepC);
      }

      C.push(batchC);
    }

    return C;
  }

  _selectiveScan(xProj, dt, A, B, C) {
    /**
     * 选择性扫描
     * 
     * 参数:
     *   xProj: 投影后的输入
     *   dt: 时间步长
     *   A: A 矩阵
     *   B: B 矩阵
     *   C: C 矩阵
     * 
     * 返回:
     *   扫描结果
     */
    const batchSize = xProj.length;
    const seqLen = xProj[0].length;
    const dInner = this.dInner;

    const y = [];

    for (let b = 0; b < batchSize; b++) {
      const batchY = [];
      let h = Array.from({ length: this.config.dState }, () => 0);

      for (let t = 0; t < seqLen; t++) {
        // 更新状态
        const dtValue = dt[b][t];
        const Bt = B[b][t];
        const Ct = C[b][t];

        // h_t = (1 - dt * A) * h_{t-1} + dt * B * x_t
        const newH = [];

        for (let s = 0; s < this.config.dState; s++) {
          const a = A[s][s];
          const hPrev = h[s];
          const b = Bt[s];
          const x = xProj[b][t][s + 1 + 2 * this.config.dState];

          const hNew = (1 - dtValue * a) * hPrev + dtValue * b * x;
          newH.push(hNew);
        }

        h = newH;

        // 计算 y_t = C * h_t + D * x_t
        const yt = [];

        for (let i = 0; i < dInner; i++) {
          let sum = 0;

          for (let s = 0; s < this.config.dState; s++) {
            sum += Ct[s] * h[s];
          }

          // 添加 D * x 项
          sum += this.D[i] * xProj[b][t][i];

          yt.push(sum);
        }

        batchY.push(yt);
      }

      y.push(batchY);
    }

    return y;
  }
}

// 使用示例
function mambaExample() {
  const mamba = new MambaBlock({
    dModel: 512,
    dState: 16,
    dConv: 4,
    expand: 2
  });

  // 模拟输入 [batch=2, seqLen=10, dModel=512]
  const x = Array.from({ length: 2 }, () => 
    Array.from({ length: 10 }, () => 
      Array.from({ length: 512 }, () => Math.random())
    )
  );

  // 前向传播
  const output = mamba.forward(x);
  console.log('输出形状:', output.length, output[0].length, output[0][0].length);
  console.log('输出示例:', output[0][0].slice(0, 5));
}
```

### RWKV

RWKV 是一种结合了 RNN 和 Transformer 优点的架构，具有线性复杂度和高效的长上下文处理能力。

```javascript
// architectures/ssm/rwkv.js
class RWKVBlock {
  constructor(config) {
    /**
     * RWKV 块
     * 
     * 参数:
     *   config: 配置对象
     */
    this.config = {
      dModel: config.dModel || 512,
      nHead: config.nHead || 8,
      dHead: config.dHead || 64,
      ...config
    };

    this.receptance = this._initializeLinear(this.config.dModel, this.config.dModel);
    this.key = this._initializeLinear(this.config.dModel, this.config.dModel);
    this.value = this._initializeLinear(this.config.dModel, this.config.dModel);
    this.gate = this._initializeLinear(this.config.dModel, this.config.dModel);
    this.output = this._initializeLinear(this.config.dModel, this.config.dModel);

    this.state = null;
  }

  _initializeLinear(dIn, dOut) {
    /**
     * 初始化线性层权重
     * 
     * 参数:
     *   dIn: 输入维度
     *   dOut: 输出维度
     * 
     * 返回:
     *   权重矩阵
     */
    return {
      weight: Array.from({ length: dOut * dIn }, () => (Math.random() - 0.5) * 0.1),
      bias: Array.from({ length: dOut }, () => 0)
    };
  }

  forward(x, state = null) {
    /**
     * 前向传播
     * 
     * 参数:
     *   x: 输入序列 [batch, seqLen, dModel]
     *   state: 初始状态
     * 
     * 返回:
     *   输出序列和新状态
     */
    const batchSize = x.length;
    const seqLen = x[0].length;

    // 初始化状态
    if (state === null) {
      state = this._initializeState(batchSize);
    }

    const outputs = [];

    for (let t = 0; t < seqLen; t++) {
      // 获取当前时间步的输入
      const xt = x.map(b => b[t]);

      // 计算 R, K, V, G
      const r = this._linear(xt, this.receptance);
      const k = this._linear(xt, this.key);
      const v = this._linear(xt, this.value);
      const g = this._linear(xt, this.gate);

      // 应用激活函数
      const rAct = this._sigmoid(r);
      const kAct = this._sigmoid(k);
      const gAct = this._sigmoid(g);

      // 更新状态
      const newState = this._updateState(kAct, v, state);

      // 计算 WKV (Weighted Key-Value)
      const wkv = this._computeWKV(rAct, newState);

      // 应用门控
      const gated = wkv.map((w, i) => w * gAct[i]);

      // 输出投影
      const output = this._linear(gated, this.output);

      outputs.push(output);
      state = newState;
    }

    // 转置输出 [seqLen, batch, dModel] -> [batch, seqLen, dModel]
    const transposed = Array.from({ length: batchSize }, () => 
      Array.from({ length: seqLen }, () => Array.from({ length: this.config.dModel }, () => 0))
    );

    for (let t = 0; t < seqLen; t++) {
      for (let b = 0; b < batchSize; b++) {
        for (let d = 0; d < this.config.dModel; d++) {
          transposed[b][t][d] = outputs[t][b][d];
        }
      }
    }

    return {
      output: transposed,
      state
    };
  }

  _initializeState(batchSize) {
    /**
     * 初始化状态
     * 
     * 参数:
     *   batchSize: 批次大小
     * 
     * 返回:
     *   初始状态
     */
    return {
      numerator: Array.from({ length: batchSize }, () => 
        Array.from({ length: this.config.dModel }, () => 0)
      ),
      denominator: Array.from({ length: batchSize }, () => 
        Array.from({ length: this.config.dModel }, () => 0)
      )
    };
  }

  _linear(x, linearLayer) {
    /**
     * 线性变换
     * 
     * 参数:
     *   x: 输入
     *   linearLayer: 线性层
     * 
     * 返回:
     *   输出
     */
    const batchSize = x.length;
    const dIn = x[0].length;
    const dOut = linearLayer.bias.length;

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      const batchOutput = [];

      for (let o = 0; o < dOut; o++) {
        let sum = linearLayer.bias[o];

        for (let i = 0; i < dIn; i++) {
          sum += x[b][i] * linearLayer.weight[o * dIn + i];
        }

        batchOutput.push(sum);
      }

      output.push(batchOutput);
    }

    return output;
  }

  _sigmoid(x) {
    /**
     * Sigmoid 激活函数
     * 
     * 参数:
     *   x: 输入值
     * 
     * 返回:
     *   激活值
     */
    return 1 / (1 + Math.exp(-x));
  }

  _updateState(k, v, state) {
    /**
     * 更新状态
     * 
     * 参数:
     *   k: Key
     *   v: Value
     *   state: 当前状态
     * 
     * 返回:
     *   新状态
     */
    const batchSize = k.length;
    const dModel = k[0].length;

    const newNumerator = Array.from({ length: batchSize }, () => 
      Array.from({ length: dModel }, () => 0)
    );

    const newDenominator = Array.from({ length: batchSize }, () => 
      Array.from({ length: dModel }, () => 0)
    );

    for (let b = 0; b < batchSize; b++) {
      for (let d = 0; d < dModel; d++) {
        const decay = Math.exp(-Math.exp(k[b][d]));
        const numerator = state.numerator[b][d] * decay + v[b][d];
        const denominator = state.denominator[b][d] * decay + 1;

        newNumerator[b][d] = numerator;
        newDenominator[b][d] = denominator;
      }
    }

    return {
      numerator: newNumerator,
      denominator: newDenominator
    };
  }

  _computeWKV(r, state) {
    /**
     * 计算 WKV
     * 
     * 参数:
     *   r: Receptance
     *   state: 状态
     * 
     * 返回:
     *   WKV
     */
    const batchSize = r.length;
    const dModel = r[0].length;

    const wkv = [];

    for (let b = 0; b < batchSize; b++) {
      const batchWkv = [];

      for (let d = 0; d < dModel; d++) {
        const numerator = state.numerator[b][d];
        const denominator = Math.max(state.denominator[b][d], 1e-8);
        const value = (numerator / denominator) * r[b][d];
        batchWkv.push(value);
      }

      wkv.push(batchWkv);
    }

    return wkv;
  }
}

function rwkvExample() {
  const rwkv = new RWKVBlock({

    nHead: 8,
    dHead: 64
  });

  // 模拟输入 [batch=2, seqLen=10, dModel=512]

    Array.from({ length: 10 }, () => 
      Array.from({ length: 512 }, () => Math.random())
    )
  );

  // 前向传播

  console.log('状态:', result.state);
}
```

### 与 Transformer 对比

状态空间模型与 Transformer 的对比分析。

```javascript
// architectures/ssm/comparison.js
class ArchitectureComparison {
  constructor() {
    /**
     * 架构对比器
     */
  }

  compareComplexity(seqLen, dModel) {
    /**
     * 比较计算复杂度
     * 
     * 参数:
     *   seqLen: 序列长度
     *   dModel: 模型维度
     * 
     * 返回:
     *   复杂度对比
     */
    return {
      transformer: {
        attention: seqLen * seqLen * dModel, // O(n²)
        ffn: seqLen * dModel * dModel * 4, // O(n)
        total: seqLen * seqLen * dModel + seqLen * dModel * dModel * 4
      },
      ssm: {
        stateUpdate: seqLen * dModel * dModel, // O(n)
        output: seqLen * dModel * dModel, // O(n)
        total: seqLen * dModel * dModel * 2 // O(n)
      }
    };
  }

  compareMemory(seqLen, dModel) {
    /**
     * 比较内存占用
     * 
     * 参数:
     *   seqLen: 序列长度
     *   dModel: 模型维度
     * 
     * 返回:
     *   内存对比
     */
    return {
      transformer: {
        attentionMatrix: seqLen * seqLen * 4, // 4 bytes per float
        kvCache: seqLen * dModel * 2 * 4,
        total: seqLen * seqLen * 4 + seqLen * dModel * 2 * 4
      },
      ssm: {
        state: dModel * dModel * 4,
        total: dModel * dModel * 4
      }
    };
  }

  comparePerformance(seqLens, dModel) {
    /**
     * 比较性能
     * 
     * 参数:
     *   seqLens: 序列长度数组
     *   dModel: 模型维度
     * 
     * 返回:
     *   性能对比
     */
    const results = [];

    for (const seqLen of seqLens) {
      const complexity = this.compareComplexity(seqLen, dModel);
      const memory = this.compareMemory(seqLen, dModel);

      results.push({
        seqLen,
        transformer: {
          flops: complexity.transformer.total,
          memory: memory.transformer.total
        },
        ssm: {
          flops: complexity.ssm.total,
          memory: memory.ssm.total
        },
        speedup: complexity.transformer.total / complexity.ssm.total,
        memoryReduction: memory.transformer.total / memory.ssm.total
      });
    }

    return results;
  }

  generateReport(comparisonResults) {
    /**
     * 生成对比报告
     * 
     * 参数:
     *   comparisonResults: 对比结果
     * 
     * 返回:
     *   报告对象
     */
    const summary = {
      averageSpeedup: 0,
      averageMemoryReduction: 0,
      bestForLongContext: null,
      bestForShortContext: null
    };

    const speedups = comparisonResults.map(r => r.speedup);
    const memoryReductions = comparisonResults.map(r => r.memoryReduction);

    summary.averageSpeedup = speedups.reduce((sum, s) => sum + s, 0) / speedups.length;
    summary.averageMemoryReduction = memoryReductions.reduce((sum, m) => sum + m, 0) / memoryReductions.length;

    // 找出最适合长上下文的架构
    const longContextResults = comparisonResults.filter(r => r.seqLen > 1000);
    if (longContextResults.length > 0) {
      const bestLong = longContextResults.reduce((best, current) => 
        current.speedup > best.speedup ? current : best
      );
      summary.bestForLongContext = {
        seqLen: bestLong.seqLen,
        architecture: 'SSM',
        speedup: bestLong.speedup,
        memoryReduction: bestLong.memoryReduction
      };
    }

    return {
      summary,
      details: comparisonResults
    };
  }
}

// 使用示例
function architectureComparisonExample() {
  const comparison = new ArchitectureComparison();

  // 比较复杂度

  console.log('复杂度对比:', complexity);

  // 比较内存
  const memory = comparison.compareMemory(1000, 512);
  console.log('内存对比:', memory);

  // 比较性能
  const seqLens = [100, 500, 1000, 5000, 10000];
  const performance = comparison.comparePerformance(seqLens, 512);
  console.log('性能对比:', performance);

  // 生成报告
  const report = comparison.generateReport(performance);
  console.log('对比报告:', report);
}
```

## ⚡ 线性 Attention

线性 Attention 机制通过近似计算 Attention，将复杂度从 O(n²) 降低到 O(n)，显著提高长序列处理效率。

### FlashAttention

FlashAttention 是一种高效的 Attention 实现算法，通过分块计算和内存优化大幅提升性能。

```javascript
// architectures/attention/flash-attention.js
class FlashAttention {
  constructor(config) {
    /**
     * Flash Attention
     * 
     * 参数:
     *   config: 配置对象
     */
    this.config = {
      blockSize: config.blockSize || 64,
      causal: config.causal || false,
      ...config
    };
  }

  forward(Q, K, V) {
    /**
     * 前向传播
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    const [batchSize, nHeads, seqLen, dHead] = this._getShape(Q);

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      for (let h = 0; h < nHeads; h++) {
        const headOutput = this._computeAttentionHead(
          Q[b][h],
          K[b][h],
          V[b][h]
        );
        output.push(headOutput);
      }
    }

    return this._reshapeOutput(output, batchSize, nHeads, seqLen, dHead);
  }

  _getShape(tensor) {
    /**
     * 获取张量形状
     * 
     * 参数:
     *   tensor: 张量
     * 
     * 返回:
     *   形状数组
     */
    return [tensor.length, tensor[0].length, tensor[0][0].length, tensor[0][0][0].length];
  }

  _computeAttentionHead(Qh, Kh, Vh) {
    /**
     * 计算 Attention 头
     * 
     * 参数:
     *   Qh: Query 头 [seqLen, dHead]
     *   Kh: Key 头 [seqLen, dHead]
     *   Vh: Value 头 [seqLen, dHead]
     * 
     * 返回:
     *   输出 [seqLen, dHead]
     */
    const seqLen = Qh.length;
    const dHead = Qh[0].length;
    const blockSize = this.config.blockSize;

    const output = Array.from({ length: seqLen }, () => Array.from({ length: dHead }, () => 0));

    // 分块计算
    for (let i = 0; i < seqLen; i += blockSize) {
      const iEnd = Math.min(i + blockSize, seqLen);

      for (let j = 0; j < seqLen; j += blockSize) {
        const jEnd = Math.min(j + blockSize, seqLen);

        // Causal mask
        if (this.config.causal && jEnd <= i) {
          continue;
        }

        // 计算当前块的 Attention
        const blockOutput = this._computeAttentionBlock(
          Qh.slice(i, iEnd),
          Kh.slice(j, jEnd),
          Vh.slice(j, jEnd),
          i,
          j
        );

        // 累加到输出
        for (let bi = 0; bi < blockOutput.length; bi++) {
          for (let d = 0; d < dHead; d++) {
            output[i + bi][d] += blockOutput[bi][d];
          }
        }
      }
    }

    return output;
  }

  _computeAttentionBlock(QBlock, KBlock, VBlock, iStart, jStart) {
    /**
     * 计算 Attention 块
     * 
     * 参数:
     *   QBlock: Query 块
     *   KBlock: Key 块
     *   VBlock: Value 块
     *   iStart: Query 起始位置
     *   jStart: Key 起始位置
     * 
     * 返回:
     *   块输出
     */
    const blockSize = QBlock.length;
    const dHead = QBlock[0].length;

    // 计算 QK^T
    const scores = Array.from({ length: blockSize }, () => Array.from({ length: KBlock.length }, () => 0));

    for (let i = 0; i < blockSize; i++) {
      for (let j = 0; j < KBlock.length; j++) {
        let score = 0;

        for (let d = 0; d < dHead; d++) {
          score += QBlock[i][d] * KBlock[j][d];
        }

        scores[i][j] = score / Math.sqrt(dHead);

        // Causal mask
        if (this.config.causal && (iStart + i) < (jStart + j)) {
          scores[i][j] = -Infinity;
        }
      }
    }

    // Softmax
    const expScores = scores.map(row => {
      const maxScore = Math.max(...row);
      const expRow = row.map(s => Math.exp(s - maxScore));
      const sumExp = expRow.reduce((sum, e) => sum + e, 0);
      return expRow.map(e => e / sumExp);
    });

    // 计算 Attention * V
    const blockOutput = Array.from({ length: blockSize }, () => Array.from({ length: dHead }, () => 0));

    for (let i = 0; i < blockSize; i++) {
      for (let d = 0; d < dHead; d++) {
        let sum = 0;

        for (let j = 0; j < KBlock.length; j++) {
          sum += expScores[i][j] * VBlock[j][d];
        }

        blockOutput[i][d] = sum;
      }
    }

    return blockOutput;
  }

  _reshapeOutput(headOutputs, batchSize, nHeads, seqLen, dHead) {
    /**
     * 重塑输出
     * 
     * 参数:
     *   headOutputs: 头输出数组
     *   batchSize: 批次大小
     *   nHeads: 头数
     *   seqLen: 序列长度
     *   dHead: 头维度
     * 
     * 返回:
     *   重塑后的输出
     */
    const output = [];

    for (let b = 0; b < batchSize; b++) {
      const batchOutput = [];

      for (let h = 0; h < nHeads; h++) {
        const headIndex = b * nHeads + h;
        batchOutput.push(headOutputs[headIndex]);
      }

      output.push(batchOutput);
    }

    return output;
  }
}
```

### Efficient Attention

Efficient Attention 使用各种近似方法来降低 Attention 的计算复杂度。

```javascript
// architectures/attention/efficient-attention.js
class EfficientAttention {
  constructor(config) {
    /**
     * Efficient Attention
     * 
     * 参数:
     *   config: 配置对象
     */
    this.config = {
      method: config.method || 'linear', // 'linear', 'low-rank', 'sparse'
      rank: config.rank || 32,
      sparsity: config.sparsity || 0.1,
      ...config
    };
  }

  forward(Q, K, V) {
    /**
     * 前向传播
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    switch (this.config.method) {
      case 'linear':
        return this._linearAttention(Q, K, V);
      case 'low-rank':
        return this._lowRankAttention(Q, K, V);
      case 'sparse':
        return this._sparseAttention(Q, K, V);
      default:
        throw new Error(`Unknown method: ${this.config.method}`);
    }
  }

  _linearAttention(Q, K, V) {
    /**
     * 线性 Attention
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    const [batchSize, nHeads, seqLen, dHead] = this._getShape(Q);

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      for (let h = 0; h < nHeads; h++) {
        const Qh = Q[b][h];
        const Kh = K[b][h];
        const Vh = V[b][h];

        // 线性 Attention: Q * (K^T * V)
        const KVt = this._matrixMultiplyTranspose(Kh, Vh);
        const headOutput = this._matrixMultiply(Qh, KVt);

        output.push(headOutput);
      }
    }

    return this._reshapeOutput(output, batchSize, nHeads, seqLen, dHead);
  }

  _lowRankAttention(Q, K, V) {
    /**
     * 低秩 Attention
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    const [batchSize, nHeads, seqLen, dHead] = this._getShape(Q);
    const rank = this.config.rank;

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      for (let h = 0; h < nHeads; h++) {
        const Qh = Q[b][h];
        const Kh = K[b][h];
        const Vh = V[b][h];

        // 低秩分解
        const KLow = this._lowRankDecompose(Kh, rank);
        const VLow = this._lowRankDecompose(Vh, rank);

        // 计算低秩 Attention
        const scores = this._matrixMultiply(Qh, this._transpose(KLow));
        const headOutput = this._matrixMultiply(scores, VLow);

        output.push(headOutput);
      }
    }

    return this._reshapeOutput(output, batchSize, nHeads, seqLen, dHead);
  }

  _sparseAttention(Q, K, V) {
    /**
     * 稀疏 Attention
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    const [batchSize, nHeads, seqLen, dHead] = this._getShape(Q);
    const sparsity = this.config.sparsity;

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      for (let h = 0; h < nHeads; h++) {
        const Qh = Q[b][h];
        const Kh = K[b][h];
        const Vh = V[b][h];

        // 计算稀疏 Attention
        const sparseIndices = this._getSparseIndices(seqLen, sparsity);
        const headOutput = this._computeSparseAttention(Qh, Kh, Vh, sparseIndices);

        output.push(headOutput);
      }
    }

    return this._reshapeOutput(output, batchSize, nHeads, seqLen, dHead);
  }

  _getSparseIndices(seqLen, sparsity) {
    /**
     * 获取稀疏索引
     * 
     * 参数:
     *   seqLen: 序列长度
     *   sparsity: 稀疏度
     * 
     * 返回:
     *   稀疏索引数组
     */
    const numIndices = Math.floor(seqLen * sparsity);
    const indices = [];

    for (let i = 0; i < seqLen; i++) {
      // 局部窗口
      const windowSize = Math.floor(seqLen * sparsity / 2);
      const start = Math.max(0, i - windowSize);
      const end = Math.min(seqLen, i + windowSize + 1);

      for (let j = start; j < end; j++) {
        indices.push([i, j]);
      }
    }

    return indices;
  }

  _computeSparseAttention(Qh, Kh, Vh, sparseIndices) {
    /**
     * 计算稀疏 Attention
     * 
     * 参数:
     *   Qh: Query 头
     *   Kh: Key 头
     *   Vh: Value 头
     *   sparseIndices: 稀疏索引
     * 
     * 返回:
     *   输出
     */
    const seqLen = Qh.length;
    const dHead = Qh[0].length;

    const output = Array.from({ length: seqLen }, () => Array.from({ length: dHead }, () => 0));

    for (const [i, j] of sparseIndices) {
      let score = 0;

      for (let d = 0; d < dHead; d++) {
        score += Qh[i][d] * Kh[j][d];
      }

      score /= Math.sqrt(dHead);

      for (let d = 0; d < dHead; d++) {
        output[i][d] += score * Vh[j][d];
      }
    }

    return output;
  }

  _getShape(tensor) {
    /**
     * 获取张量形状
     * 
     * 参数:
     *   tensor: 张量
     * 
     * 返回:
     *   形状数组
     */
    return [tensor.length, tensor[0].length, tensor[0][0].length, tensor[0][0][0].length];
  }

  _matrixMultiply(A, B) {
    /**
     * 矩阵乘法
     * 
     * 参数:
     *   A: 矩阵 A
     *   B: 矩阵 B
     * 
     * 返回:
     *   结果矩阵
     */
    const rowsA = A.length;
    const colsA = A[0].length;
    const colsB = B[0].length;

    const result = Array.from({ length: rowsA }, () => Array.from({ length: colsB }, () => 0));

    for (let i = 0; i < rowsA; i++) {
      for (let j = 0; j < colsB; j++) {
        let sum = 0;

        for (let k = 0; k < colsA; k++) {
          sum += A[i][k] * B[k][j];
        }

        result[i][j] = sum;
      }
    }

    return result;
  }

  _matrixMultiplyTranspose(A, B) {
    /**
     * 矩阵乘以转置矩阵
     * 
     * 参数:
     *   A: 矩阵 A
     *   B: 矩阵 B
     * 
     * 返回:
     *   结果矩阵
     */
    const rowsA = A.length;
    const colsA = A[0].length;
    const rowsB = B.length;

    const result = Array.from({ length: rowsA }, () => Array.from({ length: rowsB }, () => 0));

    for (let i = 0; i < rowsA; i++) {
      for (let j = 0; j < rowsB; j++) {
        let sum = 0;

        for (let k = 0; k < colsA; k++) {
          sum += A[i][k] * B[j][k];
        }

        result[i][j] = sum;
      }
    }

    return result;
  }

  _transpose(A) {
    /**
     * 转置矩阵
     * 
     * 参数:
     *   A: 矩阵 A
     * 
     * 返回:
     *   转置矩阵
     */
    const rows = A.length;
    const cols = A[0].length;

    const result = Array.from({ length: cols }, () => Array.from({ length: rows }, () => 0));

    for (let i = 0; i < rows; i++) {
      for (let j = 0; j < cols; j++) {
        result[j][i] = A[i][j];
      }
    }

    return result;
  }

  _lowRankDecompose(matrix, rank) {
    /**
     * 低秩分解
     * 
     * 参数:
     *   matrix: 矩阵
     *   rank: 秩
     * 
     * 返回:
     *   低秩矩阵
     */
    // 简化实现：截断
    const rows = matrix.length;
    const cols = matrix[0].length;

    const result = Array.from({ length: rows }, () => Array.from({ length: rank }, () => 0));

    for (let i = 0; i < rows; i++) {
      for (let j = 0; j < rank; j++) {
        result[i][j] = matrix[i][j];
      }
    }

    return result;
  }

  _reshapeOutput(headOutputs, batchSize, nHeads, seqLen, dHead) {
    /**
     * 重塑输出
     * 
     * 参数:
     *   headOutputs: 头输出数组
     *   batchSize: 批次大小
     *   nHeads: 头数
     *   seqLen: 序列长度
     *   dHead: 头维度
     * 
     * 返回:
     *   重塑后的输出
     */
    const output = [];

    for (let b = 0; b < batchSize; b++) {
      const batchOutput = [];

      for (let h = 0; h < nHeads; h++) {
        const headIndex = b * nHeads + h;
        batchOutput.push(headOutputs[headIndex]);
      }

      output.push(batchOutput);
    }

    return output;
  }
}
```

### 长上下文处理

长上下文处理技术使模型能够处理更长的输入序列。

```javascript
// architectures/attention/long-context.js
class LongContextProcessor {
  constructor(config) {
    /**
     * 长上下文处理器
     * 
     * 参数:
     *   config: 配置对象
     */
    this.config = {
      maxContextLength: config.maxContextLength || 100000,
      chunkSize: config.chunkSize || 4096,
      overlap: config.overlap || 256,
      method: config.method || 'sliding-window', // 'sliding-window', 'hierarchical', 'retrieval'
      ...config
    };

    this.kvCache = null;
  }

  processLongContext(Q, K, V) {
    /**
     * 处理长上下文
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    switch (this.config.method) {
      case 'sliding-window':
        return this._slidingWindowAttention(Q, K, V);
      case 'hierarchical':
        return this._hierarchicalAttention(Q, K, V);
      case 'retrieval':
        return this._retrievalBasedAttention(Q, K, V);
      default:
        throw new Error(`Unknown method: ${this.config.method}`);
    }
  }

  _slidingWindowAttention(Q, K, V) {
    /**
     * 滑动窗口 Attention
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    const [batchSize, nHeads, seqLen, dHead] = this._getShape(Q);
    const windowSize = this.config.chunkSize;

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      for (let h = 0; h < nHeads; h++) {
        const Qh = Q[b][h];
        const Kh = K[b][h];
        const Vh = V[b][h];

        const headOutput = [];

        for (let i = 0; i < seqLen; i++) {
          // 滑动窗口
          const start = Math.max(0, i - windowSize);
          const end = Math.min(seqLen, i + windowSize + 1);

          const QWindow = Qh.slice(i, i + 1);
          const KWindow = Kh.slice(start, end);
          const VWindow = Vh.slice(start, end);

          // 计算 Attention
          const attentionOutput = this._computeAttention(QWindow, KWindow, VWindow);
          headOutput.push(attentionOutput[0]);
        }

        output.push(headOutput);
      }
    }

    return this._reshapeOutput(output, batchSize, nHeads, seqLen, dHead);
  }

  _hierarchicalAttention(Q, K, V) {
    /**
     * 分层 Attention
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    const [batchSize, nHeads, seqLen, dHead] = this._getShape(Q);
    const chunkSize = this.config.chunkSize;

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      for (let h = 0; h < nHeads; h++) {
        const Qh = Q[b][h];
        const Kh = K[b][h];
        const Vh = V[b][h];

        // 分块处理
        const chunks = [];
        for (let i = 0; i < seqLen; i += chunkSize) {
          const end = Math.min(i + chunkSize, seqLen);
          chunks.push({
            start: i,
            end: end,
            Q: Qh.slice(i, end),
            K: Kh.slice(i, end),
            V: Vh.slice(i, end)
          });
        }

        // 计算分层 Attention
        const headOutput = [];

        for (let i = 0; i < seqLen; i++) {
          // 找到当前块
          const currentChunk = chunks.find(c => i >= c.start && i < c.end);

          // 局部 Attention
          const localOutput = this._computeAttention(
            Qh.slice(i, i + 1),
            currentChunk.K,
            currentChunk.V
          );

          // 全局 Attention（使用块摘要）
          const globalOutput = this._computeGlobalAttention(
            Qh.slice(i, i + 1),
            chunks
          );

          // 组合输出
          const combined = localOutput[0].map((val, idx) => 
            val + globalOutput[0][idx]
          );

          headOutput.push(combined);
        }

        output.push(headOutput);
      }
    }

    return this._reshapeOutput(output, batchSize, nHeads, seqLen, dHead);
  }

  _computeGlobalAttention(Q, chunks) {
    /**
     * 计算全局 Attention
     * 
     * 参数:
     *   Q: Query
     *   chunks: 块数组
     * 
     * 返回:
     *   Attention 输出
     */
    // 计算每个块的摘要
    const chunkSummaries = chunks.map(chunk => {
      const KMean = this._computeMean(chunk.K);
      const VMean = this._computeMean(chunk.V);
      return { K: KMean, V: VMean };
    });

    // 使用摘要计算 Attention
    const KGlobal = chunkSummaries.map(s => s.K);
    const VGlobal = chunkSummaries.map(s => s.V);

    return this._computeAttention(Q, KGlobal, VGlobal);
  }

  _computeMean(matrix) {
    /**
     * 计算矩阵均值
     * 
     * 参数:
     *   matrix: 矩阵
     * 
     * 返回:
     *   均值向量
     */
    const rows = matrix.length;
    const cols = matrix[0].length;

    const mean = Array.from({ length: cols }, () => 0);

    for (let j = 0; j < cols; j++) {
      let sum = 0;

      for (let i = 0; i < rows; i++) {
        sum += matrix[i][j];
      }

      mean[j] = sum / rows;
    }

    return mean;
  }

  _retrievalBasedAttention(Q, K, V) {
    /**
     * 基于检索的 Attention
     * 
     * 参数:
     *   Q: Query [batch, nHeads, seqLen, dHead]
     *   K: Key [batch, nHeads, seqLen, dHead]
     *   V: Value [batch, nHeads, seqLen, dHead]
     * 
     * 返回:
     *   输出 [batch, nHeads, seqLen, dHead]
     */
    const [batchSize, nHeads, seqLen, dHead] = this._getShape(Q);
    const topK = 128; // 检索的 top-k

    const output = [];

    for (let b = 0; b < batchSize; b++) {
      for (let h = 0; h < nHeads; h++) {
        const Qh = Q[b][h];
        const Kh = K[b][h];
        const Vh = V[b][h];

        const headOutput = [];

        for (let i = 0; i < seqLen; i++) {
          // 检索最相关的 Key
          const topKIndices = this._retrieveTopK(Qh[i], Kh, topK);

          // 使用检索到的 Key 计算 Attention
          const KRetrieved = topKIndices.map(idx => Kh[idx]);
          const VRetrieved = topKIndices.map(idx => Vh[idx]);

          const attentionOutput = this._computeAttention(
            Qh.slice(i, i + 1),
            KRetrieved,
            VRetrieved
          );

          headOutput.push(attentionOutput[0]);
        }

        output.push(headOutput);
      }
    }

    return this._reshapeOutput(output, batchSize, nHeads, seqLen, dHead);
  }

  _retrieveTopK(query, keys, topK) {
    /**
     * 检索 top-k
     * 
     * 参数:
     *   query: 查询向量
     *   keys: Key 矩阵
     *   topK: top-k 数量
     * 
     * 返回:
     *   索引数组
     */
    const scores = [];

    for (let i = 0; i < keys.length; i++) {
      let score = 0;

      for (let d = 0; d < query.length && d < keys[i].length; d++) {
        score += query[d] * keys[i][d];
      }

      scores.push({ index: i, score });
    }

    scores.sort((a, b) => b.score - a.score);

    return scores.slice(0, topK).map(s => s.index);
  }

  _computeAttention(Q, K, V) {
    /**
     * 计算 Attention
     * 
     * 参数:
     *   Q: Query
     *   K: Key
     *   V: Value
     * 
     * 返回:
     *   Attention 输出
     */
    const dHead = Q[0].length;

    // 计算 QK^T
    const scores = [];

    for (let i = 0; i < Q.length; i++) {
      const row = [];

      for (let j = 0; j < K.length; j++) {
        let score = 0;

        for (let d = 0; d < dHead; d++) {
          score += Q[i][d] * K[j][d];
        }

        score /= Math.sqrt(dHead);
        row.push(score);
      }

      scores.push(row);
    }

    // Softmax
    const expScores = scores.map(row => {
      const maxScore = Math.max(...row);
      const expRow = row.map(s => Math.exp(s - maxScore));
      const sumExp = expRow.reduce((sum, e) => sum + e, 0);
      return expRow.map(e => e / sumExp);
    });

    // 计算 Attention * V
    const output = [];

    for (let i = 0; i < Q.length; i++) {
      const row = [];

      for (let d = 0; d < dHead; d++) {
        let sum = 0;

        for (let j = 0; j < V.length; j++) {
          sum += expScores[i][j] * V[j][d];
        }

        row.push(sum);
      }

      output.push(row);
    }

    return output;
  }

  _getShape(tensor) {
    /**
     * 获取张量形状
     * 
     * 参数:
     *   tensor: 张量
     * 
     * 返回:
     *   形状数组
     */
    return [tensor.length, tensor[0].length, tensor[0][0].length, tensor[0][0][0].length];
  }

  _reshapeOutput(headOutputs, batchSize, nHeads, seqLen, dHead) {
    /**
     * 重塑输出
     * 
     * 参数:
     *   headOutputs: 头输出数组
     *   batchSize: 批次大小
     *   nHeads: 头数
     *   seqLen: 序列长度
     *   dHead: 头维度
     * 
     * 返回:
     *   重塑后的输出
     */
    const output = [];

    for (let b = 0; b < batchSize; b++) {
      const batchOutput = [];

      for (let h = 0; h < nHeads; h++) {
        const headIndex = b * nHeads + h;
        batchOutput.push(headOutputs[headIndex]);
      }

      output.push(batchOutput);
    }

    return output;
  }
}
```

## 📊 技术对比

### MoE vs Dense Model

| 特性 | MoE | Dense Model |
|------|-----|-------------|
| **参数量** | 大（总）小（激活） | 大（全部激活） |
| **计算量** | 低（只计算部分） | 高（全部计算） |
| **推理速度** | 快 | 慢 |
| **训练难度** | 高（需要负载均衡） | 中 |
| **适用场景** | 大规模模型 | 小规模模型 |

### SSM vs Transformer

| 特性 | SSM | Transformer |
|------|-----|-------------|
| **复杂度** | O(n) | O(n²) |
| **长上下文** | 优秀 | 受限 |
| **内存占用** | 低 | 高 |
| **训练稳定性** | 中 | 高 |
| **成熟度** | 新兴 | 成熟 |

### Linear Attention Methods

| 方法 | 复杂度 | 精度 | 适用场景 |
|------|--------|------|---------|
| **FlashAttention** | O(n²) | 高 | 通用 |
| **Linear Attention** | O(n) | 中 | 超长序列 |
| **Low-Rank Attention** | O(n·r) | 中 | 内存受限 |
| **Sparse Attention** | O(n·k) | 中 | 局部依赖 |
