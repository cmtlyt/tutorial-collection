```python
# Python 示例：蛋白质折叠预测
import torch
import torch.nn as nn
from typing import Dict, List, Tuple, Optional
import numpy as np

class ProteinStructurePredictor(nn.Module):
    """蛋白质结构预测器"""
    
    def __init__(
        self,
        amino_acid_dim: int = 20,
        hidden_dim: int = 256,
        num_layers: int = 6
    ):
        super().__init__()
        
        # 氨基酸编码
        self.amino_acid_embedding = nn.Embedding(20, amino_acid_dim)
        
        # 注意力层
        self.attention_layers = nn.ModuleList([
            nn.MultiheadAttention(hidden_dim, num_heads=8)
            for _ in range(num_layers)
        ])
        
        # 结构预测头
        self.structure_head = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 3)  # x, y, z 坐标
        )
        
        # 距离预测头
        self.distance_head = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 64)  # 64 个距离区间
        )
    
    def forward(
        self,
        sequence: torch.Tensor,
        attention_mask: Optional[torch.Tensor] = None
    ) -> Dict:
        """
        前向传播
        
        参数:
            sequence: 氨基酸序列 [batch_size, seq_len]
            attention_mask: 注意力掩码 [batch_size, seq_len]
        
        返回:
            predictions: 包含结构坐标和距离矩阵的字典
        """
        # 编码氨基酸序列
        x = self.amino_acid_embedding(sequence)
        
        # 通过注意力层
        for attention_layer in self.attention_layers:
            x, _ = attention_layer(x, x, x, key_padding_mask=attention_mask)
        
        # 预测结构坐标
        structure_coords = self.structure_head(x)
        
        # 预测距离矩阵
        distance_logits = self.distance_head(x)
        
        return {
            'structure_coords': structure_coords,
            'distance_logits': distance_logits
        }
    
    def predict_structure(
        self,
        sequence: str,
        num_samples: int = 5
    ) -> Dict:
        """预测蛋白质结构"""
        # 将氨基酸序列转换为张量
        aa_to_idx = self._get_amino_acid_mapping()
        sequence_tensor = torch.tensor([
            aa_to_idx.get(aa, 0) for aa in sequence
        ]).unsqueeze(0)
        
        # 多次采样以获得置信度
        predictions = []
        for _ in range(num_samples):
            with torch.no_grad():
                pred = self.forward(sequence_tensor)
                predictions.append(pred)
        
        # 计算平均和标准差
        mean_coords = torch.stack([
            p['structure_coords'] for p in predictions
        ]).mean(dim=0)
        
        std_coords = torch.stack([
            p['structure_coords'] for p in predictions
        ]).std(dim=0)
        
        return {
            'mean_coords': mean_coords,
            'std_coords': std_coords,
            'confidence': 1.0 / (1.0 + std_coords.mean())
        }
    
    def _get_amino_acid_mapping(self) -> Dict[str, int]:
        """获取氨基酸到索引的映射"""
        amino_acids = list('ACDEFGHIKLMNPQRSTVWY')
        return {aa: i for i, aa in enumerate(amino_acids)}

class DrugDiscoveryAgent:
    """药物发现智能体"""
    
    def __init__(
        self,
        protein_predictor: ProteinStructurePredictor,
        molecule_generator: nn.Module
    ):
        self.protein_predictor = protein_predictor
        self.molecule_generator = molecule_generator
    
    def discover_drug(
        self,
        target_protein: str,
        num_candidates: int = 100
    ) -> List[Dict]:
        """发现药物候选"""
        # 预测目标蛋白质结构
        protein_structure = self.protein_predictor.predict_structure(
            target_protein
        )
        
        # 生成候选分子
        candidates = []
        for _ in range(num_candidates):
            molecule = self._generate_molecule(protein_structure)
            binding_affinity = self._predict_binding_affinity(
                molecule,
                protein_structure
            )
            
            candidates.append({
                'molecule': molecule,
                'binding_affinity': binding_affinity,
                'protein_confidence': protein_structure['confidence']
            })
        
        # 按结合亲和力排序
        candidates.sort(
            key=lambda x: x['binding_affinity'],
            reverse=True
        )
        
        return candidates[:10]  # 返回前10个候选
    
    def _generate_molecule(
        self,
        protein_structure: Dict
    ) -> str:
        """生成分子"""
        # 简化的分子生成
        # 实际应用中应该使用更复杂的生成模型
        return "CC(=O)OC1=CC=CC=C1C(=O)O"
    
    def _predict_binding_affinity(
        self,
        molecule: str,
        protein_structure: Dict
    ) -> float:
        """预测结合亲和力"""
        # 简化的预测
```

```python
# Python 示例：材料发现
import torch
import torch.nn as nn
from typing import Dict, List, Tuple

class MaterialPropertyPredictor(nn.Module):
    """材料属性预测器"""
    
    def __init__(
        self,
        element_dim: int = 118,
        hidden_dim: int = 256
    ):
        super().__init__()
        
        # 元素嵌入
        self.element_embedding = nn.Embedding(element_dim, hidden_dim)
        
        # 图神经网络
        self.gnn_layers = nn.ModuleList([
            self._create_gnn_layer(hidden_dim)
            for _ in range(4)
        ])
        
        # 属性预测头
        self.property_heads = nn.ModuleDict({
            'bandgap': nn.Linear(hidden_dim, 1),
            'stability': nn.Linear(hidden_dim, 1),
            'conductivity': nn.Linear(hidden_dim, 1)
        })
    
    def _create_gnn_layer(self, hidden_dim: int) -> nn.Module:
        """创建图神经网络层"""
        return nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim)
        )
    
    def forward(
        self,
        elements: torch.Tensor,
        adjacency: torch.Tensor
    ) -> Dict:
        """
        前向传播
        
        参数:
            elements: 元素索引 [num_elements]
            adjacency: 邻接矩阵 [num_elements, num_elements]
        
        返回:
            properties: 材料属性字典
        """
        # 编码元素
        x = self.element_embedding(elements)
        
        # 通过图神经网络
        for gnn_layer in self.gnn_layers:
            # 消息传递
            messages = torch.matmul(adjacency, x)
            x = gnn_layer(messages) + x  # 残差连接
        
        # 全局池化
        global_features = x.mean(dim=0)
        
        # 预测属性
        properties = {}
        for prop_name, head in self.property_heads.items():
            properties[prop_name] = head(global_features)
        
        return properties

class MaterialDiscoverySystem:
    """材料发现系统"""
    
    def __init__(
        self,
        property_predictor: MaterialPropertyPredictor,
        target_properties: Dict
    ):
        self.property_predictor = property_predictor
        self.target_properties = target_properties
    
    def discover_materials(
        self,
        search_space: List[List[int]],
        max_iterations: int = 1000
    ) -> List[Dict]:
        """发现材料"""
        best_materials = []
        
        for iteration in range(max_iterations):
            # 随机采样候选材料
            candidate = self._sample_material(search_space)
            
            # 预测属性
            properties = self._predict_properties(candidate)
            
            # 评估是否满足目标
            score = self._evaluate_properties(properties)
            
            if score > 0.8:
                best_materials.append({
                    'composition': candidate,
                    'properties': properties,
                    'score': score
                })
        
        # 按分数排序
        best_materials.sort(
            key=lambda x: x['score'],
            reverse=True
        )
        
        return best_materials[:10]
    
    def _sample_material(
        self,
        search_space: List[List[int]]
    ) -> List[int]:
        """采样材料"""
        # 简化的采样策略
        # 实际应用中应该使用更复杂的搜索算法
        return [
            np.random.choice(elements)
            for elements in search_space
        ]
    
    def _predict_properties(
        self,
        composition: List[int]
    ) -> Dict:
        """预测材料属性"""
        # 构建邻接矩阵
        num_elements = len(composition)
        adjacency = torch.eye(num_elements)
        
        # 预测属性
        with torch.no_grad():
            properties = self.property_predictor(
                torch.tensor(composition),
                adjacency
            )
        
        return {
            name: value.item()
            for name, value in properties.items()
        }
    
    def _evaluate_properties(
        self,
        properties: Dict
    ) -> float:
        """评估材料属性"""
        score = 0.0
        
        for prop_name, target_value in self.target_properties.items():
            predicted_value = properties.get(prop_name, 0.0)
            diff = abs(predicted_value - target_value)
            score += max(0, 1.0 - diff)
        
        return score / len(self.target_properties)
```

```python
import torch
import torch.nn as nn
from typing import Dict, List, Optional

class CreativeContentGenerator(nn.Module):
    """创意内容生成器"""
    
    def __init__(
        self,
        vocab_size: int,
        embedding_dim: int = 512,
        hidden_dim: int = 1024,
        num_layers: int = 12
    ):
        super().__init__()
        
        # 词嵌入
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        
        # Transformer 解码器
        decoder_layer = nn.TransformerDecoderLayer(
            d_model=embedding_dim,
            nhead=8,
            dim_feedforward=hidden_dim
        )
        
        self.transformer_decoder = nn.TransformerDecoder(
            decoder_layer,
            num_layers=num_layers
        )
        
        # 输出层
        self.output_layer = nn.Linear(embedding_dim, vocab_size)
    
    def forward(
        self,
        input_ids: torch.Tensor,
        attention_mask: Optional[torch.Tensor] = None
    ) -> torch.Tensor:
        """前向传播"""
        # 嵌入
        x = self.embedding(input_ids)
        
        # Transformer 解码
        x = self.transformer_decoder(
            x, x, tgt_mask=attention_mask
        )
        
        # 输出
        logits = self.output_layer(x)
        
        return logits
    
    def generate(
        self,
        prompt: str,
        max_length: int = 512,
        temperature: float = 1.0,
        top_k: int = 50
    ) -> str:
        """生成内容"""
        # 将提示转换为 token IDs
        input_ids = self._tokenize(prompt)
        
        # 生成
        generated_ids = []
        for _ in range(max_length):
            with torch.no_grad():
                logits = self.forward(input_ids)
                
                # 采样
                next_token = self._sample(
                    logits[:, -1, :],
                    temperature,
                    top_k
                )
                
                generated_ids.append(next_token)
                input_ids = torch.cat([
                    input_ids,
                    next_token.unsqueeze(0)
                ], dim=1)
                
                # 检查结束符
                if next_token == self._get_eos_token():
                    break
        
        # 解码
        generated_text = self._decode(generated_ids)
        
        return generated_text
    
    def _sample(
        self,
        logits: torch.Tensor,
        temperature: float,
        top_k: int
    ) -> torch.Tensor:
        """采样下一个 token"""
        # 应用温度
        logits = logits / temperature
        
        # Top-k 采样
        if top_k > 0:
            top_k_logits, top_k_indices = torch.topk(logits, top_k)
            logits = torch.full_like(logits, float('-inf'))
            logits.scatter_(1, top_k_indices, top_k_logits)
        
        # 采样
        probs = torch.softmax(logits, dim=-1)
        next_token = torch.multinomial(probs, num_samples=1)
        
        return next_token.squeeze()
    
    def _tokenize(self, text: str) -> torch.Tensor:
        """分词"""
        # 简化的分词
        tokens = list(text)
        return torch.tensor([
            ord(token) % 10000 for token in tokens
        ]).unsqueeze(0)
    
    def _decode(self, token_ids: List[int]) -> str:
        """解码"""
        # 简化的解码
        return ''.join([chr(tid) for tid in token_ids])
    
    def _get_eos_token(self) -> int:
        """获取结束符 token"""
        return 0

class CreativeAssistant:
    """创意助手"""
    
    def __init__(
        self,
        text_generator: CreativeContentGenerator,
        image_generator: nn.Module,
        music_generator: nn.Module
    ):
        self.text_generator = text_generator
        self.image_generator = image_generator
        self.music_generator = music_generator
    
    def assist_creation(
        self,
        prompt: str,
        modality: str = 'text'
    ) -> Dict:
        """辅助创作"""
        if modality == 'text':
            generated_text = self.text_generator.generate(prompt)
            return {
                'modality': 'text',
                'content': generated_text,
                'prompt': prompt
            }
        
        elif modality == 'image':
            generated_image = self._generate_image(prompt)
            return {
                'modality': 'image',
                'content': generated_image,
                'prompt': prompt
            }
        
        elif modality == 'music':
            generated_music = self._generate_music(prompt)
            return {
                'modality': 'music',
                'content': generated_music,
                'prompt': prompt
            }
        
        else:
            return {'error': f'Unknown modality: {modality}'}
    
    def _generate_image(self, prompt: str) -> str:
        """生成图像"""
        # 简化的图像生成
        return f"Generated image based on: {prompt}"
    
    def _generate_music(self, prompt: str) -> str:
        """生成音乐"""
        # 简化的音乐生成
        return f"Generated music based on: {prompt}"
```

```python
import torch
import torch.nn as nn
from typing import Dict, List, Optional
from dataclasses import dataclass

@dataclass
class StudentProfile:
    """学生档案"""
    student_id: str
    knowledge_level: Dict[str, float]
    learning_style: str
    preferences: Dict[str, any]

class AdaptiveLearningSystem:
    """自适应学习系统"""
    
    def __init__(
        self,
        knowledge_tracer: nn.Module,
        content_recommender: nn.Module
    ):
        self.knowledge_tracer = knowledge_tracer
        self.content_recommender = content_recommender
        self.student_profiles: Dict[str, StudentProfile] = {}
    
    def register_student(
        self,
        student_id: str,
        initial_knowledge: Dict[str, float]
    ):
        """注册学生"""
        profile = StudentProfile(
            student_id=student_id,
            knowledge_level=initial_knowledge,
            learning_style='visual',
            preferences={}
        )
        self.student_profiles[student_id] = profile
    
    def update_knowledge(
        self,
        student_id: str,
        topic: str,
        performance: float
    ):
        """更新知识水平"""
        if student_id in self.student_profiles:
            profile = self.student_profiles[student_id]
            profile.knowledge_level[topic] = performance
    
    def recommend_content(
        self,
        student_id: str,
        current_topic: str
    ) -> Dict:
        """推荐学习内容"""
        if student_id not in self.student_profiles:
            return {'error': 'Student not found'}
        
        profile = self.student_profiles[student_id]
        
        # 获取推荐
        with torch.no_grad():
            recommendations = self.content_recommender(
                profile.knowledge_level,
                current_topic
            )
        
        return {
            'student_id': student_id,
            'recommendations': recommendations,
            'current_knowledge': profile.knowledge_level
        }
    
    def generate_personalized_path(
        self,

        # 基于知识图谱生成路径
        path = self._generate_learning_path(
            profile.knowledge_level,

        for i, topic in enumerate(topics):
            path.append({
                'step': i + 1,
                'topic': topic,
                'estimated_time': '30 minutes',
                'prerequisites': topics[:i]
            })
        
        return path
```

```python
# Python 示例：医疗诊断辅助
import torch
import torch.nn as nn
from typing import Dict, List, Optional
from enum import Enum

class MedicalCondition(Enum):
    """医疗状况"""
    NORMAL = "normal"
    DISEASE_A = "disease_a"
    DISEASE_B = "disease_b"
    DISEASE_C = "disease_c"

class MedicalImageAnalyzer(nn.Module):
    """医学图像分析器"""
    
    def __init__(
        self,
        num_conditions: int = 4
    ):
        super().__init__()
        
        # 图像编码器
        self.image_encoder = nn.Sequential(
            nn.Conv2d(3, 64, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(128, 256, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.AdaptiveAvgPool2d((1, 1))
        )
        
        # 诊断头
        self.diagnosis_head = nn.Sequential(
            nn.Linear(256, 128),
            nn.ReLU(),
            nn.Dropout(0.5),
            nn.Linear(128, num_conditions)
        )
    
    def forward(self, image: torch.Tensor) -> torch.Tensor:
        """前向传播"""
        # 编码图像
        features = self.image_encoder(image)
        features = features.view(features.size(0), -1)
        
        # 诊断
        logits = self.diagnosis_head(features)
        
        return logits
    
    def diagnose(
        self,
        image: torch.Tensor,
        return_confidence: bool = True
    ) -> Dict:
        """诊断"""
        with torch.no_grad():
            logits = self.forward(image)
            probs = torch.softmax(logits, dim=1)
        
        # 获取最可能的诊断
        max_prob, predicted_class = probs.max(dim=1)
        
        diagnosis = {
            'predicted_condition': MedicalCondition(predicted_class.item()).value,
            'confidence': max_prob.item(),
            'all_probabilities': probs.squeeze().tolist()
        }
        
        if return_confidence:
            diagnosis['confidence_intervals'] = self._compute_confidence_intervals(
                probs
            )
        
        return diagnosis
    
    def _compute_confidence_intervals(
        self,
        probs: torch.Tensor
    ) -> List[Dict]:
        """计算置信区间"""
        intervals = []
        for i, prob in enumerate(probs.squeeze()):
            intervals.append({
                'condition': MedicalCondition(i).value,
                'probability': prob.item(),
                'lower_bound': max(0, prob.item() - 0.05),
                'upper_bound': min(1, prob.item() + 0.05)
            })
        
        return intervals

class MedicalAssistant:
    """医疗助手"""
    
    def __init__(
        self,
        image_analyzer: MedicalImageAnalyzer,
        knowledge_base: Dict
    ):
        self.image_analyzer = image_analyzer
        self.knowledge_base = knowledge_base
    
    def assist_diagnosis(
        self,
        patient_data: Dict,
        medical_image: torch.Tensor
    ) -> Dict:
        """辅助诊断"""
        # 分析医学图像
        image_diagnosis = self.image_analyzer.diagnose(medical_image)
        
        # 结合患者数据
        combined_diagnosis = self._combine_diagnoses(
            patient_data,
            image_diagnosis
        )
        
        # 获取治疗建议
        treatment_recommendations = self._get_treatment_recommendations(
            combined_diagnosis
        )
        
        return {
            'image_diagnosis': image_diagnosis,
            'combined_diagnosis': combined_diagnosis,
            'treatment_recommendations': treatment_recommendations
        }
    
    def _combine_diagnoses(
        self,
        patient_data: Dict,
        image_diagnosis: Dict
    ) -> Dict:
        """结合诊断"""
        # 简化的结合逻辑
        # 实际应用中应该使用更复杂的融合方法
        return {
            'primary_diagnosis': image_diagnosis['predicted_condition'],
            'confidence': image_diagnosis['confidence'],
            'supporting_evidence': patient_data.get('symptoms', [])
        }
    
    def _get_treatment_recommendations(
        self,
        diagnosis: Dict
    ) -> List[Dict]:
        """获取治疗建议"""
        condition = diagnosis['primary_diagnosis']
        
        # 从知识库获取治疗建议
        recommendations = self.knowledge_base.get(condition, [])
        
        return recommendations
```
