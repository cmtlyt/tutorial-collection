## 📖 概述

工具集成是 Agent 系统的核心能力之一，通过将各种外部工具和 API 集成到 Agent 中，使其能够执行复杂的任务，如搜索信息、处理数据、调用服务等。本章节将详细介绍常用工具的使用方法以及自定义工具的开发规范。

## 🎯 为什么需要工具集成？

### Agent 的局限性

1. **知识截止**
   - 模型训练数据有时间限制
   - 无法获取最新信息
   - 缺乏实时数据访问能力

2. **执行能力**
   - 无法直接执行代码
   - 无法访问外部系统
   - 缺乏实际操作能力

3. **专业领域**
   - 通用模型在专业领域表现有限
   - 需要领域专业工具
   - 复杂计算需要专用工具

### 工具集成的价值

1. **扩展能力**
   - 访问实时信息
   - 执行实际操作
   - 处理专业任务

2. **增强智能**
   - 结合外部知识
   - 提升决策质量
   - 支持复杂推理

3. **实用价值**
   - 解决实际问题
   - 提高工作效率
   - 支持业务流程

```javascript
import { SerpAPI } from 'langchain/tools';

class GoogleSearchTool {
  constructor(apiKey) {
    this.tool = new SerpAPI(apiKey, {
      location: 'United States',
      hl: 'en',
      gl: 'us',
    });
  }

  async search(query, numResults = 5) {
    /**
     * 执行 Google 搜索
     * 
     * 参数:
     *   query: 搜索查询
     *   numResults: 返回结果数量
     * 
     * 返回:
     *   搜索结果数组
     */
    try {
      const results = await this.tool.call(query);
      
      // 解析结果
      const searchResults = [];
      if (results.organic_results) {
        for (let i = 0; i < Math.min(numResults, results.organic_results.length); i++) {
          const result = results.organic_results[i];
          searchResults.push({
            title: result.title,
            link: result.link,
            snippet: result.snippet,
            position: i + 1
          });
        }
      }
      
      return searchResults;
    } catch (error) {
      console.error('Google search error:', error);
      throw new Error(`搜索失败: ${error.message}`);
    }
  }

  async searchWithDate(query, dateRange = 'd7') {
    /**
     * 带时间范围的搜索
     * 
     * 参数:
     *   query: 搜索查询
     *   dateRange: 时间范围 (d1, d7, m1, y1)
     * 
     * 返回:
     *   搜索结果数组
     */
    const results = await this.tool.call(`${query} tbs=qdr:${dateRange}`);
    return this.parseResults(results);
  }

  async searchNews(query) {
    /**
     * 新闻搜索
     * 
     * 参数:
     *   query: 新闻查询
     * 
     * 返回:
     *   新闻结果数组
     */
    const results = await this.tool.call(`${query} tbm=nws`);
    return this.parseResults(results);
  }

  parseResults(results) {
    /**
     * 解析搜索结果
     */
    const searchResults = [];
    if (results.organic_results) {
      results.organic_results.forEach((result, index) => {
        searchResults.push({
          title: result.title,
          link: result.link,
          snippet: result.snippet,
          position: index + 1
        });
      });
    }
    return searchResults;
  }
}
```

```javascript
class BingSearchTool {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.endpoint = 'https://api.bing.microsoft.com/v7.0/search';
  }

  async search(query, count = 10, offset = 0) {
    /**
     * 执行 Bing 搜索
     * 
     * 参数:
     *   query: 搜索查询
     *   count: 返回结果数量
     *   offset: 结果偏移量
     * 
     * 返回:
     *   搜索结果数组
     */
    try {
      const response = await fetch(
        `${this.endpoint}?q=${encodeURIComponent(query)}&count=${count}&offset=${offset}`,
        {
          headers: {
            'Ocp-Apim-Subscription-Key': this.apiKey,
          },
        }
      );

      const data = await response.json();
      
      if (!data.webPages || !data.webPages.value) {
        return [];
      }

      return data.webPages.value.map((result, index) => ({
        title: result.name,
        link: result.url,
        snippet: result.snippet,
        position: offset + index + 1,
        displayUrl: result.displayUrl
      }));
    } catch (error) {
      console.error('Bing search error:', error);
      throw new Error(`搜索失败: ${error.message}`);
```

```javascript
class CodeInterpreterTool {
  constructor() {
    this.sandbox = null;
  }

  async executeCode(code, language = 'javascript') {
    /**
     * 执行代码
     * 
     * 参数:
     *   code: 要执行的代码
     *   language: 编程语言
     * 
     * 返回:
     *   执行结果
     */
    try {
      switch (language) {
        case 'javascript':
          return this.executeJavaScript(code);
        case 'python':
          return this.executePython(code);
        default:
          throw new Error(`不支持的语言: ${language}`);
      }
    } catch (error) {
      console.error('Code execution error:', error);
      throw new Error(`代码执行失败: ${error.message}`);
    }
  }

  executeJavaScript(code) {
    /**
     * 执行 JavaScript 代码
     * 
     * 参数:
     *   code: JavaScript 代码
     * 
     * 返回:
     *   执行结果
     */
    // 创建安全的执行环境
    const sandbox = {
      console: {
        log: (...args) => this.logOutput(args),
        error: (...args) => this.logError(args),
      },
      Math: Math,
      Array: Array,
      Object: Object,
      String: String,
      Number: Number,
      Date: Date,
      JSON: JSON,
    };

    // 执行代码
    const fn = new Function(
      ...Object.keys(sandbox),
      `
      try {
        return ${code};
      } catch (error) {
        throw error;
      }
      `
    );

    const result = fn(...Object.values(sandbox));
    return {
      success: true,
      result: result,
      output: this.getOutput()
    };
  }

  async executePython(code) {
    /**
     * 执行 Python 代码
     * 
     * 参数:
     *   code: Python 代码
     * 
     * 返回:
     *   执行结果
     */
    // 这里需要调用 Python 解释器
    // 实际实现可能使用 child_process 或其他 Python 运行时
    const { spawn } = require('child_process');
    
    return new Promise((resolve, reject) => {
      const python = spawn('python', ['-c', code]);
      
      let output = '';
      let error = '';
      
      python.stdout.on('data', (data) => {
        output += data.toString();
      });
      
      python.stderr.on('data', (data) => {
        error += data.toString();
      });
      
      python.on('close', (code) => {
        if (code === 0) {
          resolve({
            success: true,
            result: output.trim(),
            output: output.trim()
          });
        } else {
          reject(new Error(`Python 执行失败: ${error}`));
        }
      });
    });
  }

  logOutput(args) {
    /**
     * 记录输出
     */
    if (!this.output) {
      this.output = [];
    }
    this.output.push(args.map(arg => 
      typeof arg === 'object' ? JSON.stringify(arg, null, 2) : String(arg)
    ).join(' '));
  }

  logError(args) {
    /**
     * 记录错误
     */
    if (!this.error) {
      this.error = [];
    }
    this.error.push(args.map(arg => 
      typeof arg === 'object' ? JSON.stringify(arg, null, 2) : String(arg)
    ).join(' '));
  }

  getOutput() {
    /**
     * 获取输出
     */
    return this.output || [];
  }

  getError() {
    /**
     * 获取错误
     */
    return this.error || [];
  }

  reset() {
    /**
     * 重置解释器
     */
    this.output = [];
    this.error = [];
  }
}

// 使用示例
async function codeInterpreterExample() {
  const interpreter = new CodeInterpreterTool();
  
  // 执行 JavaScript 代码
  const jsResult = await interpreter.executeCode(`
    const numbers = [1, 2, 3, 4, 5];
    const sum = numbers.reduce((a, b) => a + b, 0);
    const average = sum / numbers.length;
    console.log('Sum:', sum);
    console.log('Average:', average);
    { sum, average }
  `, 'javascript');
  
  console.log('JavaScript 结果:', jsResult);
  
  // 执行 Python 代码
  const pythonResult = await interpreter.executeCode(`
    import math
    numbers = [1, 2, 3, 4, 5]
    sum_result = sum(numbers)
    average = sum_result / len(numbers)
    print(f'Sum: {sum_result}')
    print(f'Average: {average}')
    {'sum': sum_result, 'average': average}
  `, 'python');
  
  console.log('Python 结果:', pythonResult);
}
```

```javascript
class GitHubTool {
  constructor(token) {
    this.token = token;
    this.baseUrl = 'https://api.github.com';
  }

  async searchRepositories(query, language = '', sort = 'stars', order = 'desc') {
    /**
     * 搜索仓库
     * 
     * 参数:
     *   query: 搜索查询
     *   language: 编程语言
     *   sort: 排序方式 (stars, forks, updated)
     *   order: 排序顺序 (asc, desc)
     * 
     * 返回:
     *   仓库列表
     */
    try {
      const searchQuery = language ? `${query} language:${language}` : query;
      const response = await fetch(
        `${this.baseUrl}/search/repositories?q=${encodeURIComponent(searchQuery)}&sort=${sort}&order=${order}&per_page=10`,
        {
          headers: {
            'Authorization': `token ${this.token}`,
            'Accept': 'application/vnd.github.v3+json',
          },
        }
      );

      const data = await response.json();
      
      if (!data.items) {
        return [];
      }

      return data.items.map(repo => ({
        id: repo.id,
        name: repo.name,
        fullName: repo.full_name,
        description: repo.description,
        url: repo.html_url,
        stars: repo.stargazers_count,
        forks: repo.forks_count,
        language: repo.language,
        updatedAt: repo.updated_at,
        owner: {
          login: repo.owner.login,
          type: repo.owner.type,
        }
      }));
    } catch (error) {
      console.error('GitHub search error:', error);
      throw new Error(`仓库搜索失败: ${error.message}`);
    }
  }

  async searchCode(query, language = '', sort = 'indexed', order = 'desc') {
    /**
     * 搜索代码
     * 
     * 参数:
     *   query: 搜索查询
     *   language: 编程语言
     *   sort: 排序方式 (indexed)
     *   order: 排序顺序 (asc, desc)
     * 
     * 返回:
     *   代码结果列表
     */
    try {
      const searchQuery = language ? `${query} language:${language}` : query;
      const response = await fetch(
        `${this.baseUrl}/search/code?q=${encodeURIComponent(searchQuery)}&sort=${sort}&order=${order}&per_page=10`,
        {
          headers: {
            'Authorization': `token ${this.token}`,
            'Accept': 'application/vnd.github.v3+json',
          },
        }
      );

      const data = await response.json();
      
      if (!data.items) {
        return [];
      }

      return data.items.map(item => ({
        name: item.name,
        path: item.path,
        htmlUrl: item.html_url,
        gitUrl: item.git_url,
        repository: {
          name: item.repository.name,
          fullName: item.repository.full_name,
          url: item.repository.html_url,
        }
      }));
    } catch (error) {
      console.error('GitHub code search error:', error);
      throw new Error(`代码搜索失败: ${error.message}`);
    }
  }

  async getRepositoryContents(owner, repo, path = '') {
    /**
     * 获取仓库内容
     * 
     * 参数:
     *   owner: 仓库所有者
     *   repo: 仓库名称
     *   path: 文件路径
     * 
     * 返回:
     *   文件/目录内容
     */
    try {
      const response = await fetch(
        `${this.baseUrl}/repos/${owner}/${repo}/contents/${path}`,
        {
          headers: {
            'Authorization': `token ${this.token}`,
            'Accept': 'application/vnd.github.v3+json',
          },
        }
      );

      const data = await response.json();
      
      if (Array.isArray(data)) {
        // 目录
        return data.map(item => ({
          name: item.name,
          type: item.type,
          path: item.path,
          size: item.size,
          url: item.html_url,
        }));
      } else {
        // 文件
        return {
          name: data.name,
          type: data.type,
          path: data.path,
          size: data.size,
          url: data.html_url,
          content: data.content,
          encoding: data.encoding,
        };
      }
    } catch (error) {
      console.error('GitHub get contents error:', error);
      throw new Error(`获取仓库内容失败: ${error.message}`);
    }
  }

  async getFileContent(owner, repo, path, ref = 'main') {
    /**
     * 获取文件内容
     * 
     * 参数:
     *   owner: 仓库所有者
     *   repo: 仓库名称
     *   path: 文件路径
     *   ref: 分支或标签
     * 
     * 返回:
     *   文件内容
     */
    try {
      const response = await fetch(
        `${this.baseUrl}/repos/${owner}/${repo}/contents/${path}?ref=${ref}`,
        {
          headers: {
            'Authorization': `token ${this.token}`,
            'Accept': 'application/vnd.github.v3+json',
          },
        }
      );

      const data = await response.json();
      
      // 解码 Base64 内容
      const content = Buffer.from(data.content, 'base64').toString('utf-8');
      
      return {
        name: data.name,
        path: data.path,
        size: data.size,
        content: content,
        sha: data.sha,
      };
    } catch (error) {
      console.error('GitHub get file content error:', error);
      throw new Error(`获取文件内容失败: ${error.message}`);
    }
  }
}
```

```javascript
     * 参数:
     *   headers: 请求头对象

    this.headers = { ...this.headers, ...headers };
  }

  setAuth(token, type = 'Bearer') {
    /**
     * 设置认证信息
     * 
     * 参数:
     *   token: 认证令牌
     *   type: 认证类型 (Bearer, Basic, etc.)
     */

  async get(endpoint, params = {}, headers = {}) {
    /**
     * GET 请求
     * 
     * 参数:
     *   endpoint: 端点路径
     *   params: 查询参数
     *   headers: 额外的请求头
     * 
     * 返回:
     *   响应数据

        method: 'GET',
        headers: { ...this.headers, ...headers },
        signal: AbortSignal.timeout(this.timeout),
      });

      return this.handleResponse(response);
    } catch (error) {
      console.error('GET request error:', error);
      throw this.handleError(error);
    }
  }

     * POST 请求

     *   endpoint: 端点路径

     * 返回:
     *   响应数据
     */
    try {

        body: JSON.stringify(data),

    /**
     * PATCH 请求
     * 
     * 参数:
     *   endpoint: 端点路径
     *   data: 请求数据
     *   headers: 额外的请求头
     * 
     * 返回:
     *   响应数据
     */
    try {
      const url = this.buildURL(endpoint);
      const response = await fetch(url, {
        method: 'PATCH',
        headers: { ...this.headers, ...headers },
        body: JSON.stringify(data),
        signal: AbortSignal.timeout(this.timeout),
      });

      return this.handleResponse(response);
    } catch (error) {
      console.error('PATCH request error:', error);
      throw this.handleError(error);
    }
  }

  buildURL(endpoint, params = {}) {
    /**
     * 构建完整 URL
     * 
     * 参数:
     *   endpoint: 端点路径
     *   params: 查询参数
     * 
     * 返回:
     *   完整 URL
     */
    let url = `${this.baseURL}${endpoint}`;
    
    if (Object.keys(params).length > 0) {
      const queryString = new URLSearchParams(params).toString();
      url += `?${queryString}`;
    }
    
    return url;
  }

  async handleResponse(response) {
    /**
     * 处理响应
     * 
     * 参数:
     *   response: 响应对象
     * 
     * 返回:
     *   解析后的响应数据
     */
    const contentType = response.headers.get('content-type');
    let data;

    if (contentType && contentType.includes('application/json')) {
      data = await response.json();
    } else {
      data = await response.text();
    }

    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${data.message || response.statusText}`);
    }

    return {
      status: response.status,
      statusText: response.statusText,
      data: data,
      headers: Object.fromEntries(response.headers.entries()),
    };
  }

  handleError(error) {
    /**
     * 处理错误
     * 
     * 参数:
     *   error: 错误对象
     * 
     * 返回:
     *   标准化错误对象
     */
    if (error.name === 'AbortError') {
      return new Error('请求超时');
    }
    
    if (error.message.includes('Failed to fetch')) {
      return new Error('网络连接失败');
    }
    
    return error;
  }
}
```

```javascript
/**
 * 工具接口定义
 * 所有自定义工具都应该实现这个接口
 */
class BaseTool {
  constructor() {
    this.name = '';
    this.description = '';
    this.parameters = {};
    this.required = [];
  }

  /**
   * 执行工具
   * 
   * @param {Object} inputs - 输入参数
   * @returns {Promise<Object>} - 执行结果
   */
  async execute(inputs) {
    throw new Error('execute 方法必须被实现');
  }

  /**
   * 验证输入参数
   * 
   * @param {Object} inputs - 输入参数
   * @returns {Object} - 验证结果 { valid: boolean, errors: Array }
   */
  validateInputs(inputs) {
    const errors = [];

    // 检查必需参数
    for (const param of this.required) {
      if (!(param in inputs) || inputs[param] === undefined || inputs[param] === null) {
        errors.push(`缺少必需参数: ${param}`);
      }
    }
```

```javascript
class ParameterValidator {
  constructor() {
    this.validators = {};
  }

  /**
   * 添加验证器
   * 
   * @param {String} name - 参数名称
   * @param {Function} validator - 验证函数
   */
  addValidator(name, validator) {
    this.validators[name] = validator;
  }

  /**
   * 验证参数
   * 
   * @param {Object} params - 参数对象
   * @param {Object} schema - 验证模式
   * @returns {Object} - 验证结果
   */
  validate(params, schema) {
    const errors = [];
    const warnings = [];

    // 验证必需参数
    if (schema.required) {
      for (const requiredParam of schema.required) {
        if (!(requiredParam in params)) {
          errors.push(`缺少必需参数: ${requiredParam}`);
        }
      }
    }

    // 验证参数类型和约束
    if (schema.properties) {
      for (const [paramName, paramSchema] of Object.entries(schema.properties)) {
        if (paramName in params) {
          const value = params[paramName];
          const validation = this.validateParameter(paramName, value, paramSchema);
          
          if (!validation.valid) {
            errors.push(...validation.errors);
          }
          
          if (validation.warnings && validation.warnings.length > 0) {
            warnings.push(...validation.warnings);
          }
        }
      }
    }

    // 应用自定义验证器
    for (const [validatorName, validator] of Object.entries(this.validators)) {
      const result = validator(params);
      if (!result.valid) {
        errors.push(...result.errors);
      }
    }

    return {
      valid: errors.length === 0,
      errors: errors,
      warnings: warnings,
    };
  }

  /**
   * 验证单个参数
   * 
   * @param {String} name - 参数名称
   * @param {*} value - 参数值
   * @param {Object} schema - 参数模式
   * @returns {Object} - 验证结果
   */
  validateParameter(name, value, schema) {
    const errors = [];
    const warnings = [];

    // 类型验证
    if (schema.type) {
      if (!this.validateType(value, schema.type)) {
        errors.push(`参数 ${name} 类型错误，期望 ${schema.type}`);
        return { valid: false, errors, warnings };
      }
    }

    // 枚举验证
    if (schema.enum) {
      if (!schema.enum.includes(value)) {
        errors.push(`参数 ${name} 必须是以下值之一: ${schema.enum.join(', ')}`);
      }
    }

    // 范围验证
    if (schema.minimum !== undefined) {
      if (value < schema.minimum) {
        errors.push(`参数 ${name} 不能小于 ${schema.minimum}`);
      }
    }

    if (schema.maximum !== undefined) {
      if (value > schema.maximum) {
        errors.push(`参数 ${name} 不能大于 ${schema.maximum}`);
      }
    }

    // 字符串长度验证
    if (schema.minLength !== undefined && typeof value === 'string') {
      if (value.length < schema.minLength) {
        errors.push(`参数 ${name} 长度不能少于 ${schema.minLength}`);
      }
    }

    if (schema.maxLength !== undefined && typeof value === 'string') {
      if (value.length > schema.maxLength) {
        errors.push(`参数 ${name} 长度不能超过 ${schema.maxLength}`);
      }
    }

    // 数组长度验证
    if (schema.minItems !== undefined && Array.isArray(value)) {
      if (value.length < schema.minItems) {
        errors.push(`参数 ${name} 数量不能少于 ${schema.minItems}`);
      }
    }

    if (schema.maxItems !== undefined && Array.isArray(value)) {
      if (value.length > schema.maxItems) {
        errors.push(`参数 ${name} 数量不能超过 ${schema.maxItems}`);
      }
    }

    // 正则验证
    if (schema.pattern && typeof value === 'string') {
      const regex = new RegExp(schema.pattern);
      if (!regex.test(value)) {
        errors.push(`参数 ${name} 格式不正确`);
      }
    }

    // 自定义验证
    if (schema.validator) {
      const customValidation = schema.validator(value);
      if (!customValidation.valid) {
        errors.push(...customValidation.errors);
      }
    }

    return {
      valid: errors.length === 0,
      errors,
      warnings,
    };
  }

  /**
   * 验证类型
   * 
   * @param {*} value - 值
   * @param {String} type - 类型
   * @returns {Boolean} - 是否匹配
   */
  validateType(value, type) {
    const typeMap = {
      'string': (v) => typeof v === 'string',
      'number': (v) => typeof v === 'number' && !isNaN(v),
      'integer': (v) => Number.isInteger(v),
      'boolean': (v) => typeof v === 'boolean',
      'array': (v) => Array.isArray(v),
      'object': (v) => typeof v === 'object' && v !== null && !Array.isArray(v),
      'null': (v) => v === null,
    };

    const validator = typeMap[type];
    return validator ? validator(value) : false;
  }
}
```

```javascript
class ToolError extends Error {

    super(message);

    this.details = details;

  }

    return {
      name: this.name,

      code: this.code,

    };

class ErrorHandler {
  constructor() {
    this.errorHandlers = {};
    this.retryConfig = {
      maxRetries: 3,
      retryDelay: 1000,
      backoffMultiplier: 2,
    };
  }

  /**
   * 注册错误处理器
   * 
   * @param {String} errorCode - 错误代码
   * @param {Function} handler - 处理函数
   */
  registerHandler(errorCode, handler) {
    this.errorHandlers[errorCode] = handler;
  }

  /**
   * 处理错误
   * 
   * @param {Error} error - 错误对象
   * @param {Object} context - 上下文信息
   * @returns {Object} - 处理结果
   */
  async handleError(error, context = {}) {
    console.error('Error occurred:', error);

    // 确定错误代码
    const errorCode = this.getErrorCode(error);
    
    // 获取错误处理器
    const handler = this.errorHandlers[errorCode] || this.defaultHandler;
    
    // 执行错误处理
    const result = await handler(error, context);
    
    return result;
  }

  /**
   * 获取错误代码
   * 
   * @param {Error} error - 错误对象
   * @returns {String} - 错误代码
   */
  getErrorCode(error) {
    if (error instanceof ToolError) {
      return error.code;
    }
    
    // 根据错误类型确定代码
    if (error.message.includes('timeout')) {
      return 'TIMEOUT_ERROR';
    }
    
    if (error.message.includes('network') || error.message.includes('fetch')) {
      return 'NETWORK_ERROR';
    }
    
    if (error.message.includes('validation')) {
      return 'VALIDATION_ERROR';
    }
    
    if (error.message.includes('authentication') || error.message.includes('authorization')) {
      return 'AUTH_ERROR';
    }
    
    return 'UNKNOWN_ERROR';
  }

  /**
   * 默认错误处理器
   * 
   * @param {Error} error - 错误对象
   * @param {Object} context - 上下文信息
   * @returns {Object} - 处理结果
   */
  async defaultHandler(error, context) {
    return {
      success: false,
      error: error.message,
      code: this.getErrorCode(error),
      shouldRetry: false,
      timestamp: new Date().toISOString(),
    };
  }

  /**
   * 带重试的执行
   * 
   * @param {Function} fn - 要执行的函数
   * @param {Object} options - 重试选项
   * @returns {*} - 执行结果
   */
  async executeWithRetry(fn, options = {}) {
    const config = { ...this.retryConfig, ...options };
    let lastError;
    let delay = config.retryDelay;

    for (let attempt = 0; attempt <= config.maxRetries; attempt++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error;
        
        // 检查是否应该重试
        const errorCode = this.getErrorCode(error);
        const shouldRetry = await this.shouldRetry(errorCode, attempt, config);
        
        if (!shouldRetry) {
          throw error;
        }
        
        console.log(`Attempt ${attempt + 1} failed, retrying in ${delay}ms...`);
        
        // 等待后重试
        await this.sleep(delay);
        delay *= config.backoffMultiplier;
      }
    }

    throw lastError;
  }

  /**
   * 判断是否应该重试
   * 
   * @param {String} errorCode - 错误代码
   * @param {Number} attempt - 当前尝试次数
   * @param {Object} config - 重试配置
   * @returns {Boolean} - 是否应该重试
   */
  async shouldRetry(errorCode, attempt, config) {
    const retryableErrors = ['TIMEOUT_ERROR', 'NETWORK_ERROR', 'RATE_LIMIT_ERROR'];
    
    if (!retryableErrors.includes(errorCode)) {
      return false;
    }
    
    if (attempt >= config.maxRetries) {
      return false;
    }
    
    return true;
  }

  /**
   * 睡眠指定时间
   * 
   * @param {Number} ms - 毫秒数
   * @returns {Promise<void>}
   */
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  /**
   * 创建包装函数
   * 
   * @param {Function} fn - 要包装的函数
   * @param {String} toolName - 工具名称
   * @returns {Function} - 包装后的函数
   */
  wrapFunction(fn, toolName) {
    return async (...args) => {
      try {
        return await this.executeWithRetry(() => fn(...args));
      } catch (error) {
        const result = await this.handleError(error, {
          toolName: toolName,
          arguments: args,
        });
        throw new ToolError(result.error, result.code, result);
      }
    };
  }
}
```

```javascript
class ToolChain {
  constructor() {
    this.tools = [];
    this.context = {};
  }

  /**
   * 添加工具到链中
   * 
   * @param {BaseTool} tool - 工具实例
   * @param {Function} inputMapper - 输入映射函数
   * @param {Function} outputMapper - 输出映射函数
   * @returns {ToolChain} - 链实例
   */
  addTool(tool, inputMapper = null, outputMapper = null) {
    this.tools.push({
      tool: tool,
      inputMapper: inputMapper || ((ctx) => ctx),
      outputMapper: outputMapper || ((result) => result),
    });
    return this;
  }

  /**
   * 执行工具链
   * 
   * @param {Object} initialContext - 初始上下文
   * @returns {Promise<Object>} - 最终结果
   */
  async execute(initialContext = {}) {
    let currentContext = { ...initialContext };
    
    for (const { tool, inputMapper, outputMapper } of this.tools) {
      try {
        // 映射输入
        const inputs = inputMapper(currentContext);
        
        // 执行工具
        const result = await tool.execute(inputs);
        
        // 映射输出并更新上下文
        const mappedResult = outputMapper(result);
        currentContext = { ...currentContext, ...mappedResult };
        
        // 添加执行历史
        if (!currentContext.executionHistory) {
          currentContext.executionHistory = [];
        }
        currentContext.executionHistory.push({
          tool: tool.name,
          inputs: inputs,
          result: result,
          timestamp: new Date().toISOString(),
        });
        
      } catch (error) {
        console.error(`Tool ${tool.name} execution error:`, error);
        throw new Error(`工具链执行失败在 ${tool.name}: ${error.message}`);
      }
    }
    
    return currentContext;
  }

  /**
   * 并行执行工具
   * 
   * @param {Array<BaseTool>} tools - 工具数组
   * @param {Function} inputMapper - 输入映射函数
   * @param {Function} outputMapper - 输出映射函数
   * @returns {ToolChain} - 链实例
   */
  addParallelTools(tools, inputMapper = null, outputMapper = null) {
    this.tools.push({
      tools: tools,
      inputMapper: inputMapper || ((ctx) => ctx),
      outputMapper: outputMapper || ((results) => ({ parallelResults: results })),
      parallel: true,
    });
    return this;
  }

  /**
   * 条件执行
   * 
   * @param {Function} condition - 条件函数
   * @param {BaseTool} tool - 工具实例
   * @param {Function} inputMapper - 输入映射函数
   * @param {Function} outputMapper - 输出映射函数
   * @returns {ToolChain} - 链实例
   */
  addConditionalTool(condition, tool, inputMapper = null, outputMapper = null) {
    this.tools.push({
      tool: tool,
      condition: condition,
      inputMapper: inputMapper || ((ctx) => ctx),
      outputMapper: outputMapper || ((result) => result),
      conditional: true,
    });
    return this;
  }
}

ToolChain.prototype.execute = async function(initialContext = {}) {
  let currentContext = { ...initialContext };
  
  for (const step of this.tools) {
    try {
      if (step.parallel) {
        // 并行执行多个工具
        const inputs = step.inputMapper(currentContext);
        const promises = step.tools.map(tool => tool.execute(inputs));
        const results = await Promise.all(promises);
        const mappedResult = step.outputMapper(results);

      } else if (step.conditional) {
        // 条件执行
        if (step.condition(currentContext)) {
          const inputs = step.inputMapper(currentContext);
          const result = await step.tool.execute(inputs);
          const mappedResult = step.outputMapper(result);
          currentContext = { ...currentContext, ...mappedResult };

      } else {
        // 顺序执行
        const inputs = step.inputMapper(currentContext);
        const result = await step.tool.execute(inputs);
        const mappedResult = step.outputMapper(result);
        currentContext = { ...currentContext, ...mappedResult };
      }
      
      // 添加执行历史
```

```javascript
   * @param {String} name - 工具名称
   * @returns {BaseTool} - 工具实例
   */
  getTool(name) {
    return this.toolRegistry.get(name);
  }

  /**
   * 创建组合工具
   * 
   * @param {String} name - 组合工具名称
   * @param {Array} toolSequence - 工具序列
   * @returns {BaseTool} - 组合工具
   */
  createComposedTool(name, toolSequence) {
    class ComposedTool extends BaseTool {
      constructor(toolRegistry, sequence) {
        super();
        this.name = name;
        this.description = `组合工具: ${sequence.map(s => s.tool).join(' -> ')}`;
        this.toolRegistry = toolRegistry;
        this.sequence = sequence;
        this.parameters = this.inferParameters();
        this.required = this.inferRequiredParameters();
      }

      inferParameters() {
        // 从第一个工具推断参数
        const firstStep = this.sequence[0];
        const tool = this.toolRegistry.get(firstStep.tool);
        return tool ? tool.parameters : {};
      }

      inferRequiredParameters() {
        const firstStep = this.sequence[0];
        const tool = this.toolRegistry.get(firstStep.tool);
        return tool ? tool.required : [];
      }

      async execute(inputs) {
        let currentInputs = inputs;
        let results = {};

        for (const step of this.sequence) {
          const tool = this.toolRegistry.get(step.tool);
          if (!tool) {
            throw new Error(`工具 ${step.tool} 未找到`);
          }

          // 应用输入转换
          if (step.inputTransform) {
            currentInputs = step.inputTransform(currentInputs, results);
          }

          // 执行工具
          const result = await tool.execute(currentInputs);
          results[step.tool] = result;

          // 应用输出转换
          if (step.outputTransform) {
            results = step.outputTransform(result, results);
          }

          // 准备下一步的输入
          currentInputs = { ...currentInputs, ...result };
        }

        return this.formatOutput(results);
      }
    }

    return new ComposedTool(this.toolRegistry, toolSequence);
  }

  /**
   * 创建并行工具
   * 
   * @param {String} name - 并行工具名称
   * @param {Array} tools - 工具数组
   * @param {Function} resultMerger - 结果合并函数
   * @returns {BaseTool} - 并行工具
   */
  createParallelTool(name, tools, resultMerger = null) {
    class ParallelTool extends BaseTool {
      constructor(toolRegistry, toolNames, merger) {
        super();
        this.name = name;
        this.description = `并行工具: ${toolNames.join(', ')}`;
        this.toolRegistry = toolRegistry;
        this.toolNames = toolNames;
        this.resultMerger = merger || this.defaultMerger;
        this.parameters = this.mergeParameters();
        this.required = this.mergeRequiredParameters();
      }

      mergeParameters() {
        const merged = {};
        for (const toolName of this.toolNames) {
          const tool = this.toolRegistry.get(toolName);
          if (tool) {
            Object.assign(merged, tool.parameters);
          }
        }
        return merged;
      }

      mergeRequiredParameters() {
        const merged = [];
        for (const toolName of this.toolNames) {
          const tool = this.toolRegistry.get(toolName);
          if (tool) {
            merged.push(...tool.required);
          }
        }
        return [...new Set(merged)]; // 去重
      }

      defaultMerger(results) {
        return { parallelResults: results };
      }

      async execute(inputs) {
        const promises = this.toolNames.map(toolName => {
          const tool = this.toolRegistry.get(toolName);
          return tool ? tool.execute(inputs) : null;
        });

        const results = await Promise.all(promises);
        const mergedResult = this.resultMerger(results);

        return this.formatOutput(mergedResult);
      }
    }

    return new ParallelTool(this.toolRegistry, tools, resultMerger);
  }

  /**
   * 创建条件工具
   * 
   * @param {String} name - 条件工具名称
   * @param {Function} condition - 条件函数
   * @param {BaseTool} trueTool - 条件为真时执行的工具
   * @param {BaseTool} falseTool - 条件为假时执行的工具
   * @returns {BaseTool} - 条件工具
   */
  createConditionalTool(name, condition, trueTool, falseTool = null) {
    class ConditionalTool extends BaseTool {
      constructor(condition, trueTool, falseTool) {
        super();
        this.name = name;
        this.description = '条件执行工具';
        this.condition = condition;
        this.trueTool = trueTool;
        this.falseTool = falseTool;
        this.parameters = trueTool.parameters;
        this.required = trueTool.required;
      }

      async execute(inputs) {
        if (this.condition(inputs)) {
          return await this.trueTool.execute(inputs);
        } else if (this.falseTool) {
          return await this.falseTool.execute(inputs);
        } else {
          return this.formatOutput({ skipped: true });
        }
      }
    }

    return new ConditionalTool(condition, trueTool, falseTool);
  }
}

// 使用示例
async function toolComposerExample() {
  const composer = new ToolComposer();
  
  // 注册工具
  composer.registerTool('weather', new WeatherTool('api-key'));
  composer.registerTool('search', new GoogleSearchTool('api-key'));
  composer.registerTool('code', new CodeInterpreterTool());
  
  // 创建组合工具
  const weatherSearchTool = composer.createComposedTool('weather_search', [
    {
      tool: 'weather',
      inputTransform: (inputs) => ({ city: inputs.city }),
      outputTransform: (result) => ({ weatherData: result }),
    },
    {
      tool: 'search',
      inputTransform: (inputs, results) => ({
        query: `${inputs.city} 旅游攻略`,
      }),
      outputTransform: (result) => ({ searchData: result }),
    },
  ]);
  
  // 创建并行工具
  const parallelWeatherTool = composer.createParallelTool(
    'parallel_weather',
    ['weather', 'search'],
    (results) => ({
      weather: results[0],
      search: results[1],
    })
  );
  
  // 使用组合工具
  const composedResult = await weatherSearchTool.execute({ city: '北京' });
  console.log('组合工具结果:', composedResult);
  
  // 使用并行工具
  const parallelResult = await parallelWeatherTool.execute({ city: '上海' });
  console.log('并行工具结果:', parallelResult);
}
```
