# Agent 规划与执行

> **版本**：v1.2  
> **最后更新**：2026-04-16

## 📖 概述

规划与执行是 Agent 实现复杂任务的核心能力。通过将复杂任务分解为可执行的子任务，智能地安排执行顺序，并在执行过程中动态调整策略，Agent 能够高效地完成各种复杂目标。本章节将详细介绍 Agent 规划算法、执行策略和错误处理机制。

## 🎯 为什么需要规划与执行？

### 无规划 Agent 的局限性

1. **任务复杂性**
   - 无法处理复杂的多步骤任务
   - 缺乏任务分解能力
   - 难以处理依赖关系

2. **执行效率**
   - 串行执行导致效率低下
   - 无法利用并行机会
   - 资源利用率低

3. **容错能力**
   - 缺乏错误恢复机制
   - 无法处理失败情况
   - 缺少降级策略

### 规划与执行的价值

1. **任务分解**
   - 将复杂任务拆解为子任务
   - 明确任务依赖关系
   - 优化执行顺序

2. **高效执行**
   - 支持并行执行
   - 动态资源调度
   - 实时进度监控

3. **健壮性**
   - 自动错误恢复
   - 智能重试机制
   - 优雅降级策略

class TaskDecomposer {
  constructor() {
    this.decompositionStrategies = [];
  }

  /**
   * 分解任务
   * 
   * @param {Object} task - 任务对象
   * @returns {Promise<Array>} - 子任务数组
   */
  async decompose(task) {
    const decomposition = {
      originalTask: task,
      subtasks: [],
      dependencies: [],
      estimatedTime: 0,
      complexity: 'medium',
    };

    // 根据任务类型选择分解策略
    const strategy = this.selectDecompositionStrategy(task);
    
    if (strategy) {
      const result = await strategy.decompose(task);
      decomposition.subtasks = result.subtasks;
      decomposition.dependencies = result.dependencies;
      decomposition.estimatedTime = result.estimatedTime;
      decomposition.complexity = result.complexity;
    }

    return decomposition;
  }

  /**
   * 选择分解策略
   * 
   * @param {Object} task - 任务对象
   * @returns {Object} - 分解策略
   */
  selectDecompositionStrategy(task) {
    for (const strategy of this.decompositionStrategies) {
      if (strategy.canHandle(task)) {
        return strategy;
      }
    }
    return null;
  }

  /**
   * 添加分解策略
   * 
   * @param {Object} strategy - 分解策略
   */
  addDecompositionStrategy(strategy) {
    this.decompositionStrategies.push(strategy);
  }
}

class HierarchicalTaskDecomposer {
  constructor(maxDepth = 3) {
    this.maxDepth = maxDepth;
    this.decompositionRules = [];
  }

  /**
   * 层次化分解任务
   * 
   * @param {Object} task - 任务对象
   * @param {Number} depth - 当前深度
   * @returns {Promise<Object>} - 分解结果
   */
  async decompose(task, depth = 0) {
    if (depth >= this.maxDepth) {
      return {
        task: task,
        subtasks: [],
        isLeaf: true,
      };
    }

    const decomposition = {
      task: task,
      subtasks: [],
      isLeaf: false,
      depth: depth,
    };

    // 查找适用的分解规则
    const rule = this.findApplicableRule(task);
    
    if (rule) {
      const subtaskSpecs = await rule.apply(task);
      
      for (const spec of subtaskSpecs) {
        const subtask = {
          id: `${task.id}-${spec.id}`,
          description: spec.description,
          type: spec.type,
          priority: spec.priority || 'medium',
          estimatedTime: spec.estimatedTime || 60,
        };

        // 递归分解子任务
        const subtaskDecomposition = await this.decompose(subtask, depth + 1);
        decomposition.subtasks.push(subtaskDecomposition);
      }
    } else {
      decomposition.isLeaf = true;
    }

    return decomposition;
  }

  /**
   * 查找适用的分解规则
   * 
   * @param {Object} task - 任务对象
   * @returns {Object} - 分解规则
   */
  findApplicableRule(task) {
    for (const rule of this.decompositionRules) {
      if (rule.matches(task)) {
        return rule;
      }
    }
    return null;
  }

  /**
   * 添加分解规则
   * 
   * @param {Object} rule - 分解规则
   */
  addDecompositionRule(rule) {
    this.decompositionRules.push(rule);
  }

  /**
   * 获取所有叶子任务
   * 
   * @param {Object} decomposition - 分解结果
   * @returns {Array} - 叶子任务数组
   */
  getLeafTasks(decomposition) {
    if (decomposition.isLeaf) {
      return [decomposition.task];
    }

    const leafTasks = [];
    for (const subtask of decomposition.subtasks) {
      leafTasks.push(...this.getLeafTasks(subtask));
    }
    return leafTasks;
  }

  /**
   * 获取执行计划
   * 
   * @param {Object} decomposition - 分解结果
   * @returns {Array} - 执行计划
   */
  getExecutionPlan(decomposition) {
    const plan = [];
    this.buildExecutionPlan(decomposition, plan);
    return plan;
  }

  buildExecutionPlan(decomposition, plan, dependencies = []) {
    if (decomposition.isLeaf) {
      plan.push({
        task: decomposition.task,
        dependencies: [...dependencies],
      });
    } else {
      for (const subtask of decomposition.subtasks) {
        const newDependencies = [...dependencies];
        if (decomposition.task.id !== subtask.task.id) {
          newDependencies.push(decomposition.task.id);
        }
        this.buildExecutionPlan(subtask, plan, newDependencies);
      }
    }
  }
}

### 2. 子任务排序

#### 拓扑排序

```javascript
class TaskScheduler {
  constructor() {
    this.tasks = new Map();
    this.dependencies = [];
  }

  /**
   * 添加任务
   * 
   * @param {Object} task - 任务对象
   */
  addTask(task) {
    this.tasks.set(task.id, task);
  }

  /**
   * 添加依赖关系
   * 
   * @param {String} from - 依赖任务 ID
   * @param {String} to - 被依赖任务 ID
   */
  addDependency(from, to) {
    this.dependencies.push({ from, to });
  }

  /**
   * 拓扑排序
   * 
   * @returns {Array} - 排序后的任务 ID 数组
   */
  topologicalSort() {
    const inDegree = new Map();
    const adjacencyList = new Map();
    
    // 初始化
    for (const taskId of this.tasks.keys()) {
      inDegree.set(taskId, 0);
      adjacencyList.set(taskId, []);
    }
    
    // 构建邻接表和入度
    for (const dep of this.dependencies) {
      adjacencyList.get(dep.from).push(dep.to);
      inDegree.set(dep.to, (inDegree.get(dep.to) || 0) + 1);
    }
    
    // Kahn 算法
    const queue = [];
    const result = [];
    
    // 找到入度为 0 的节点
    for (const [taskId, degree] of inDegree.entries()) {
      if (degree === 0) {
        queue.push(taskId);
      }
    }
    
    while (queue.length > 0) {
      const current = queue.shift();
      result.push(current);
      
      // 减少邻居的入度
      for (const neighbor of adjacencyList.get(current)) {
        inDegree.set(neighbor, inDegree.get(neighbor) - 1);
        
        if (inDegree.get(neighbor) === 0) {
          queue.push(neighbor);
        }
      }
    }
    
    // 检查是否有环
    if (result.length !== this.tasks.size) {
      throw new Error('任务依赖关系中存在循环依赖');
    }
    
    return result;
  }

  /**
   * 基于优先级的排序
   * 
   * @returns {Array} - 排序后的任务数组
   */
  prioritySort() {
    const sortedTasks = Array.from(this.tasks.values());
    
    const priorityOrder = {
      'critical': 0,
      'high': 1,
      'medium': 2,
      'low': 3,
    };
    
    sortedTasks.sort((a, b) => {
      const priorityA = priorityOrder[a.priority] || 2;
      const priorityB = priorityOrder[b.priority] || 2;
      
      if (priorityA !== priorityB) {
        return priorityA - priorityB;
      }
      
      // 优先级相同，按估计时间排序
      return (a.estimatedTime || 0) - (b.estimatedTime || 0);
    });
    
    return sortedTasks;
  }

  /**
   * 混合排序（拓扑 + 优先级）
   * 
   * @returns {Array} - 排序后的任务数组
   */
  hybridSort() {
    try {
      // 先进行拓扑排序
      const topologicalOrder = this.topologicalSort();
      
      // 按拓扑顺序分组
      const levels = this.groupByLevel(topologicalOrder);
      
      // 在每个层级内按优先级排序
      const result = [];
      for (const level of levels) {
        const levelTasks = level.map(id => this.tasks.get(id));
        const sortedLevel = this.sortByPriority(levelTasks);
        result.push(...sortedLevel);
      }
      
      return result;
    } catch (error) {
      // 如果拓扑排序失败（有环），使用优先级排序
      console.warn('拓扑排序失败，使用优先级排序:', error.message);
      return this.prioritySort();
    }
  }

  /**
   * 按层级分组
   * 
   * @param {Array} order - 拓扑顺序
   * @returns {Array} - 层级数组
   */
  groupByLevel(order) {
    const levels = [];
    const inLevel = new Set();
    
    for (const taskId of order) {
      const dependencies = this.getDependencies(taskId);
      
      // 检查所有依赖是否已经在某个层级中
      const dependenciesPlaced = dependencies.every(depId => 
        inLevel.has(depId)
      );
      
      if (dependenciesPlaced) {
        // 找到可以放置的层级
        let placed = false;
        for (const level of levels) {
          if (!level.includes(taskId)) {
            level.push(taskId);
            inLevel.add(taskId);
            placed = true;
            break;
          }
        }
        
        if (!placed) {
          levels.push([taskId]);
          inLevel.add(taskId);
        }
      }
    }
    
    return levels;
  }

  /**
   * 获取任务的依赖
   * 
   * @param {String} taskId - 任务 ID
   * @returns {Array} - 依赖任务 ID 数组
   */
  getDependencies(taskId) {
    const dependencies = [];
    
    for (const dep of this.dependencies) {
      if (dep.to === taskId) {
        dependencies.push(dep.from);
      }
    }
    
    return dependencies;
  }

  /**
   * 按优先级排序
   * 
   * @param {Array} tasks - 任务数组
   * @returns {Array} - 排序后的任务数组
   */
  sortByPriority(tasks) {
    const priorityOrder = {
      'critical': 0,
      'high': 1,
      'medium': 2,
      'low': 3,
    };
    
    return tasks.sort((a, b) => {
      const priorityA = priorityOrder[a.priority] || 2;
      const priorityB = priorityOrder[b.priority] || 2;
      return priorityA - priorityB;
    });
  }

  /**
   * 检测循环依赖
   * 
   * @returns {Boolean} - 是否存在循环依赖
   */
  hasCycle() {
    try {
      this.topologicalSort();
      return false;
    } catch (error) {
      return true;
    }
  }

  /**
   * 获取关键路径
   * 
   * @returns {Array} - 关键路径上的任务 ID 数组
   */
  getCriticalPath() {
    // 构建任务图
    const graph = this.buildTaskGraph();
    
    // 计算最早开始时间
    const earliestStart = this.calculateEarliestStart(graph);
    
    // 计算最晚开始时间
    const latestStart = this.calculateLatestStart(graph, earliestStart);
    
    // 找出关键路径上的任务
    const criticalPath = [];
    for (const [taskId, earliest] of earliestStart.entries()) {
      const latest = latestStart.get(taskId);
      if (earliest === latest) {
        criticalPath.push(taskId);
      }
    }
    
    return criticalPath;
  }

  buildTaskGraph() {
    const graph = new Map();
    
    for (const taskId of this.tasks.keys()) {
      graph.set(taskId, {
        task: this.tasks.get(taskId),
        dependencies: [],
        dependents: [],
      });
    }
    
    for (const dep of this.dependencies) {
      const fromNode = graph.get(dep.from);
      const toNode = graph.get(dep.to);
      
      fromNode.dependents.push(dep.to);
      toNode.dependencies.push(dep.from);
    }
    
    return graph;
  }

  calculateEarliestStart(graph) {
    const earliestStart = new Map();
    
    // 初始化
    for (const [taskId, node] of graph.entries()) {
      if (node.dependencies.length === 0) {
        earliestStart.set(taskId, 0);
      } else {
        earliestStart.set(taskId, -Infinity);
      }
    }
    
    // 拓扑顺序计算
    const visited = new Set();
    const queue = [];
    
    for (const [taskId, start] of earliestStart.entries()) {
      if (start === 0) {
        queue.push(taskId);
      }
    }
    
    while (queue.length > 0) {
      const current = queue.shift();
      visited.add(current);
      
      const currentNode = graph.get(current);
      const currentStart = earliestStart.get(current);
      const currentDuration = currentNode.task.estimatedTime || 0;
      const currentFinish = currentStart + currentDuration;
      
      for (const dependentId of currentNode.dependents) {
        const dependentNode = graph.get(dependentId);
        const maxStart = Math.max(
          earliestStart.get(dependentId),
          currentFinish
        );
        earliestStart.set(dependentId, maxStart);
        
        if (!visited.has(dependentId)) {
          queue.push(dependentId);
        }
      }
    }
    
    return earliestStart;
  }

  calculateLatestStart(graph, earliestStart) {
    const latestStart = new Map();
    
    // 找到最大完成时间
    const maxFinish = Math.max(...earliestStart.values());
    
    // 初始化
    for (const [taskId, node] of graph.entries()) {
      if (node.dependents.length === 0) {
        const duration = node.task.estimatedTime || 0;
        latestStart.set(taskId, maxFinish - duration);
      } else {
        latestStart.set(taskId, Infinity);
      }
    }
    
    // 反向拓扑顺序计算
    const visited = new Set();
    const queue = [];
    
    for (const [taskId, start] of latestStart.entries()) {
      if (start !== Infinity) {
        queue.push(taskId);
      }
    }
    
    while (queue.length > 0) {
      const current = queue.shift();
      visited.add(current);
      
      const currentNode = graph.get(current);
      const currentLatest = latestStart.get(current);
      
      for (const dependencyId of currentNode.dependencies) {
        const dependencyNode = graph.get(dependencyId);
        const duration = dependencyNode.task.estimatedTime || 0;
        const minStart = Math.min(
          latestStart.get(dependencyId),
          currentLatest - duration
        );
        latestStart.set(dependencyId, minStart);
        
        if (!visited.has(dependencyId)) {
          queue.push(dependencyId);
        }
      }
    }
    
    return latestStart;
  }
}

class DependencyManager {
  constructor() {
    this.dependencies = new Map();
    this.cycles = [];
  }

  /**
   * 添加依赖关系
   * 
   * @param {String} from - 依赖任务 ID
   * @param {String} to - 被依赖任务 ID
   * @param {Object} metadata - 元数据
   */
  addDependency(from, to, metadata = {}) {
    if (!this.dependencies.has(from)) {
      this.dependencies.set(from, new Set());
    }
    
    this.dependencies.get(from).add(to);
    
    // 检查是否产生循环依赖
    if (this.detectCycle()) {
      this.dependencies.get(from).delete(to);
      throw new Error(`添加依赖 ${from} -> ${to} 会产生循环依赖`);
    }
  }

  /**
   * 移除依赖关系
   * 
   * @param {String} from - 依赖任务 ID
   * @param {String} to - 被依赖任务 ID
   */
  removeDependency(from, to) {
    if (this.dependencies.has(from)) {
      this.dependencies.get(from).delete(to);
    }
  }

  /**
   * 获取直接依赖
   * 
   * @param {String} taskId - 任务 ID
   * @returns {Set} - 直接依赖集合
   */
  getDirectDependencies(taskId) {
    return this.dependencies.get(taskId) || new Set();
  }

  /**
   * 获取所有依赖（传递闭包）
   * 
   * @param {String} taskId - 任务 ID
   * @returns {Set} - 所有依赖集合
   */
  getAllDependencies(taskId) {
    const allDependencies = new Set();
    const visited = new Set();
    
    this.collectDependencies(taskId, allDependencies, visited);
    
    return allDependencies;
  }

  /**
   * 收集依赖
   * 
   * @param {String} taskId - 任务 ID
   * @param {Set} dependencies - 依赖集合
   * @param {Set} visited - 已访问集合
   */
  collectDependencies(taskId, dependencies, visited) {
    if (visited.has(taskId)) {
      return;
    }
    
    visited.add(taskId);
    
    const directDeps = this.getDirectDependencies(taskId);
    for (const depId of directDeps) {
      dependencies.add(depId);
      this.collectDependencies(depId, dependencies, visited);
    }
  }

  /**
   * 获取依赖者
   * 
   * @param {String} taskId - 任务 ID
   * @returns {Set} - 依赖者集合
   */
  getDependents(taskId) {
    const dependents = new Set();
    
    for (const [from, deps] of this.dependencies.entries()) {
      if (deps.has(taskId)) {
        dependents.add(from);
      }
    }
    
    return dependents;
  }

  /**
   * 检测循环依赖
   * 
   * @returns {Boolean} - 是否存在循环依赖
   */
  detectCycle() {
    const visited = new Set();
    const recursionStack = new Set();
    const cycles = [];
    
    for (const taskId of this.dependencies.keys()) {
      if (!visited.has(taskId)) {
        this.detectCycleDFS(taskId, visited, recursionStack, cycles);
      }
    }
    
    this.cycles = cycles;
    return cycles.length > 0;
  }

  /**
   * 深度优先搜索检测循环
   * 
   * @param {String} taskId - 任务 ID
   * @param {Set} visited - 已访问集合
   * @param {Set} recursionStack - 递归栈
   * @param {Array} cycles - 循环数组
   */
  detectCycleDFS(taskId, visited, recursionStack, cycles) {
    visited.add(taskId);
    recursionStack.add(taskId);
    
    const dependencies = this.getDirectDependencies(taskId);
    for (const depId of dependencies) {
      if (!visited.has(depId)) {
        this.detectCycleDFS(depId, visited, recursionStack, cycles);
      } else if (recursionStack.has(depId)) {
        // 发现循环
        cycles.push([...recursionStack, depId]);
      }
    }
    
    recursionStack.delete(taskId);
  }

  /**
   * 获取循环依赖
   * 
   * @returns {Array} - 循环依赖数组
   */
  getCycles() {
    return this.cycles;
  }

  /**
   * 解决循环依赖
   * 
   * @param {Array} cycle - 循环路径
   * @returns {String} - 被移除的依赖
   */
  resolveCycle(cycle) {
    // 移除循环中的最后一个依赖
    const lastDep = cycle[cycle.length - 2];
    const current = cycle[cycle.length - 1];
    
    this.removeDependency(lastDep, current);
    
    return `${lastDep} -> ${current}`;
  }

  /**
   * 验证依赖关系
   * 
   * @param {Array} taskIds - 所有任务 ID
   * @returns {Object} - 验证结果
   */
  validate(taskIds) {
    const result = {
      valid: true,
      errors: [],
      warnings: [],
    };
    
    // 检查循环依赖
    if (this.detectCycle()) {
      result.valid = false;
      result.errors.push({
        type: 'cycle',
        message: '存在循环依赖',
        cycles: this.getCycles(),
      });
    }
    
    // 检查孤立任务
    const connected = new Set();
    for (const [from, deps] of this.dependencies.entries()) {
      connected.add(from);
      for (const dep of deps) {
        connected.add(dep);
      }
    }
    
    const isolated = taskIds.filter(id => !connected.has(id));
    if (isolated.length > 0) {
      result.warnings.push({
        type: 'isolated',
        message: '存在孤立任务',
        tasks: isolated,
      });
    }
    
    return result;
  }

  /**
   * 可视化依赖关系
   * 
   * @returns {String} - 依赖关系图
   */
  visualize() {
    let graph = '';
    
    for (const [from, deps] of this.dependencies.entries()) {
      for (const to of deps) {
        graph += `${from} -> ${to}\n`;
      }
    }
    
    return graph;
  }

  /**
   * 导出为 JSON
   * 
   * @returns {Object} - JSON 对象
   */
  toJSON() {
    const json = {};
    
    for (const [from, deps] of this.dependencies.entries()) {
      json[from] = Array.from(deps);
    }
    
    return json;
  }

  /**
   * 从 JSON 导入
   * 
   * @param {Object} json - JSON 对象
   */
  fromJSON(json) {
    this.dependencies.clear();
    
    for (const [from, deps] of Object.entries(json)) {
      for (const to of deps) {
        this.addDependency(from, to);
      }
    }
  }

  /**
   * 清空所有依赖
   */
  clear() {
    this.dependencies.clear();
    this.cycles = [];
  }
}

      totalTasks: this.tasks.length,
      successCount: results.filter(r => r.success).length,

      results: results,
      duration: endTime - startTime,
    };
  }

   * 执行单个任务

   * @param {Object} task - 任务对象

        }
      }
    }
    
    // 执行任务

  getResult(taskId) {

class ParallelExecutor {
  constructor(maxConcurrency = 4) {
    this.maxConcurrency = maxConcurrency;
    this.tasks = [];
    this.results = new Map();
    this.errors = new Map();
    this.runningTasks = 0;
  }

  /**
   * 添加任务
   * 
   * @param {Object} task - 任务对象
   */
  addTask(task) {
    this.tasks.push(task);
  }

  /**
   * 执行所有任务
   * 
   * @returns {Promise<Object>} - 执行结果
   */
  async executeAll() {
    const startTime = Date.now();
    const results = [];
    const taskQueue = [...this.tasks];
    const executing = new Set();

    const executeNext = async () => {
      if (taskQueue.length === 0 || executing.size >= this.maxConcurrency) {
        return;
      }

      const task = taskQueue.shift();
      const taskPromise = this.executeTask(task).then(
        result => {
          results.push({
            taskId: task.id,
            success: true,
            result: result,
          });
          this.results.set(task.id, result);
        },
        error => {
          results.push({
            taskId: task.id,
            success: false,
            error: error.message,
          });
          this.errors.set(task.id, error);
        }
      ).finally(() => {
        executing.delete(taskPromise);
      });

      executing.add(taskPromise);
      await executeNext();
    };

    // 启动初始任务
    const initialPromises = [];
    for (let i = 0; i < this.maxConcurrency && taskQueue.length > 0; i++) {
      initialPromises.push(executeNext());
    }

    await Promise.all(initialPromises);

    // 等待所有任务完成
    await Promise.all(executing);

    const endTime = Date.now();

    return {
      totalTasks: this.tasks.length,
      successCount: results.filter(r => r.success).length,
      failureCount: results.filter(r => !r.success).length,
      results: results,
      duration: endTime - startTime,
    };
  }

  /**
   * 执行单个任务
   * 
   * @param {Object} task - 任务对象
   * @returns {Promise<*>} - 任务结果
   */
  async executeTask(task) {
    console.log(`执行任务: ${task.description}`);
    
    // 检查任务依赖
    if (task.dependencies && task.dependencies.length > 0) {
      for (const depId of task.dependencies) {
        if (!this.results.has(depId)) {
          throw new Error(`依赖任务 ${depId} 未完成`);
        }
      }
    }
    
    // 执行任务
    const result = await task.execute();
    
    console.log(`任务完成: ${task.description}`);
    
    return result;
  }

  /**
   * 执行任务组
   * 
   * @param {Array} taskGroup - 任务组
   * @returns {Promise<Array>} - 结果数组
   */
  async executeGroup(taskGroup) {
    const promises = taskGroup.map(task => 
      this.executeTask(task).catch(error => ({
        taskId: task.id,
        success: false,
        error: error.message,
      }))
    );

    return Promise.all(promises);
  }

  /**
   * 批量执行
   * 
   * @param {Array} tasks - 任务数组
   * @param {Number} batchSize - 批次大小
   * @returns {Promise<Object>} - 执行结果
   */
  async executeBatch(tasks, batchSize = 10) {
    const results = [];
    const startTime = Date.now();

    for (let i = 0; i < tasks.length; i += batchSize) {
      const batch = tasks.slice(i, i + batchSize);
      const batchResults = await this.executeGroup(batch);
      results.push(...batchResults);
    }

    const endTime = Date.now();

    return {
      totalTasks: tasks.length,
      successCount: results.filter(r => r.success).length,
      failureCount: results.filter(r => !r.success).length,
      results: results,
      duration: endTime - startTime,
    };
  }

  /**
   * 获取进度
   * 
   * @returns {Object} - 进度信息
   */
  getProgress() {
    const completed = this.results.size + this.errors.size;
    const total = this.tasks.length;
    
    return {
      completed: completed,
      total: total,
      percentage: total > 0 ? (completed / total) * 100 : 0,
      running: this.runningTasks,
    };
  }

  /**
   * 重置执行器
   */
  reset() {
    this.tasks = [];
    this.results.clear();
    this.errors.clear();
    this.runningTasks = 0;
  }
}

class DynamicExecutor {
  constructor(config = {}) {
    this.maxConcurrency = config.maxConcurrency || 4;
    this.adaptiveConcurrency = config.adaptiveConcurrency || false;
    this.tasks = [];
    this.results = new Map();
    this.errors = new Map();
    this.performanceMetrics = {
      averageExecutionTime: 0,
      successRate: 1.0,
      resourceUsage: 0.5,
    };
  }

  /**
   * 添加任务
   * 
   * @param {Object} task - 任务对象
   */
  addTask(task) {
    this.tasks.push(task);
  }

  /**
   * 执行所有任务（动态调整）
   * 
   * @returns {Promise<Object>} - 执行结果
   */
  async executeAll() {
    const startTime = Date.now();
    const results = [];
    const taskQueue = [...this.tasks];
    const executing = new Set();
    let currentConcurrency = this.maxConcurrency;

    const executeNext = async () => {
      if (taskQueue.length === 0) {
        return;
      }

      // 动态调整并发数
      if (this.adaptiveConcurrency) {
        currentConcurrency = this.calculateOptimalConcurrency();
      }

      if (executing.size >= currentConcurrency) {
        return;
      }

      const task = taskQueue.shift();
      const taskStartTime = Date.now();

      const taskPromise = this.executeTask(task).then(
        result => {
          const executionTime = Date.now() - taskStartTime;
          this.updatePerformanceMetrics(executionTime, true);
          
          results.push({
            taskId: task.id,
            success: true,
            result: result,
            executionTime: executionTime,
          });
          this.results.set(task.id, result);
        },
        error => {
          const executionTime = Date.now() - taskStartTime;
          this.updatePerformanceMetrics(executionTime, false);
          
          results.push({
            taskId: task.id,
            success: false,
            error: error.message,
            executionTime: executionTime,
          });
          this.errors.set(task.id, error);
        }
      ).finally(() => {
        executing.delete(taskPromise);
      });

      executing.add(taskPromise);
      await executeNext();
    };

    // 启动任务
    const initialPromises = [];
    for (let i = 0; i < currentConcurrency && taskQueue.length > 0; i++) {
      initialPromises.push(executeNext());
    }

    await Promise.all(initialPromises);
    await Promise.all(executing);

    const endTime = Date.now();

    return {
      totalTasks: this.tasks.length,
      successCount: results.filter(r => r.success).length,
      failureCount: results.filter(r => !r.success).length,
      results: results,
      duration: endTime - startTime,
      performanceMetrics: this.performanceMetrics,
    };
  }

  /**
   * 计算最优并发数
   * 
   * @returns {Number} - 最优并发数
   */
  calculateOptimalConcurrency() {
    const { averageExecutionTime, successRate, resourceUsage } = this.performanceMetrics;
    
    // 基于性能指标动态调整
    let optimalConcurrency = this.maxConcurrency;
    
    // 如果平均执行时间很长，降低并发数
    if (averageExecutionTime > 5000) {
      optimalConcurrency = Math.max(1, Math.floor(this.maxConcurrency * 0.5));
    }
    
    // 如果成功率低，降低并发数
    if (successRate < 0.8) {
      optimalConcurrency = Math.max(1, Math.floor(this.maxConcurrency * 0.7));
    }
    
    // 如果资源使用率高，降低并发数
    if (resourceUsage > 0.8) {
      optimalConcurrency = Math.max(1, Math.floor(this.maxConcurrency * 0.6));
    }
    
    return optimalConcurrency;
  }

  /**
   * 更新性能指标
   * 
   * @param {Number} executionTime - 执行时间
   * @param {Boolean} success - 是否成功
   */
  updatePerformanceMetrics(executionTime, success) {
    // 更新平均执行时间
    const completedTasks = this.results.size + this.errors.size;
    this.performanceMetrics.averageExecutionTime = 
      (this.performanceMetrics.averageExecutionTime * (completedTasks - 1) + executionTime) / completedTasks;
    
    // 更新成功率
    const successCount = this.results.size;
    this.performanceMetrics.successRate = successCount / completedTasks;
    
    // 更新资源使用（基于实际系统指标）
    this.performanceMetrics.resourceUsage = this.calculateResourceUsage();
  }

  /**
   * 计算资源使用率
   * 
   * @returns {Number} - 资源使用率 (0-1)
   */
  calculateResourceUsage() {
    // 基于多个因素计算资源使用率
    const cpuUsage = this.getCPUUsage();
    const memoryUsage = this.getMemoryUsage();
    const networkUsage = this.getNetworkUsage();
    
    // 加权平均
    return (cpuUsage * 0.4 + memoryUsage * 0.4 + networkUsage * 0.2);
  }

  /**
   * 获取 CPU 使用率
   * 
   * @returns {Number} - CPU 使用率 (0-1)
   */
  getCPUUsage() {
    // 在实际环境中，可以使用系统监控库获取真实 CPU 使用率
    // 这里基于并发任务数进行估算
    const runningTasks = this.results.size + this.errors.size;
    const usage = runningTasks / (this.maxConcurrency * 1.5);
    return Math.min(usage, 1.0);
  }

  /**
   * 获取内存使用率
   * 
   * @returns {Number} - 内存使用率 (0-1)
   */
  getMemoryUsage() {
    // 在实际环境中，可以使用系统监控库获取真实内存使用率
    // 这里基于任务结果大小进行估算
    let totalSize = 0;
    
    for (const result of this.results.values()) {
      totalSize += JSON.stringify(result).length;
    }
    
    // 假设最大内存使用为 10MB
    const usage = totalSize / (10 * 1024 * 1024);
    return Math.min(usage, 1.0);
  }

  /**
   * 获取网络使用率
   * 
   * @returns {Number} - 网络使用率 (0-1)
   */
  getNetworkUsage() {
    // 在实际环境中，可以使用网络监控库获取真实网络使用率
    // 这里基于任务类型进行估算
    let networkTasks = 0;
    const totalTasks = this.tasks.length;
    
    for (const task of this.tasks) {
      if (task.type === 'api_call' || task.type === 'download' || task.type === 'upload') {
        networkTasks++;
      }
    }
    
    const usage = totalTasks > 0 ? networkTasks / totalTasks : 0;
    return Math.min(usage, 1.0);
  }

  /**
   * 重新调度任务
   * 
   * @param {Array} failedTasks - 失败任务数组
   */
  async reschedule(failedTasks) {
    console.log(`重新调度 ${failedTasks.length} 个失败任务`);
    
    // 根据失败原因调整任务
    const rescheduledTasks = failedTasks.map(task => {
      return {
        ...task,
        retryCount: (task.retryCount || 0) + 1,
        priority: this.increasePriority(task.priority),
      };
    });
    
    // 重新执行
    const executor = new DynamicExecutor({
      maxConcurrency: Math.max(1, Math.floor(this.maxConcurrency * 0.5)),
      adaptiveConcurrency: this.adaptiveConcurrency,
    });
    
    rescheduledTasks.forEach(task => executor.addTask(task));
    
    return await executor.executeAll();
  }

  /**
   * 提高优先级
   * 
   * @param {String} priority - 当前优先级
   * @returns {String} - 提高后的优先级
   */
  increasePriority(priority) {
    const priorityOrder = ['low', 'medium', 'high', 'critical'];
    const currentIndex = priorityOrder.indexOf(priority);
    
    if (currentIndex < priorityOrder.length - 1) {
      return priorityOrder[currentIndex + 1];
    }
    
    return priority;
  }

  /**
   * 执行单个任务
   * 
   * @param {Object} task - 任务对象
   * @returns {Promise<*>} - 任务结果
   */
  async executeTask(task) {
    console.log(`执行任务: ${task.description} (优先级: ${task.priority})`);
    
    // 检查任务依赖
    if (task.dependencies && task.dependencies.length > 0) {
      for (const depId of task.dependencies) {
        if (!this.results.has(depId)) {
          throw new Error(`依赖任务 ${depId} 未完成`);
        }
      }
    }
    
    // 执行任务
    const result = await task.execute();
    
    console.log(`任务完成: ${task.description}`);
    
    return result;
  }

  /**
   * 获取实时状态
   * 
   * @returns {Object} - 实时状态
   */
  getStatus() {
    const completed = this.results.size + this.errors.size;
    const total = this.tasks.length;
    
    return {
      progress: {
        completed: completed,
        total: total,
        percentage: total > 0 ? (completed / total) * 100 : 0,
      },
      performance: this.performanceMetrics,
      concurrency: this.calculateOptimalConcurrency(),
    };
  }

  /**
   * 重置执行器
   */
  reset() {
    this.tasks = [];
    this.results.clear();
    this.errors.clear();
    this.performanceMetrics = {
      averageExecutionTime: 0,
      successRate: 1.0,
      resourceUsage: 0.5,
    };
  }
}

class RetryManager {
  constructor(config = {}) {
    this.maxRetries = config.maxRetries || 3;
    this.retryDelay = config.retryDelay || 1000;
    this.backoffMultiplier = config.backoffMultiplier || 2;
    this.retryableErrors = config.retryableErrors || [
      'NETWORK_ERROR',
      'TIMEOUT_ERROR',
      'RATE_LIMIT_ERROR',
      'TEMPORARY_ERROR',
    ];
  }

  /**
   * 执行带重试的任务
   * 
   * @param {Function} task - 任务函数
   * @param {Object} options - 选项
   * @returns {Promise<*>} - 任务结果
   */
  async executeWithRetry(task, options = {}) {
    const maxRetries = options.maxRetries || this.maxRetries;
    const retryDelay = options.retryDelay || this.retryDelay;
    const backoffMultiplier = options.backoffMultiplier || this.backoffMultiplier;
    
    let lastError;
    let currentDelay = retryDelay;
    
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        return await task();
      } catch (error) {
        lastError = error;
        
        // 检查是否应该重试
        if (!this.shouldRetry(error, attempt, maxRetries)) {
          throw error;
        }
        
        console.log(`任务失败，第 ${attempt + 1} 次重试，等待 ${currentDelay}ms`);
        
        // 等待后重试
        await this.sleep(currentDelay);
        currentDelay *= backoffMultiplier;
      }
    }
    
    throw lastError;
  }

  /**
   * 判断是否应该重试
   * 
   * @param {Error} error - 错误对象
   * @param {Number} attempt - 当前尝试次数
   * @param {Number} maxRetries - 最大重试次数
   * @returns {Boolean} - 是否应该重试
   */
  shouldRetry(error, attempt, maxRetries) {
    // 检查错误类型
    const errorCode = this.getErrorCode(error);
    if (!this.retryableErrors.includes(errorCode)) {
      return false;
    }
    
    // 检查重试次数
    if (attempt >= maxRetries) {
      return false;
    }
    
    return true;
  }

  /**
   * 获取错误代码
   * 
   * @param {Error} error - 错误对象
   * @returns {String} - 错误代码
   */
  getErrorCode(error) {
    if (error.code) {
      return error.code;
    }
    
    // 根据错误消息推断错误类型
    const message = error.message.toLowerCase();
    
    if (message.includes('timeout') || message.includes('超时')) {
      return 'TIMEOUT_ERROR';
    }
    
    if (message.includes('network') || message.includes('网络')) {
      return 'NETWORK_ERROR';
    }
    
    if (message.includes('rate limit') || message.includes('限流')) {
      return 'RATE_LIMIT_ERROR';
    }
    
    return 'UNKNOWN_ERROR';
  }

  /**
   * 睡眠指定时间
   * 
   * @param {Number} ms - 毫秒数
   * @returns {Promise<void>}
   */
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  /**
   * 添加可重试错误类型
   * 
   * @param {String} errorType - 错误类型
   */
  addRetryableError(errorType) {
    if (!this.retryableErrors.includes(errorType)) {
      this.retryableErrors.push(errorType);
    }
  }

  /**
   * 移除可重试错误类型
   * 
   * @param {String} errorType - 错误类型
   */
  removeRetryableError(errorType) {
    const index = this.retryableErrors.indexOf(errorType);
    if (index > -1) {
      this.retryableErrors.splice(index, 1);
    }
  }
}

class RollbackManager {

    this.currentCheckpoint = null;
  }

  /**
   * 创建检查点
   * 
   * @param {String} checkpointId - 检查点 ID
   * @param {Object} state - 状态对象
   */
  createCheckpoint(checkpointId, state) {
    const checkpoint = {
      id: checkpointId,
      state: JSON.parse(JSON.stringify(state)), // 深拷贝
      timestamp: Date.now(),
    };
    
    this.checkpoints.set(checkpointId, checkpoint);
    this.currentCheckpoint = checkpointId;
    
    console.log(`创建检查点: ${checkpointId}`);
  }

  /**
   * 回滚到检查点
   * 
   * @param {String} checkpointId - 检查点 ID
   * @returns {Object} - 恢复的状态
   */
  rollback(checkpointId) {
    const checkpoint = this.checkpoints.get(checkpointId);
    
    if (!checkpoint) {
      throw new Error(`检查点 ${checkpointId} 不存在`);
    }
    
    console.log(`回滚到检查点: ${checkpointId}`);
    
    this.currentCheckpoint = checkpointId;
    
    return JSON.parse(JSON.stringify(checkpoint.state));
  }

  /**
   * 回滚到上一个检查点

  rollbackToPrevious() {

    const currentIndex = checkpoints.indexOf(this.currentCheckpoint);
    
    if (currentIndex <= 0) {
      throw new Error('没有上一个检查点');
    }
    
    const previousCheckpointId = checkpoints[currentIndex - 1];
    return this.rollback(previousCheckpointId);
  }

  /**
   * 删除检查点
   * 
   * @param {String} checkpointId - 检查点 ID
   */
  deleteCheckpoint(checkpointId) {

  getCheckpoints() {

class DegradationManager {
  constructor() {
    this.strategies = new Map();
    this.currentLevel = 'normal';
    this.levels = ['normal', 'degraded', 'minimal'];
  }

  /**
   * 注册降级策略
   * 
   * @param {String} name - 策略名称
   * @param {Object} strategy - 策略对象
   */
  registerStrategy(name, strategy) {
    this.strategies.set(name, {
      ...strategy,
      currentLevel: 'normal',
    });
  }

  /**
   * 执行策略
   * 
   * @param {String} name - 策略名称
   * @param {Object} context - 上下文对象
   * @returns {Promise<*>} - 执行结果
   */
  async execute(name, context = {}) {
    const strategy = this.strategies.get(name);
    
    if (!strategy) {
      throw new Error(`策略 ${name} 不存在`);
    }
    
    // 根据当前降级级别选择实现
    const implementation = strategy[this.currentLevel];
    
    if (!implementation) {
      throw new Error(`策略 ${name} 在级别 ${this.currentLevel} 没有实现`);
    }
    
    return await implementation(context);
  }

  /**
   * 降级到指定级别
   * 
   * @param {String} level - 降级级别
   */
  degradeTo(level) {
    const levelIndex = this.levels.indexOf(level);
    
    if (levelIndex === -1) {
      throw new Error(`无效的降级级别: ${level}`);
    }
    
    const currentIndex = this.levels.indexOf(this.currentLevel);
    
    if (levelIndex < currentIndex) {
      console.log(`降级从 ${this.currentLevel} 到 ${level}`);
      this.currentLevel = level;
    }
  }

  /**
   * 升级到指定级别
   * 
   * @param {String} level - 升级级别
   */
  upgradeTo(level) {
    const levelIndex = this.levels.indexOf(level);
    
    if (levelIndex === -1) {
      throw new Error(`无效的升级级别: ${level}`);
    }
    
    const currentIndex = this.levels.indexOf(this.currentLevel);
    
    if (levelIndex > currentIndex) {
      console.log(`升级从 ${this.currentLevel} 到 ${level}`);
      this.currentLevel = level;
    }
  }

  /**
   * 恢复正常级别
   */
  restoreNormal() {
    this.upgradeTo('normal');
  }

  /**
   * 获取当前级别
   * 
   * @returns {String} - 当前级别
   */
  getCurrentLevel() {
    return this.currentLevel;
  }

  /**
   * 监控并自动调整
   * 
   * @param {Function} healthCheck - 健康检查函数
   * @param {Number} interval - 检查间隔
   */
  async monitor(healthCheck, interval = 5000) {
    const monitorInterval = setInterval(async () => {
      try {
        const health = await healthCheck();
        
        if (health.status === 'healthy') {
          this.restoreNormal();
        } else if (health.status === 'degraded') {
          this.degradeTo('degraded');
        } else if (health.status === 'critical') {
          this.degradeTo('minimal');
        }
      } catch (error) {
        console.error('健康检查失败:', error);
        this.degradeTo('minimal');
      }
    }, interval);
    
    return () => clearInterval(monitorInterval);
  }
}

    },
  });

  // 注册推荐策略
  manager.registerStrategy('recommendation', {
    normal: async (context) => {
      console.log('执行完整推荐');
      // 基于用户历史的个性化推荐
      return { recommendations: ['item1', 'item2', 'item3'], quality: 'high' };
    },
    degraded: async (context) => {
      console.log('执行降级推荐（基于热门）');
      // 基于热门项目的推荐
      return { recommendations: ['popular-item1', 'popular-item2'], quality: 'medium' };
    },
    minimal: async (context) => {
      console.log('执行最小推荐（随机）');
      // 随机推荐
      return { recommendations: ['random-item'], quality: 'low' };
    },
  });

  // 正常执行完整搜索
  const searchResult = await manager.execute('search', { query: 'test' });
  console.log('搜索结果:', searchResult);}
```

## 📊 最佳实践

### 1. 规划最佳实践

- **任务分解粒度**：保持适度的任务粒度，避免过细或过粗
- **依赖管理**：明确任务依赖关系，避免循环依赖
- **优先级设置**：合理设置任务优先级，确保关键任务优先执行
- **时间估算**：准确估算任务时间，为调度提供依据

### 2. 执行最佳实践

- **并发控制**：根据系统资源合理设置并发数
- **进度监控**：实时监控任务执行进度
- **资源管理**：有效管理系统资源，避免资源耗尽
- **性能优化**：根据实际执行情况动态调整策略

### 3. 错误处理最佳实践

- **重试策略**：合理设置重试次数和间隔
- **错误分类**：区分可恢复和不可恢复错误
- **日志记录**：详细记录错误信息，便于排查
- **降级方案**：准备完善的降级方案

### 4. 监控与维护

- **性能监控**：监控系统性能指标
- **错误追踪**：追踪和分析错误模式
- **容量规划**：根据历史数据规划系统容量
- **自动化测试**：建立自动化测试体系
