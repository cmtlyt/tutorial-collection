## 🎯 Semantic Kernel

### 概述

Semantic Kernel 是微软开发的企业级 AI 框架，专注于将 LLM 集成到企业应用中。它提供了插件化的架构，支持多种编程语言（C#、Python、Java）。

### 核心特性

- **插件系统**：通过插件扩展功能
- **企业级支持**：安全、合规、可扩展
- **多语言支持**：C#、Python、Java
- **微软生态集成**：与 Azure、Office 365 深度集成

### 基础使用

```python
import semantic_kernel as sk
from semantic_kernel.connectors.ai.open_ai import OpenAIChatCompletion

# 1. 创建内核
kernel = sk.Kernel()

# 2. 添加 LLM
kernel.add_chat_service(
    "gpt-4",
    OpenAIChatCompletion(
        "gpt-4",
        api_key="your-api-key",
        org_id="your-org-id"
    )
)

# 3. 定义插件
from semantic_kernel.plugin_definition import (
    kernel_function,
    kernel_function_context_parameter
)

class WeatherPlugin:
    @kernel_function(
        description="获取指定城市的天气",
        name="get_weather"
    )
    def get_weather(self, city: str) -> str:
        """获取天气信息"""
        # 实现天气查询逻辑
        return f"{city} 今天天气晴朗，温度 25°C"

# 4. 注册插件
kernel.import_plugin(WeatherPlugin(), "weather")

# 5. 创建 Agent
from semantic_kernel.orchestration import sk_function

@sk_function(
    description="帮助用户查询天气",
    name="weather_assistant"
)
async def weather_assistant(question: str) -> str:
    """天气助手"""
    # 使用内核执行
    result = await kernel.run_async(
        kernel.plugins["weather"]["get_weather"],
        input_str=question
    )
    return str(result)

# 6. 执行
result = await weather_assistant("北京")
print(result)
```

### 插件开发

```python
from semantic_kernel.plugin_definition import kernel_function

class CalculatorPlugin:
    @kernel_function(
        description="执行加法运算",
        name="add"
    )
    def add(self, a: float, b: float) -> float:
        """加法"""
        return a + b
    
    @kernel_function(
        description="执行减法运算",
        name="subtract"
    )
    def subtract(self, a: float, b: float) -> float:
        """减法"""
        return a - b
    
    @kernel_function(
        description="执行乘法运算",
        name="multiply"
    )
    def multiply(self, a: float, b: float) -> float:
        """乘法"""
        return a * b
    
    @kernel_function(
        description="执行除法运算",
        name="divide"
    )
    def divide(self, a: float, b: float) -> float:
        """除法"""
        if b == 0:
            raise ValueError("除数不能为零")
        return a / b

# 注册插件
kernel.import_plugin(CalculatorPlugin(), "calculator")

# 使用插件
result = await kernel.run_async(
    kernel.plugins["calculator"]["add"],
    input_str="10",  # 第一个参数
    input_str2="20"  # 第二个参数
)
```

## 🚀 MetaGPT

### 概述

MetaGPT 是一个专注于软件开发的 Agent 框架，它模拟了真实的软件开发团队，包括产品经理、架构师、工程师、测试工程师等角色。

### 核心特性

- **角色扮演**：模拟真实团队角色
- **标准化流程**：遵循软件开发最佳实践
- **代码生成**：自动生成高质量代码
- **文档生成**：自动生成技术文档

### 基础使用

```python
import asyncio
from metagpt.roles import (
    ProductManager,
    Architect,
    Engineer,
    Tester
)
from metagpt.schema import Message

async def main():
    # 1. 创建角色
    pm = ProductManager()
    architect = Architect()
    engineer = Engineer()
    tester = Tester()
    
    # 2. 定义任务
    task = "开发一个简单的博客系统，包括文章发布、评论和用户管理功能"
    
    # 3. 产品经理分析需求
    pm_msg = await pm.run(Message(content=task))
    print(f"产品需求: {pm_msg.content}")
    
    # 4. 架构师设计架构
    arch_msg = await architect.run(pm_msg)
    print(f"架构设计: {arch_msg.content}")
    
    # 5. 工程师开发代码
    eng_msg = await engineer.run(arch_msg)
    print(f"代码实现: {eng_msg.content}")
    
    # 6. 测试工程师测试
    test_msg = await tester.run(eng_msg)
    print(f"测试结果: {test_msg.content}")

# 运行
asyncio.run(main())
```

## 🔄 BabyAGI

### 概述

BabyAGI 是一个简单的任务自动化框架，它通过不断创建、优先级排序和执行任务来实现自动化目标。

### 核心特性

- **任务循环**：自动创建和执行任务
- **优先级排序**：智能排序任务
- **上下文管理**：维护任务上下文
- **简单易用**：代码简洁，易于理解

### 基础使用

```python
import os
from collections import deque
from typing import Dict, List

import openai

# 配置
openai.api_key = "your-api-key"

class BabyAGI:
    def __init__(self, objective: str):
        self.objective = objective
        self.task_list = deque()
        self.completed_tasks = []
    
    def create_task(self, task_description: str) -> str:
        """创建新任务"""
        self.task_list.append({
            "task_name": task_description,
            "status": "pending"
        })
        return f"任务已创建: {task_description}"
    
    def prioritize_tasks(self) -> List[Dict]:
        """优先级排序任务"""
        # 使用 LLM 对任务进行排序
        tasks = list(self.task_list)
        if not tasks:
            return []
        
        # 简单的优先级排序（实际应用中可以使用 LLM）
        sorted_tasks = sorted(tasks, key=lambda x: len(x["task_name"]))
        return sorted_tasks
    
    def execute_task(self, task: Dict) -> str:
        """执行任务"""
        task_name = task["task_name"]
        
        # 使用 LLM 执行任务
        response = openai.ChatCompletion.create(
            model="gpt-4",
            messages=[
                {
                    "role": "system",
                    "content": f"你是一个任务执行助手。你的目标是: {self.objective}"
                },
                {
                    "role": "user",
                    "content": f"执行任务: {task_name}"
                }
            ]
        )
        
        result = response.choices[0].message.content
        
        # 标记任务为完成
        self.completed_tasks.append({
            "task_name": task_name,
            "result": result
        })
        
        return result
    
    def run(self, max_iterations: int = 10):
        """运行 BabyAGI"""
        # 创建初始任务
        self.create_task(f"分析目标: {self.objective}")
        
        iteration = 0
        while iteration < max_iterations:
            print(f"\n=== 迭代 {iteration + 1} ===")
            
            # 优先级排序任务
            sorted_tasks = self.prioritize_tasks()
            
            if not sorted_tasks:
                print("没有更多任务")
                break
            
            # 执行第一个任务
            task = sorted_tasks[0]
            print(f"执行任务: {task['task_name']}")
            
            result = self.execute_task(task)
            print(f"任务结果: {result}")
            
            # 从任务列表中移除
            if self.task_list:
                self.task_list.popleft()
            
            # 根据结果创建新任务
            self.create_task(f"基于结果创建新任务: {result[:50]}...")
            
            iteration += 1
        
        print("\n=== 完成 ===")
        print(f"完成的任务数: {len(self.completed_tasks)}")

# 使用
if __name__ == "__main__":
    baby_agi = BabyAGI("研究 AI Agent 的最新进展")
    baby_agi.run(max_iterations=5)
```

### 高级功能

```python
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings

class AdvancedBabyAGI(BabyAGI):
    """高级 BabyAGI"""
    
    def __init__(self, objective: str):
        super().__init__(objective)
        self.embeddings = OpenAIEmbeddings()
        self.vector_store = FAISS.from_texts(
            ["初始上下文"],
            self.embeddings
        )
    
    def add_context(self, text: str):
        """添加上下文"""
        self.vector_store.add_texts([text])
    
    def get_relevant_context(self, query: str) -> str:
        """获取相关上下文"""
        docs = self.vector_store.similarity_search(query, k=3)
        return "\n".join([doc.page_content for doc in docs])
    
    def execute_task(self, task: Dict) -> str:
        """执行任务（带上下文）"""
        task_name = task["task_name"]
        
        # 获取相关上下文
        context = self.get_relevant_context(task_name)
        
        # 使用 LLM 执行任务
        response = openai.ChatCompletion.create(
            model="gpt-4",
            messages=[
                {
                    "role": "system",
                    "content": f"""你是一个任务执行助手。
目标: {self.objective}

上下文:
{context}"""
                },
                {
                    "role": "user",
                    "content": f"执行任务: {task_name}"
                }
            ]
        )
        
        result = response.choices[0].message.content
        
        # 添加结果到上下文
        self.add_context(f"任务: {task_name}\n结果: {result}")
        
        # 标记任务为完成
        self.completed_tasks.append({
            "task_name": task_name,
            "result": result
        })
        
        return result
```

## 🌐 AgentScope

### 概述

AgentScope 是阿里巴巴开发的多模态 Agent 框架，专注于易用性和多模态能力。

### 核心特性

- **多模态支持**：文本、图像、音频等
- **易用性**：简单的 API 设计
- **可视化**：支持可视化调试
- **分布式**：支持分布式部署

### 基础使用

```python
from agentscope import Agent, Msg

# 1. 创建 Agent
agent = Agent(
    name="助手",
    system_prompt="你是一个有帮助的助手。",
    model_name="gpt-4",
    api_key="your-api-key"
)

# 2. 发送消息
response = agent(Msg(role="user", content="你好！"))
print(response.content)

# 3. 多轮对话
history = []
for i in range(3):
    user_input = input("用户: ")
    response = agent(Msg(role="user", content=user_input, history=history))
    print(f"助手: {response.content}")
    history.append(response)
```

```python
from agentscope.tools import tool

def search_web(query: str) -> str:
```
