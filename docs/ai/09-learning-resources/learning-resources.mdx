```python
# 学习进度跟踪示例
from typing import Dict, List
from datetime import datetime

class LearningTracker:
    """学习进度跟踪器"""
    
    def __init__(self):
        self.courses: Dict[str, Dict] = {}
        self.completed_lessons: List[str] = []
    
    def register_course(
        self,
        course_name: str,
        total_lessons: int,
        estimated_hours: float
    ):
        """注册课程"""
        self.courses[course_name] = {
            'total_lessons': total_lessons,
            'estimated_hours': estimated_hours,
            'completed_lessons': 0,
            'started_at': None,
            'completed_at': None
        }
    
    def start_course(self, course_name: str):
        """开始课程"""
        if course_name in self.courses:
            self.courses[course_name]['started_at'] = datetime.now()
    
    def complete_lesson(self, course_name: str, lesson_id: str):
        """完成课程"""
        if course_name in self.courses:
            self.courses[course_name]['completed_lessons'] += 1
            self.completed_lessons.append(f"{course_name}:{lesson_id}")
    
    def get_progress(self, course_name: str) -> Dict:
        """获取进度"""
        if course_name in self.courses:
            course = self.courses[course_name]
            progress = course['completed_lessons'] / course['total_lessons']
            
            return {
                'course': course_name,
                'progress': progress,
                'completed_lessons': course['completed_lessons'],
                'total_lessons': course['total_lessons'],
                'percentage': progress * 100
            }
        
        return {'error': 'Course not found'}
```

```python
# Hugging Face 模型使用示例
from transformers import pipeline, AutoTokenizer, AutoModel

class HuggingFaceModelManager:
    """Hugging Face 模型管理器"""
    
    def __init__(self):
        self.models = {}
        self.pipelines = {}
    
    def load_pipeline(
        self,
        task: str,
        model_name: str = None
    ):
        """加载 pipeline"""
        if model_name:
            self.pipelines[task] = pipeline(
                task,
                model=model_name
            )
        else:
            self.pipelines[task] = pipeline(task)
    
    def use_pipeline(
        self,
        task: str,
        inputs: str,
        **kwargs
    ) -> Dict:
        """使用 pipeline"""
        if task in self.pipelines:
            result = self.pipelines[task](inputs, **kwargs)
            return {
                'task': task,
                'inputs': inputs,
                'result': result
            }
        
        return {'error': 'Pipeline not loaded'}
    
    def load_model(
        self,
        model_name: str,
        tokenizer_name: str = None
    ):
        """加载模型"""
        tokenizer_name = tokenizer_name or model_name
        
        tokenizer = AutoTokenizer.from_pretrained(tokenizer_name)
        model = AutoModel.from_pretrained(model_name)
        
        self.models[model_name] = {
            'model': model,
            'tokenizer': tokenizer
        }
```

```python
# Hugging Face Hub 使用示例
from huggingface_hub import HfApi, login
from typing import Dict, List

class HuggingFaceHubManager:
    """Hugging Face Hub 管理器"""
    
    def __init__(self, token: str = None):
        self.api = HfApi()
        if token:
            login(token)
    
    def search_models(
        self,
        query: str,
        limit: int = 10
    ) -> List[Dict]:
        """搜索模型"""
        models = self.api.list_models(
            search=query,
            limit=limit
        )
        
        return [
            {
                'model_id': model.modelId,
                'downloads': model.downloads,
                'likes': model.likes,
                'tags': model.tags
            }
            for model in models
        ]
    
    def upload_model(
        self,
        model_path: str,
        repo_id: str,
        private: bool = False
    ) -> Dict:
        """上传模型"""
        try:
            self.api.upload_folder(
                folder_path=model_path,
                repo_id=repo_id,
                private=private
            )
            
            return {
                'status': 'success',
                'repo_id': repo_id,
                'private': private
            }
        except Exception as e:
            return {
                'status': 'error',
                'message': str(e)
            }
    
    def create_space(
        self,
        name: str,
        sdk: str = "gradio"
    ) -> Dict:
        """创建 Space"""
        try:
            self.api.create_repo(
                repo_id=name,
                repo_type="space",
                space_sdk=sdk
            )
            
            return {
                'status': 'success',
                'space_name': name,
                'sdk': sdk
            }
        except Exception as e:
            return {
                'status': 'error',
                'message': str(e)
            }
```

```python
# Colab 环境检测和配置
import torch
import os

class ColabEnvironment:
    """Colab 环境管理"""
    
    def __init__(self):
        self.is_colab = self._check_colab()
        self.gpu_available = self._check_gpu()
    
    def _check_colab(self) -> bool:
        """检查是否在 Colab 环境"""
        return 'COLAB_GPU' in os.environ
    
    def _check_gpu(self) -> bool:
        """检查 GPU 是否可用"""
        return torch.cuda.is_available()
    
    def get_gpu_info(self) -> Dict:
        """获取 GPU 信息"""
        if self.gpu_available:
            return {
                'gpu_available': True,
                'gpu_name': torch.cuda.get_device_name(0),
                'gpu_memory': torch.cuda.get_device_properties(0).total_memory / 1024**3
            }
        
        return {
            'gpu_available': False,
            'message': 'GPU not available'
        }
    
    def mount_drive(self):
        """挂载 Google Drive"""
        if self.is_colab:
            from google.colab import drive
            drive.mount('/content/drive')
            return {'status': 'success', 'mount_point': '/content/drive'}
        
        return {'status': 'error', 'message': 'Not in Colab environment'}
```

```python
# arXiv 论文管理示例
import arxiv
from typing import List, Dict

class ArxivPaperManager:
    """arXiv 论文管理器"""
    
    def __init__(self):
        self.client = arxiv.Client()
        self.saved_papers: List[Dict] = []
    
    def search_papers(
        self,
        query: str,
        max_results: int = 10,
        sort_by: str = "relevance"
    ) -> List[Dict]:
        """搜索论文"""
        search = arxiv.Search(
            query=query,
            max_results=max_results,
            sort_by=arxiv.SortCriterion.Relevance
        )
        
        papers = []
        for result in self.client.results(search):
            paper = {
                'title': result.title,
                'authors': [author.name for author in result.authors],
                'published': result.published,
                'summary': result.summary,
                'pdf_url': result.pdf_url,
                'entry_id': result.entry_id
            }
            papers.append(paper)
        
        return papers
    
    def save_paper(self, paper: Dict):
        """保存论文"""
        self.saved_papers.append(paper)
    
    def get_saved_papers(self) -> List[Dict]:
        """获取已保存的论文"""
        return self.saved_papers
    
    def filter_by_category(
        self,
        category: str
    ) -> List[Dict]:
        """按类别筛选"""
        return [
            paper for paper in self.saved_papers
            if category in paper['entry_id']
        ]
```

### LangChain

LangChain 是 LLM 应用开发框架。

#### 核心组件
- **Chains**：链式调用
- **Agents**：智能代理
- **Memory**：记忆管理
- **Tools**：工具集成

#### 使用示例
```python
# LangChain 基础使用
from langchain.chains import LLMChain
from langchain.prompts import PromptTemplate
from langchain.llms import OpenAI

class LangChainApplication:
    """LangChain 应用"""
    
    def __init__(self, api_key: str):
        self.llm = OpenAI(temperature=0.7, openai_api_key=api_key)
        self.chains = {}
    
    def create_chain(
        self,
        name: str,
        template: str,
        input_variables: List[str]
    ):
        """创建 Chain"""
        prompt = PromptTemplate(
            template=template,
            input_variables=input_variables
        )
        
        chain = LLMChain(llm=self.llm, prompt=prompt)
        self.chains[name] = chain
    
    def run_chain(
        self,
        name: str,
        inputs: Dict
    ) -> str:
        """运行 Chain"""
        if name in self.chains:
            return self.chains[name].run(**inputs)
        
        return "Chain not found"
```

```python
# vLLM 推理示例
from vllm import LLM, SamplingParams

class VLLMInference:
    """vLLM 推理引擎"""
    
    def __init__(self, model_name: str):
        self.llm = LLM(model=model_name)
        self.sampling_params = SamplingParams(
            temperature=0.8,
            top_p=0.95,
            max_tokens=1000
        )
    
    def generate(
        self,
        prompts: List[str]
    ) -> List[str]:
        """生成文本"""
        outputs = self.llm.generate(
            prompts,
            self.sampling_params
        )
        
        return [
            output.outputs[0].text
            for output in outputs
        ]
    
    def batch_generate(
        self,
        prompts: List[str],
        batch_size: int = 8
    ) -> List[str]:
        """批量生成"""
        all_outputs = []
        
        for i in range(0, len(prompts), batch_size):
            batch = prompts[i:i + batch_size]
            outputs = self.generate(batch)
            all_outputs.extend(outputs)
        
        return all_outputs
```

### 循序渐进
从基础概念开始，逐步深入到高级应用。

```python
# 学习路径规划
from typing import List, Dict
from enum import Enum

class LearningLevel(Enum):
    """学习级别"""
    BEGINNER = "beginner"
    INTERMEDIATE = "intermediate"
    ADVANCED = "advanced"
    EXPERT = "expert"

class LearningPath:
    """学习路径"""
    
    def __init__(self):
        self.stages: List[Dict] = []
    
    def add_stage(
        self,
        name: str,
        level: LearningLevel,
        topics: List[str],
        resources: List[str],
        estimated_hours: float
    ):
        """添加学习阶段"""
        stage = {
            'name': name,
            'level': level.value,
            'topics': topics,
            'resources': resources,
            'estimated_hours': estimated_hours,
            'completed': False
        }
        self.stages.append(stage)
    
    def get_current_stage(self) -> Dict:
        """获取当前阶段"""
        for stage in self.stages:
            if not stage['completed']:
                return stage
        
        return {'message': 'All stages completed'}
    
    def complete_stage(self, stage_name: str):
        """完成阶段"""
        for stage in self.stages:
            if stage['name'] == stage_name:
                stage['completed'] = True
                break
    
    def get_progress(self) -> Dict:
        """获取进度"""
        total = len(self.stages)
        completed = sum(1 for stage in self.stages if stage['completed'])
        
        return {
            'total_stages': total,
            'completed_stages': completed,
            'progress': completed / total if total > 0 else 0,
            'percentage': (completed / total * 100) if total > 0 else 0
        }
```
