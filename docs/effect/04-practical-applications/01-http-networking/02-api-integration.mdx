## 🎯 REST API

### 基本 REST 操作

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// REST API 客户端
const RestClient = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const baseUrl = 'https://api.example.com'
  
  return {
    // GET - 获取资源
    get: (path: string) => Effect.gen(function* () {
      const response = yield* httpClient.get(`${baseUrl}${path}`)
      return yield* HttpClient.response.json(response)
    }),
    
    // POST - 创建资源
    post: (path: string, data: any) => Effect.gen(function* () {
      const response = yield* httpClient.post(`${baseUrl}${path}`, {
        body: HttpClient.body.json(data)
      })
      return yield* HttpClient.response.json(response)
    }),
    
    // PUT - 更新资源
    put: (path: string, data: any) => Effect.gen(function* () {
      const response = yield* httpClient.put(`${baseUrl}${path}`, {
        body: HttpClient.body.json(data)
      })
      return yield* HttpClient.response.json(response)
    }),
    
    // PATCH - 部分更新资源
    patch: (path: string, data: any) => Effect.gen(function* () {
      const response = yield* httpClient.patch(`${baseUrl}${path}`, {
        body: HttpClient.body.json(data)
      })
      return yield* HttpClient.response.json(response)
    }),
    
    // DELETE - 删除资源
    delete: (path: string) => Effect.gen(function* () {
      const response = yield* httpClient.delete(`${baseUrl}${path}`)
      return yield* HttpClient.response.json(response)
    })
  }
})

// 使用 REST 客户端
const program = Effect.gen(function* () {
  const restClient = yield* RestClient
  
  // 获取所有用户
  const users = yield* restClient.get('/users')
  console.log('Users:', users)
  
  // 创建新用户
  const newUser = yield* restClient.post('/users', {
    name: 'Alice',
    email: 'alice@example.com'
  })
  console.log('New user:', newUser)
  
  // 更新用户
  const updatedUser = yield* restClient.put(`/users/${newUser.id}`, {
    name: 'Alice Updated',
    email: 'alice.updated@example.com'
  })
  console.log('Updated user:', updatedUser)
  
  // 删除用户
  const deletedUser = yield* restClient.delete(`/users/${newUser.id}`)
  console.log('Deleted user:', deletedUser)
  
  return { users, newUser, updatedUser, deletedUser }
})

Effect.runPromise(
  program.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### RESTful 资源操作

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// RESTful 资源客户端
const createResourceClient = <T>(resourceName: string) =>
  Effect.gen(function* () {
    const httpClient = yield* HttpClient.HttpClient
    const baseUrl = 'https://api.example.com'
    
    return {
      // 获取所有资源
      getAll: () => Effect.gen(function* () {
        const response = yield* httpClient.get(`${baseUrl}/${resourceName}`)
        return yield* HttpClient.response.json(response)
      }),
      
      // 获取单个资源
      getById: (id: number) => Effect.gen(function* () {
        const response = yield* httpClient.get(`${baseUrl}/${resourceName}/${id}`)
        return yield* HttpClient.response.json(response)
      }),
      
      // 创建资源
      create: (data: Partial<T>) => Effect.gen(function* () {
        const response = yield* httpClient.post(`${baseUrl}/${resourceName}`, {
          body: HttpClient.body.json(data)
        })
        return yield* HttpClient.response.json(response)
      }),
      
      // 更新资源
      update: (id: number, data: Partial<T>) => Effect.gen(function* () {
        const response = yield* httpClient.put(`${baseUrl}/${resourceName}/${id}`, {
          body: HttpClient.body.json(data)
        })
        return yield* HttpClient.response.json(response)
      }),
      
      // 部分更新资源
      patch: (id: number, data: Partial<T>) => Effect.gen(function* () {
        const response = yield* httpClient.patch(`${baseUrl}/${resourceName}/${id}`, {
          body: HttpClient.body.json(data)
        })
        return yield* HttpClient.response.json(response)
      }),
      
      // 删除资源
      delete: (id: number) => Effect.gen(function* () {
        const response = yield* httpClient.delete(`${baseUrl}/${resourceName}/${id}`)
        return yield* HttpClient.response.json(response)
      })
    }
  })

// 使用资源客户端
const program = Effect.gen(function* () {
  const userClient = yield* createResourceClient('users')
  
  // 获取所有用户
  const users = yield* userClient.getAll()
  console.log('All users:', users)
  
  // 获取单个用户
  const user = yield* userClient.getById(1)
  console.log('User:', user)
  
  // 创建用户
  const newUser = yield* userClient.create({
    name: 'Alice',
    email: 'alice@example.com'
  })
  console.log('New user:', newUser)
  
  return { users, user, newUser }
})

Effect.runPromise(
  program.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

## 💡 GraphQL

### GraphQL 客户端

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// GraphQL 客户端
const GraphQLClient = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const baseUrl = 'https://api.example.com/graphql'
  
  return {
    // 执行查询
    query: (query: string, variables?: any) => Effect.gen(function* () {
      const response = yield* httpClient.post(baseUrl, {
        body: HttpClient.body.json({
          query,
          variables
        })
      })
      
      const result = yield* HttpClient.response.json(response)
      
      if (result.errors) {
        return Effect.fail(result.errors)
      }
      
      return result.data
    }),
    
    // 执行变更
    mutate: (mutation: string, variables?: any) => Effect.gen(function* () {
      const response = yield* httpClient.post(baseUrl, {
        body: HttpClient.body.json({
          query: mutation,
          variables
        })
      })
      
      const result = yield* HttpClient.response.json(response)
      
      if (result.errors) {
        return Effect.fail(result.errors)
      }
      
      return result.data
    })
  }
})

// 使用 GraphQL 客户端
const program = Effect.gen(function* () {
  const graphqlClient = yield* GraphQLClient
  
  // 查询用户
  const query = `
    query GetUsers($limit: Int) {
      users(limit: $limit) {
        id
        name
        email
      }
    }
  `
  
  const users = yield* graphqlClient.query(query, { limit: 10 })
  console.log('Users:', users)
  
  // 创建用户
  const mutation = `
    mutation CreateUser($name: String!, $email: String!) {
      createUser(name: $name, email: $email) {
        id
        name
        email
      }
    }
  `
  
  const newUser = yield* graphqlClient.mutate(mutation, {
    name: 'Alice',
    email: 'alice@example.com'
  })
  console.log('New user:', newUser)
  
  return { users, newUser }
})

Effect.runPromise(
  program.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### GraphQL 订阅

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// GraphQL 订阅
const GraphQLSubscription = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const baseUrl = 'https://api.example.com/graphql'
  
  return {
    // 订阅事件
    subscribe: (subscription: string, variables?: any) => Effect.gen(function* () {
      const response = yield* httpClient.post(baseUrl, {
        body: HttpClient.body.json({
          query: subscription,
          variables
        }),
        headers: {
          'Sec-WebSocket-Protocol': 'graphql-ws'
        }
      })
      
      return yield* HttpClient.response.json(response)
    })
  }
})

// 使用 GraphQL 订阅
const program = Effect.gen(function* () {
  const graphqlSubscription = yield* GraphQLSubscription
  
  // 订阅用户创建事件
  const subscription = `
    subscription OnUserCreated {
      userCreated {
        id
        name
        email
        createdAt
      }
    }
  `
  
  const subscriptionResult = yield* graphqlSubscription.subscribe(subscription)
  console.log('Subscription result:', subscriptionResult)
  
  return subscriptionResult
})

Effect.runPromise(
  program.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

## 🚀 WebSocket

### WebSocket 连接

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// WebSocket 客户端
const WebSocketClient = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  
  return {
    // 建立 WebSocket 连接
    connect: (url: string) => Effect.gen(function* () {
      const ws = yield* Effect.tryPromise(() =>
        Promise.resolve(new WebSocket(url))
      )
      
      return {
        ws,
        send: (message: string) => Effect.sync(() => {
          ws.send(message)
        }),
        onMessage: (callback: (message: any) => void) => Effect.sync(() => {
          ws.onmessage = (event) => callback(JSON.parse(event.data))
        }),
        onClose: (callback: () => void) => Effect.sync(() => {
          ws.onclose = callback
        }),
        onError: (callback: (error: any) => void) => Effect.sync(() => {
          ws.onerror = callback
        }),
        close: Effect.sync(() => {
          ws.close()
        })
      }
    })
  }
})

// 使用 WebSocket 客户端
const program = Effect.gen(function* () {
  const wsClient = yield* WebSocketClient
  
  const connection = yield* wsClient.connect('ws://localhost:8080')
  
  // 发送消息
  yield* connection.send(JSON.stringify({
    type: 'greeting',
    message: 'Hello, Server!'
  }))
  
  // 监听消息
  yield* connection.onMessage((message) => {
    console.log('Received message:', message)
  })
  
  // 监听关闭
  yield* connection.onClose(() => {
    console.log('WebSocket connection closed')
  })
  
  // 监听错误
  yield* connection.onError((error) => {
    console.error('WebSocket error:', error)
  })
  
  // 关闭连接
  yield* connection.close
  
  return connection
})

Effect.runPromise(
  program.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### WebSocket 消息处理

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// WebSocket 消息处理器
const WebSocketMessageHandler = Effect.gen(function* () {
  const wsClient = yield* WebSocketClient
  
  return {
    // 处理不同类型的消息
    handleMessage: (connection: any) => Effect.gen(function* () {
      yield* connection.onMessage((message: any) => {
        switch (message.type) {
          case 'chat':
            console.log('Chat message:', message.content)
            break
          case 'notification':
            console.log('Notification:', message.content)
            break
          case 'error':
            console.error('Error message:', message.content)
            break
          default:
            console.log('Unknown message type:', message.type)
        }
      })
    }),
    
    // 发送不同类型的消息
    sendChatMessage: (connection: any, content: string) =>
      connection.send(JSON.stringify({
        type: 'chat',
        content
      })),
    
    sendNotification: (connection: any, content: string) =>
      connection.send(JSON.stringify({
        type: 'notification',
        content
      }))
  }
})

// 使用消息处理器
const program = Effect.gen(function* () {
  const messageHandler = yield* WebSocketMessageHandler
  
  const connection = yield* messageHandler.connect('ws://localhost:8080')
  
  // 处理消息
  yield* messageHandler.handleMessage(connection)
  
  // 发送聊天消息
  yield* messageHandler.sendChatMessage(connection, 'Hello, everyone!')
  
  // 发送通知
  yield* messageHandler.sendNotification(connection, 'New user joined!')
  
  return connection
})

Effect.runPromise(
  program.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

## 📝 实际应用示例

### REST API 完整示例

```js
import { Effect, HttpClient, Context } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 用户服务
interface UserService {
  getAllUsers: () => Effect.Effect<never, never, any[]>
  getUserById: (id: number) => Effect.Effect<never, never, any>
  createUser: (data: any) => Effect.Effect<never, never, any>
  updateUser: (id: number, data: any) => Effect.Effect<never, never, any>
  deleteUser: (id: number) => Effect.Effect<never, never, any>
}

const UserService = Context.GenericTag<UserService>('UserService')

// 实现 UserService
const makeUserService = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const baseUrl = 'https://api.example.com'
  
  return {
    getAllUsers: () => Effect.gen(function* () {
      const response = yield* httpClient.get(`${baseUrl}/users`)
      return yield* HttpClient.response.json(response)
    }),
    
    getUserById: (id: number) => Effect.gen(function* () {
      const response = yield* httpClient.get(`${baseUrl}/users/${id}`)
      return yield* HttpClient.response.json(response)
    }),
    
    createUser: (data: any) => Effect.gen(function* () {
      const response = yield* httpClient.post(`${baseUrl}/users`, {
        body: HttpClient.body.json(data)
      })
      return yield* HttpClient.response.json(response)
    }),
    
    updateUser: (id: number, data: any) => Effect.gen(function* () {
      const response = yield* httpClient.put(`${baseUrl}/users/${id}`, {
        body: HttpClient.body.json(data)
      })
      return yield* HttpClient.response.json(response)
    }),
    
    deleteUser: (id: number) => Effect.gen(function* () {
      const response = yield* httpClient.delete(`${baseUrl}/users/${id}`)
      return yield* HttpClient.response.json(response)
    })
  }
})

// 使用 UserService
const program = Effect.gen(function* () {
  const userService = yield* UserService
  
  // 获取所有用户
  const users = yield* userService.getAllUsers()
  console.log('All users:', users)
  
  // 创建用户
  const newUser = yield* userService.createUser({
    name: 'Alice',
    email: 'alice@example.com'
  })
  console.log('Created user:', newUser)
  
  // 更新用户
  const updatedUser = yield* userService.updateUser(newUser.id, {
    name: 'Alice Updated'
  })
  console.log('Updated user:', updatedUser)
  
  // 删除用户
  const deletedUser = yield* userService.deleteUser(newUser.id)
  console.log('Deleted user:', deletedUser)
  
  return { users, newUser, updatedUser, deletedUser }
})

Effect.runPromise(
  program.pipe(
    Effect.provideService(UserService, Effect.runSync(makeUserService)),
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### GraphQL 完整示例

```js
import { Effect, HttpClient, Context } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// GraphQL 服务
interface GraphQLService {
  query: (query: string, variables?: any) => Effect.Effect<never, never, any>
  mutate: (mutation: string, variables?: any) => Effect.Effect<never, never, any>
}

const GraphQLService = Context.GenericTag<GraphQLService>('GraphQLService')

// 实现 GraphQLService
const makeGraphQLService = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const baseUrl = 'https://api.example.com/graphql'
  
  const execute = (queryOrMutation: string, variables?: any) => Effect.gen(function* () {
    const response = yield* httpClient.post(baseUrl, {
      body: HttpClient.body.json({
        query: queryOrMutation,
        variables
      })
    })
    
    const result = yield* HttpClient.response.json(response)
    
    if (result.errors) {
      return Effect.fail(result.errors)
    }
    
    return result.data
  })
  
  return {
    query: (query: string, variables?: any) => execute(query, variables),
    mutate: (mutation: string, variables?: any) => execute(mutation, variables)
  }
})

// 使用 GraphQLService
const program = Effect.gen(function* () {
  const graphqlService = yield* GraphQLService
  
  // 查询用户
  const query = `
    query GetUsers($limit: Int) {
      users(limit: $limit) {
        id
        name
        email
      }
    }
  `
  
  const users = yield* graphqlService.query(query, { limit: 10 })
  console.log('Users:', users)
  
  // 创建用户

      createUser(name: $name, email: $email) {
        id
```

### WebSocket 聊天应用

```js
import { Effect, HttpClient, Context } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 聊天服务
interface ChatService {
  connect: () => Effect.Effect<never, never, any>
  sendMessage: (message: string) => Effect.Effect<never, never, void>
  onMessage: (callback: (message: any) => void) => Effect.Effect<never, never, void>
  disconnect: () => Effect.Effect<never, never, void>
}

const ChatService = Context.GenericTag<ChatService>('ChatService')

// 实现 ChatService
const makeChatService = Effect.gen(function* () {
  let connection: any = null
  
  return {
    connect: () => Effect.gen(function* () {
      const ws = yield* Effect.tryPromise(() =>
        Promise.resolve(new WebSocket('ws://localhost:8080/chat'))
      )
      
      connection = {
        ws,
        send: (message: string) => Effect.sync(() => {
          if (connection && connection.ws.readyState === WebSocket.OPEN) {
            connection.ws.send(message)
          }
        }),
        onMessage: (callback: (message: any) => void) => Effect.sync(() => {
          if (connection) {
            connection.ws.onmessage = (event: MessageEvent) => {
              callback(JSON.parse(event.data))
            }
          }
        }),
        close: Effect.sync(() => {
          if (connection) {
            connection.ws.close()
            connection = null
          }
        })
      }
      
      return connection
    }),
    
    sendMessage: (message: string) => Effect.gen(function* () {
      if (connection) {
        yield* connection.send(JSON.stringify({
          type: 'chat',
          content: message,
          timestamp: Date.now()
        }))
      }
    }),
    
    onMessage: (callback: (message: any) => void) => Effect.gen(function* () {
      if (connection) {
        yield* connection.onMessage(callback)
      }
    }),
    
    disconnect: () => Effect.gen(function* () {
      if (connection) {
        yield* connection.close()
      }
    })
  }
})

// 使用 ChatService
const program = Effect.gen(function* () {
  const chatService = yield* ChatService
  
  // 连接到聊天服务器
  yield* chatService.connect()
  console.log('Connected to chat server')
  
  // 监听消息
  yield* chatService.onMessage((message) => {
    console.log('Received message:', message)
  })
  
  // 发送消息
  yield* chatService.sendMessage('Hello, everyone!')
  
  // 等待一段时间后断开连接
  yield* Effect.sleep(5000)
  yield* chatService.disconnect()
  console.log('Disconnected from chat server')
})

Effect.runPromise(
  program.pipe(
    Effect.provideService(ChatService, Effect.runSync(makeChatService)),
    Effect.provide(NodeHttpClient.layer)
  )
)
```
