# HTTP 客户端

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了强大的 HTTP 客户端功能，通过 `@effect/platform` 包提供跨平台的 HTTP 请求能力。

## 🎯 @effect/platform - 平台基础包

### 安装与设置

```js
// 安装依赖
// npm install @effect/platform

// 基本导入
import { Effect } from 'effect'
import { HttpClient } from '@effect/platform'
import { NodeHttpClient } from '@effect/platform-node'
```

### 平台客户端

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 创建 HTTP 客户端
const program = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  
  return httpClient
})

// 在 Node.js 环境中运行
Effect.runSync(
  program.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

## 💡 HttpClient - HTTP 客户端

### 基本请求

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 基本 GET 请求
const getRequest = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  
  const response = yield* httpClient.get('https://api.example.com/data')
  const data = yield* HttpClient.response.json(response)
  
  console.log('Response data:', data)
  return data
})

Effect.runPromise(
  getRequest.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### POST 请求

```js
      email: 'alice@example.com'
    })

  const data = yield* HttpClient.response.json(response)
  console.log('Created user:', data)
  return data
})

Effect.runPromise(
  postRequest.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

## 🎯 响应处理与错误处理

### 响应处理

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 响应处理
const handleResponse = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  
  const response = yield* httpClient.get('https://api.example.com/data')
  
  // 获取响应状态
  console.log('Status:', response.status)
  console.log('Status Text:', response.statusText)
  
  // 获取响应头
  console.log('Headers:', response.headers)
  
  // 获取 JSON 响应
  const jsonData = yield* HttpClient.response.json(response)
  console.log('JSON Data:', jsonData)
  
  // 获取文本响应
  const textData = yield* HttpClient.response.text(response)
  console.log('Text Data:', textData)
  
  return jsonData
})

Effect.runPromise(
  handleResponse.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### 错误处理

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 错误处理
const handleErrors = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  
  const response = yield* httpClient.get('https://api.example.com/not-found')
    .pipe(
      Effect.catchAll((error) => {
        console.error('Request failed:', error)
        return Effect.succeed({ error: error.message })
      })
    )
  
  return response
})

// 基于状态码的错误处理
const handleStatusErrors = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  
  const response = yield* httpClient.get('https://api.example.com/data')
  
  if (response.status >= 400) {
    return Effect.fail(`HTTP Error: ${response.status}`)
  }
  
  return yield* HttpClient.response.json(response)
}).pipe(
  Effect.catchAll((error) => Effect.succeed({ error: error.toString() }))
)

Effect.runPromise(
  handleErrors.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### 重试机制

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 重试机制
const requestWithRetry = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  
  const response = yield* httpClient.get('https://api.example.com/unstable')
    .pipe(
      Effect.retry({
        while: (error) => error._tag === 'RequestError',
        times: 3,
        schedule: Effect.scheduleSpaced('1s')
      })
    )
  
  return yield* HttpClient.response.json(response)
})

Effect.runPromise(
  requestWithRetry.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

## 📝 实际应用示例

### API 客户端封装

```js
import { Effect, HttpClient, Context } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 定义 API 客户端接口
interface ApiClient {
  getUsers: () => Effect.Effect<never, never, any>
  getUser: (id: number) => Effect.Effect<never, never, any>
  createUser: (data: any) => Effect.Effect<never, never, any>
  updateUser: (id: number, data: any) => Effect.Effect<never, never, any>
  deleteUser: (id: number) => Effect.Effect<never, never, any>
}

const ApiClient = Context.GenericTag<ApiClient>('ApiClient')

// 实现 API 客户端
const makeApiClient = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const baseUrl = 'https://api.example.com'
  
  return {
    getUsers: () => Effect.gen(function* () {
      const response = yield* httpClient.get(`${baseUrl}/users`)
      return yield* HttpClient.response.json(response)
    }),
    
    getUser: (id: number) => Effect.gen(function* () {
      const response = yield* httpClient.get(`${baseUrl}/users/${id}`)
      return yield* HttpClient.response.json(response)
    }),
    
    createUser: (data: any) => Effect.gen(function* () {
      const response = yield* httpClient.post(`${baseUrl}/users`, {
        body: HttpClient.body.json(data)
      })
      return yield* HttpClient.response.json(response)
    }),
    
    updateUser: (id: number, data: any) => Effect.gen(function* () {
      const response = yield* httpClient.put(`${baseUrl}/users/${id}`, {
        body: HttpClient.body.json(data)
      })
      return yield* HttpClient.response.json(response)
    }),
    
    deleteUser: (id: number) => Effect.gen(function* () {
      const response = yield* httpClient.delete(`${baseUrl}/users/${id}`)
      return yield* HttpClient.response.json(response)
    })
  }
})

// 使用 API 客户端
const program = Effect.gen(function* () {
  const apiClient = yield* ApiClient
  
  // 获取所有用户
  const users = yield* apiClient.getUsers()
  console.log('Users:', users)
  
  // 创建用户
  const newUser = yield* apiClient.createUser({
    name: 'Alice',
    email: 'alice@example.com'
  })
  console.log('New user:', newUser)
  
  return { users, newUser }
})

Effect.runPromise(
  program.pipe(
    Effect.provideService(ApiClient, Effect.runSync(makeApiClient)),
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### 认证处理

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// Bearer Token 认证
const bearerAuth = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const token = 'your-bearer-token'

      'Authorization': `Bearer ${token}`
    }
  })
  
  return yield* HttpClient.response.json(response)
})

// API Key 认证
const apiKeyAuth = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const apiKey = 'your-api-key'
  
  const response = yield* httpClient.get('https://api.example.com/data', {
    headers: {
      'X-API-Key': apiKey
    }
  })
  
  return yield* HttpClient.response.json(response)
})

// Basic Auth 认证
const basicAuth = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  const username = 'user'
  const password = 'pass'
  const credentials = btoa(`${username}:${password}`)
  
  const response = yield* httpClient.get('https://api.example.com/protected', {
    headers: {
      'Authorization': `Basic ${credentials}`
    }
  })
  
  return yield* HttpClient.response.json(response)
})

Effect.runPromise(
  bearerAuth.pipe(
    Effect.provide(NodeHttpClient.layer)
  )
)
```

### 文件上传

```js
import { Effect, HttpClient } from 'effect'
import { NodeHttpClient } from '@effect/platform-node'

// 文件上传
const uploadFile = Effect.gen(function* () {
  const httpClient = yield* HttpClient.HttpClient
  
  const formData = new FormData()
  formData.append('file', new Blob(['file content']), 'example.txt')
  formData.append('description', 'File description')
  
  const response = yield* httpClient.post('https://api.example.com/upload', {
    body: HttpClient.body.formData(formData)
  })
  
  return yield* HttpClient.response.json(response)
})
```
