# 交互式 CLI

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了强大的交互式 CLI 功能，支持用户输入、选择菜单、进度显示、彩色输出等。

## 🎯 用户输入

### 基本输入

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 简单输入提示
const promptUser = Effect.gen(function* () {
  yield* Console.log('Please enter your name:')
  
  // 注意：实际应用中需要使用适当的输入库
  // 这里演示基本概念
  const name = 'Alice'  // 模拟用户输入
  
  console.log(`Hello, ${name}!`)
  return name
})

Effect.runSync(
  promptUser.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

### 确认提示

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 确认提示
const confirmAction = Effect.gen(function* () {

    console.log('Continuing...')
    return true
  } else {
    console.log('Cancelled')
    return false
  }
})

Effect.runSync(
  confirmAction.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

## 💡 选择菜单

### 基本选择

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 选择菜单
const selectOption = Effect.gen(function* () {
  const options = [
    { id: 1, name: 'Option 1' },
    { id: 2, name: 'Option 2' },
    { id: 3, name: 'Option 3' }
  ]
  
  yield* Console.log('Please select an option:')
  options.forEach((option, index) => {
    console.log(`${index + 1}. ${option.name}`)
  })
  
  // 模拟用户选择
  const selected = 1
  const option = options[selected - 1]
  
  console.log(`You selected: ${option.name}`)
  return option
})

Effect.runSync(
  selectOption.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

### 多项选择

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 多项选择
const selectMultiple = Effect.gen(function* () {
  const options = [
    { id: 1, name: 'Feature A' },
    { id: 2, name: 'Feature B' },
    { id: 3, name: 'Feature C' },
    { id: 4, name: 'Feature D' }
  ]
  
  yield* Console.log('Select features (comma-separated):')
  options.forEach((option, index) => {
    console.log(`${index + 1}. ${option.name}`)
  })
  
  // 模拟用户选择
  const selected = '1,3'
  const indices = selected.split(',').map(Number)
  const selectedOptions = indices.map(i => options[i - 1])
  
  console.log('Selected features:')
  selectedOptions.forEach(option => {
    console.log(`- ${option.name}`)
  })
  
  return selectedOptions
})

Effect.runSync(
  selectMultiple.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

## 🚀 进度显示

### 基本进度条

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 进度条
const showProgress = Effect.gen(function* () {
  const total = 100
  
  for (let i = 0; i <= total; i++) {
    const progress = Math.floor((i / total) * 100)
    const bar = '█'.repeat(Math.floor(progress / 2)) + '░'.repeat(50 - Math.floor(progress / 2))
    
    // 清除当前行并显示进度
    process.stdout.write(`\r[${bar}] ${progress}%`)
    
    yield* Effect.sleep(50)
  }
  
  console.log('\nComplete!')
})

Effect.runSync(
  showProgress.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

### 多任务进度

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 多任务进度
const showMultiProgress = Effect.gen(function* () {
  const tasks = [
    { name: 'Task 1', progress: 0 },
    { name: 'Task 2', progress: 0 },
    { name: 'Task 3', progress: 0 }
  ]
  
  for (let i = 0; i <= 100; i++) {
    console.clear()
    console.log('Progress:')
    
    tasks.forEach(task => {
      task.progress = Math.min(100, task.progress + Math.random() * 5)
      const bar = '█'.repeat(Math.floor(task.progress / 2)) + '░'.repeat(50 - Math.floor(task.progress / 2))
      console.log(`${task.name}: [${bar}] ${Math.floor(task.progress)}%`)
    })
    
    yield* Effect.sleep(100)
  }
  
  console.log('\nAll tasks complete!')
})

Effect.runSync(
  showMultiProgress.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

### 日志级别

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 日志级别
const logLevel = (level: string, message: string) => {
  const colors = {
    error: '\x1b[31m',
    warn: '\x1b[33m',
    info: '\x1b[34m',
    debug: '\x1b[36m',
    reset: '\x1b[0m'
  }
  
  const color = colors[level] || colors.reset
  console.log(`${color}[${level.toUpperCase()}]${colors.reset} ${message}`)
}

// 使用日志级别
const logging = Effect.gen(function* () {
  logLevel('error', 'Something went wrong')
  logLevel('warn', 'This is a warning')
  logLevel('info', 'Information message')
  logLevel('debug', 'Debug information')
})

Effect.runSync(
  logging.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

## 📝 实际应用示例

### 交互式向导

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 交互式向导
const runWizard = Effect.gen(function* () {
  // 步骤 1: 项目名称
  yield* Console.log('Step 1: Enter project name')
  const projectName = 'my-project'  // 模拟输入
  
  // 步骤 2: 选择模板
  yield* Console.log('Step 2: Select template')
  const templates = ['Basic', 'Advanced', 'Custom']
  templates.forEach((t, i) => console.log(`${i + 1}. ${t}`))
  const template = templates[0]  // 模拟选择
  
  // 步骤 3: 选择功能
  yield* Console.log('Step 3: Select features')
  const features = ['TypeScript', 'Testing', 'Linting', 'CI/CD']
  features.forEach((f, i) => console.log(`${i + 1}. ${f}`))
  const selectedFeatures = [features[0], features[1]]  // 模拟选择
  
  // 确认
  yield* Console.log('\nConfiguration:')
  console.log(`Project name: ${projectName}`)
  console.log(`Template: ${template}`)
  console.log(`Features: ${selectedFeatures.join(', ')}`)
  
  yield* Console.log('\nProceed? (y/n)')
  const confirmed = true  // 模拟确认
  
  if (confirmed) {
    console.log('\nCreating project...')
    yield* Effect.sleep(1000)
    console.log('Project created successfully!')
  } else {
    console.log('\nCancelled')
  }
})

Effect.runSync(
  runWizard.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

### 文件上传进度

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 文件上传进度
const uploadFile = Effect.gen(function* () {
  const fileName = 'large-file.zip'
  const fileSize = 1024 * 1024 * 100  // 100MB
  const chunkSize = 1024 * 1024  // 1MB chunks
  const totalChunks = Math.ceil(fileSize / chunkSize)
  
  console.log(`Uploading ${fileName} (${(fileSize / 1024 / 1024).toFixed(2)} MB)`)
  
  let uploaded = 0
  for (let i = 0; i < totalChunks; i++) {
    uploaded += chunkSize
    const progress = Math.floor((uploaded / fileSize) * 100)
    const uploadedMB = (uploaded / 1024 / 1024).toFixed(2)
    
    const bar = '█'.repeat(Math.floor(progress / 2)) + '░'.repeat(50 - Math.floor(progress / 2))
    process.stdout.write(`\r[${bar}] ${progress}% (${uploadedMB}/${(fileSize / 1024 / 1024).toFixed(2)} MB)`)
    
    yield* Effect.sleep(50)
  }
  
  console.log('\nUpload complete!')
})

Effect.runSync(
  uploadFile.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```

### 交互式搜索

```js
import { Effect } from 'effect'
import { Console } from 'effect'
import { NodeContext } from '@effect/platform-node'

// 交互式搜索
const interactiveSearch = Effect.gen(function* () {
  const items = [
    'Apple',
    'Banana',
    'Cherry',
    'Date',
    'Elderberry',
    'Fig',
    'Grape',
    'Honeydew'
  ]
  
  yield* Console.log('Search items (type to filter):')
  
  // 模拟搜索输入
  const searchTerm = 'a'
  
  const filtered = items.filter(item =>
    item.toLowerCase().includes(searchTerm.toLowerCase())
  )
  
  if (filtered.length === 0) {
    console.log('No results found')
  } else {
    console.log('\nResults:')
    filtered.forEach((item, index) => {
      console.log(`${index + 1}. ${item}`)
    })
  }
})

Effect.runSync(
  interactiveSearch.pipe(
    Effect.provide(NodeContext.layer)
  )
)
```
