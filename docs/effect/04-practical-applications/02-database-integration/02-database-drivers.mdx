# 数据库驱动

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 支持多种数据库驱动，包括 PostgreSQL、MySQL、SQLite、SQL Server 等。

## 🎯 PostgreSQL

### 基本连接

```js
import { Effect, Sql } from 'effect'
import { PgClient } from '@effect/sql-pg'

// PostgreSQL 连接
const program = Effect.gen(function* () {
  const sql = yield* Sql.client({
    transform: ['names']
  })
  
  // 执行查询
  const result = yield* Sql`
    SELECT version()
  `
  
  console.log('PostgreSQL version:', result)
  return result
})

Effect.runSync(
  program.pipe(
    Effect.provide(
      PgClient.layer({
        database: 'mydb',
        host: 'localhost',
        port: 5432,
        user: 'postgres',
        password: 'password'
      })
    )
  )
)
```

### 连接字符串

```js
import { Effect, Sql } from 'effect'
import { PgClient } from '@effect/sql-pg'

// 使用连接字符串

    transform: ['names']

  return result

Effect.runSync(
  program.pipe(
    Effect.provide(

        connectionString: 'postgresql://postgres:password@localhost:5432/mydb'

  )
)
```

### 高级配置

```js
import { Effect, Sql } from 'effect'
import { PgClient } from '@effect/sql-pg'

// PostgreSQL 高级配置
const program = Effect.gen(function* () {
  const sql = yield* Sql.client({
    transform: ['names'],
    schema: 'public'  // 默认 schema
  })
  
  return sql
})

Effect.runSync(
  program.pipe(
    Effect.provide(
      PgClient.layer({
        database: 'mydb',
        host: 'localhost',
        port: 5432,
        user: 'postgres',
        password: 'password',
        ssl: true,  // 启用 SSL
        max: 20,    // 最大连接数
        idleTimeout: 30000,
        connectionTimeout: 10000
      })
    )
  )
)
```

### 内存数据库

```js

import { SqliteClient } from '@effect/sql-sqlite'

// SQLite 内存数据库
const program = Effect.gen(function* () {
  const sql = yield* Sql.client({
    transform: ['names']
  })
  
  // 创建表
  yield* Sql`
    CREATE TABLE IF NOT EXISTS users (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      email TEXT NOT NULL
    )
  `
  
  // 插入数据
  yield* Sql`
    INSERT INTO users (name, email) VALUES (${'Alice'}, ${'alice@example.com'})
  `
  
  // 查询数据

  console.log('Users:', users)
  
  return users
})

Effect.runSync(
  program.pipe(
    Effect.provide(
      SqliteClient.layer({
        filename: ':memory:'
```

## 📝 实际应用示例

### 多数据库支持

```js
import { Effect, Sql, Context } from 'effect'
import { PgClient } from '@effect/sql-pg'
import { MysqlClient } from '@effect/sql-mysql'
import { SqliteClient } from '@effect/sql-sqlite'

// 多数据库管理器
interface MultiDbManager {
  postgresql: Sql.SqlClient
  mysql: Sql.SqlClient
  sqlite: Sql.SqlClient
}

const MultiDbManager = Context.GenericTag<MultiDbManager>('MultiDbManager')

// 创建多数据库管理器
const makeMultiDbManager = Effect.gen(function* () {
  const postgresql = yield* Sql.client({
    transform: ['names']
  })
  
  const mysql = yield* Sql.client({
    transform: ['names']
  })
  
  const sqlite = yield* Sql.client({
    transform: ['names']
  })
  
  return {
    postgresql,
    mysql,
    sqlite
  }
})

// 使用多数据库管理器
const program = Effect.gen(function* () {
  const dbManager = yield* MultiDbManager
  
  // 在 PostgreSQL 中查询
  const pgResult = yield* Sql.with(dbManager.postgresql)(
    Sql`SELECT version()`
  )
  console.log('PostgreSQL:', pgResult)
  
  // 在 MySQL 中查询
  const mysqlResult = yield* Sql.with(dbManager.mysql)(
    Sql`SELECT VERSION()`
  )
  console.log('MySQL:', mysqlResult)
  
  // 在 SQLite 中查询
  const sqliteResult = yield* Sql.with(dbManager.sqlite)(
    Sql`SELECT sqlite_version()`
  )
  console.log('SQLite:', sqliteResult)
  
  return { pgResult, mysqlResult, sqliteResult }
})

Effect.runSync(
  program.pipe(
    Effect.provideService(MultiDbManager, Effect.runSync(makeMultiDbManager)),
    Effect.provide(
      PgClient.layer({
        database: 'mydb',
        host: 'localhost',
        port: 5432,
        user: 'postgres',
        password: 'password'
      })
    ),
    Effect.provide(
      MysqlClient.layer({
        database: 'mydb',
        host: 'localhost',
        port: 3306,
        user: 'root',
        password: 'password'
      })
    ),
    Effect.provide(
      SqliteClient.layer({
        filename: './database.db'
      })
    )
  )
)
```

### 数据库迁移

```js
import { Effect, Sql } from 'effect'
import { PgClient } from '@effect/sql-pg'

// 数据库迁移
const migrations = [
  {
    version: 1,
    name: 'create_users_table',
    up: Sql`
      CREATE TABLE IF NOT EXISTS users (
        id SERIAL PRIMARY KEY,
        name VARCHAR(255) NOT NULL,
        email VARCHAR(255) UNIQUE NOT NULL,
        age INTEGER,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
      )
    `,
    down: Sql`DROP TABLE IF EXISTS users`
  },
  {
    version: 2,
    name: 'create_orders_table',
    up: Sql`
      CREATE TABLE IF NOT EXISTS orders (
        id SERIAL PRIMARY KEY,
        user_id INTEGER REFERENCES users(id),
        total_amount DECIMAL(10, 2),
        status VARCHAR(50),
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
      )
    `,
    down: Sql`DROP TABLE IF EXISTS orders`
  }
]

// 运行迁移
const runMigrations = Effect.gen(function* () {
  const sql = yield* Sql.client({
    transform: ['names']
  })
  
  // 创建迁移表
  yield* Sql`
    CREATE TABLE IF NOT EXISTS migrations (
      version INTEGER PRIMARY KEY,
      name VARCHAR(255) NOT NULL,
      applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )
  `
  
  // 获取已应用的迁移
  const appliedMigrations = yield* Sql`
    SELECT version FROM migrations ORDER BY version
  `
  
  const appliedVersions = new Set(appliedMigrations.map(m => m.version))
  
  // 运行未应用的迁移
  for (const migration of migrations) {
    if (!appliedVersions.has(migration.version)) {
      console.log(`Running migration: ${migration.name}`)
      
      yield* Sql.withTransaction(
        Effect.gen(function* () {
          yield* migration.up
          
          yield* Sql`
            INSERT INTO migrations (version, name)
            VALUES (${migration.version}, ${migration.name})
          `
        })
      )
      
      console.log(`Migration ${migration.name} completed`)
    }
  }
  
  return 'Migrations completed'
})

Effect.runSync(
  runMigrations.pipe(
    Effect.provide(
      PgClient.layer({
        database: 'mydb',
        host: 'localhost',
        port: 5432,
        user: 'postgres',
        password: 'password'
      })
    )
  )
)
```

### 读写分离

```js
import { Effect, Sql, Context } from 'effect'
import { PgClient } from '@effect/sql-pg'

// 读写分离管理器
interface ReadWriteManager {
  read: Sql.SqlClient
  write: Sql.SqlClient
}

const ReadWriteManager = Context.GenericTag<ReadWriteManager>('ReadWriteManager')

// 创建读写分离管理器
const makeReadWriteManager = Effect.gen(function* () {
  const read = yield* Sql.client({
    transform: ['names']
  })
  
  const write = yield* Sql.client({
    transform: ['names']
  })
  
  return {
    read,
    write
  }
})

// 使用读写分离
const program = Effect.gen(function* () {
  const rwManager = yield* ReadWriteManager
  
  // 读操作使用读库
  const users = yield* Sql.with(rwManager.read)(
    Sql`SELECT * FROM users`
  )
  console.log('Read users:', users)
  
  // 写操作使用写库
  yield* Sql.with(rwManager.write)(
    Sql`INSERT INTO users (name, email) VALUES (${'Alice'}, ${'alice@example.com'})`
  )
  console.log('Inserted user')
  
  return users
})

Effect.runSync(
  program.pipe(
    Effect.provideService(ReadWriteManager, Effect.runSync(makeReadWriteManager)),
    Effect.provide(
      PgClient.layer({
        database: 'mydb',
        host: 'localhost',
        port: 5432,
        user: 'postgres',
        password: 'password'
      })
    )
  )
)
```
