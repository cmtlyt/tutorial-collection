# 路径操作

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了跨平台的路径操作功能，通过 `@effect/platform` 包提供路径解析、路径组合、路径规范化、路径匹配等功能。

## 🎯 路径解析

### 基本路径解析

```js
import { Effect, Path } from 'effect'

// 路径解析
const parsePath = Effect.gen(function* () {
  const path = '/home/user/documents/file.txt'
  
  // 解析路径
  const parsed = Path.parse(path)
  console.log('Parsed path:', parsed)
  
  return parsed
})

// 获取路径信息
const getPathInfo = Effect.gen(function* () {
  const path = '/home/user/documents/file.txt'
  
  // 获取目录名
  const dirname = Path.dirname(path)
  console.log('Directory:', dirname)
  
  // 获取文件名
  const basename = Path.basename(path)
  console.log('Basename:', basename)
  
  // 获取扩展名
  const extname = Path.extname(path)
  console.log('Extension:', extname)
  
  // 获取不带扩展名的文件名
  const name = Path.basename(path, extname)
  console.log('Name without extension:', name)
  
  return { dirname, basename, extname, name }
})

Effect.runSync(getPathInfo)
```

### 相对路径与绝对路径

```js
import { Effect, Path } from 'effect'

// 路径转换
const convertPath = Effect.gen(function* () {
  // 解析为绝对路径
  const absolute = Path.resolve('./file.txt')
  console.log('Absolute path:', absolute)
  
  // 转换为相对路径
  const relative = Path.relative('/home/user', '/home/user/documents/file.txt')
  console.log('Relative path:', relative)
  
  // 检查是否为绝对路径
  const isAbsolute = Path.isAbsolute('/home/user/file.txt')
  console.log('Is absolute:', isAbsolute)
  
  // 检查是否为相对路径
  const isRelative = Path.isRelative('./file.txt')
  console.log('Is relative:', isRelative)
  
  return { absolute, relative, isAbsolute, isRelative }
})

Effect.runSync(convertPath)
```

## 💡 路径组合

### 基本路径组合

```js
import { Effect, Path } from 'effect'

// 路径组合
const joinPaths = Effect.gen(function* () {
  // 组合路径
  const joined = Path.join('/home', 'user', 'documents', 'file.txt')
  console.log('Joined path:', joined)
  
  // 使用变量组合
  const baseDir = '/home/user'
  const subDir = 'documents'
  const filename = 'file.txt'
  const fullpath = Path.join(baseDir, subDir, filename)
  console.log('Full path:', fullpath)
  
  return { joined, fullpath }
})

// 规范化路径
const normalizePaths = Effect.gen(function* () {
  // 规范化路径
  const normalized = Path.normalize('/home/user/../user/./documents/file.txt')
  console.log('Normalized path:', normalized)
  
  // 处理重复分隔符
  const cleaned = Path.normalize('/home//user/documents///file.txt')
  console.log('Cleaned path:', cleaned)
  
  return { normalized, cleaned }
})

Effect.runSync(joinPaths)
```

### 路径解析与组合

```js
import { Effect, Path } from 'effect'

// 路径解析与组合
const parseAndJoin = Effect.gen(function* () {
  const baseDir = '/home/user'
  const relativePath = './documents/../documents/file.txt'
  
  // 解析相对路径
  const resolved = Path.resolve(baseDir, relativePath)
  console.log('Resolved path:', resolved)
  
  // 组合路径并规范化
  const combined = Path.normalize(Path.join(baseDir, 'documents', 'file.txt'))
  console.log('Combined path:', combined)
  
  return { resolved, combined }
})

Effect.runSync(parseAndJoin)
```

## 🚀 路径规范化

### 基本规范化

```js
import { Effect, Path } from 'effect'

// 路径规范化
const normalizePath = Effect.gen(function* () {
  // 规范化路径
  const paths = [
    '/home/user/../user/./documents/file.txt',
    './documents/../documents/file.txt',
    '/home//user/documents///file.txt',
    'documents/./file.txt'
  ]
  
  const normalized = paths.map(path => Path.normalize(path))
  console.log('Normalized paths:', normalized)
  
  return normalized
})

// 处理特殊路径
const handleSpecialPaths = Effect.gen(function* () {
  // 当前目录
  const current = Path.normalize('.')
  console.log('Current directory:', current)
  
  // 父目录
  const parent = Path.normalize('..')
  console.log('Parent directory:', parent)
  
  // 根目录
  const root = Path.normalize('/')
  console.log('Root directory:', root)
  
  return { current, parent, root }
})

Effect.runSync(normalizePath)
```

## 🎯 路径匹配

### 基本匹配

```js
import { Effect, Path } from 'effect'

// 路径匹配
const matchPath = Effect.gen(function* () {
  const pattern = '/home/user/**/*.txt'
  const path = '/home/user/documents/file.txt'
  
  // 简单匹配
  const matches = Path.match(pattern, path)
  console.log('Matches:', matches)
  
  return matches
})

// 通配符匹配
const wildcardMatch = Effect.gen(function* () {
  const patterns = [
    '/home/user/*.txt',
    '/home/user/**/*.md',
    '/home/**/file.txt'
  ]
  
  const paths = [
    '/home/user/file.txt',
    '/home/user/documents/file.txt',
    '/home/user/documents/readme.md',
    '/home/other/file.txt'
  ]
  
  const results = paths.map(path => ({
    path,
    matches: patterns.some(pattern => Path.match(pattern, path))
  }))
  
  console.log('Match results:', results)
  return results
})

Effect.runSync(matchPath)
```

### 模式匹配

```js
import { Effect, Path } from 'effect'

// 模式匹配
const patternMatch = Effect.gen(function* () {
  const patterns = [
    { pattern: '**/*.js', type: 'javascript' },
    { pattern: '**/*.ts', type: 'typescript' },
    { pattern: '**/*.md', type: 'markdown' },
    { pattern: '**/*.txt', type: 'text' }
  ]
  
  const files = [
    'src/index.js',
    'src/utils.ts',
    'docs/readme.md',
    'data/file.txt'
  ]
  
  const results = files.map(file => {
    const matched = patterns.find(({ pattern }) => Path.match(pattern, file))
    return {
      file,
      type: matched?.type || 'unknown'
    }
  })
  
  console.log('File types:', results)
  return results
})

Effect.runSync(patternMatch)
```

## 📝 实际应用示例

### 文件查找

```js
import { Effect, FileSystem, Path } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 文件查找器
const FileFinder = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  return {
    // 查找文件
    find: (dir: string, pattern: string) => Effect.gen(function* () {
      const entries = yield* fs.readDirectory(dir)
      const matches = []
      
      for (const entry of entries) {
        const fullPath = Path.join(dir, entry.name)
        
        if (entry.isDirectory) {
          // 递归查找
          const subMatches = yield* this.find(fullPath, pattern)
          matches.push(...subMatches)
        } else if (Path.match(pattern, entry.name)) {
          matches.push(fullPath)
        }
      }
      
      return matches
    }),
    
    // 查找特定扩展名的文件
    findByExtension: (dir: string, extension: string) => Effect.gen(function* () {
      const pattern = `**/*${extension}`
      return yield* this.find(dir, pattern)
    })
  }
})

// 使用文件查找器
const program = Effect.gen(function* () {
  const fileFinder = yield* FileFinder
  
  // 查找所有 JavaScript 文件
  const jsFiles = yield* fileFinder.findByExtension('./src', '.js')
  console.log('JavaScript files:', jsFiles)
  
  // 查找所有 Markdown 文件
  const mdFiles = yield* fileFinder.findByExtension('./docs', '.md')
  console.log('Markdown files:', mdFiles)
  
  return { jsFiles, mdFiles }
})

Effect.runSync(
  program.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```

### 路径验证

```js
import { Effect, Path, FileSystem } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 路径验证器
const PathValidator = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  return {
    // 验证路径是否存在
    exists: (path: string) => fs.exists(path),
    
    // 验证是否为文件
    isFile: (path: string) => Effect.gen(function* () {
      const exists = yield* fs.exists(path)
      if (!exists) {
        return Effect.succeed(false)
      }
      
      const stat = yield* fs.stat(path)
      return Effect.succeed(stat.isFile)
    }),
    
    // 验证是否为目录
    isDirectory: (path: string) => Effect.gen(function* () {
      const exists = yield* fs.exists(path)
      if (!exists) {
        return Effect.succeed(false)
      }
      
      const stat = yield* fs.stat(path)
      return Effect.succeed(stat.isDirectory)
    }),
    
    // 验证路径是否可访问
    isAccessible: (path: string) => Effect.gen(function* () {
      try {
        yield* fs.access(path)
        return Effect.succeed(true)
      } catch {
        return Effect.succeed(false)
      }
    })
  }
})

// 使用路径验证器
const program = Effect.gen(function* () {
  const validator = yield* PathValidator
  
  // 验证路径
  const path = './example.txt'
  const exists = yield* validator.exists(path)
  console.log('Exists:', exists)
  
  const isFile = yield* validator.isFile(path)
  console.log('Is file:', isFile)
  
  const isDir = yield* validator.isDirectory(path)
  console.log('Is directory:', isDir)
  
  return { exists, isFile, isDir }
})

Effect.runSync(
  program.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```

### 路径构建器

```js
import { Effect, Path } from 'effect'

// 路径构建器
const PathBuilder = Effect.gen(function* () {
  return {
    // 构建项目路径
    buildProjectPath: (basePath: string, projectName: string) => {
      return Path.join(basePath, 'projects', projectName)
    },
    
    // 构建资源路径
    buildResourcePath: (basePath: string, type: string, name: string) => {
      return Path.join(basePath, 'resources', type, name)
    },
    
    // 构建缓存路径
    buildCachePath: (basePath: string, key: string) => {
      const hash = key.split('').reduce((acc, char) => acc + char.charCodeAt(0), 0)
      return Path.join(basePath, 'cache', hash.toString(), key)
    },
    
    // 构建日志路径
    buildLogPath: (basePath: string, date: Date) => {
      const year = date.getFullYear()
      const month = String(date.getMonth() + 1).padStart(2, '0')
      const day = String(date.getDate()).padStart(2, '0')
      return Path.join(basePath, 'logs', year, month, `${day}.log`)
    }
  }
})

// 使用路径构建器
const program = Effect.gen(function* () {
  const pathBuilder = yield* PathBuilder
  
  const basePath = '/home/user'
  
  // 构建各种路径
  const projectPath = pathBuilder.buildProjectPath(basePath, 'my-project')
  console.log('Project path:', projectPath)
  
  const resourcePath = pathBuilder.buildResourcePath(basePath, 'images', 'logo.png')
  console.log('Resource path:', resourcePath)
  
  const cachePath = pathBuilder.buildCachePath(basePath, 'user:123')
  console.log('Cache path:', cachePath)
  
  const logPath = pathBuilder.buildLogPath(basePath, new Date())
  console.log('Log path:', logPath)
  
  return { projectPath, resourcePath, cachePath, logPath }
})

Effect.runSync(program)
```
