# 文件操作

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了跨平台的文件系统抽象，通过 `@effect/platform` 包提供统一的文件操作接口。

## 🎯 @effect/platform - 平台文件系统

### 安装与设置

```js
// 安装依赖
// npm install @effect/platform
// npm install @effect/platform-node

// 基本导入
import { Effect } from 'effect'
import { FileSystem } from '@effect/platform'
import { NodeFileSystem } from '@effect/platform-node'
```

### 基本文件系统访问

```js
import { Effect, FileSystem } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 访问文件系统
const program = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  return fs
})

// 在 Node.js 环境中运行
Effect.runSync(
  program.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```

## 💡 FileSystem 抽象

### 文件系统接口

```js
import { Effect, FileSystem } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 文件系统操作
const program = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  return {
    // 读取文件
    readFile: (path: string) => fs.readFileString(path),
    
    // 写入文件
    writeFile: (path: string, content: string) => fs.writeFileString(path, content),
    
    // 检查文件是否存在
    exists: (path: string) => fs.exists(path),
    
    // 删除文件
    remove: (path: string) => fs.remove(path),
    
    // 创建目录
    makeDirectory: (path: string) => fs.makeDirectory(path),
    
    // 读取目录
    readDirectory: (path: string) => fs.readDirectory(path)
  }
})

Effect.runSync(
  program.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```

## 🎯 目录与权限管理

### 目录操作

```js
import { Effect, FileSystem } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 创建目录
const createDirectory = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  // 创建单个目录
  yield* fs.makeDirectory('./new-dir')
  console.log('Directory created')
  
  // 创建嵌套目录
  yield* fs.makeDirectory('./parent/child/grandchild', { recursive: true })
  console.log('Nested directories created')
})

// 读取目录
const readDirectory = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  const entries = yield* fs.readDirectory('./')
  console.log('Directory entries:', entries)
  
  return entries
})

// 删除目录
const deleteDirectory = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  yield* fs.remove('./directory-to-delete', { recursive: true })
  console.log('Directory deleted')
})

Effect.runSync(
  createDirectory.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```

### 文件权限

```js
import { Effect, FileSystem } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 检查文件权限
const checkPermissions = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  const exists = yield* fs.exists('./example.txt')
  console.log('File exists:', exists)
  
  return exists
})

// 设置文件权限
const setPermissions = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  // 设置文件权限（使用八进制权限模式）
  // 0o644 = rw-r--r-- (所有者可读写，组和其他用户只读)
  yield* fs.chmod('./example.txt', 0o644)
  console.log('File permissions set to 644')
  
  // 设置可执行权限
  // 0o755 = rwxr-xr-x (所有者可读写执行，组和其他用户可读执行)
  yield* fs.chmod('./script.sh', 0o755)
  console.log('File permissions set to 755')
})

Effect.runSync(
  checkPermissions.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```

## 📝 实际应用示例

### 配置文件管理

```js
import { Effect, FileSystem } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 配置文件管理器
const ConfigManager = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  const configPath = './config.json'
  
  return {
    // 读取配置
    readConfig: Effect.gen(function* () {
      const exists = yield* fs.exists(configPath)
      
      if (!exists) {
        return Effect.succeed({})
      }
      
      const content = yield* fs.readFileString(configPath)
      return JSON.parse(content)
    }),
    
    // 写入配置
    writeConfig: (config: any) => Effect.gen(function* () {
      const content = JSON.stringify(config, null, 2)
      yield* fs.writeFileString(configPath, content)
    }),
    
    // 更新配置
    updateConfig: (updater: (config: any) => any) => Effect.gen(function* () {
      const config = yield* this.readConfig
      const updated = updater(config)
      yield* this.writeConfig(updated)
      return updated
    })
  }
})

// 使用配置管理器
const program = Effect.gen(function* () {
  const configManager = yield* ConfigManager
  
  // 读取配置
  const config = yield* configManager.readConfig
  console.log('Current config:', config)
  
  // 更新配置
  const updated = yield* configManager.updateConfig((config) => ({
    ...config,
    version: '1.0.0',
    updatedAt: new Date().toISOString()
  }))
  console.log('Updated config:', updated)
  
  return config
})

Effect.runSync(
  program.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```

### 日志文件管理

```js
import { Effect, FileSystem } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 日志管理器
const LogManager = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  const logPath = './app.log'
  
  return {
    // 写入日志
    writeLog: (message: string, level: string = 'INFO') => Effect.gen(function* () {
      const timestamp = new Date().toISOString()
      const logEntry = `[${timestamp}] [${level}] ${message}\n`
      
      const exists = yield* fs.exists(logPath)
      const existingContent = exists ? yield* fs.readFileString(logPath) : ''
      
      yield* fs.writeFileString(logPath, existingContent + logEntry)
    }),
    
    // 读取日志
    readLogs: Effect.gen(function* () {
      const exists = yield* fs.exists(logPath)
      
      if (!exists) {
        return Effect.succeed([])
      }
      
      const content = yield* fs.readFileString(logPath)
      return content.split('\n').filter(line => line.trim())
    }),
    
    // 清空日志
    clearLogs: Effect.gen(function* () {
      yield* fs.writeFileString(logPath, '')
    })
  }
})

// 使用日志管理器
const program = Effect.gen(function* () {
  const logManager = yield* LogManager
  
  // 写入日志
  yield* logManager.writeLog('Application started')
  yield* logManager.writeLog('Processing data', 'DEBUG')
  yield* logManager.writeLog('Error occurred', 'ERROR')
  
  // 读取日志
  const logs = yield* logManager.readLogs
  console.log('Logs:', logs)
  
  return logs
})

Effect.runSync(
  program.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```

### 文件监控

```js
import { Effect, FileSystem } from 'effect'
import { NodeFileSystem } from '@effect/platform-node'

// 文件监控器
const FileWatcher = Effect.gen(function* () {
  const fs = yield* FileSystem.FileSystem
  
  return {
    // 监控文件变化
    watch: (path: string, callback: (event: any) => void) => Effect.gen(function* () {
      console.log(`Watching file: ${path}`)
      
      // 简化的监控实现
      // 实际应用中可能需要使用平台特定的 API
      let lastModified = 0
      
      const check = Effect.gen(function* () {
        const exists = yield* fs.exists(path)
        
        if (exists) {
          const stat = yield* fs.stat(path)
          
          if (stat.mtimeMs !== lastModified) {
            lastModified = stat.mtimeMs
            callback({ type: 'change', path })
          }
        }
      })
      
      // 定期检查
      yield* Effect.repeat(check, {
        schedule: Effect.scheduleSpaced('1s')
      })
    })
  }
})

// 使用文件监控器
const program = Effect.gen(function* () {
  const fileWatcher = yield* FileWatcher
  
  // 监控文件变化
  yield* fileWatcher.watch('./example.txt', (event) => {
    console.log('File changed:', event)
  })
})

Effect.runSync(
  program.pipe(
    Effect.provide(NodeFileSystem.layer)
  )
)
```
