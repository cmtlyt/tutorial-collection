### 依赖版本管理

```js
import { Effect } from 'effect'

// 依赖版本检查工具
const DependencyChecker = {
  versions: {
    node: '>=18.0.0',
    npm: '>=9.0.0',
    effect: '^3.0.0'
  },
  
  checkNodeVersion: Effect.gen(function* () {
    const version = process.version
    const required = DependencyChecker.versions.node
    
    console.log(`Node version: ${version}`)
    console.log(`Required: ${required}`)
    
    if (!version.startsWith('v18') && !version.startsWith('v20')) {
      return Effect.fail(new Error(`Node version ${version} does not meet requirement ${required}`))
    }
    
    return Effect.succeed(version)
  }),
  
  checkEffectVersion: Effect.gen(function* () {
    const packageJson = yield* Effect.tryPromise(() => 
      import('../package.json', { assert: { type: 'json' } })
    )
    
    const effectVersion = packageJson.default.dependencies.effect
    const required = DependencyChecker.versions.effect
    
    console.log(`Effect version: ${effectVersion}`)
    console.log(`Required: ${required}`)
    
    return Effect.succeed(effectVersion)
  }),
  
  checkAll: Effect.gen(function* () {
    console.log('Checking dependencies...\n')
    
    const nodeResult = yield* DependencyChecker.checkNodeVersion
    const effectResult = yield* DependencyChecker.checkEffectVersion
    
    console.log('\n✓ All dependencies meet requirements')
    
    return {
      node: nodeResult,
      effect: effectResult
    }
  })
}

Effect.runSync(DependencyChecker.checkAll)
```

### 构建脚本

```js
import { Effect } from 'effect'
import { exec } from 'child_process'
import { promisify } from 'util'

const execAsync = promisify(exec)

// 构建工具
const BuildTool = {
  clean: Effect.gen(function* () {
    console.log('Cleaning dist directory...')
    yield* Effect.tryPromise(() => execAsync('rm -rf dist'))
    console.log('✓ Clean completed')
  }),
  
  typeCheck: Effect.gen(function* () {
    console.log('Running type check...')
    yield* Effect.tryPromise(() => execAsync('tsc --noEmit'))
    console.log('✓ Type check passed')
  }),
  
  lint: Effect.gen(function* () {
    console.log('Running linter...')
    yield* Effect.tryPromise(() => execAsync('eslint src --ext .ts,.tsx'))
    console.log('✓ Lint passed')
  }),
  
  build: Effect.gen(function* () {
    console.log('Building project...')
    yield* Effect.tryPromise(() => execAsync('vite build'))
    console.log('✓ Build completed')
  }),
  
  fullBuild: Effect.gen(function* () {
    console.log('Starting full build...\n')
    
    yield* BuildTool.clean
    yield* BuildTool.typeCheck
    yield* BuildTool.lint
    yield* BuildTool.build
    
    console.log('\n✓ Full build completed successfully')
  })
}

Effect.runSync(BuildTool.fullBuild)
```

## 🔧 环境变量

### 环境配置

```js
import { Effect } from 'effect'

// 环境配置工具
const EnvConfig = {
  development: {
    apiUrl: 'http://localhost:3000',
    logLevel: 'debug',
    enableDebugTools: true
  },
  
  production: {
    apiUrl: 'https://api.example.com',
    logLevel: 'error',
    enableDebugTools: false
  },
  
  test: {
    apiUrl: 'http://localhost:3000',
    logLevel: 'silent',
    enableDebugTools: false
  },
  
  load: Effect.gen(function* () {
    const env = process.env.NODE_ENV || 'development'
    const config = EnvConfig[env]
    
    console.log(`Loading config for: ${env}`)
    console.log(`API URL: ${config.apiUrl}`)
    console.log(`Log Level: ${config.logLevel}`)
    console.log(`Debug Tools: ${config.enableDebugTools}`)
    
    return config
  }),
  
  validate: (config) => {
    if (!config.apiUrl) {
      throw new Error('API URL is required')
    }
    
    if (!['debug', 'info', 'warn', 'error', 'silent'].includes(config.logLevel)) {
      throw new Error('Invalid log level')
    }
    
    return true
  }
}

Effect.runSync(
  EnvConfig.load.pipe(
    Effect.tap(EnvConfig.validate)
  )
)
```

## 📝 实际应用示例

### 完整的开发环境配置

```js
import { Effect } from 'effect'
import { exec } from 'child_process'
import { promisify } from 'util'
import { readFileSync } from 'fs'
import { resolve } from 'path'

const execAsync = promisify(exec)

// 完整的开发环境配置
const DevEnvironment = {
  config: {
    env: process.env.NODE_ENV || 'development',
    port: process.env.PORT || 3000,
    apiUrl: process.env.API_URL || 'http://localhost:3000'
  },
  
  checkDependencies: Effect.gen(function* () {
    console.log('Checking dependencies...')
    
    try {
      yield* Effect.tryPromise(() => execAsync('npm list --depth=0'))
      console.log('✓ Dependencies installed')
    } catch (error) {
      console.log('⚠ Some dependencies may be missing')
      console.log('Run: npm install')
    }
  }),
  
  checkConfig: Effect.gen(function* () {
    console.log('Checking configuration...')
    
    const configPath = resolve(process.cwd(), 'tsconfig.json')
    const config = JSON.parse(readFileSync(configPath, 'utf-8'))
    
    if (!config.compilerOptions) {
      throw new Error('Invalid tsconfig.json')
    }
    
    console.log('✓ Configuration valid')
    return config
  }),
  
  setup: Effect.gen(function* () {
    console.log('Setting up development environment...\n')
    
    yield* DevEnvironment.checkDependencies
    yield* DevEnvironment.checkConfig
    
    console.log('\n✓ Development environment ready')
    console.log(`Environment: ${DevEnvironment.config.env}`)
    console.log(`Port: ${DevEnvironment.config.port}`)
    console.log(`API URL: ${DevEnvironment.config.apiUrl}`)
    
    return DevEnvironment.config
  }),
  
  startDevServer: Effect.gen(function* () {
    console.log('\nStarting development server...')
    console.log(`http://localhost:${DevEnvironment.config.port}`)
    
    yield* Effect.never
  })
}

// 启动开发环境
Effect.runSync(
  Effect.gen(function* () {
    yield* DevEnvironment.setup
    yield* DevEnvironment.startDevServer
  })
)
```

### 项目初始化脚本

```js
import { Effect } from 'effect'
import { exec } from 'child_process'
import { promisify } from 'util'
import { mkdirSync, writeFileSync } from 'fs'
import { resolve } from 'path'

const execAsync = promisify(exec)

// 项目初始化工具
const ProjectInit = {
  createDirectory: (path) => Effect.gen(function* () {
    console.log(`Creating directory: ${path}`)
    mkdirSync(path, { recursive: true })
  }),
  
  createFile: (path, content) => Effect.gen(function* () {
    console.log(`Creating file: ${path}`)
    writeFileSync(path, content, 'utf-8')
  }),
  
  initGit: Effect.gen(function* () {
    console.log('Initializing Git repository...')
    yield* Effect.tryPromise(() => execAsync('git init'))
    console.log('✓ Git initialized')
  }),
  
  installDependencies: Effect.gen(function* () {
    console.log('Installing dependencies...')
    yield* Effect.tryPromise(() => execAsync('npm install'))
    console.log('✓ Dependencies installed')
  }),
  
  init: (projectName) => Effect.gen(function* () {
    const projectPath = resolve(process.cwd(), projectName)
    
    console.log(`Creating Effect project: ${projectName}\n`)
    
    // 创建目录结构
    yield* ProjectInit.createDirectory(projectPath)
    yield* ProjectInit.createDirectory(`${projectPath}/src`)
    yield* ProjectInit.createDirectory(`${projectPath}/tests`)
    
    // 创建 package.json
    yield* ProjectInit.createFile(
      `${projectPath}/package.json`,
      JSON.stringify({
        name: projectName,
        version: '1.0.0',
        type: 'module',
        scripts: {
          dev: 'vite',
          build: 'tsc && vite build',
          test: 'vitest'
        },
        dependencies: {
          effect: '^3.0.0'
        },
        devDependencies: {
          typescript: '^5.0.0',
          vite: '^5.0.0',
          vitest: '^1.0.0'
        }
      }, null, 2)
    )
    
    // 创建 tsconfig.json
    yield* ProjectInit.createFile(
      `${projectPath}/tsconfig.json`,
      JSON.stringify({
        compilerOptions: {
          target: 'ES2022',
          module: 'ESNext',
          strict: true,
          moduleResolution: 'bundler',
          esModuleInterop: true,
          skipLibCheck: true
        },
        include: ['src/**/*']
      }, null, 2)
    )
    
    // 创建示例文件
    yield* ProjectInit.createFile(
      `${projectPath}/src/index.ts`,
      `import { Effect } from 'effect'

const program = Effect.gen(function* () {
  console.log('Hello from Effect!')
  return 'Success'
})

Effect.runSync(program)
`
    )
    
    // 初始化 Git
    process.chdir(projectPath)
    yield* ProjectInit.initGit
    
    // 安装依赖
    yield* ProjectInit.installDependencies
    
    console.log('\n✓ Project initialized successfully!')
    console.log(`\nNext steps:`)
    console.log(`  cd ${projectName}`)
    console.log(`  npm run dev`)
  })
}

// 使用示例

```
