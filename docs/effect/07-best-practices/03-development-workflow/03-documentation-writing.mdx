import { Effect } from 'effect'

/**
 * 计算订单总价
 * 
 * 根据订单中的商品数量和单价计算总价，并应用折扣
 * 
 * @param {Array<{price: number, quantity: number}>} items - 订单商品列表
 * @param {number} discountRate - 折扣率（0-1）
 * @returns {number} 订单总价
 * 
 * @example
 * const total = calculateOrderTotal([
 *   { price: 100, quantity: 2 },
 *   { price: 50, quantity: 1 }
 * ], 0.1)
 * console.log(total) // 225
 */
const calculateOrderTotal = (items, discountRate = 0) => {
  const subtotal = items.reduce((sum, item) => sum + item.price * item.quantity, 0)
  return subtotal * (1 - discountRate)
}

/**
 * 格式化日期为本地字符串
 * 
 * @param {Date} date - 日期对象
 * @param {string} [locale='zh-CN'] - 本地化设置
 * @returns {string} 格式化后的日期字符串
 * 
 * @example
 * const formatted = formatDate(new Date(), 'zh-CN')
 * console.log(formatted) // "2024-01-01"
 */
const formatDate = (date, locale = 'zh-CN') => {
  return date.toLocaleDateString(locale)
}

/**
 * 深度克隆对象
 * 
 * 创建对象的深拷贝，支持嵌套对象和数组
 * 
 * @param {any} obj - 要克隆的对象
 * @returns {any} 克隆后的对象
 * 
 * @example
 * const cloned = deepClone({ a: { b: 1 } })
 * cloned.a.b = 2
 * console.log(original.a.b) // 1
 */
const deepClone = (obj) => {
  if (obj === null || typeof obj !== 'object') {
    return obj
  }
  
  if (obj instanceof Date) {
    return new Date(obj)
  }
  
  if (obj instanceof Array) {
    return obj.map(item => deepClone(item))
  }
  
  const cloned = {}
  for (const key in obj) {
    if (obj.hasOwnProperty(key)) {
      cloned[key] = deepClone(obj[key])
    }
  }
  
  return cloned
}

### API 文档生成

```js
import { Effect } from 'effect'

/**
 * API 文档生成器
 * 
 * 自动生成 API 文档，包括方法签名、参数说明、返回值、示例等
 */
const ApiDocGenerator = {
  /**
   * 生成方法文档
   * 
   * @param {string} methodName - 方法名称
   * @param {Object} method - 方法对象
   * @returns {string} 生成的文档字符串
   */
  generateMethodDoc: (methodName, method) => {
    const lines = []
    
    lines.push(`### ${methodName}`)
    lines.push('')
    
    if (method.description) {
      lines.push(method.description)
      lines.push('')
    }
    
    if (method.params) {
      lines.push('#### 参数')
      lines.push('')
      method.params.forEach(param => {
        lines.push(`- **${param.name}** (\`${param.type}\`): ${param.description}`)
      })
      lines.push('')
    }
    
    if (method.returns) {
      lines.push('#### 返回值')
      lines.push('')
      lines.push(`\`${method.returns.type}\`: ${method.returns.description}`)
      lines.push('')
    }
    
    if (method.example) {
      lines.push('#### 示例')
      lines.push('')
      lines.push('```js')
      lines.push(method.example)
      lines.push('```')
      lines.push('')
    }
    
    return lines.join('\n')
  },
  
  /**
   * 生成模块文档
   * 
   * @param {string} moduleName - 模块名称
   * @param {Object} module - 模块对象
   * @returns {string} 生成的文档字符串
   */
  generateModuleDoc: (moduleName, module) => {
    const lines = []
    
    lines.push(`# ${moduleName}`)
    lines.push('')
    
    if (module.description) {
      lines.push(module.description)
      lines.push('')
    }
    
    lines.push('## 方法')
    lines.push('')
    
    Object.entries(module.methods).forEach(([methodName, method]) => {
      lines.push(ApiDocGenerator.generateMethodDoc(methodName, method))
    })
    
    return lines.join('\n')
  }
}

// 使用示例
const userServiceModule = {
  description: '用户服务模块，提供用户相关的业务逻辑',
  methods: {
    getUser: {
      description: '根据用户 ID 获取用户信息',
      params: [
        { name: 'userId', type: 'number', description: '用户 ID' }
      ],
      returns: { type: 'User', description: '用户信息' },
      example: 'const user = yield* UserService.getUser(1)'
    },
    createUser: {
      description: '创建新用户',
      params: [
        { name: 'name', type: 'string', description: '用户名称' },

      ],
      returns: { type: 'User', description: '创建的用户' },
      example: 'const user = yield* UserService.createUser({ name: "John", email: "john@example.com" })'
    }
  }
}

### 文档生成工具

```js
import { Effect } from 'effect'
import { readFileSync, writeFileSync } from 'fs'
import { resolve } from 'path'

/**
 * 文档生成工具
 * 
 * 从源代码中提取注释并生成文档
 */
const DocGenerator = {
  /**
   * 提取 JSDoc 注释
   * 
   * @param {string} code - 源代码
   * @returns {Array} 注释列表
   */
  extractComments: (code) => {
    const comments = []
    const regex = /\/\*\*([\s\S]*?)\*\//g
    let match
    
    while ((match = regex.exec(code)) !== null) {
      comments.push(match[1].trim())
    }
    
    return comments
  },
  
  /**
   * 解析 JSDoc 注释
   * 
   * @param {string} comment - JSDoc 注释
   * @returns {Object} 解析后的对象
   */
  parseComment: (comment) => {
    const lines = comment.split('\n').map(line => line.trim().replace(/^\*\s?/, ''))
    const result = {
      description: '',
      params: [],
      returns: null,
      example: null
    }
    
    let currentSection = 'description'
    
    for (const line of lines) {
      if (line.startsWith('@param')) {
        currentSection = 'params'
        const parts = line.replace('@param', '').trim().split(/\s+/)
        result.params.push({
          name: parts[0],
          type: parts[1]?.replace(/[{}]/g, ''),
          description: parts.slice(2).join(' ')
        })
      } else if (line.startsWith('@returns')) {
        currentSection = 'returns'
        const parts = line.replace('@returns', '').trim().split(/\s+/)
        result.returns = {
          type: parts[0]?.replace(/[{}]/g, ''),
          description: parts.slice(1).join(' ')
        }
      } else if (line.startsWith('@example')) {
        currentSection = 'example'
        result.example = ''
      } else if (currentSection === 'example') {
        result.example += line + '\n'
      } else if (currentSection === 'description' && line) {
        result.description += line + '\n'
      }
    }
    
    return result
  },
  
  /**
   * 生成文档
   * 
   * @param {string} sourcePath - 源代码路径
   * @param {string} outputPath - 输出路径
   */
  generate: (sourcePath, outputPath) => Effect.gen(function* () {
    const code = readFileSync(sourcePath, 'utf-8')
    const comments = DocGenerator.extractComments(code)
    
    const docs = comments.map(comment => {
      const parsed = DocGenerator.parseComment(comment)
      return `### 函数说明\n\n${parsed.description}\n\n**参数**\n\n${parsed.params.map(p => `- \`${p.name}\` (\`${p.type}\`): ${p.description}`).join('\n')}\n\n**返回值**\n\n\`${parsed.returns?.type}\`: ${parsed.returns?.description}\n\n**示例**\n\n\`\`\`js\n${parsed.example?.trim()}\n\`\`\``
    }).join('\n\n---\n\n')
    
    writeFileSync(outputPath, docs, 'utf-8')
    console.log(`Documentation generated: ${outputPath}`)
  })
}

import { Effect } from 'effect'

/**
 * 综合文档系统
 * 
 * 提供完整的文档生成和管理功能
 */
const DocSystem = {
  /**
   * 生成类型文档
   * 
   * @param {Object} types - 类型定义
   * @returns {string} 类型文档
   */
  generateTypeDoc: (types) => {
    const sections = []
    
    sections.push('## 类型定义')
    sections.push('')
    
    Object.entries(types).forEach(([name, type]) => {
      sections.push(`### ${name}`)
      sections.push('')
      sections.push(type.description)
      sections.push('')
      
      if (type.properties) {
        sections.push('#### 属性')
        sections.push('')
        Object.entries(type.properties).forEach(([propName, prop]) => {
          sections.push(`- **${propName}** (\`${prop.type}\`): ${prop.description}`)
        })
        sections.push('')
      }
    })
    
    return sections.join('\n')
  },
  
  /**
   * 生成函数文档
   * 
   * @param {Object} functions - 函数定义
   * @returns {string} 函数文档
   */
  generateFunctionDoc: (functions) => {
    const sections = []
    
    sections.push('## 函数')
    sections.push('')
    
    Object.entries(functions).forEach(([name, func]) => {
      sections.push(`### ${name}`)
      sections.push('')
      sections.push(func.description)
      sections.push('')
      
      if (func.params) {
        sections.push('#### 参数')
        sections.push('')
        func.params.forEach(param => {
          sections.push(`- **${param.name}** (\`${param.type}\`): ${param.description}`)
        })
        sections.push('')
      }
      
      if (func.returns) {
        sections.push('#### 返回值')
        sections.push('')
        sections.push(`\`${func.returns.type}\`: ${func.returns.description}`)
        sections.push('')
      }
      
      if (func.example) {
        sections.push('#### 示例')
        sections.push('')
        sections.push('```js')
        sections.push(func.example)
        sections.push('```')
        sections.push('')
      }
    })
    
    return sections.join('\n')
  },
  
  /**
   * 生成完整文档
   * 
   * @param {Object} config - 文档配置
   * @returns {string} 完整文档
   */
  generate: (config) => {
    const sections = []
    
    // 标题
    sections.push(`# ${config.title}`)
    sections.push('')
    
    if (config.description) {
      sections.push(config.description)
      sections.push('')
    }
    
    // 目录
    sections.push('## 目录')
    sections.push('')
    config.sections.forEach(section => {
      sections.push(`- [${section.name}](#${section.id})`)
    })
    sections.push('')
    
    // 类型文档
    if (config.types) {
      sections.push(DocSystem.generateTypeDoc(config.types))
    }
    
    // 函数文档
    if (config.functions) {
      sections.push(DocSystem.generateFunctionDoc(config.functions))
    }
    
    return sections.join('\n')
  }
}
