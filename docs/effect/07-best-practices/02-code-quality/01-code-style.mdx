## 🎯 命名规范

### 基本命名规范

```js
import { Effect } from 'effect'

// 基本命名规范
const program = Effect.gen(function* () {
  // 变量命名：使用 camelCase
  const userName = 'John Doe'
  const userEmail = 'john@example.com'
  const isValidEmail = true
  
  // 函数命名：使用动词开头，camelCase
  const getUserById = (id) => Effect.gen(function* () {
    console.log('Getting user by id:', id)
    return Effect.succeed({ id, name: 'John Doe' })
  })
  
  const validateUserData = (data) => Effect.gen(function* () {
    console.log('Validating user data:', data)
    return Effect.succeed(true)
  })
  
  // 常量命名：使用 UPPER_SNAKE_CASE
  const MAX_RETRY_COUNT = 3
  const DEFAULT_TIMEOUT = 5000
  const API_BASE_URL = 'https://api.example.com'
  
  // 类型/接口命名：使用 PascalCase
  const User = {
    id: 1,
    name: 'John Doe',
    email: 'john@example.com'
  }
  
  const UserService = {
    create: (data) => Effect.succeed({ id: 1, ...data }),
    get: (id) => Effect.succeed({ id, name: 'John Doe' })
  }
  
  // 私有变量：使用 _ 前缀
  const _internalState = new Map()
  const _logger = (message) => console.log(`[Internal] ${message}`)
  
  console.log('Variables:', { userName, userEmail, isValidEmail })
  console.log('Constants:', { MAX_RETRY_COUNT, DEFAULT_TIMEOUT, API_BASE_URL })
  
  return { userName, userEmail, isValidEmail }
})

Effect.runSync(program)
```

### Effect 命名规范

```js
import { Effect } from 'effect'

// Effect 命名规范
const program = Effect.gen(function* () {
  // Effect 程序命名：使用描述性的动词+名词
  const fetchUserData = (userId) => Effect.gen(function* () {
    console.log('Fetching user data for:', userId)
    yield* Effect.sleep(100)
    return Effect.succeed({ id: userId, name: 'John Doe' })
  })
  
  const validateUserInput = (input) => Effect.gen(function* () {
    console.log('Validating user input:', input)
    return Effect.succeed(input.length > 0)
  })
  
  const saveUserToDatabase = (user) => Effect.gen(function* () {
    console.log('Saving user to database:', user)
    yield* Effect.sleep(100)
    return Effect.succeed({ ...user, saved: true })
  })
  
  // 使用 Effect 程序
  const user = yield* fetchUserData(1)
  console.log('Fetched user:', user)
  
  const isValid = yield* validateUserInput('test')
  console.log('Is valid:', isValid)
  
  const saved = yield* saveUserToDatabase(user)
  console.log('Saved user:', saved)
  
  return { user, isValid, saved }
})

Effect.runSync(program)
```

## 💡 代码组织

### 文件组织

```js
// 文件组织示例

// 1. 导入语句
import { Effect } from 'effect'
import { pipe } from 'effect/Function'

// 2. 类型定义
interface User {
  id: number
  name: string
  email: string
}

interface CreateUserInput {
  name: string
  email: string
}

// 3. 常量定义
const MAX_NAME_LENGTH = 100
const MIN_NAME_LENGTH = 2
const EMAIL_REGEX = /^[^\s@]+@[^\s@]+\.[^\s@]+$/

// 4. 辅助函数
const validateEmail = (email: string): boolean => {
  return EMAIL_REGEX.test(email)
}

const validateName = (name: string): boolean => {
  return name.length >= MIN_NAME_LENGTH && name.length <= MAX_NAME_LENGTH
}

// 5. Effect 程序
const validateUserInput = (input: CreateUserInput): Effect.Effect<never, Error, CreateUserInput> => {
  if (!validateName(input.name)) {
    return Effect.fail(new Error('Invalid name'))
  }
  
  if (!validateEmail(input.email)) {
    return Effect.fail(new Error('Invalid email'))
  }
  
  return Effect.succeed(input)
}

// 6. 服务定义
const UserService = {
  create: (input: CreateUserInput): Effect.Effect<never, Error, User> => 
    Effect.gen(function* () {
      const validated = yield* validateUserInput(input)
      
      const user: User = {
        id: Date.now(),
        name: validated.name,
        email: validated.email
      }
      
      return user
    })
}

// 7. 导出
export { User, CreateUserInput, UserService }
```

### 模块组织

```js
import { Effect } from 'effect'

// 模块组织示例
const program = Effect.gen(function* () {
  // 按功能组织代码
  const UserModule = {
    // 类型定义
    Types: {
      User: {
        id: 1,
        name: 'John Doe',
        email: 'john@example.com'
      },
      
      CreateUserInput: {
        name: 'John Doe',
        email: 'john@example.com'
      }
    },
    
    // 验证逻辑
    Validation: {
      validateName: (name) => Effect.gen(function* () {
        if (!name || name.length < 2) {
          return Effect.fail(new Error('Name is too short'))
        }
        return Effect.succeed(name)
      }),
      
      validateEmail: (email) => Effect.gen(function* () {
        if (!email || !email.includes('@')) {
          return Effect.fail(new Error('Invalid email'))
        }
        return Effect.succeed(email)
      })
    },
    
    // 业务逻辑
    Services: {
      create: (input) => Effect.gen(function* () {
        const name = yield* UserModule.Validation.validateName(input.name)
        const email = yield* UserModule.Validation.validateEmail(input.email)
        
        return {
          id: Date.now(),
          name,
          email
        }
      })
    }
  }
  
  // 使用模块
  const user = yield* UserModule.Services.create({
    name: 'John Doe',
    email: 'john@example.com'
  })
  
  console.log('Created user:', user)
  
  return user
})

Effect.runSync(program)
```

## 🚀 注释文档

### 基本注释

```js
import { Effect } from 'effect'

// 基本注释示例
const program = Effect.gen(function* () {
  /**
   * 计算两个数的和
   * @param {number} a - 第一个数
   * @param {number} b - 第二个数
   * @returns {Effect.Effect<never, never, number>} 两个数的和
   */
  const add = (a, b) => Effect.succeed(a + b)
  
  /**
   * 验证用户输入
   * @param {Object} input - 用户输入数据
   * @param {string} input.name - 用户名
   * @param {string} input.email - 用户邮箱
   * @returns {Effect.Effect<never, Error, Object>} 验证结果
   * @throws {Error} 当输入无效时抛出错误
   */
  const validateInput = (input) => Effect.gen(function* () {
    if (!input.name || input.name.length < 2) {
      return Effect.fail(new Error('Name is too short'))
    }
    
    if (!input.email || !input.email.includes('@')) {
      return Effect.fail(new Error('Invalid email'))
    }
    
    return Effect.succeed(input)
  })
  
  /**
   * 获取用户信息
   * @param {number} userId - 用户ID
   * @returns {Effect.Effect<never, Error, Object>} 用户信息
   * 
   * @example
   * ```js
   * const user = yield* getUser(1)
   * console.log(user)
   * ```
   */
  const getUser = (userId) => Effect.gen(function* () {
    console.log(`Fetching user ${userId}`)
    yield* Effect.sleep(100)
    return Effect.succeed({ id: userId, name: 'John Doe' })
  })
  
  // 使用注释
  const sum = yield* add(1, 2)
  console.log('Sum:', sum)
  
  const user = yield* getUser(1)
  console.log('User:', user)
  
  return { sum, user }
})

Effect.runSync(program)
```

## 🎯 类型定义

### 基本类型定义

```js
import { Effect } from 'effect'

// 基本类型定义
const program = Effect.gen(function* () {
  // 定义类型
  /**
   * @typedef {Object} User
   * @property {number} id - 用户ID
   * @property {string} name - 用户名
   * @property {string} email - 用户邮箱
   */
  
  /**
   * @typedef {Object} CreateUserInput
   * @property {string} name - 用户名
   * @property {string} email - 用户邮箱
   */
  
  /**
   * @typedef {Object} ValidationError
   * @property {string} field - 错误字段
   * @property {string} message - 错误消息
   */
  
  // 使用类型
  /**
   * 验证用户输入
   * @param {CreateUserInput} input - 用户输入
   * @returns {Effect.Effect<never, ValidationError, CreateUserInput>} 验证结果
   */
  const validateInput = (input) => Effect.gen(function* () {
    if (!input.name || input.name.length < 2) {
      return Effect.fail({ field: 'name', message: 'Name is too short' })
    }
    
    if (!input.email || !input.email.includes('@')) {
      return Effect.fail({ field: 'email', message: 'Invalid email' })
    }
    
    return Effect.succeed(input)
  })
  
  /**
   * 创建用户
   * @param {CreateUserInput} input - 用户输入
   * @returns {Effect.Effect<never, Error, User>} 创建的用户
   */
  const createUser = (input) => Effect.gen(function* () {
    const validated = yield* validateInput(input)
    
    return {
      id: Date.now(),
      name: validated.name,
      email: validated.email
    }
  })
  
  // 使用类型定义的函数
  const user = yield* createUser({
    name: 'John Doe',
    email: 'john@example.com'
  })
  
  console.log('Created user:', user)
  
  return user
})

Effect.runSync(program)
```

### 高级类型定义

```js
import { Effect } from 'effect'

// 高级类型定义
const program = Effect.gen(function* () {
  // 定义泛型类型
  /**
   * @template T
   * @typedef {Object} ServiceResponse
   * @property {boolean} success - 是否成功
   * @property {T} [data] - 响应数据
   * @property {string} [error] - 错误信息
   */
  
  /**
   * @template T
   * @typedef {Object} PaginatedResponse
   * @property {T[]} items - 数据项
   * @property {number} total - 总数
   * @property {number} page - 当前页
   * @property {number} pageSize - 每页大小
   */
  
  /**
   * @template T
   * @param {T[]} items - 数据项
   * @param {number} page - 当前页
   * @param {number} pageSize - 每页大小
   * @returns {PaginatedResponse<T>} 分页响应
   */
  const paginate = (items, page = 1, pageSize = 10) => {
    const start = (page - 1) * pageSize
    const end = start + pageSize
    const paginatedItems = items.slice(start, end)
    
    return {
      items: paginatedItems,
      total: items.length,
      page,
      pageSize
    }
  }
  
  // 使用泛型类型
  const users = [
    { id: 1, name: 'John Doe' },
    { id: 2, name: 'Jane Smith' },
    { id: 3, name: 'Bob Johnson' }
  ]
  
  const paginatedUsers = paginate(users, 1, 2)
  console.log('Paginated users:', paginatedUsers)
  
  return paginatedUsers
})

Effect.runSync(program)
```

## 📝 实际应用示例

### 综合代码风格示例

```js
import { Effect } from 'effect'

// 综合代码风格示例
const program = Effect.gen(function* () {
  /**
   * 用户管理模块
   * @namespace UserManagement
   */
  
  const UserManagement = {
    // ==================== 类型定义 ====================
    /**
     * @typedef {Object} User
     * @property {number} id - 用户ID
     * @property {string} name - 用户名
     * @property {string} email - 用户邮箱
     * @property {Date} createdAt - 创建时间
     */
    
    /**
     * @typedef {Object} CreateUserInput
     * @property {string} name - 用户名
     * @property {string} email - 用户邮箱
     */
    
    // ==================== 常量定义 ====================
    Constants: {
      MIN_NAME_LENGTH: 2,
      MAX_NAME_LENGTH: 100,
      MAX_EMAIL_LENGTH: 255,
      DEFAULT_PAGE_SIZE: 20
    },
    
    // ==================== 验证逻辑 ====================
    Validation: {
      /**
       * 验证用户名
       * @param {string} name - 用户名
       * @returns {Effect.Effect<never, Error, string>} 验证结果
       */
      validateName: (name) => Effect.gen(function* () {
        if (!name) {
          return Effect.fail(new Error('Name is required'))
        }
        
        if (name.length < UserManagement.Constants.MIN_NAME_LENGTH) {
          return Effect.fail(new Error('Name is too short'))
        }
        
        if (name.length > UserManagement.Constants.MAX_NAME_LENGTH) {
          return Effect.fail(new Error('Name is too long'))
        }
        
        return Effect.succeed(name)
      }),
      
      /**
       * 验证邮箱
       * @param {string} email - 邮箱地址
       * @returns {Effect.Effect<never, Error, string>} 验证结果
       */
      validateEmail: (email) => Effect.gen(function* () {
        if (!email) {
          return Effect.fail(new Error('Email is required'))
        }
        
        if (!email.includes('@')) {
          return Effect.fail(new Error('Invalid email format'))
        }
        
        if (email.length > UserManagement.Constants.MAX_EMAIL_LENGTH) {
          return Effect.fail(new Error('Email is too long'))
        }
        
        return Effect.succeed(email)
      })
    },
    
    // ==================== 数据访问 ====================
    Repository: {
      _users: new Map(),
      
      /**
       * 保存用户
       * @param {User} user - 用户数据
       * @returns {Effect.Effect<never, never, User>} 保存的用户
       */
      save: (user) => Effect.gen(function* () {
        console.log(`[Repository] Saving user: ${user.id}`)
        UserManagement.Repository._users.set(user.id, user)
        return user
      }),
      
      /**
       * 根据ID查找用户
       * @param {number} id - 用户ID
       * @returns {Effect.Effect<never, Error, User>} 用户数据
       */
      findById: (id) => Effect.gen(function* () {
        const user = UserManagement.Repository._users.get(id)
        
        if (!user) {
          return Effect.fail(new Error('User not found'))
        }
        
        return user
      })
    },
    
    // ==================== 服务层 ====================
    Services: {
      /**
       * 创建用户
       * @param {CreateUserInput} input - 用户输入
       * @returns {Effect.Effect<never, Error, User>} 创建的用户
       */
      create: (input) => Effect.gen(function* () {
        console.log('[Service] Creating user')
        
        // 验证输入
        const name = yield* UserManagement.Validation.validateName(input.name)
        const email = yield* UserManagement.Validation.validateEmail(input.email)
        
        // 创建用户
        const user = {
          id: Date.now(),
          name,
          email,
          createdAt: new Date()
        }
        
        // 保存用户
        const saved = yield* UserManagement.Repository.save(user)
        
        return saved
      }),
      
      /**
       * 获取用户
       * @param {number} id - 用户ID
       * @returns {Effect.Effect<never, Error, User>} 用户数据
       */
      get: (id) => Effect.gen(function* () {
        console.log(`[Service] Getting user: ${id}`)
        return yield* UserManagement.Repository.findById(id)
      })
    }
  }
  
  // 使用综合模块
  const user = yield* UserManagement.Services.create({
    name: 'John Doe',
    email: 'john@example.com'
  })
  
  console.log('Created user:', user)
  
  const retrieved = yield* UserManagement.Services.get(user.id)
  console.log('Retrieved user:', retrieved)
  
  return { user, retrieved }
})

Effect.runSync(program)
```
