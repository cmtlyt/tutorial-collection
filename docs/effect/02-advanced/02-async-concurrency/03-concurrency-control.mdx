# 并发控制

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了强大的并发控制能力，让你能够精确控制并发执行的策略、限制和模式。

## 🎯 并发限制

### Effect.all 的并发控制

```js
import { Effect } from 'effect'

// 创建多个任务
const tasks = Array.from({ length: 10 }, (_, i) =>
  Effect.gen(function* () {
    console.log(`Task ${i} started`)
    yield* Effect.sleep(1000)
    console.log(`Task ${i} completed`)
    return i
  })
)

// 无限制并发
const unlimited = Effect.all(tasks, {
  concurrency: 'unbounded'
})

// 限制并发数量
const limited = Effect.all(tasks, {
  concurrency: 3  // 最多同时运行 3 个任务
})

Effect.runPromise(limited)
```

### Effect.forEach 的并发控制

```js
import { Effect } from 'effect'

const items = Array.from({ length: 10 }, (_, i) => i)

// 串行执行
const sequential = Effect.forEach(items, item =>
  Effect.gen(function* () {
    console.log(`Processing ${item}`)
    yield* Effect.sleep(500)
    return item * 2
  }),
  { concurrency: 1 }
)

// 并发执行
const parallel = Effect.forEach(items, item =>
  Effect.gen(function* () {
    console.log(`Processing ${item}`)
    yield* Effect.sleep(500)
    return item * 2
  }),
  { concurrency: 5 }
)

Effect.runPromise(parallel)
```

## 💡 并发策略

### Effect.race - 竞争执行

```js
import { Effect } from 'effect'

// 多个 Effect 竞争，第一个完成的胜出
const task1 = Effect.gen(function* () {
  yield* Effect.sleep(2000)
  return 'Task 1'
})

const task2 = Effect.gen(function* () {
  yield* Effect.sleep(1000)
  return 'Task 2'
})

const task3 = Effect.gen(function* () {
  yield* Effect.sleep(3000)
  return 'Task 3'
})

const winner = Effect.race([task1, task2, task3])

Effect.runPromise(winner).then(result => {
  console.log(result)  // 'Task 2' (最快的)
})
```

### Effect.merge - 合并执行

```js
import { Effect } from 'effect'

// 合并多个 Effect 的结果
const effect1 = Effect.succeed([1, 2])
const effect2 = Effect.succeed([3, 4])
const effect3 = Effect.succeed([5, 6])

const merged = Effect.all([effect1, effect2, effect3]).pipe(
  Effect.map(results => results.flat())
)

Effect.runSync(merged)  // [1, 2, 3, 4, 5, 6]
```

### Effect.allWith - 带配置的并行

```js
import { Effect } from 'effect'

const tasks = [
  Effect.succeed(1),
  Effect.succeed(2),
  Effect.succeed(3)
]

// 使用 allWith 进行精细控制
const controlled = Effect.allWith(tasks, {
  concurrency: 2,  // 限制并发
  discard: false,  // 保留所有结果
  mode: 'all'      // 等待所有完成
})

Effect.runSync(controlled)  // [1, 2, 3]
```

## 🚀 背压控制

### 限制生产速度

```js
import { Effect } from 'effect'

// 生产者-消费者模式
const producer = Effect.gen(function* () {
  for (let i = 0; i < 10; i++) {
    console.log(`Producing item ${i}`)
    yield* Effect.sleep(100)
    yield* Effect.sync(() => queue.push(i))
  }
})

const consumer = Effect.gen(function* () {
  while (true) {
    const item = queue.shift()
    if (item !== undefined) {
      console.log(`Consuming item ${item}`)
      yield* Effect.sleep(500)
    } else {
      yield* Effect.sleep(100)
    }
  }
})

// 使用 Fiber 实现背压控制
const queue: number[] = []

const program = Effect.gen(function* () {
  const producerFiber = yield* Effect.fork(producer)
  const consumerFiber = yield* Effect.fork(consumer)
  
  yield* Effect.fromFiber(producerFiber)
  yield* Effect.fromFiber(consumerFiber).pipe(
    Effect.interrupt
  )
})

Effect.runPromise(program)
```

### 信号量模式

```js
import { Effect } from 'effect'

// 创建信号量
const createSemaphore = (max: number) => {
  let available = max
  const waiters: ((value: void) => void)[] = []
  
  return {
    acquire: Effect.sync(() => {
      if (available > 0) {
        available--
        return Effect.succeed(void 0)
      } else {
        return Effect.async<void>((resume) => {
          waiters.push(() => {
            available--
            resume(Effect.succeed(void 0))
          })
        })
      }
    }),
    release: Effect.sync(() => {
      const waiter = waiters.shift()
      if (waiter) {
        waiter()
      } else {
        available++
      }
    })
  }
}

// 使用信号量限制并发
const semaphore = createSemaphore(3)

const task = (id: number) =>
  Effect.gen(function* () {
    yield* semaphore.acquire
    console.log(`Task ${id} started`)
    yield* Effect.sleep(1000)
    console.log(`Task ${id} completed`)
    yield* semaphore.release
    return id
  })

const tasks = Array.from({ length: 10 }, (_, i) => task(i))

Effect.runPromise(Effect.all(tasks, { concurrency: 'unbounded' }))
```

## 📝 实际应用示例

### 批处理 API 请求

```js
import { Effect } from 'effect'

// 批量处理 API 请求
const fetchUser = (id: string) =>
  Effect.tryPromise(() =>
    fetch(`/api/users/${id}`).then(res => res.json())
  )

const userIds = Array.from({ length: 100 }, (_, i) => String(i))

// 分批处理，每批 10 个
const batchProcess = Effect.gen(function* () {
  const batches = []
  for (let i = 0; i < userIds.length; i += 10) {
    batches.push(userIds.slice(i, i + 10))
  }
  
  const results = yield* Effect.forEach(
    batches,
    batch =>
      Effect.forEach(batch, id => fetchUser(id), {
        concurrency: 5
      }).pipe(
        Effect.map(users => users.length)
      ),
    { concurrency: 1 }
  )
  
  return results.reduce((sum, count) => sum + count, 0)
})

Effect.runPromise(batchProcess).then(total => {
  console.log(`Fetched ${total} users`)
})
```

### 速率限制

```js
import { Effect } from 'effect'

// 创建速率限制器
const createRateLimiter = (maxRequests: number, windowMs: number) => {
  let count = 0
  let resetTime = Date.now() + windowMs
  
  return {
    acquire: Effect.sync(() => {
      const now = Date.now()
      if (now >= resetTime) {
        count = 0
        resetTime = now + windowMs
      }
      
      if (count < maxRequests) {
        count++
        return Effect.succeed(void 0)
      } else {
        const waitTime = resetTime - now
        return Effect.sleep(waitTime).pipe(
          Effect.andThen(() => {
            count = 0
            resetTime = Date.now() + windowMs
            count++
          })
        )
      }
    })
  }
}

// 使用速率限制器
const rateLimiter = createRateLimiter(10, 1000)  // 每秒最多 10 个请求

const fetchWithRateLimit = (url: string) =>
  Effect.gen(function* () {
    yield* rateLimiter.acquire
    return yield* Effect.tryPromise(() => fetch(url))
  })

const urls = Array.from({ length: 20 }, (_, i) => `/api/data/${i}`)

Effect.runPromise(
  Effect.forEach(urls, url => fetchWithRateLimit(url), {
    concurrency: 'unbounded'
  })
)
```

### 资源池

```js
import { Effect } from 'effect'

// 创建资源池
const createPool = <T>(factory: () => T, size: number) => {
  const available: T[] = Array.from({ length: size }, factory)
  const waiters: ((resource: T) => void)[] = []
  
  return {
    acquire: Effect.sync(() => {
      const resource = available.shift()
      if (resource) {
        return Effect.succeed(resource)
      } else {
        return Effect.async<T>((resume) => {
          waiters.push((resource) => {
            resume(Effect.succeed(resource))
          })
        })
      }
    }),
    release: (resource: T) => Effect.sync(() => {
      const waiter = waiters.shift()
      if (waiter) {
        waiter(resource)
      } else {
        available.push(resource)
      }
    })
  }
}

// 使用资源池
const pool = createPool(() => ({ id: Math.random() }), 3)

const useResource = () =>
  Effect.gen(function* () {
    const resource = yield* pool.acquire
    console.log(`Using resource ${resource.id}`)
    yield* Effect.sleep(1000)
    pool.release(resource)
    return resource.id
  })

Effect.runPromise(
  Effect.forEach(Array.from({ length: 10 }), () => useResource(), {
    concurrency: 'unbounded'
  })
)
```
