# OpenAI 集成

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了与 OpenAI 的深度集成，通过 `@effect/ai-openai` 包提供 GPT 模型集成、对话管理和流式响应处理等功能。

## 🎯 @effect/ai-openai - OpenAI 适配器

### 安装与设置

```js
// 安装依赖
// npm install @effect/ai @effect/ai-openai

// 基本导入
import { Effect } from 'effect'
import { AI } from '@effect/ai'
import { OpenAI } from '@effect/ai-openai'

// 配置 OpenAI
const program = Effect.gen(function* () {
  const openai = yield* OpenAI.make({
    apiKey: 'your-api-key'
  })
  
  console.log('OpenAI configured')
  return openai
})

Effect.runSync(program)
```

### 基本使用

```js
import { Effect } from 'effect'
import { OpenAI } from '@effect/ai-openai'

// 基本 OpenAI 调用

Effect.runSync(program)
```

### 模型选择

```js
import { Effect } from 'effect'
import { OpenAI } from '@effect/ai-openai'

// 模型选择
const program = Effect.gen(function* () {
  const openai = yield* OpenAI.make({
    apiKey: 'your-api-key'
  })
  
  const models = ['gpt-4', 'gpt-4-turbo', 'gpt-3.5-turbo']
  
  const responses = yield* Effect.all(
    models.map(model => Effect.gen(function* () {
      const response = yield* openai.generate({
        model: model,
        messages: [
          { role: 'user', content: 'Say hello in one word' }
        ]
      })
      
      return { model, response: response.text }
    })),
    { concurrency: 1 }
  )
  
  console.log('Model comparisons:', responses)
  
  return responses
})

Effect.runSync(program)
```

### 对话历史管理

```js
import { Effect } from 'effect'
import { OpenAI } from '@effect/ai-openai'

// 对话历史管理
const program = Effect.gen(function* () {
  const openai = yield* OpenAI.make({
    apiKey: 'your-api-key'
  })
  
  // 对话历史
  const history = []
  
  // 添加消息到历史
  const addMessage = (role, content) => {
    history.push({ role, content })
  }
  
  // 发送消息
  const sendMessage = (content) => Effect.gen(function* () {
    addMessage('user', content)
    
    const response = yield* openai.generate({
      model: 'gpt-4',
      messages: history
    })
    
    addMessage('assistant', response.text)
    
    return response.text
  })
  
  // 对话
  const response1 = yield* sendMessage('What is Effect?')
  console.log('Response 1:', response1)
  
  const response2 = yield* sendMessage('Can you explain more about it?')
  console.log('Response 2:', response2)
  
  const response3 = yield* sendMessage('What did you just say?')
  console.log('Response 3:', response3)
  
  return history
})

Effect.runSync(program)
```

### 上下文管理

```js
import { Effect } from 'effect'
import { OpenAI } from '@effect/ai-openai'

// 上下文管理
const program = Effect.gen(function* () {
  const openai = yield* OpenAI.make({
    apiKey: 'your-api-key'
  })
  
  const context = {
    systemPrompt: 'You are a helpful assistant specializing in Effect programming.',
    maxHistory: 5,
    history: []
  }
  
  const manageContext = (messages) => {
    // 保留最近的对话
    const recent = messages.slice(-context.maxHistory)
    return [
      { role: 'system', content: context.systemPrompt },
      ...recent
    ]
  }
  
  const chat = (message) => Effect.gen(function* () {
    context.history.push({ role: 'user', content: message })
    
    const response = yield* openai.generate({
      model: 'gpt-4',
      messages: manageContext(context.history)
    })
    
    context.history.push({ role: 'assistant', content: response.text })
    
    return response.text
  })
  
  // 对话
  const response1 = yield* chat('What is Effect?')
  console.log('Response 1:', response1)
  
  const response2 = yield* chat('How do I handle errors?')
  console.log('Response 2:', response2)
  
  return context.history
})

Effect.runSync(program)
```

## 🎯 流式响应处理

### 基本流式响应

```js
import { Effect } from 'effect'
import { OpenAI } from '@effect/ai-openai'

// 基本流式响应
const program = Effect.gen(function* () {
  const openai = yield* OpenAI.make({
    apiKey: 'your-api-key'
  })
  
  const stream = yield* openai.stream({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'Tell me a short story' }
    ]
  })
  
  // 处理流式响应
  const chunks = []
  for (const chunk of stream) {
    chunks.push(chunk.content)
    process.stdout.write(chunk.content)
  }
  
  console.log('\n\nStream completed')
  console.log('Total chunks:', chunks.length)
  
  return chunks
})

Effect.runSync(program)
```

### 流式响应收集

```js
import { Effect } from 'effect'
import { OpenAI } from '@effect/ai-openai'

// 流式响应收集
const program = Effect.gen(function* () {
  const openai = yield* OpenAI.make({
    apiKey: 'your-api-key'
  })
  
  const stream = yield* openai.stream({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'Explain functional programming' }
    ]
  })
  
  // 收集流式响应
  let fullText = ''
  let chunkCount = 0
  
  for (const chunk of stream) {
    fullText += chunk.content
    chunkCount++
    
    // 显示进度
    process.stdout.write('.')
    if (chunkCount % 10 === 0) {
      process.stdout.write(` (${chunkCount})\n`)
    }
  }
  
  console.log('\n\nFull text:', fullText)
  console.log('Total chunks:', chunkCount)
  
  return { text: fullText, chunks: chunkCount }
})

Effect.runSync(program)
```

### 流式响应处理

```js
import { Effect } from 'effect'
import { OpenAI } from '@effect/ai-openai'

// 流式响应处理
const program = Effect.gen(function* () {
  const openai = yield* OpenAI.make({
    apiKey: 'your-api-key'
  })
  
  const stream = yield* openai.stream({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'List 5 programming languages' }
    ]
  })
  
  // 处理流式响应
  const lines = []
  let currentLine = ''
  
  for (const chunk of stream) {
    currentLine += chunk.content
    
    // 检测换行
    if (chunk.content.includes('\n')) {
      const parts = currentLine.split('\n')
      lines.push(...parts.filter(line => line.trim()))
      currentLine = parts[parts.length - 1]
    }
  }
  
  if (currentLine.trim()) {
    lines.push(currentLine)
  }
  
  console.log('Extracted lines:', lines)
  
  return lines
})

Effect.runSync(program)
```

## 📝 实际应用示例

### 智能客服

```js
import { Effect } from 'effect'
import { OpenAI } from '@effect/ai-openai'

// 智能客服
const program = Effect.gen(function* () {
  const openai = yield* OpenAI.make({
    apiKey: 'your-api-key'
  })
  
  const customerService = {
    systemPrompt: 'You are a helpful customer service assistant. Be polite and helpful.',
    history: []
  }
  
  const handleQuery = (query) => Effect.gen(function* () {
    customerService.history.push({ role: 'user', content: query })
    
    const response = yield* openai.generate({
      model: 'gpt-4',
      messages: [
        { role: 'system', content: customerService.systemPrompt },
        ...customerService.history.slice(-5)
      ]
    })
    
    customerService.history.push({ role: 'assistant', content: response.text })
    
    return response.text
  })
  
  // 处理客户查询
  const queries = [
    'How do I reset my password?',
    'What are your business hours?',
    'I need help with my order'
  ]
  
  const responses = yield* Effect.all(
    queries.map(query => handleQuery(query)),
    { concurrency: 1 }
  )
  
  console.log('Customer service responses:', responses)
  
  return responses
})

Effect.runSync(program)
```

// 代码助手
const program = Effect.gen(function* () {
