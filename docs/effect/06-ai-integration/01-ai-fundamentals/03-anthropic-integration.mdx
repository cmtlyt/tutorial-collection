# Anthropic 集成

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了与 Anthropic 的深度集成，通过 `@effect/ai-anthropic` 包提供 Claude 模型集成、消息处理和工具调用支持等功能。

## 🎯 @effect/ai-anthropic - Anthropic 适配器

### 安装与设置

```js
// 安装依赖
// npm install @effect/ai @effect/ai-anthropic

// 基本导入
import { Effect } from 'effect'
import { AI } from '@effect/ai'
import { Anthropic } from '@effect/ai-anthropic'

// 配置 Anthropic
const program = Effect.gen(function* () {
  const anthropic = yield* Anthropic.make({
    apiKey: 'your-api-key'
  })
  
  console.log('Anthropic configured')
  return anthropic
})

Effect.runSync(program)
```

### 基本使用

```js
import { Effect } from 'effect'
import { Anthropic } from '@effect/ai-anthropic'

// 基本 Anthropic 调用

    messages: [
      { role: 'user', content: 'Hello!' }
    ]

Effect.runSync(program)
```

### 模型比较

```js
import { Effect } from 'effect'
import { Anthropic } from '@effect/ai-anthropic'

// 模型比较
const program = Effect.gen(function* () {
  const anthropic = yield* Anthropic.make({
    apiKey: 'your-api-key'
  })
  
  const models = [
    'claude-3-opus-20240229',
    'claude-3-sonnet-20240229',
    'claude-3-haiku-20240307'
  ]
  
  const prompt = 'What is the meaning of life? Answer in one sentence.'
  
  const responses = yield* Effect.all(
    models.map(model => Effect.gen(function* () {
      const start = Date.now()
      
      const response = yield* anthropic.generate({
        model: model,
        messages: [
          { role: 'user', content: prompt }
        ]
      })
      
      const end = Date.now()
      const duration = end - start
      
      return {
        model,
        response: response.text,
        duration,
        usage: response.usage
      }
    })),
    { concurrency: 1 }
  )
  
  console.log('Model comparisons:', responses)
  
  return responses
})

Effect.runSync(program)
```

## 🎯 工具调用支持

### 基本工具调用

```js
import { Effect } from 'effect'
import { Anthropic } from '@effect/ai-anthropic'

// 基本工具调用
const program = Effect.gen(function* () {

      name: 'get_weather',
      description: 'Get the current weather for a location',

    model: 'claude-3-opus-20240229',
    messages: [
      { role: 'user', content: 'What is the weather in New York?' }
    ],
    tools: tools
  })
  
  console.log('Response:', response)
  
  return response
})

Effect.runSync(program)
```

### 工具执行

```js
import { Effect } from 'effect'
import { Anthropic } from '@effect/ai-anthropic'

// 工具执行
const program = Effect.gen(function* () {
  const anthropic = yield* Anthropic.make({
    apiKey: 'your-api-key'
  })
  
  // 定义工具
  const tools = [
    {
      name: 'calculate',
      description: 'Perform a calculation',
      input_schema: {
        type: 'object',
        properties: {
          expression: {
            type: 'string',
            description: 'The mathematical expression to evaluate'
          }
        },
        required: ['expression']
      }
    }
  ]
  
  // 工具执行函数
  const executeTool = (toolName, toolInput) => Effect.gen(function* () {
    if (toolName === 'calculate') {
      try {
        const result = eval(toolInput.expression)
        return { result }
      } catch (error) {
        return { error: error.message }
      }
    }
    return { error: 'Unknown tool' }
  })
  
  // 调用工具
  const response = yield* anthropic.generate({
    model: 'claude-3-opus-20240229',
    messages: [
      { role: 'user', content: 'What is 2 + 2 * 3?' }
    ],
    tools: tools
  })
  
  // 执行工具
  if (response.toolCalls) {
    for (const toolCall of response.toolCalls) {
      const result = yield* executeTool(toolCall.name, toolCall.input)
      console.log('Tool result:', result)
    }
  }
  
  return response
})

Effect.runSync(program)
```

### 多工具调用

```js
import { Effect } from 'effect'
import { Anthropic } from '@effect/ai-anthropic'

// 多工具调用
const program = Effect.gen(function* () {
  const anthropic = yield* Anthropic.make({
    apiKey: 'your-api-key'
  })
  
  // 定义多个工具
  const tools = [
    {
      name: 'get_time',
      description: 'Get the current time',
      input_schema: {
        type: 'object',
        properties: {},
        required: []
      }
    },
    {
      name: 'get_date',
      description: 'Get the current date',
      input_schema: {
        type: 'object',
        properties: {},
        required: []
      }
    }
  ]
  
  // 工具执行函数
  const executeTool = (toolName) => Effect.gen(function* () {
    if (toolName === 'get_time') {
      return { time: new Date().toLocaleTimeString() }
    } else if (toolName === 'get_date') {
      return { date: new Date().toLocaleDateString() }
    }
    return { error: 'Unknown tool' }
  })
  
  // 调用工具
  const response = yield* anthropic.generate({
    model: 'claude-3-opus-20240229',
    messages: [
      { role: 'user', content: 'What is the current time and date?' }
    ],
    tools: tools
  })
  
  // 执行工具
  if (response.toolCalls) {
    const results = yield* Effect.all(
      response.toolCalls.map(toolCall => executeTool(toolCall.name)),
      { concurrency: 'unbounded' }
    )
    console.log('Tool results:', results)
  }
  
  return response
})

Effect.runSync(program)
```

## 📝 实际应用示例

### 智能分析

```js
import { Effect } from 'effect'
import { Anthropic } from '@effect/ai-anthropic'

// 智能分析
const program = Effect.gen(function* () {
  const anthropic = yield* Anthropic.make({
    apiKey: 'your-api-key'
  })
  
  const analyzeText = (text, analysisType) => Effect.gen(function* () {
    const prompts = {
      sentiment: 'Analyze the sentiment of this text. Is it positive, negative, or neutral?',
      summary: 'Provide a brief summary of this text.',
      keywords: 'Extract the main keywords from this text.'
    }
    
    const response = yield* anthropic.generate({
      model: 'claude-3-opus-20240229',
      messages: [
        {
          role: 'user',
          content: `${prompts[analysisType]}\n\nText: ${text}`
        }
      ]
    })
    
    return response.text
  })
  
  const text = 'I absolutely love this product! It has exceeded all my expectations.'
  
  const sentiment = yield* analyzeText(text, 'sentiment')
  console.log('Sentiment:', sentiment)
  
  const summary = yield* analyzeText(text, 'summary')
  console.log('Summary:', summary)
  
  const keywords = yield* analyzeText(text, 'keywords')
  console.log('Keywords:', keywords)
  
  return { sentiment, summary, keywords }
})

Effect.runSync(program)
```

import { Effect } from 'effect'

// 内容生成
const program = Effect.gen(function* () {

    apiKey: 'your-api-key'

      blog: 'Write a blog post about this topic. Include an introduction, main points, and conclusion.',
      email: 'Write a professional email about this topic.',

        }
      ],
      maxTokens: 500
    })
    
    return response.text
  })
  
  const topic = 'The benefits of meditation'
  
  const blogPost = yield* generateContent(topic, 'blog')
  console.log('Blog Post:\n', blogPost)
  
  const email = yield* generateContent(topic, 'email')
  console.log('\nEmail:\n', email)
  
  const social = yield* generateContent(topic, 'social')

Effect.runSync(program)
