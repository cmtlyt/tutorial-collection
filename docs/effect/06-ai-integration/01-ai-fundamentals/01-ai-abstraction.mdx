# AI 抽象

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了强大的 AI 抽象层，通过 `@effect/ai` 包提供统一的 AI 模型接口定义、提示工程模式和响应处理机制。

## 🎯 @effect/ai - AI 基础包

### 安装与设置

```js
// 安装依赖
// npm install @effect/ai

// 基本导入
import { Effect } from 'effect'
import { AI } from '@effect/ai'
```

### 基本使用

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 基本 AI 调用
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 生成文本
  const response = yield* ai.generate({
    model: 'gpt-4',
    prompt: 'Hello, how are you?'
  })
  
  console.log('AI Response:', response.text)
  
  return response
})

Effect.runSync(program)
```

## 💡 AI 模型接口定义

### 模型接口

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 定义模型接口
interface AIModel {
  generate: (input: AI.GenerateInput) => Effect.Effect<never, AI.AIError, AI.GenerateOutput>
  stream: (input: AI.GenerateInput) => Effect.Effect<never, AI.AIError, AI.StreamOutput>
}

// 使用模型接口
const program = Effect.gen(function* () {
  const model = yield* AI.AI
  
  const input = {
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'What is Effect?' }
    ]
  }
  
  const output = yield* model.generate(input)
  console.log('Model output:', output)
  
  return output
})

Effect.runSync(program)
```

### 模型配置

```js

    ]

  console.log('Response:', output.text)
  
  return output
})

Effect.runSync(program)
```

### 链式提示

```js

const program = Effect.gen(function* () {

})

Effect.runSync(program)
```

## 🎯 响应处理机制

### 基本响应处理

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 基本响应处理
const program = Effect.gen(function* () {
  const model = yield* AI.AI
  
  const response = yield* model.generate({
    model: 'gpt-4',
    prompt: 'What is 2 + 2?'
  })
  
  // 处理响应
  const text = response.text
  const usage = response.usage
  const finishReason = response.finishReason
  
  console.log('Response:', text)
  console.log('Usage:', usage)
  console.log('Finish reason:', finishReason)
  
  return { text, usage, finishReason }
})

Effect.runSync(program)
```

### 结构化响应

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 结构化响应
const program = Effect.gen(function* () {
  const model = yield* AI.AI
  
  const response = yield* model.generate({
    model: 'gpt-4',
    prompt: 'List 3 programming languages with their year of creation in JSON format',
    responseFormat: { type: 'json_object' }
  })
  
  // 解析 JSON 响应
  const data = JSON.parse(response.text)
  console.log('Languages:', data)
  
  return data
})

Effect.runSync(program)
```

### 错误处理

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 错误处理
const program = Effect.gen(function* () {
  const model = yield* AI.AI
  
  const response = yield* model.generate({
    model: 'gpt-4',
    prompt: 'What is the meaning of life?'
  }).pipe(
    Effect.catchAll((error) => {
      console.error('AI Error:', error.message)
      return Effect.succeed({
        text: 'Sorry, I encountered an error.',
        usage: { promptTokens: 0, completionTokens: 0, totalTokens: 0 }
      })
    })
  )
  
  console.log('Response:', response.text)
  
  return response
})

Effect.runSync(program)
```

## 📝 实际应用示例

### 智能助手

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 智能助手
const program = Effect.gen(function* () {
  const model = yield* AI.AI
  
  // 定义助手角色
  const assistant = {
    name: 'Effect Helper',
    role: 'system',
    content: 'You are an expert in Effect programming. Help users understand and use Effect effectively.'
  }
  
  // 用户问题
  const questions = [
    'What is Effect?',
    'How do I handle errors in Effect?',
    'What is a Fiber?'
  ]
  
  // 回答问题
  const answers = yield* Effect.all(
    questions.map(question => Effect.gen(function* () {
      const response = yield* model.generate({
        model: 'gpt-4',
        messages: [
          assistant,
          { role: 'user', content: question }
        ]
      })
      
      console.log(`\nQ: ${question}`)
      console.log(`A: ${response.text}`)
      
      return { question, answer: response.text }
    })),
    { concurrency: 1 }
  )
  
  return answers
})

Effect.runSync(program)
```

### 代码生成

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 代码生成
const program = Effect.gen(function* () {
  const model = yield* AI.AI
  
  const generateCode = (description, language) => Effect.gen(function* () {
    const prompt = `Write ${language} code to: ${description}\n\nProvide only the code, no explanation.`
    
    const response = yield* model.generate({
      model: 'gpt-4',
      prompt: prompt,
      temperature: 0.3
    })
    
    return response.text
  })
  
  // 生成代码示例
  const code1 = yield* generateCode('create a function that adds two numbers', 'JavaScript')
  console.log('JavaScript code:\n', code1)
  
  const code2 = yield* generateCode('create a function that reverses a string', 'Python')
  console.log('\nPython code:\n', code2)
  
  return { code1, code2 }
})

Effect.runSync(program)
```

### 文本处理

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 文本处理
const program = Effect.gen(function* () {
  const model = yield* AI.AI
  
  const text = 'Effect is a powerful TypeScript library for functional programming.'
  
  // 摘要
  const summary = yield* model.generate({
    model: 'gpt-4',
    prompt: `Summarize this text in one sentence: ${text}`
  })
  console.log('Summary:', summary.text)
  
  // 翻译
  const translation = yield* model.generate({
    model: 'gpt-4',
    prompt: `Translate this to Chinese: ${text}`
  })
  console.log('Translation:', translation.text)
  
  // 提取关键词
  const keywords = yield* model.generate({
    model: 'gpt-4',
    prompt: `Extract 3 keywords from this text: ${text}`
  })
  console.log('Keywords:', keywords.text)
  
  return { summary, translation, keywords }
})

Effect.runSync(program)
```
