> **版本**：v1.0

## 📚 概念介绍

Effect 提供了强大的 AI 工具系统，支持工具定义、工具执行、工具链和错误处理，使 AI 能够调用外部服务和执行复杂任务。

## 🎯 工具定义

### 基本工具定义

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 基本工具定义
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 定义工具
  const tool = {
    name: 'get_current_time',
    description: 'Get the current time',
    parameters: {
      type: 'object',
      properties: {},
      required: []
    }
  }
  
  // 使用工具
  const response = yield* ai.generate({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'What time is it?' }
    ],
    tools: [tool]
  })
  
  console.log('Response:', response)
  
  return response
})

Effect.runSync(program)
```

## 💡 工具执行

### 基本工具执行

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 基本工具执行
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 定义工具
  const tool = {
    name: 'get_current_time',
    description: 'Get the current time',
    parameters: {
      type: 'object',
      properties: {},
      required: []
    }
  }
  
  // 工具执行函数
  const executeTool = (toolName, toolInput) => Effect.gen(function* () {
    if (toolName === 'get_current_time') {
      const time = new Date().toLocaleTimeString()
      return { time }
    }
    return { error: 'Unknown tool' }
  })
  
  // 生成响应
  const response = yield* ai.generate({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'What time is it?' }
    ],
    tools: [tool]
  })
  
  // 执行工具
  if (response.toolCalls) {
    for (const toolCall of response.toolCalls) {
      const result = yield* executeTool(toolCall.name, toolCall.input)
      console.log('Tool result:', result)
    }
  }
  
  return response
})

Effect.runSync(program)
```

### 参数化工具执行

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 参数化工具执行
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 定义工具
  const tool = {
    name: 'calculate',
    description: 'Perform a mathematical calculation',
    parameters: {
      type: 'object',
      properties: {
        expression: {
          type: 'string',
          description: 'The mathematical expression to evaluate'
        }
      },
      required: ['expression']
    }
  }
  
  // 工具执行函数
  const executeTool = (toolName, toolInput) => Effect.gen(function* () {
    if (toolName === 'calculate') {
      try {
        const result = eval(toolInput.expression)
        return { result }
      } catch (error) {
        return { error: error.message }
      }
    }
    return { error: 'Unknown tool' }
  })
  
  // 生成响应
  const response = yield* ai.generate({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'What is 2 + 2 * 3?' }
    ],
    tools: [tool]
  })
  
  // 执行工具
  if (response.toolCalls) {
    for (const toolCall of response.toolCalls) {
      const result = yield* executeTool(toolCall.name, toolCall.input)
      console.log('Tool result:', result)
      
      // 将工具结果返回给 AI
      const followUp = yield* ai.generate({
        model: 'gpt-4',
        messages: [
          { role: 'user', content: 'What is 2 + 2 * 3?' },
          response,
          {
            role: 'tool',
            toolCallId: toolCall.id,
            content: JSON.stringify(result)
          }
        ]
      })
      
      console.log('Follow-up response:', followUp.text)
    }
  }
  
  return response
})

Effect.runSync(program)
```

### 并行工具执行

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 并行工具执行
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 定义多个工具
  const tools = [
    {
      name: 'get_time',
      description: 'Get the current time',
      parameters: {
        type: 'object',
        properties: {},
        required: []
      }
    },
    {
      name: 'get_date',
      description: 'Get the current date',
      parameters: {
        type: 'object',
        properties: {},
        required: []
      }
    }
  ]
  
  // 工具执行函数
  const executeTool = (toolName) => Effect.gen(function* () {
    if (toolName === 'get_time') {
      return { time: new Date().toLocaleTimeString() }
    } else if (toolName === 'get_date') {
      return { date: new Date().toLocaleDateString() }
    }
    return { error: 'Unknown tool' }
  })
  
  // 生成响应
  const response = yield* ai.generate({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'What is the current time and date?' }
    ],
    tools: tools
  })
  
  // 并行执行工具
  if (response.toolCalls) {
    const results = yield* Effect.all(
      response.toolCalls.map(toolCall => Effect.gen(function* () {
        const result = yield* executeTool(toolCall.name)
        return {
          toolCallId: toolCall.id,
          result
        }
      })),
      { concurrency: 'unbounded' }
    )
    
    console.log('Tool results:', results)
  }
  
  return response
})

Effect.runSync(program)
```

## 🚀 工具链

### 顺序工具链

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 顺序工具链
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 定义工具
  const tools = [
    {
      name: 'get_location',
      description: 'Get the location of a user',
      parameters: {
        type: 'object',
        properties: {
          userId: {
            type: 'string',
            description: 'The user ID'
          }
        },
        required: ['userId']
      }
    },
    {
      name: 'get_weather',
      description: 'Get the weather for a location',
      parameters: {
        type: 'object',
        properties: {
          location: {
            type: 'string',
            description: 'The location'
          }
        },
        required: ['location']
      }
    }
  ]
  
  // 工具执行函数
  const executeTool = (toolName, toolInput) => Effect.gen(function* () {
    if (toolName === 'get_location') {
      return { location: 'New York, NY' }
    } else if (toolName === 'get_weather') {
      return { weather: 'Sunny, 72°F' }
    }
    return { error: 'Unknown tool' }
  })
  
  // 生成响应
  const response = yield* ai.generate({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'What is the weather for user 123?' }
    ],
    tools: tools
  })
  
  // 执行工具链
  let currentResponse = response
  let messages = [
    { role: 'user', content: 'What is the weather for user 123?' },
    response
  ]
  
  while (currentResponse.toolCalls) {
    const toolResults = yield* Effect.all(
      currentResponse.toolCalls.map(toolCall => Effect.gen(function* () {
        const result = yield* executeTool(toolCall.name, toolCall.input)
        return {
          role: 'tool',
          toolCallId: toolCall.id,
          content: JSON.stringify(result)
        }
      })),
      { concurrency: 'unbounded' }
    )
    
    messages.push(...toolResults)
    
    currentResponse = yield* ai.generate({
      model: 'gpt-4',
      messages: messages
    })
    
    messages.push(currentResponse)
  }
  
  console.log('Final response:', currentResponse.text)
  
  return currentResponse
})

Effect.runSync(program)
```

## 🎯 错误处理

### 工具执行错误

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 工具执行错误
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 定义工具
  const tool = {
    name: 'divide',
    description: 'Divide two numbers',
    parameters: {
      type: 'object',
      properties: {
        a: { type: 'number', description: 'The numerator' },
        b: { type: 'number', description: 'The denominator' }
      },
      required: ['a', 'b']
    }
  }
  
  // 工具执行函数
  const executeTool = (toolName, toolInput) => Effect.gen(function* () {
    if (toolName === 'divide') {
      if (toolInput.b === 0) {
        return { error: 'Division by zero' }
      }
      const result = toolInput.a / toolInput.b
      return { result }
    }
    return { error: 'Unknown tool' }
  })
  
  // 生成响应
  const response = yield* ai.generate({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'What is 10 divided by 0?' }
    ],
    tools: [tool]
  })
  
  // 执行工具
  if (response.toolCalls) {
    for (const toolCall of response.toolCalls) {
      const result = yield* executeTool(toolCall.name, toolCall.input)
      
      if (result.error) {
        console.log('Tool error:', result.error)
        
        // 将错误返回给 AI
        const followUp = yield* ai.generate({
          model: 'gpt-4',
          messages: [
            { role: 'user', content: 'What is 10 divided by 0?' },
            response,
            {
              role: 'tool',
              toolCallId: toolCall.id,
              content: JSON.stringify(result)
            }
          ]
        })
        
        console.log('Error response:', followUp.text)
      }
    }
  }
  
  return response
})

Effect.runSync(program)
```

### 重试机制

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 重试机制
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 定义工具
  const tool = {
    name: 'unstable_operation',
    description: 'An operation that may fail',
    parameters: {
      type: 'object',
      properties: {},
      required: []
    }
  }
  
  // 工具执行函数（带重试）
  const executeToolWithRetry = (toolName, maxRetries = 3) => Effect.gen(function* () {
    let attempts = 0
    
    while (attempts < maxRetries) {
      attempts++
      
      try {
        // 模拟不稳定的操作
        if (Math.random() > 0.5) {
          return { success: true, data: 'Operation completed' }
        } else {
          throw new Error('Operation failed')
        }
      } catch (error) {
        console.log(`Attempt ${attempts} failed:`, error.message)
        
        if (attempts >= maxRetries) {
          return { error: 'Max retries exceeded' }
        }
        
        yield* Effect.sleep(1000)
      }
    }
  })
  
  // 生成响应
  const response = yield* ai.generate({
    model: 'gpt-4',
    messages: [
      { role: 'user', content: 'Perform the unstable operation' }
    ],
    tools: [tool]
  })
  
  // 执行工具（带重试）
  if (response.toolCalls) {
    for (const toolCall of response.toolCalls) {
      const result = yield* executeToolWithRetry(toolCall.name)
      console.log('Tool result:', result)
    }
  }
  
  return response
})

Effect.runSync(program)
```

```js
import { Effect } from 'effect'
import { AI } from '@effect/ai'

// 智能助手工具链
const program = Effect.gen(function* () {
  const ai = yield* AI.AI
  
  // 定义工具集
  const tools = [
    {
      name: 'search_database',
      description: 'Search the database for information',
      parameters: {
        type: 'object',
        properties: {
          query: {
            type: 'string',
            description: 'The search query'
          }
        },
        required: ['query']
      }
    },
    {
      name: 'calculate',
      description: 'Perform calculations',
      parameters: {
        type: 'object',
        properties: {
          expression: {
            type: 'string',
            description: 'The mathematical expression'
          }
        },
        required: ['expression']
      }
    },
    {
      name: 'send_notification',
      description: 'Send a notification to a user',
      parameters: {
        type: 'object',
        properties: {
          userId: {
            type: 'string',
            description: 'The user ID'
          },
          message: {
            type: 'string',
            description: 'The notification message'
          }
        },
        required: ['userId', 'message']
      }
    }
  ]
  
  // 工具执行函数
  const executeTool = (toolName, toolInput) => Effect.gen(function* () {
    if (toolName === 'search_database') {
      return { results: [`Result for: ${toolInput.query}`] }
    } else if (toolName === 'calculate') {
      const result = eval(toolInput.expression)
      return { result }
    } else if (toolName === 'send_notification') {
      return { success: true, notificationId: '123' }
    }
    return { error: 'Unknown tool' }
  })
  
  // 处理用户请求
  const processRequest = (userMessage) => Effect.gen(function* () {
    let messages = [
      { role: 'user', content: userMessage }
    ]
    
    let currentResponse = yield* ai.generate({
      model: 'gpt-4',
      messages: messages,
      tools: tools
    })
    
    messages.push(currentResponse)
    
    // 执行工具链
    while (currentResponse.toolCalls) {
      const toolResults = yield* Effect.all(
        currentResponse.toolCalls.map(toolCall => Effect.gen(function* () {
          const result = yield* executeTool(toolCall.name, toolCall.input)
          return {
            role: 'tool',
            toolCallId: toolCall.id,
            content: JSON.stringify(result)
          }
        })),
        { concurrency: 'unbounded' }
      )
      
      messages.push(...toolResults)
      
      currentResponse = yield* ai.generate({
        model: 'gpt-4',
        messages: messages
      })
      
      messages.push(currentResponse)
    }
    
    return currentResponse.text
  })
  
  // 处理多个请求
  const requests = [
    'Search for information about Effect',
    'Calculate 2 + 2 * 3',
    'Send a notification to user 123'
  ]
  
  const responses = yield* Effect.all(
    requests.map(request => processRequest(request)),
    { concurrency: 1 }
  )
  
  console.log('Responses:', responses)
  
  return responses
})

Effect.runSync(program)
```
