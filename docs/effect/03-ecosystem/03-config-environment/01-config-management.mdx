# 配置管理

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了强大的配置管理功能，支持类型安全的配置读取、环境变量处理、配置验证和默认值设置。

## 🎯 Effect.Config - 配置读取

### 基本配置读取

```js
import { Effect, Config } from 'effect'

// 基本配置读取
const program = Effect.gen(function* () {
  // 读取字符串配置
  const apiUrl = yield* Config.string('API_URL')
  console.log('API URL:', apiUrl)
  
  // 读取数字配置
  const port = yield* Config.number('PORT')
  console.log('Port:', port)
  
  // 读取布尔配置
  const debug = yield* Config.boolean('DEBUG')
  console.log('Debug:', debug)
  
  return { apiUrl, port, debug }
})

// 运行程序并提供环境变量
Effect.runSync(
  program.pipe(
    Effect.provideService(Config.Config, {
      string: (key) => Effect.sync(() => {
        const env = {
          API_URL: 'https://api.example.com',
          PORT: '3000',
          DEBUG: 'true'
        }
        return env[key] || ''
      }),
      number: (key) => Effect.sync(() => {
        const env = { PORT: '3000' }
        return parseInt(env[key] || '0')
      }),
      boolean: (key) => Effect.sync(() => {
        const env = { DEBUG: 'true' }
        return env[key] === 'true'
      })
    })
  )
)
```

## 🎯 配置验证与默认值

### 配置验证

```js
import { Effect, Config } from 'effect'

// 配置验证
const validatePort = (port: number) => {
  if (port < 1 || port > 65535) {
    throw new Error(`Invalid port: ${port}. Port must be between 1 and 65535`)
  }
  return port
}

const validateEmail = (email: string) => {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
  if (!emailRegex.test(email)) {
    throw new Error(`Invalid email: ${email}`)
  }
  return email
}

const program = Effect.gen(function* () {
  const port = yield* Config.number('PORT')
  const validatedPort = yield* Effect.sync(() => validatePort(port))
  
  const adminEmail = yield* Config.string('ADMIN_EMAIL')
  const validatedEmail = yield* Effect.sync(() => validateEmail(adminEmail))
  
  console.log('Validated Port:', validatedPort)
  console.log('Validated Email:', validatedEmail)
  
  return { port: validatedPort, email: validatedEmail }
})

Effect.runSync(
  program.pipe(
    Effect.provideService(Config.Config, {
      string: (key) => Effect.sync(() => {
        const env = { ADMIN_EMAIL: 'admin@example.com' }
        return env[key] || ''
      }),
      number: (key) => Effect.sync(() => {
        const env = { PORT: '3000' }
        return parseInt(env[key] || '0')
      }),
      boolean: (key) => Effect.sync(() => false)
    })
  )
)
```

### 配置默认值

```js
import { Effect, Config } from 'effect'

// 配置默认值
const program = Effect.gen(function* () {
  // 带默认值的配置读取
  const port = yield* Effect.all([
    Config.number('PORT'),
    Effect.succeed(3000)
  ]).pipe(
    Effect.map(([port, defaultPort]) => port || defaultPort)
  )
  
  const debug = yield* Effect.all([
    Config.boolean('DEBUG'),
    Effect.succeed(false)
  ]).pipe(
    Effect.map(([debug, defaultDebug]) => debug ?? defaultDebug)
  )
  
  const timeout = yield* Effect.all([
    Config.number('TIMEOUT'),
    Effect.succeed(5000)
  ]).pipe(
    Effect.map(([timeout, defaultTimeout]) => timeout || defaultTimeout)
  )
  
  console.log('Port:', port)
  console.log('Debug:', debug)
  console.log('Timeout:', timeout)
  
  return { port, debug, timeout }
})

Effect.runSync(
  program.pipe(
    Effect.provideService(Config.Config, {
      string: (key) => Effect.sync(() => ''),
      number: (key) => Effect.sync(() => 0),
      boolean: (key) => Effect.sync(() => false)
    })
  )
)
```

```js
const createAppConfig = () => Effect.gen(function* () {
```

### 多环境配置

```js
import { Effect, Config } from 'effect'

// 多环境配置
const getEnvironmentConfig = (environment: string) => {
  const defaults = {
    port: 3000,
    debug: false,
    timeout: 5000
  }
  
  const envConfigs = {
    development: {
      port: 3000,
      debug: true,
      timeout: 10000
    },
    production: {
      port: 8080,
      debug: false,
      timeout: 5000
    },
    test: {
      port: 3001,
      debug: true,
      timeout: 3000
    }
  }
  
  return envConfigs[environment] || defaults
}

const program = Effect.gen(function* () {
  const environment = yield* Config.string('ENVIRONMENT')
  const envConfig = getEnvironmentConfig(environment)
  
  const port = yield* Config.number('PORT').pipe(
    Effect.catchAll(() => Effect.succeed(envConfig.port))
  )
  
  const debug = yield* Config.boolean('DEBUG').pipe(
    Effect.catchAll(() => Effect.succeed(envConfig.debug))
  )
  
  const timeout = yield* Config.number('TIMEOUT').pipe(
    Effect.catchAll(() => Effect.succeed(envConfig.timeout))
  )
  
  const config = {
    environment,
    port,
    debug,
    timeout
  }
  
  console.log('Environment Config:', config)
  return config
})

Effect.runSync(
  program.pipe(
    Effect.provideService(Config.Config, {
      string: (key) => Effect.sync(() => {
        const env = { ENVIRONMENT: 'production' }
        return env[key] || ''
      }),
      number: (key) => Effect.sync(() => 0),
      boolean: (key) => Effect.sync(() => false)
    })
  )
)
```

### 配置热重载

```js
import { Effect, Config } from 'effect'

// 配置热重载
const createConfigWatcher = () => Effect.gen(function* () {
  const config: Record<string, any> = {}
  const listeners: Array<(config: any) => void> = []
  
  const getConfig = (key: string) => Effect.sync(() => config[key])
  
  const setConfig = (key: string, value: any) => Effect.sync(() => {
    config[key] = value
    console.log(`Config updated: ${key} = ${value}`)
    
    // 通知所有监听器
    listeners.forEach(listener => listener({ ...config }))
  })
  
  const onChange = (callback: (config: any) => void) => Effect.sync(() => {
    listeners.push(callback)
  })
  
  return {
    getConfig,
    setConfig,
    onChange
  }
})

const program = Effect.gen(function* () {
  const watcher = yield* createConfigWatcher()
  
  // 初始化配置
  yield* watcher.setConfig('PORT', 3000)
  yield* watcher.setConfig('DEBUG', true)
  
  // 监听配置变化
  yield* watcher.onChange((newConfig) => {
    console.log('Config changed:', newConfig)
  })
  
  // 更新配置
  yield* watcher.setConfig('PORT', 8080)
  yield* watcher.setConfig('DEBUG', false)
  
  const port = yield* watcher.getConfig('PORT')
  const debug = yield* watcher.getConfig('DEBUG')
  
  console.log('Current config:', { port, debug })
})

Effect.runSync(program)
```
