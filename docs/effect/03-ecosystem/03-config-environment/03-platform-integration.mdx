# 平台集成

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了跨平台的能力，支持 Node.js、浏览器、Bun 等不同平台，并提供统一的抽象接口。

## 🎯 Node.js 平台

### 文件系统操作

```js
import { Effect } from 'effect'

// 文件系统操作
const program = Effect.gen(function* () {
  // 读取文件
  const readFile = Effect.tryPromise(() =>
    import('fs').then(fs => fs.promises.readFile('example.txt', 'utf-8'))
  )
  
  const content = yield* readFile
  console.log('File content:', content)
  
  // 写入文件
  const writeFile = Effect.tryPromise(() =>
    import('fs').then(fs => fs.promises.writeFile('output.txt', 'Hello, Effect!'))
  )
  
  yield* writeFile
  console.log('File written successfully')
  
  return content
})

Effect.runPromise(program)
```

### HTTP 服务器

```js
import { Effect } from 'effect'

// HTTP 服务器
const createServer = Effect.gen(function* () {
  const http = yield* Effect.tryPromise(() => import('http'))
  
  const server = http.createServer((req, res) => {
    res.writeHead(200, { 'Content-Type': 'text/plain' })
    res.end('Hello from Effect!')
  })
  
  const listen = Effect.tryPromise(() =>
    new Promise((resolve) => {
      server.listen(3000, () => resolve('Server running on port 3000'))
    })
  )
  
  yield* listen
  console.log('Server started')
  
  return server
})

Effect.runPromise(createServer)
```

### 环境变量

```js
import { Effect } from 'effect'
```

```js

// DOM 操作
const program = Effect.gen(function* () {
  // 获取元素

    const element = document.getElementById('app')
    if (!element) {
      throw new Error('Element not found')
    }
    return element
  })
  
  const appElement = yield* getElement
  
  // 设置内容
  const setContent = Effect.sync(() => {
    appElement.innerHTML = '<h1>Hello from Effect!</h1>'
  })
  
  yield* setContent
  console.log('DOM updated')
  
  return appElement
})

// 在浏览器中运行
if (typeof window !== 'undefined') {
  Effect.runPromise(program)
}
```

### 统一文件操作

```js
import { Effect } from 'effect'

// 统一文件操作接口
interface FileSystem {
  readFile: (path: string) => Effect.Effect<never, never, string>
  writeFile: (path: string, content: string) => Effect.Effect<never, never, void>
}

// Node.js 文件系统
const nodeFileSystem: FileSystem = {
  readFile: (path: string) => Effect.tryPromise(() =>
    import('fs').then(fs => fs.promises.readFile(path, 'utf-8'))
  ),
  writeFile: (path: string, content: string) => Effect.tryPromise(() =>
    import('fs').then(fs => fs.promises.writeFile(path, content))
  )
}

// 浏览器文件系统（使用 IndexedDB）
const browserFileSystem: FileSystem = {
  readFile: (path: string) => Effect.tryPromise(async () => {
    const db = await indexedDB.open('FileSystem', 1)
    const tx = db.transaction('files', 'readonly')
    const store = tx.objectStore('files')
    const result = await store.get(path)
    return result?.content || ''
  }),
  writeFile: (path: string, content: string) => Effect.tryPromise(async () => {
    const db = await indexedDB.open('FileSystem', 1)
    const tx = db.transaction('files', 'readwrite')
    const store = tx.objectStore('files')
    await store.put({ path, content })
  })
}

// Bun 文件系统
const bunFileSystem: FileSystem = {
  readFile: (path: string) => Effect.tryPromise(() =>
    Bun.file(path).text()
  ),
  writeFile: (path: string, content: string) => Effect.sync(() => {
    Bun.write(path, content)
  })
}

// 获取当前平台的文件系统
const getFileSystem = Effect.sync((): FileSystem => {
  if (typeof Bun !== 'undefined') {
    return bunFileSystem
  } else if (typeof window !== 'undefined') {
    return browserFileSystem
  } else {
    return nodeFileSystem
  }
})

// 使用统一文件系统
const program = Effect.gen(function* () {
  const fs = yield* getFileSystem
  
  yield* fs.writeFile('test.txt', 'Hello from Effect!')
  const content = yield* fs.readFile('test.txt')
  
  console.log('File content:', content)
  return content
})

Effect.runPromise(program)
```

### 统一网络请求

```js
import { Effect } from 'effect'

// 统一网络请求接口
interface HttpClient {
  get: (url: string) => Effect.Effect<never, never, any>
  post: (url: string, data: any) => Effect.Effect<never, never, any>
}

// Node.js HTTP 客户端
const nodeHttpClient: HttpClient = {
  get: (url: string) => Effect.tryPromise(async () => {
    const response = await fetch(url)
    return response.json()
  }),
  post: (url: string, data: any) => Effect.tryPromise(async () => {
    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    })
    return response.json()
  })
}

// 浏览器 HTTP 客户端
const browserHttpClient: HttpClient = {
  get: (url: string) => Effect.tryPromise(async () => {
    const response = await fetch(url)
    return response.json()
  }),
  post: (url: string, data: any) => Effect.tryPromise(async () => {
    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    })
    return response.json()
  })
}

// Bun HTTP 客户端
const bunHttpClient: HttpClient = {
  get: (url: string) => Effect.tryPromise(async () => {
    const response = await fetch(url)
    return response.json()
  }),
  post: (url: string, data: any) => Effect.tryPromise(async () => {
    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    })
    return response.json()
  })
}

// 获取当前平台的 HTTP 客户端
const getHttpClient = Effect.sync((): HttpClient => {
  if (typeof Bun !== 'undefined') {
    return bunHttpClient
  } else if (typeof window !== 'undefined') {
    return browserHttpClient
  } else {
    return nodeHttpClient
  }
})

// 使用统一 HTTP 客户端
const program = Effect.gen(function* () {
  const http = yield* getHttpClient
  
  const data = yield* http.get('https://api.example.com/data')
  console.log('GET response:', data)
  
  const result = yield* http.post('https://api.example.com/create', { name: 'Alice' })
  console.log('POST response:', result)
  
  return { data, result }
})

Effect.runPromise(program)
```

### 适配器模式

```js
import { Effect } from 'effect'

// 适配器模式
interface StorageAdapter {
  get: (key: string) => Effect.Effect<never, never, string | null>
  set: (key: string, value: string) => Effect.Effect<never, never, void>
  remove: (key: string) => Effect.Effect<never, never, void>
}

// LocalStorage 适配器
const localStorageAdapter: StorageAdapter = {
  get: (key: string) => Effect.sync(() => {
    if (typeof window === 'undefined') return null
    return localStorage.getItem(key)
  }),
  set: (key: string, value: string) => Effect.sync(() => {
    if (typeof window === 'undefined') return
    localStorage.setItem(key, value)
  }),
  remove: (key: string) => Effect.sync(() => {
    if (typeof window === 'undefined') return
    localStorage.removeItem(key)
  })
}

// Memory 适配器
const memoryAdapter: StorageAdapter = {
  get: (key: string) => Effect.sync(() => {
    const storage = memoryAdapter.storage
    return storage[key] || null
  }),
  set: (key: string, value: string) => Effect.sync(() => {
    memoryAdapter.storage[key] = value
  }),
  remove: (key: string) => Effect.sync(() => {
    delete memoryAdapter.storage[key]
  })
}
memoryAdapter.storage = {} as Record<string, string>

// 选择适配器
const getStorageAdapter = Effect.sync((): StorageAdapter => {
  if (typeof window !== 'undefined' && typeof localStorage !== 'undefined') {
    return localStorageAdapter
  }
  return memoryAdapter
})

// 使用适配器
const program = Effect.gen(function* () {
  const storage = yield* getStorageAdapter
  
  yield* storage.set('user', 'Alice')
  const user = yield* storage.get('user')
  console.log('User:', user)
  
  yield* storage.remove('user')
  const removedUser = yield* storage.get('user')
  console.log('Removed user:', removedUser)
})

Effect.runPromise(program)
```

```js
// 环境特定配置
const getEnvironmentConfig = Effect.sync(() => {
  const platform = typeof Bun !== 'undefined' ? 'bun' :
                  typeof window !== 'undefined' ? 'browser' : 'node'
  
  const configs = {
    node: {
      database: 'postgresql://localhost:5432/mydb',
      cache: 'redis://localhost:6379',
      storage: '/var/data'
    },
    browser: {
      api: 'https://api.example.com',
      storage: 'indexeddb',
      cache: 'localstorage'
    },
    bun: {
      database: 'sqlite://bun.sqlite',
      storage: './data',
      cache: 'memory'
    }
  }
  
  return configs[platform]
})

const program = Effect.gen(function* () {
  const config = yield* getEnvironmentConfig
  console.log('Environment config:', config)
  
  return config
})

Effect.runSync(program)
```
