# 字符串处理

> **版本**：v1.0

## 📚 概念介绍

Effect 提供了强大的字符串处理能力，包括模板字符串、格式化、正则表达式和文本解析等功能。

## 🎯 模板字符串

### 基本模板字符串

```js
import { Effect } from 'effect'

// 基本模板字符串
const name = 'Alice'
const age = 30

const message = `Hello, ${name}! You are ${age} years old.`
console.log(message)
// Hello, Alice! You are 30 years old.

// 在 Effect 中使用
const program = Effect.gen(function* () {
  const user = { name: 'Alice', age: 30 }
  const greeting = `Hello, ${user.name}!`
  console.log(greeting)
  return greeting
})

Effect.runSync(program)
```

### 表达式插值

```js
import { Effect } from 'effect'

// 表达式插值
const a = 10
const b = 20

const result = `The sum of ${a} and ${b} is ${a + b}`

  return message

Effect.runSync(program)
```

## 💡 字符串格式化

### 基本格式化

```js
import { Effect } from 'effect'

// 基本格式化
const formatName = (first: string, last: string) => {
  return `${first.charAt(0).toUpperCase()}${first.slice(1).toLowerCase()} ${last.toUpperCase()}`
}

console.log(formatName('alice', 'smith'))
// Alice SMITH

// 在 Effect 中使用
const program = Effect.gen(function* () {
  const user = { firstName: 'alice', lastName: 'smith' }
  const formattedName = formatName(user.firstName, user.lastName)
  console.log(formattedName)
  return formattedName
})

Effect.runSync(program)
```

### 数字格式化

```js
import { Effect } from 'effect'

// 数字格式化
const formatNumber = (num: number, decimals: number = 2) => {
  return num.toFixed(decimals)
}

const formatCurrency = (num: number, currency: string = 'USD') => {
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: currency
  }).format(num)
}

const formatPercent = (num: number) => {
  return new Intl.NumberFormat('en-US', {
    style: 'percent',
    minimumFractionDigits: 2,
    maximumFractionDigits: 2
  }).format(num)
}

console.log(formatNumber(3.14159, 2))
// 3.14

console.log(formatCurrency(1234.56))
// $1,234.56

console.log(formatPercent(0.1234))
// 12.34%

// 在 Effect 中使用
const program = Effect.gen(function* () {
  const price = 1234.56
  const discount = 0.1234
  
  console.log(formatNumber(price, 2))
  console.log(formatCurrency(price))
  console.log(formatPercent(discount))
  
  return { price, discount }
})

Effect.runSync(program)
```

## 🚀 正则表达式

### 基本匹配

```js
import { Effect } from 'effect'

// 基本匹配
const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/

const isValidEmail = (email: string) => {
  return emailRegex.test(email)
}

console.log(isValidEmail('alice@example.com'))
// true

console.log(isValidEmail('invalid-email'))
// false

// 在 Effect 中使用
const program = Effect.gen(function* () {
  const emails = [
    'alice@example.com',
    'bob@example.com',
    'invalid-email',
    'charlie@example.com'
  ]
  
  const validEmails = emails.filter(isValidEmail)
  console.log('Valid emails:', validEmails)
  return validEmails
})

Effect.runSync(program)
```

### 模式匹配

```js
import { Effect } from 'effect'

// 模式匹配
const extractUrlParts = (url: string) => {
  const urlRegex = /^(https?):\/\/([^\/]+)(\/.*)?$/
  const match = url.match(urlRegex)
  
  if (match) {
    return {
      protocol: match[1],
      domain: match[2],
      path: match[3] || '/'
    }
  }
  
  return null
}

console.log(extractUrlParts('https://example.com/path/to/resource'))
// { protocol: 'https', domain: 'example.com', path: '/path/to/resource' }

console.log(extractUrlParts('http://localhost:3000'))
// { protocol: 'http', domain: 'localhost:3000', path: '/' }

// 在 Effect 中使用
const program = Effect.gen(function* () {
  const urls = [
    'https://example.com/path',
    'http://localhost:3000',
    'https://api.example.com/v1/users'
  ]
  
  const parsedUrls = urls.map(extractUrlParts).filter(Boolean)
  console.log(parsedUrls)
  return parsedUrls
})

Effect.runSync(program)
```

### 字符串替换

```js
import { Effect } from 'effect'

// 字符串替换
const sanitizeInput = (input: string) => {
  return input
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;')
}

console.log(sanitizeInput('<script>alert("XSS")</script>'))
// &lt;script&gt;alert(&quot;XSS&quot;)&lt;/script&gt;

// 在 Effect 中使用
const program = Effect.gen(function* () {
  const userInput = '<div>Hello, <b>World</b>!</div>'
  const sanitized = sanitizeInput(userInput)
  
  console.log('Original:', userInput)
  console.log('Sanitized:', sanitized)
  
  return sanitized
})

Effect.runSync(program)
```

```js

// JSON 解析
const parseJson = Effect.tryCatch(
  (jsonString) => JSON.parse(jsonString),
  (error) => `Invalid JSON: ${error}`
)

const program = Effect.gen(function* () {
  const validJson = '{"name": "Alice", "age": 30}'
  const invalidJson = '{"name": "Alice", "age": 30'
  
  const validResult = yield* parseJson(validJson)
  console.log('Valid JSON:', validResult)
  // { name: 'Alice', age: 30 }
  
  const invalidResult = yield* parseJson(invalidJson)
  console.log('Invalid JSON:', invalidResult)
  // Invalid JSON: ...
  
  return { validResult, invalidResult }
})

Effect.runSync(program)
```

### CSV 解析

```js
import { Effect } from 'effect'

// CSV 解析
const parseCsv = (csv: string) => {
  const lines = csv.trim().split('\n')
  const headers = lines[0].split(',').map(h => h.trim())
  
  return lines.slice(1).map(line => {
    const values = line.split(',').map(v => v.trim())
    const row: Record<string, string> = {}
    
    headers.forEach((header, index) => {
      row[header] = values[index] || ''
    })
    
    return row
  })
}

const csvData = `name,age,city
Alice,30,New York
Bob,25,London
Charlie,35,Paris`

console.log(parseCsv(csvData))
// [
//   { name: 'Alice', age: '30', city: 'New York' },
//   { name: 'Bob', age: '25', city: 'London' },
//   { name: 'Charlie', age: '35', city: 'Paris' }
// ]

// 在 Effect 中使用
const program = Effect.gen(function* () {
  const csvData = `name,age,city
Alice,30,New York
Bob,25,London
Charlie,35,Paris`
  
  const parsed = parseCsv(csvData)
  console.log(parsed)
  return parsed
})

Effect.runSync(program)
```

### 模板解析

```js
import { Effect } from 'effect'

// 模板解析
const parseTemplate = (template: string, variables: Record<string, string>) => {
  return template.replace(/\{\{(\w+)\}\}/g, (match, key) => {
    return variables[key] || match
  })
}

const template = 'Hello, {{name}}! Welcome to {{location}}.'
const variables = {
  name: 'Alice',
  location: 'New York'
}

console.log(parseTemplate(template, variables))
// Hello, Alice! Welcome to New York.

// 在 Effect 中使用
const program = Effect.gen(function* () {
  const template = 'Hello, {{name}}! Welcome to {{location}}.'
  const variables = {
    name: 'Alice',
    location: 'New York'
  }
  
  const result = parseTemplate(template, variables)
  console.log(result)
  return result
})

Effect.runSync(program)
```

## 📝 实际应用示例

### 表单验证

```js
import { Effect } from 'effect'

// 表单验证
const validateEmail = (email: string) => {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
  return emailRegex.test(email) ? null : 'Invalid email format'
}

const validatePassword = (password: string) => {
  if (password.length < 8) {
    return 'Password must be at least 8 characters'
  }
  if (!/[A-Z]/.test(password)) {
    return 'Password must contain at least one uppercase letter'
  }
  if (!/[0-9]/.test(password)) {
    return 'Password must contain at least one number'
  }
  return null
}

const validateForm = (data: { email: string; password: string }) => {
  const errors: string[] = []
  
  const emailError = validateEmail(data.email)
  if (emailError) errors.push(emailError)
  
  const passwordError = validatePassword(data.password)
  if (passwordError) errors.push(passwordError)
  
  return errors
}

const program = Effect.gen(function* () {
  const formData = {
    email: 'alice@example.com',
    password: 'Password123'
  }
  
  const errors = validateForm(formData)
  
  if (errors.length > 0) {
    console.log('Validation errors:', errors)
  } else {
    console.log('Form is valid!')
  }
  
  return errors
})

Effect.runSync(program)
```

### 数据清洗

```js
import { Effect } from 'effect'

// 数据清洗
const cleanPhoneNumber = (phone: string) => {
  return phone.replace(/\D/g, '')
}

const cleanEmail = (email: string) => {
  return email.trim().toLowerCase()
}

const cleanText = (text: string) => {
  return text
    .trim()
    .replace(/\s+/g, ' ')
    .replace(/[^\w\s]/g, '')
}

const program = Effect.gen(function* () {
  const rawData = {
    phone: '(123) 456-7890',
    email: '  ALICE@EXAMPLE.COM  ',
    text: '  Hello,   World!  '
  }
  
  const cleanedData = {
    phone: cleanPhoneNumber(rawData.phone),
    email: cleanEmail(rawData.email),
    text: cleanText(rawData.text)
  }
  
  console.log('Original:', rawData)
  console.log('Cleaned:', cleanedData)
  
  return cleanedData
})

Effect.runSync(program)
```

### URL 处理

```js
import { Effect } from 'effect'

// URL 处理
const buildUrl = (baseUrl: string, path: string, params: Record<string, string>) => {
  const url = new URL(path, baseUrl)
  
  Object.entries(params).forEach(([key, value]) => {
    url.searchParams.set(key, value)
  })
  
  return url.toString()
}

const parseQueryString = (queryString: string) => {
  const params = new URLSearchParams(queryString)
  const result: Record<string, string> = {}
  
  params.forEach((value, key) => {
    result[key] = value
  })
  
  return result
}

const program = Effect.gen(function* () {
  const url = buildUrl('https://api.example.com', '/v1/users', {
    page: '1',
    limit: '10',
    sort: 'name'
  })
  
  console.log('Built URL:', url)
  // https://api.example.com/v1/users?page=1&limit=10&sort=name
  
  const queryString = 'page=1&limit=10&sort=name'
  const parsedParams = parseQueryString(queryString)
  
  console.log('Parsed params:', parsedParams)
  // { page: '1', limit: '10', sort: 'name' }
  
  return { url, parsedParams }
})

Effect.runSync(program)
```
