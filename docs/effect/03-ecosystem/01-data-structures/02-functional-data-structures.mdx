# 函数式数据结构

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

函数式数据结构提供了类型安全的方式来处理可能缺失的值、错误结果和验证场景。这些数据结构帮助编写更健壮、更可预测的代码。

## 🎯 Option 类型

### 基本概念

```js
import { Option } from 'effect'

// Option 表示可能存在或可能不存在的值
const someValue = Option.some(42)
const noValue = Option.none()

console.log(someValue)  // Option.some(42)
console.log(noValue)    // Option.none()
```

### 创建 Option

```js
import { Option } from 'effect'

// 从可能为 null/undefined 的值创建
const maybeValue = Option.fromNullable(null)
console.log(maybeValue)  // Option.none()

const definitelyValue = Option.fromNullable(42)
console.log(definitelyValue)  // Option.some(42)

// 从条件创建
const conditional = Option.liftPredicate(
  42,
  (n) => n > 0
)
console.log(conditional)  // Option.some(42)

const noMatch = Option.liftPredicate(
  -1,
  (n) => n > 0
)
console.log(noMatch)  // Option.none()
```

### 操作 Option

```js
import { Option } from 'effect'

// 映射
const mapped = Option.map(someValue, (n) => n * 2)
console.log(mapped)  // Option.some(84)

// 链式调用
const chained = Option.flatMap(someValue, (n) =>
  n > 0 ? Option.some(n * 2) : Option.none()
)
console.log(chained)  // Option.some(84)

// 过滤
const filtered = Option.filter(someValue, (n) => n > 0)
console.log(filtered)  // Option.some(42)

// 获取值或默认值
const value = Option.getOrElse(someValue, () => 0)
console.log(value)  // 42

const defaultValue = Option.getOrElse(noValue, () => 0)
console.log(defaultValue)  // 0
```

### 模式匹配

```js
import { Option } from 'effect'

// 模式匹配
const result = Option.match(someValue, {
  onSome: (value) => `Value is ${value}`,
  onNone: () => 'No value'
})
console.log(result)  // 'Value is 42'

const noneResult = Option.match(noValue, {
  onSome: (value) => `Value is ${value}`,
  onNone: () => 'No value'
})
console.log(noneResult)  // 'No value'
```

## 💡 Either 类型

### 基本概念

```js
import { Either } from 'effect'

// Either 表示可能失败的操作
const success = Either.right(42)
const failure = Either.left('Error occurred')

console.log(success)  // Either.right(42)
console.log(failure)  // Either.left('Error occurred')
```

### 创建 Either

```js
import { Either } from 'effect'

// 从可能抛出异常的函数创建
const safeDivide = (a: number, b: number) =>
  b === 0
    ? Either.left('Division by zero')
    : Either.right(a / b)

console.log(safeDivide(10, 2))  // Either.right(5)
console.log(safeDivide(10, 0))  // Either.left('Division by zero')

// 使用 tryCatch
const parseNumber = (str: string) =>
  Either.tryCatch(
    () => parseInt(str),
    () => 'Invalid number'
  )

console.log(parseNumber('42'))    // Either.right(42)
console.log(parseNumber('abc'))   // Either.left('Invalid number')
```

### 操作 Either

```js
import { Either } from 'effect'

// 映射
const mapped = Either.map(success, (n) => n * 2)
console.log(mapped)  // Either.right(84)

// 链式调用
const chained = Either.flatMap(success, (n) =>
  n > 0 ? Either.right(n * 2) : Either.left('Negative number')
)
console.log(chained)  // Either.right(84)

// 处理错误
const handled = Either.catchAll(failure, (error) =>
  Either.right(`Error: ${error}`)
)
console.log(handled)  // Either.right('Error: Error occurred')

// 获取值或默认值
const value = Either.getOrElse(success, () => 0)
console.log(value)  // 42

const defaultValue = Either.getOrElse(failure, () => 0)
console.log(defaultValue)  // 0
```

### 模式匹配

```js
import { Either } from 'effect'

// 模式匹配
const result = Either.match(success, {
  onRight: (value) => `Success: ${value}`,
  onLeft: (error) => `Error: ${error}`
})
console.log(result)  // 'Success: 42'

const errorResult = Either.match(failure, {
  onRight: (value) => `Success: ${value}`,
  onLeft: (error) => `Error: ${error}`
})
console.log(errorResult)  // 'Error: Error occurred'
```

## 🚀 Result 类型

### 基本概念

```js
import { Result } from 'effect'

// Result 是 Either 的别名，表示操作结果
const success = Result.succeed(42)
const failure = Result.fail('Error occurred')

console.log(success)  // Either.right(42)
console.log(failure)  // Either.left('Error occurred')
```

### 创建 Result

```js
import { Result } from 'effect'

// 从 Effect 创建 Result
const effectResult = Result.succeed(
  Result.try(() => {
    if (Math.random() > 0.5) {
      return 42
    }
    throw new Error('Random error')
  })
)

console.log(effectResult)
// Either.right(42) 或 Either.left(Error: Random error)

// 使用 Result.fromOption
const optionToResult = Result.fromOption(
  Option.some(42),
  () => 'Value not found'
)
console.log(optionToResult)  // Either.right(42)

const noneToResult = Result.fromOption(
  Option.none(),
  () => 'Value not found'
)
console.log(noneToResult)  // Either.left('Value not found')
```

### 操作 Result

```js
import { Result } from 'effect'

// 映射
const mapped = Result.map(success, (n) => n * 2)
console.log(mapped)  // Either.right(84)

// 链式调用
const chained = Result.flatMap(success, (n) =>
  n > 0 ? Result.succeed(n * 2) : Result.fail('Negative number')
)
console.log(chained)  // Either.right(84)

// 组合多个 Result
const combined = Result.all([
  Result.succeed(1),
  Result.succeed(2),
  Result.succeed(3)
])
console.log(combined)  // Either.right([1, 2, 3])

const failed = Result.all([
  Result.succeed(1),
  Result.fail('Error'),
  Result.succeed(3)
])
console.log(failed)  // Either.left('Error')
```

## 🎯 Validation 类型

### 基本概念

```js
import { Either } from 'effect'

// Validation 用于累积多个错误
const validateUser = (name: string, age: number) => {
  const errors: string[] = []
  
  if (name.length < 3) {
    errors.push('Name must be at least 3 characters')
  }
  
  if (age < 0 || age > 120) {
    errors.push('Age must be between 0 and 120')
  }
  
  return errors.length > 0
    ? Either.left(errors)
    : Either.right({ name, age })
}

console.log(validateUser('Alice', 30))
// Either.right({ name: 'Alice', age: 30 })

console.log(validateUser('Al', 150))
// Either.left(['Name must be at least 3 characters', 'Age must be between 0 and 120'])
```

### 累积验证

```js
import { Either, Array as Arr } from 'effect'

// 累积多个验证结果
const validateEmail = (email: string): Either.Either<string, string> => {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
  return emailRegex.test(email)
    ? Either.right(email)
    : Either.left('Invalid email format')
}

const validatePassword = (password: string): Either.Either<string, string> => {
  return password.length >= 8
    ? Either.right(password)
    : Either.left('Password must be at least 8 characters')
}

const validateForm = (email: string, password: string) => {
  const emailResult = validateEmail(email)
  const passwordResult = validatePassword(password)
  
  const errors = Arr.make(
    Either.isLeft(emailResult) ? Either.getLeft(emailResult) : null,
    Either.isLeft(passwordResult) ? Either.getLeft(passwordResult) : null
  ).filter((error): error is string => error !== null)
  
  return errors.length > 0
    ? Either.left(errors)
    : Either.right({ email, password })
}

console.log(validateForm('alice@example.com', 'password123'))
// Either.right({ email: 'alice@example.com', password: 'password123' })

console.log(validateForm('invalid-email', 'short'))
// Either.left(['Invalid email format', 'Password must be at least 8 characters'])
```

### 验证链

```js
import { Either } from 'effect'

// 验证链
const validateName = (name: string): Either.Either<string, string> => {
  return name.length >= 3
    ? Either.right(name)
    : Either.left('Name must be at least 3 characters')
}

const validateAge = (age: number): Either.Either<string, number> => {
  return age >= 0 && age <= 120
    ? Either.right(age)
    : Either.left('Age must be between 0 and 120')
}

const validateUser = (name: string, age: number) => {
  return Either.flatMap(validateName(name), (validName) =>
    Either.map(validateAge(age), (validAge) => ({
      name: validName,
      age: validAge
    }))
  )
}

console.log(validateUser('Alice', 30))
// Either.right({ name: 'Alice', age: 30 })

console.log(validateUser('Al', 150))
// Either.left('Name must be at least 3 characters')
```

## 📝 实际应用示例

### 表单验证

```js
import { Either, Array as Arr } from 'effect'

// 表单验证
interface FormData {
  email: string
  password: string
  confirmPassword: string
}

const validateEmail = (email: string): Either.Either<string, string> => {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
  return emailRegex.test(email)
    ? Either.right(email)
    : Either.left('Invalid email format')
}

const validatePassword = (password: string): Either.Either<string, string> => {
  return password.length >= 8
    ? Either.right(password)
    : Either.left('Password must be at least 8 characters')
}

const validatePasswordMatch = (
  password: string,
  confirmPassword: string
): Either.Either<string, void> => {
  return password === confirmPassword
    ? Either.right(void 0)
    : Either.left('Passwords do not match')
}

const validateForm = (data: FormData) => {
  const emailResult = validateEmail(data.email)
  const passwordResult = validatePassword(data.password)
  const matchResult = validatePasswordMatch(
    data.password,
    data.confirmPassword
  )
  
  const errors = Arr.make(
    Either.isLeft(emailResult) ? Either.getLeft(emailResult) : null,
    Either.isLeft(passwordResult) ? Either.getLeft(passwordResult) : null,
    Either.isLeft(matchResult) ? Either.getLeft(matchResult) : null
  ).filter((error): error is string => error !== null)
  
  return errors.length > 0
    ? Either.left(errors)
    : Either.right({
        email: Either.getOrThrow(emailResult),
        password: Either.getOrThrow(passwordResult)
      })
}

// 使用表单验证
const formData = {
  email: 'alice@example.com',
  password: 'password123',
  confirmPassword: 'password123'
}

const result = validateForm(formData)
console.log(result)
// Either.right({ email: 'alice@example.com', password: 'password123' })

const invalidFormData = {
  email: 'invalid-email',
  password: 'short',
  confirmPassword: 'different'
}

const invalidResult = validateForm(invalidFormData)
console.log(invalidResult)
// Either.left(['Invalid email format', 'Password must be at least 8 characters', 'Passwords do not match'])
```

### API 错误处理

```js
import { Either, Effect } from 'effect'

// API 错误处理
const fetchUser = async (id: number): Promise<Either.Either<string, any>> => {
  try {
    const response = await fetch(`/api/users/${id}`)
    
    if (!response.ok) {
      return Either.left(`HTTP error: ${response.status}`)
    }
    
    const user = await response.json()
    return Either.right(user)
  } catch (error) {
    return Either.left(`Network error: ${error.message}`)
  }
}

const fetchUserEffect = (id: number) =>
  Effect.tryPromise(() => fetchUser(id))

// 使用 API 错误处理
const program = Effect.gen(function* () {
  const result = yield* fetchUserEffect(1)
  
  const user = Either.match(result, {
    onLeft: (error) => {
      console.error('Error:', error)
      return null
    },
    onRight: (user) => {
      console.log('User:', user)
      return user
    }
  })
  
  return user
})

Effect.runPromise(program)
```

### 配置验证

```js
import { Either, Array as Arr } from 'effect'

// 配置验证
interface Config {
  apiUrl: string
  apiKey: string
  timeout: number
}

const validateApiUrl = (url: string): Either.Either<string, string> => {
  try {
    new URL(url)
    return Either.right(url)
  } catch {
    return Either.left('Invalid API URL')
  }
}

const validateApiKey = (key: string): Either.Either<string, string> => {
  return key.length >= 16
    ? Either.right(key)
    : Either.left('API key must be at least 16 characters')
}

const validateTimeout = (timeout: number): Either.Either<string, number> => {
  return timeout > 0 && timeout <= 60000
    ? Either.right(timeout)
    : Either.left('Timeout must be between 1 and 60000 ms')
}

const validateConfig = (config: Partial<Config>) => {
  const urlResult = validateApiUrl(config.apiUrl || '')
  const keyResult = validateApiKey(config.apiKey || '')
  const timeoutResult = validateTimeout(config.timeout || 0)
  
  const errors = Arr.make(
    Either.isLeft(urlResult) ? Either.getLeft(urlResult) : null,
    Either.isLeft(keyResult) ? Either.getLeft(keyResult) : null,
    Either.isLeft(timeoutResult) ? Either.getLeft(timeoutResult) : null
  ).filter((error): error is string => error !== null)
  
  return errors.length > 0
    ? Either.left(errors)
    : Either.right({
        apiUrl: Either.getOrThrow(urlResult),
        apiKey: Either.getOrThrow(keyResult),
        timeout: Either.getOrThrow(timeoutResult)
      })
}

// 使用配置验证
const validConfig = {
  apiUrl: 'https://api.example.com',
  apiKey: '1234567890123456',
  timeout: 5000
}

const result = validateConfig(validConfig)
console.log(result)
// Either.right({ apiUrl: 'https://api.example.com', apiKey: '1234567890123456', timeout: 5000 })
```
