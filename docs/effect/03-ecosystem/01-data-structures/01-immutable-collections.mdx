# 不可变集合

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了丰富的不可变集合操作，帮助你以函数式的方式处理数据。不可变集合确保数据的一致性和可预测性。

## 🎯 Array 操作

### 基本操作

```js
import { Array as Arr } from 'effect'

// 创建数组
const arr = Arr.make(1, 2, 3, 4, 5)
console.log(arr)  // [1, 2, 3, 4, 5]

// 获取长度
const length = Arr.length(arr)
console.log(length)  // 5

// 获取元素
const first = Arr.head(arr)
console.log(first)  // Option.some(1)

const last = Arr.last(arr)
console.log(last)  // Option.some(5)
```

### 映射与过滤

```js
import { Array as Arr, Option } from 'effect'

// 映射
const doubled = Arr.map(arr, (n) => n * 2)
console.log(doubled)  // [2, 4, 6, 8, 10]

// 过滤
const evens = Arr.filter(arr, (n) => n % 2 === 0)
console.log(evens)  // [2, 4]

// 过滤映射
const evensDoubled = Arr.filterMap(arr, (n) =>
  n % 2 === 0 ? Option.some(n * 2) : Option.none()
)
console.log(evensDoubled)  // [4, 8]
```

### 归约与折叠

```js
import { Array as Arr } from 'effect'

// 归约
const sum = Arr.reduce(arr, 0, (acc, n) => acc + n)
console.log(sum)  // 15

// 折叠（从左到右）
const product = Arr.reduce(arr, 1, (acc, n) => acc * n)
console.log(product)  // 120

// 折叠（从右到左）
const reversedSum = Arr.reduceRight(arr, 0, (acc, n) => acc + n)
console.log(reversedSum)  // 15
```

### 排序与分组

```js
import { Array as Arr } from 'effect'

// 排序
const sorted = Arr.sort(arr, (a, b) => a - b)
console.log(sorted)  // [1, 2, 3, 4, 5]

// 反转
const reversed = Arr.reverse(arr)
console.log(reversed)  // [5, 4, 3, 2, 1]

// 分组
const grouped = Arr.group(arr, (n) => n % 2 === 0 ? 'even' : 'odd')
console.log(grouped)
// { odd: [1, 3, 5], even: [2, 4] }
```

## 💡 Map 和 Set

### Map 操作

```js
import { HashMap as Map } from 'effect'

// 创建 Map
const map = Map.make(
  ['a', 1],
  ['b', 2],
  ['c', 3]
)

// 获取值
const value = Map.get(map, 'b')
console.log(value)  // Option.some(2)

// 设置值
const updated = Map.set(map, 'd', 4)
console.log(Map.get(updated, 'd'))  // Option.some(4)

// 删除值
const removed = Map.remove(updated, 'a')
console.log(Map.has(removed, 'a'))  // false

// 遍历
Map.forEach(map, (value, key) => {
  console.log(`${key}: ${value}`)
})
```

### Set 操作

```js
import { HashSet as Set } from 'effect'

// 创建 Set
const set = Set.make(1, 2, 3, 4, 5)

// 检查包含
const hasValue = Set.has(set, 3)
console.log(hasValue)  // true

// 添加值
const added = Set.add(set, 6)
console.log(Set.has(added, 6))  // true

// 删除值
const removed = Set.remove(added, 1)
console.log(Set.has(removed, 1))  // false

// 集合操作
const set1 = Set.make(1, 2, 3)
const set2 = Set.make(3, 4, 5)

const union = Set.union(set1, set2)
console.log(union)  // Set(1, 2, 3, 4, 5)

const intersection = Set.intersection(set1, set2)
console.log(intersection)  // Set(3)

const difference = Set.difference(set1, set2)
console.log(difference)  // Set(1, 2)
```

## 🚀 Record 操作

### 基本操作

```js
import { Record as Rec } from 'effect'

// 创建 Record
const record = {
  name: 'Alice',
  age: 30,
  city: 'New York'
}

// 获取值
const name = Rec.get(record, 'name')
console.log(name)  // 'Alice'

// 设置值
const updated = Rec.set(record, 'age', 31)
console.log(updated.age)  // 31

// 删除键
const removed = Rec.remove(updated, 'city')
console.log(removed.city)  // undefined
```

### 遍历与转换

```js
import { Record as Rec } from 'effect'

// 映射值
const uppercased = Rec.map(record, (value) => {
  if (typeof value === 'string') {
    return value.toUpperCase()
  }
  return value
})
console.log(uppercased)
// { name: 'ALICE', age: 30, city: 'NEW YORK' }

// 转换为数组
const entries = Rec.toEntries(record)
console.log(entries)
// [['name', 'Alice'], ['age', 30], ['city', 'New York']]

// 获取键
const keys = Rec.keys(record)
console.log(keys)  // ['name', 'age', 'city']

// 获取值
const values = Rec.values(record)
console.log(values)  // ['Alice', 30, 'New York']
```

## 🎯 自定义数据结构

### 栈

```js
import { Array as Arr } from 'effect'

// 栈实现
const createStack = () => ({
  stack: Arr.empty<number>(),
  
  push: function(value: number) {
    return { ...this, stack: Arr.prepend(this.stack, value) }
  },
  
  pop: function() {
    const [head, tail] = Arr.splitAt(this.stack, 1)
    return { ...this, stack: tail, value: Arr.head(head) }
  },
  
  peek: function() {
    return Arr.head(this.stack)
  },
  
  isEmpty: function() {
    return Arr.isEmpty(this.stack)
  }
})

// 使用栈
const stack = createStack()
const stack1 = stack.push(1)
const stack2 = stack1.push(2)
const stack3 = stack2.push(3)

console.log(stack3.peek())  // Option.some(3)

const { stack: newStack, value } = stack3.pop()
console.log(value)  // Option.some(3)
console.log(newStack.peek())  // Option.some(2)
```

### 队列

```js
import { Array as Arr } from 'effect'

// 队列实现
const createQueue = () => ({
  queue: Arr.empty<number>(),
  
  enqueue: function(value: number) {
    return { ...this, queue: Arr.append(this.queue, value) }
  },
  
  dequeue: function() {
    const [head, tail] = Arr.splitAt(this.queue, 1)
    return { ...this, queue: tail, value: Arr.head(head) }
  },
  
  peek: function() {
    return Arr.head(this.queue)
  },
  
  isEmpty: function() {
    return Arr.isEmpty(this.queue)
  }
})

// 使用队列
const queue = createQueue()
const queue1 = queue.enqueue(1)
const queue2 = queue1.enqueue(2)
const queue3 = queue2.enqueue(3)

console.log(queue3.peek())  // Option.some(1)

const { queue: newQueue, value } = queue3.dequeue()
console.log(value)  // Option.some(1)
console.log(newQueue.peek())  // Option.some(2)
```

### 链表

```js
import { Option } from 'effect'

// 链表实现
type ListNode<T> = {
  value: T
  next: Option.Option<ListNode<T>>
}

const createList = <T>(values: T[]): Option.Option<ListNode<T>> => {
  if (values.length === 0) {
    return Option.none()
  }
  
  const [head, ...tail] = values
  return Option.some({
    value: head,
    next: createList(tail)
  })
}

const toArray = <T>(list: Option.Option<ListNode<T>>): T[] => {
  return Option.match(list, {
    onNone: () => [],
    onSome: (node) => [node.value, ...toArray(node.next)]
  })
}

const map = <T, U>(
  list: Option.Option<ListNode<T>>,
  fn: (value: T) => U
): Option.Option<ListNode<U>> => {
  return Option.match(list, {
    onNone: () => Option.none(),
    onSome: (node) => Option.some({
      value: fn(node.value),
      next: map(node.next, fn)
    })
  })
}

// 使用链表
const list = createList([1, 2, 3, 4, 5])
console.log(toArray(list))  // [1, 2, 3, 4, 5]

const doubled = map(list, (n) => n * 2)
console.log(toArray(doubled))  // [2, 4, 6, 8, 10]
```

## 📝 实际应用示例

### 数据转换

```js
import { Array as Arr, HashMap as Map } from 'effect'

// 数据转换
const users = [
  { id: 1, name: 'Alice', role: 'admin' },
  { id: 2, name: 'Bob', role: 'user' },
  { id: 3, name: 'Charlie', role: 'admin' }
]

// 转换为 Map
const userMap = Arr.reduce(
  users,
  Map.empty(),
  (map, user) => Map.set(map, user.id, user)
)

console.log(Map.get(userMap, 1))
// Option.some({ id: 1, name: 'Alice', role: 'admin' })

// 按角色分组
const groupedByRole = Arr.group(users, (user) => user.role)
console.log(groupedByRole)
// { admin: [...], user: [...] }
```

### 数据聚合

```js
import { Array as Arr } from 'effect'

// 数据聚合
const orders = [
  { id: 1, userId: 1, amount: 100 },
  { id: 2, userId: 1, amount: 200 },
  { id: 3, userId: 2, amount: 150 },
  { id: 4, userId: 2, amount: 250 }
]

// 按用户聚合订单金额
const aggregated = Arr.group(orders, (order) => order.userId)

const totals = Arr.map(Object.entries(aggregated), ([userId, userOrders]) => {
  const total = Arr.reduce(
    userOrders,
    0,
    (sum, order) => sum + order.amount
  )
  return { userId: parseInt(userId), total }
})

console.log(totals)
// [{ userId: 1, total: 300 }, { userId: 2, total: 400 }]
```

### 数据验证

```js
import { Array as Arr, Option } from 'effect'

// 数据验证
const validateEmail = (email: string): Option.Option<string> => {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
  return emailRegex.test(email) ? Option.some(email) : Option.none()
}

const users = [
  { name: 'Alice', email: 'alice@example.com' },
  { name: 'Bob', email: 'invalid-email' },
  { name: 'Charlie', email: 'charlie@example.com' }
]

// 验证邮箱
const validatedUsers = Arr.filterMap(users, (user) =>
  Option.map(validateEmail(user.email), (email) => ({
    ...user,
    email
  }))
)

console.log(validatedUsers)
// [{ name: 'Alice', email: 'alice@example.com' }, { name: 'Charlie', email: 'charlie@example.com' }]
```
