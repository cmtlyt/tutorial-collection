# 集合操作

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了强大的集合操作工具，让你能够以声明式的方式处理数据集合。

## 🎯 遍历与折叠

### 遍历

```js
import { Effect, Array as Arr } from 'effect'

// 遍历数组
const array = [1, 2, 3, 4, 5]

// forEach - 副作用遍历
Effect.runSync(
  Effect.forEach(array, (n) =>
    Effect.sync(() => console.log(n))
  )
)
// 输出: 1, 2, 3, 4, 5

// forEachSequential - 顺序遍历
Effect.runSync(
  Effect.forEachSequential(array, (n) =>
    Effect.sync(() => console.log(n))
  )
)

// forEachConcurrent - 并发遍历
Effect.runSync(
  Effect.forEachConcurrent(array, (n) =>
    Effect.sync(() => console.log(n))
  )
)
```

### 折叠

```js
import { Array as Arr } from 'effect'

// reduce - 归约
const sum = Arr.reduce(array, 0, (acc, n) => acc + n)
console.log(sum)  // 15

// reduceRight - 从右到左归约
const product = Arr.reduceRight(array, 1, (acc, n) => acc * n)
console.log(product)  // 120

// reduceWhile - 条件归约
const partialSum = Arr.reduceWhile(
  array,
  0,
  (acc, n) => acc + n,
  (acc) => acc < 10
)
console.log(partialSum)  // 10
```

### 扫描

```js
import { Array as Arr } from 'effect'

// scan - 扫描
const runningSum = Arr.scan(array, 0, (acc, n) => acc + n)
console.log(runningSum)  // [0, 1, 3, 6, 10, 15]

// scanRight - 从右到左扫描
const runningProduct = Arr.scanRight(array, 1, (acc, n) => acc * n)
console.log(runningProduct)  // [120, 120, 60, 20, 5, 1]
```

## 💡 过滤与映射

### 过滤

```js
import { Array as Arr } from 'effect'

// filter - 过滤
const evens = Arr.filter(array, (n) => n % 2 === 0)
console.log(evens)  // [2, 4]

// filterMap - 过滤并映射
const doubledEvens = Arr.filterMap(array, (n) =>
  n % 2 === 0 ? Option.some(n * 2) : Option.none()
)
console.log(doubledEvens)  // [4, 8]

// partition - 分区
const [evens, odds] = Arr.partition(array, (n) => n % 2 === 0)
console.log(evens)  // [2, 4]
console.log(odds)   // [1, 3, 5]
```

### 映射

```js
import { Array as Arr } from 'effect'

// map - 映射
const doubled = Arr.map(array, (n) => n * 2)
console.log(doubled)  // [2, 4, 6, 8, 10]

// mapWithIndex - 带索引的映射
const withIndex = Arr.mapWithIndex(array, (n, i) => `${i}: ${n}`)
console.log(withIndex)  // ['0: 1', '1: 2', '2: 3', '3: 4', '4: 5']

// flatMap - 扁平映射
const nested = [[1, 2], [3, 4], [5, 6]]
const flattened = Arr.flatMap(nested, (arr) => arr)
console.log(flattened)  // [1, 2, 3, 4, 5, 6]
```

### 排序与查找

```js
import { Array as Arr } from 'effect'

// sort - 排序
const sorted = Arr.sort(array, (a, b) => a - b)
console.log(sorted)  // [1, 2, 3, 4, 5]

// sortBy - 按键排序
const users = [
  { name: 'Alice', age: 30 },
  { name: 'Bob', age: 25 },
  { name: 'Charlie', age: 35 }
]

const sortedByAge = Arr.sortBy(users, (user) => user.age)
console.log(sortedByAge)
// [{ name: 'Bob', age: 25 }, { name: 'Alice', age: 30 }, { name: 'Charlie', age: 35 }]

// find - 查找
const found = Arr.find(array, (n) => n > 3)
console.log(found)  // Option.some(4)

// findFirst - 查找第一个
const firstEven = Arr.findFirst(array, (n) => n % 2 === 0)
console.log(firstEven)  // Option.some(2)

// findLast - 查找最后一个
const lastEven = Arr.findLast(array, (n) => n % 2 === 0)
console.log(lastEven)  // Option.some(4)
```

## 🚀 分组与排序

### 分组

```js
import { Array as Arr } from 'effect'

// group - 分组
const grouped = Arr.group(array, (n) => n % 2 === 0 ? 'even' : 'odd')
console.log(grouped)
// { odd: [1, 3, 5], even: [2, 4] }

// groupBy - 按键分组
const products = [
  { category: 'fruit', name: 'apple' },
  { category: 'fruit', name: 'banana' },
  { category: 'vegetable', name: 'carrot' },
  { category: 'fruit', name: 'orange' }
]

const groupedByCategory = Arr.groupBy(products, (product) => product.category)
console.log(groupedByCategory)
// {
//   fruit: [{ category: 'fruit', name: 'apple' }, { category: 'fruit', name: 'banana' }, { category: 'fruit', name: 'orange' }],
//   vegetable: [{ category: 'vegetable', name: 'carrot' }]
// }

// chunk - 分块
const chunked = Arr.chunk(array, 2)
console.log(chunked)  // [[1, 2], [3, 4], [5]]
```

### 排序

```js
import { Array as Arr } from 'effect'

// sort - 排序
const sorted = Arr.sort(array, (a, b) => a - b)
console.log(sorted)  // [1, 2, 3, 4, 5]

// reverse - 反转
const reversed = Arr.reverse(array)
console.log(reversed)  // [5, 4, 3, 2, 1]

// sortBy - 多键排序
const users = [
  { name: 'Alice', age: 30 },
  { name: 'Bob', age: 25 },
  { name: 'Alice', age: 25 }
]

const sortedByNameAge = Arr.sortBy(
  users,
  (user) => user.name,
  (user) => user.age
)
console.log(sortedByNameAge)
// [{ name: 'Alice', age: 25 }, { name: 'Alice', age: 30 }, { name: 'Bob', age: 25 }]
```

## 🎯 集合转换

### 转换为其他集合

```js
import { Array as Arr, HashSet as Set, HashMap as Map } from 'effect'

// 转换为 Set
const set = Arr.fromIterable(array)
console.log(set)  // Set(1, 2, 3, 4, 5)

// 转换为 Map
const pairs = [['a', 1], ['b', 2], ['c', 3]] as const
const map = Arr.fromIterable(pairs)
console.log(map)  // Map { 'a' => 1, 'b' => 2, 'c' => 3 }

// 转换为 Record
const record = Arr.toRecord(
  [['a', 1], ['b', 2], ['c', 3]],
  (pair) => pair[0],
  (pair) => pair[1]
)
console.log(record)  // { a: 1, b: 2, c: 3 }
```

### 集合操作

```js
import { Array as Arr } from 'effect'

// union - 并集
const array1 = [1, 2, 3]
const array2 = [3, 4, 5]
const union = Arr.union(array1, array2)
console.log(union)  // [1, 2, 3, 4, 5]

// intersection - 交集
const intersection = Arr.intersection(array1, array2)
console.log(intersection)  // [3]

// difference - 差集
const difference = Arr.difference(array1, array2)
console.log(difference)  // [1, 2]

// zip - 压缩
const zipped = Arr.zip([1, 2, 3], ['a', 'b', 'c'])
console.log(zipped)  // [[1, 'a'], [2, 'b'], [3, 'c']]

// unzip - 解压
const unzipped = Arr.unzip([[1, 'a'], [2, 'b'], [3, 'c']])
console.log(unzipped)  // [[1, 2, 3], ['a', 'b', 'c']]
```

### 高级转换

```js
import { Array as Arr } from 'effect'

// flatMap - 扁平映射
const nested = [[1, 2], [3, 4], [5, 6]]
const flattened = Arr.flatMap(nested, (arr) => arr)
console.log(flattened)  // [1, 2, 3, 4, 5, 6]

// flatten - 扁平化
const deepNested = [[1, [2, 3]], [4, [5, 6]]]
const deepFlattened = Arr.flatten(deepNested)
console.log(deepFlattened)  // [1, [2, 3], 4, [5, 6]]

// distinct - 去重
const withDuplicates = [1, 2, 2, 3, 3, 3, 4, 5]
const distinct = Arr.distinct(withDuplicates)
console.log(distinct)  // [1, 2, 3, 4, 5]

// distinctBy - 按键去重
const users = [
  { id: 1, name: 'Alice' },
  { id: 2, name: 'Bob' },
  { id: 1, name: 'Alice' }
]
const distinctUsers = Arr.distinctBy(users, (user) => user.id)
console.log(distinctUsers)
// [{ id: 1, name: 'Alice' }, { id: 2, name: 'Bob' }]
```

## 📝 实际应用示例

### 数据处理管道

```js
import { Array as Arr, Option, Effect } from 'effect'

// 数据处理管道
const processNumbers = (numbers: number[]) => {
  const positive = Arr.filter(numbers, (n) => n > 0)           // 过滤正数
  const doubled = Arr.map(positive, (n) => n * 2)               // 翻倍
  const filtered = Arr.filter(doubled, (n) => n < 100)          // 过滤小于100
  const sum = Arr.reduce(filtered, 0, (acc, n) => acc + n)      // 求和
  return sum
}

const numbers = [-5, 1, 2, 3, 4, 5, 100, 200]
const result = processNumbers(numbers)
console.log(result)  // 30
```

### 数据聚合

```js
import { Array as Arr } from 'effect'

// 数据聚合
const orders = [
  { id: 1, userId: 1, amount: 100, status: 'completed' },
  { id: 2, userId: 1, amount: 200, status: 'completed' },
  { id: 3, userId: 2, amount: 150, status: 'pending' },
  { id: 4, userId: 2, amount: 250, status: 'completed' },
  { id: 5, userId: 3, amount: 300, status: 'cancelled' }
]

// 按用户聚合订单
const userOrders = Arr.groupBy(orders, (order) => order.userId)

// 计算每个用户的总金额
const userTotals = Arr.map(
  Object.entries(userOrders),
  ([userId, orders]) => {
    const total = Arr.reduce(
      orders,
      0,
      (sum, order) => sum + order.amount
    )
    return { userId: parseInt(userId), total }
  }
)

console.log(userTotals)
// [{ userId: 1, total: 300 }, { userId: 2, total: 400 }, { userId: 3, total: 300 }]

// 按状态聚合订单
const statusOrders = Arr.groupBy(orders, (order) => order.status)

// 计算每个状态的订单数和总金额
const statusStats = Arr.map(
  Object.entries(statusOrders),
  ([status, orders]) => {
    const count = Arr.length(orders)
    const total = Arr.reduce(
      orders,
      0,
      (sum, order) => sum + order.amount
    )
    return { status, count, total }
  }
)

console.log(statusStats)
// [{ status: 'completed', count: 3, total: 550 }, { status: 'pending', count: 1, total: 150 }, { status: 'cancelled', count: 1, total: 300 }]
```

### 数据转换

```js
import { Array as Arr, HashMap as Map } from 'effect'

// 数据转换
const rawData = [
  { id: 1, name: 'Alice', role: 'admin' },
  { id: 2, name: 'Bob', role: 'user' },
  { id: 3, name: 'Charlie', role: 'admin' }
]

// 转换为 Map
const userMap = Arr.reduce(
  rawData,
  Map.empty(),
  (map, user) => Map.set(map, user.id, user)
)

console.log(Map.get(userMap, 1))
// Option.some({ id: 1, name: 'Alice', role: 'admin' })

// 转换为分组数据
const groupedByRole = Arr.group(rawData, (user) => user.role)

console.log(groupedByRole)
// {
//   admin: [{ id: 1, name: 'Alice', role: 'admin' }, { id: 3, name: 'Charlie', role: 'admin' }],
//   user: [{ id: 2, name: 'Bob', role: 'user' }]
// }

// 转换为简化的数据
const simplified = Arr.map(rawData, (user) => ({
  id: user.id,
  name: user.name,
  isAdmin: user.role === 'admin'
}))

console.log(simplified)
// [{ id: 1, name: 'Alice', isAdmin: true }, { id: 2, name: 'Bob', isAdmin: false }, { id: 3, name: 'Charlie', isAdmin: true }]
```

### 复杂查询

```js
import { Array as Arr } from 'effect'

// 复杂查询
const products = [
  { id: 1, name: 'Laptop', category: 'Electronics', price: 1000, stock: 10 },
  { id: 2, name: 'Mouse', category: 'Electronics', price: 50, stock: 100 },
  { id: 3, name: 'Keyboard', category: 'Electronics', price: 100, stock: 50 },
  { id: 4, name: 'Desk', category: 'Furniture', price: 500, stock: 5 },
  { id: 5, name: 'Chair', category: 'Furniture', price: 300, stock: 20 }
]

// 查询：电子产品中价格大于100且库存大于10的产品
const electronics = Arr.filter(products, (p) => p.category === 'Electronics')
const expensive = Arr.filter(electronics, (p) => p.price > 100)
const inStock = Arr.filter(expensive, (p) => p.stock > 10)

console.log(inStock)
// [{ id: 3, name: 'Keyboard', category: 'Electronics', price: 100, stock: 50 }]

// 查询：按类别分组，计算每个类别的平均价格
const grouped = Arr.groupBy(products, (p) => p.category)
const categoryStats = Arr.map(
  Object.entries(grouped),
  ([category, items]) => {
    const avgPrice = Arr.reduce(
      items,
      0,
      (sum, item) => sum + item.price
    ) / Arr.length(items)
    return { category, avgPrice: avgPrice.toFixed(2) }
  }
)

console.log(categoryStats)
// [
//   { category: 'Electronics', avgPrice: '383.33' },
//   { category: 'Furniture', avgPrice: '400.00' }
// ]
```
