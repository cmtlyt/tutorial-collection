# 工作流模式

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了丰富的工作流模式，包括顺序工作流、并行工作流、条件分支、循环工作流等。

## 🎯 顺序工作流

### 基本顺序执行

```js
import { Effect } from 'effect'
import { ClusterWorkflow } from '@effect/cluster-workflow'

// 顺序工作流
const sequentialWorkflow = ClusterWorkflow.make('sequential-workflow', {
  execute: Effect.gen(function* () {
    // 步骤 1
    console.log('Step 1: Initialize')
    yield* Effect.sleep(500)
    
    // 步骤 2
    console.log('Step 2: Process')
    yield* Effect.sleep(500)
    
    // 步骤 3
    console.log('Step 3: Finalize')
    yield* Effect.sleep(500)
    
    return 'Sequential workflow completed'
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* sequentialWorkflow.execute
  return result
})

Effect.runSync(program)
```

### 依赖顺序

```js
import { Effect } from 'effect'
import { ClusterWorkflow } from '@effect/cluster-workflow'

// 依赖顺序工作流
const dependentWorkflow = ClusterWorkflow.make('dependent-workflow', {
  execute: Effect.gen(function* () {
    // 第一步：获取数据
    console.log('Step 1: Fetching data...')
    const data = yield* Effect.succeed({ userId: 123, name: 'Alice' })
    yield* Effect.sleep(500)
    
    // 第二步：处理数据（依赖第一步）
    console.log('Step 2: Processing data...')
    const processed = yield* Effect.succeed({ ...data, processed: true })
    yield* Effect.sleep(500)
    
    // 第三步：保存结果（依赖第二步）
    console.log('Step 3: Saving result...')
    yield* Effect.succeed({ ...processed, saved: true })
    yield* Effect.sleep(500)
    
    return 'Dependent workflow completed'
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* dependentWorkflow.execute
  return result
})

Effect.runSync(program)
```

## 💡 并行工作流

### 基本并行执行

```js
import { Effect } from 'effect'
import { ClusterWorkflow } from '@effect/cluster-workflow'

// 并行工作流
const parallelWorkflow = ClusterWorkflow.make('parallel-workflow', {
  execute: Effect.gen(function* () {
    // 并行执行多个任务
    const results = yield* Effect.all([
      Effect.gen(function* () {
        console.log('Task 1 started')
        yield* Effect.sleep(500)
        console.log('Task 1 completed')
        return 'Result 1'
      }),
      Effect.gen(function* () {
        console.log('Task 2 started')
        yield* Effect.sleep(700)
        console.log('Task 2 completed')
        return 'Result 2'
      }),
      Effect.gen(function* () {
        console.log('Task 3 started')
        yield* Effect.sleep(400)
        console.log('Task 3 completed')
        return 'Result 3'
      })
    ], { concurrency: 3 })
    
    console.log('All tasks completed:', results)
    return results
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* parallelWorkflow.execute
  return result
})

Effect.runSync(program)
```

### 竞争执行

```js

import { ClusterWorkflow } from '@effect/cluster-workflow'

// 竞争工作流
const raceWorkflow = ClusterWorkflow.make('race-workflow', {
  execute: Effect.gen(function* () {
    // 竞争执行，最快的完成
    const result = yield* Effect.race(
      Effect.gen(function* () {
        console.log('Task A started')
        yield* Effect.sleep(1000)
        console.log('Task A completed')
        return 'Task A won'
      }),
      Effect.gen(function* () {
        console.log('Task B started')
        yield* Effect.sleep(500)
        console.log('Task B completed')
        return 'Task B won'
      })
    )
    
    console.log('Race result:', result)
    return result
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* raceWorkflow.execute
  return result
})

Effect.runSync(program)
```

## 🚀 条件分支

### 基本 If-Else

```js
import { Effect } from 'effect'
import { ClusterWorkflow } from '@effect/cluster-workflow'

// 条件分支工作流
const conditionalWorkflow = ClusterWorkflow.make('conditional-workflow', {
  execute: Effect.gen(function* () {
    const condition = Math.random() > 0.5
    
    console.log('Condition:', condition)
    
    if (condition) {
      console.log('Branch A executed')
      yield* Effect.sleep(500)
      return 'Branch A result'
    } else {
      console.log('Branch B executed')
      yield* Effect.sleep(500)
      return 'Branch B result'
    }
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* conditionalWorkflow.execute
  return result
})

Effect.runSync(program)
```

### 多分支选择

```js
import { Effect } from 'effect'
import { ClusterWorkflow } from '@effect/cluster-workflow'

// 多分支工作流
const switchWorkflow = ClusterWorkflow.make('switch-workflow', {
  execute: Effect.gen(function* () {
    const value = Math.floor(Math.random() * 3)
    
    console.log('Value:', value)
    
    const result = yield* Effect.gen(function* () {
      switch (value) {
        case 0:
          console.log('Case 0: Initialize')
          yield* Effect.sleep(500)
          return 'Initialized'
        case 1:
          console.log('Case 1: Process')
          yield* Effect.sleep(500)
          return 'Processed'
        case 2:
          console.log('Case 2: Cleanup')
          yield* Effect.sleep(500)
          return 'Cleaned up'
        default:
          console.log('Default: Unknown')
          return 'Unknown'
      }
    })
    
    return result
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* switchWorkflow.execute
  return result
})

Effect.runSync(program)
```

### 递归工作流

```js
import { Effect } from 'effect'
import { ClusterWorkflow } from '@effect/cluster-workflow'

// 递归工作流
const recursiveWorkflow = ClusterWorkflow.make('recursive-workflow', {
  execute: Effect.gen(function* () {
    const recursive = (n: number): Effect.Effect<never, never, number> =>
      Effect.gen(function* () {
        if (n <= 0) {
          console.log('Base case reached')
          return 0
        }
        
        console.log(`Processing: ${n}`)
        yield* Effect.sleep(300)
        
        const result = yield* recursive(n - 1)
        return n + result
      })
    
    const result = yield* recursive(5)
    console.log('Recursive result:', result)
    
    return result
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* recursiveWorkflow.execute
  return result
})

Effect.runSync(program)
```

## 📝 实际应用示例

### 订单处理流程

```js
import { Effect } from 'effect'
import { ClusterWorkflow } from '@effect/cluster-workflow'

// 订单处理工作流
const orderProcessingWorkflow = ClusterWorkflow.make('order-processing', {
  execute: Effect.gen(function* () {
    // 步骤 1: 验证订单
    console.log('Step 1: Validating order...')
    yield* Effect.sleep(500)
    const isValid = Math.random() > 0.2
    
    if (!isValid) {
      console.log('Order validation failed')
      return 'Order rejected'
    }
    
    // 步骤 2: 处理支付
    console.log('Step 2: Processing payment...')
    yield* Effect.sleep(1000)
    const paymentSuccess = Math.random() > 0.1
    
    if (!paymentSuccess) {
      console.log('Payment failed')
      return 'Payment failed'
    }
    
    // 步骤 3: 并行处理库存和发货
    console.log('Step 3: Processing inventory and shipping...')
    yield* Effect.all([
      Effect.gen(function* () {
        console.log('  - Updating inventory...')
        yield* Effect.sleep(700)
        console.log('  - Inventory updated')
      }),
      Effect.gen(function* () {
        console.log('  - Creating shipment...')
        yield* Effect.sleep(600)
        console.log('  - Shipment created')
      })
    ], { concurrency: 2 })
    
    // 步骤 4: 发送通知
    console.log('Step 4: Sending notification...')
    yield* Effect.sleep(400)
    
    return 'Order processed successfully'
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* orderProcessingWorkflow.execute
  return result
})

Effect.runSync(program)
```

### 数据处理管道

```js
import { Effect } from 'effect'
import { ClusterWorkflow } from '@effect/cluster-workflow'

// 数据处理管道
const dataPipelineWorkflow = ClusterWorkflow.make('data-pipeline', {
  execute: Effect.gen(function* () {
    // 阶段 1: 数据提取
    console.log('Stage 1: Extracting data from sources...')
    const sources = ['source1', 'source2', 'source3']
    
    const extractedData = yield* Effect.all(
      sources.map(source =>
        Effect.gen(function* () {
          console.log(`  - Extracting from ${source}...`)
          yield* Effect.sleep(400)
          return { source, data: `data-from-${source}` }
        })
      ),
      { concurrency: 3 }
    )
    
    // 阶段 2: 数据转换
    console.log('Stage 2: Transforming data...')
    const transformedData = yield* Effect.all(
      extractedData.map(item =>
        Effect.gen(function* () {
          console.log(`  - Transforming ${item.source}...`)
          yield* Effect.sleep(300)
          return { ...item, transformed: true }
        })
      ),
      { concurrency: 3 }
    )
    
    // 阶段 3: 数据验证
    console.log('Stage 3: Validating data...')
    const validData = []
    
    for (const item of transformedData) {
      console.log(`  - Validating ${item.source}...`)
      yield* Effect.sleep(200)
      
      const isValid = Math.random() > 0.1
      if (isValid) {
        validData.push(item)
      } else {
        console.log(`  - ${item.source} validation failed`)
      }
    }
    
    // 阶段 4: 数据加载
    console.log('Stage 4: Loading data...')
    yield* Effect.all(
      validData.map(item =>
        Effect.gen(function* () {
          console.log(`  - Loading ${item.source}...`)
          yield* Effect.sleep(500)
          return { ...item, loaded: true }
        })
      ),
      { concurrency: 3 }
    )
    
    return `Pipeline completed with ${validData.length} valid items`
  })
})

// 运行工作流
const program = Effect.gen(function* () {
  const result = yield* dataPipelineWorkflow.execute
  return result
})

Effect.runSync(program)
```
