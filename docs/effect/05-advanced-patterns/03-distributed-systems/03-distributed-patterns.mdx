# 分布式模式

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了丰富的分布式模式，包括服务发现、负载均衡、分布式锁、分布式事务等。

## 🎯 服务发现

### 基本服务发现

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 服务发现
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'my-cluster',
    nodeId: 'node-1',
    address: 'localhost:3000'
  })
  
  // 注册服务
  yield* Cluster.registerService({
    name: 'user-service',
    address: 'localhost:4000',
    metadata: { version: '1.0.0' }
  })
  
  // 发现服务
  const services = yield* Cluster.discoverServices('user-service')
  console.log('Discovered services:', services)
  
  return services
})

Effect.runSync(program)
```

## 💡 负载均衡

### 基本负载均衡

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 负载均衡
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'my-cluster',
    nodeId: 'node-1',
    address: 'localhost:3000'
  })

    services: ['user-service', 'order-service']
  })

  const instance = yield* Cluster.selectServiceInstance('user-service')
  console.log('Selected instance:', instance)
  
  return instance
})

Effect.runSync(program)
```

## 🚀 分布式锁

### 基本分布式锁

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 分布式锁
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'my-cluster',
    nodeId: 'node-1',
    address: 'localhost:3000'
  })
  
  // 获取锁
  const lock = yield* Cluster.acquireLock('resource-1', {
    ttl: '30s'
  })
  
  console.log('Lock acquired')
  
  // 执行临界区代码
  yield* Effect.gen(function* () {
    console.log('Executing critical section')
    yield* Effect.sleep(1000)
  })
  
  // 释放锁
  yield* Cluster.releaseLock(lock)
  console.log('Lock released')
  
  return 'Lock operation completed'
})

Effect.runSync(program)
```

### 锁超时处理

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 锁超时处理
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'my-cluster',
    nodeId: 'node-1',
    address: 'localhost:3000'
  })
  
  // 尝试获取锁
  const lockResult = yield* Cluster.acquireLock('resource-1', {
    ttl: '10s',
    timeout: '5s'
  }).pipe(
    Effect.timeout('5s'),
    Effect.catchAll(() => Effect.succeed(null))
  )
  
  if (lockResult) {
    console.log('Lock acquired')
    
    // 执行临界区代码
    yield* Effect.gen(function* () {
      console.log('Executing critical section')
      yield* Effect.sleep(2000)
    })
    
    // 释放锁
    yield* Cluster.releaseLock(lockResult)
    console.log('Lock released')
  } else {
    console.log('Failed to acquire lock')
  }
  
  return 'Lock operation completed'
})

Effect.runSync(program)
```

## 🎯 分布式事务

### 基本分布式事务

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 分布式事务
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'my-cluster',
    nodeId: 'node-1',
    address: 'localhost:3000'
  })
  
  // 开始事务
  const transaction = yield* Cluster.beginTransaction({
    timeout: '30s'
  })
  
  console.log('Transaction started')
  
  // 执行事务操作
  yield* Effect.gen(function* () {
    // 操作 1
    console.log('Executing operation 1')
    yield* Effect.sleep(500)
    
    // 操作 2
    console.log('Executing operation 2')
    yield* Effect.sleep(500)
    
    // 操作 3
    console.log('Executing operation 3')
    yield* Effect.sleep(500)
  })
  
  // 提交事务
  yield* Cluster.commitTransaction(transaction)
  console.log('Transaction committed')
  
  return 'Transaction completed'
})

Effect.runSync(program)
```

### 事务回滚

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 事务回滚
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'my-cluster',
    nodeId: 'node-1',
    address: 'localhost:3000'
  })
  
  // 开始事务
  const transaction = yield* Cluster.beginTransaction()
  
  console.log('Transaction started')
  
  // 执行事务操作
  const result = yield* Effect.gen(function* () {
    // 操作 1
    console.log('Executing operation 1')
    yield* Effect.sleep(500)
    
    // 操作 2 - 可能失败
    console.log('Executing operation 2')
    if (Math.random() > 0.5) {
      throw new Error('Operation 2 failed')
    }
    
    // 操作 3
    console.log('Executing operation 3')
    yield* Effect.sleep(500)
    
    return 'Success'
  }).pipe(
    Effect.catchAll((error) => {
      console.error('Transaction failed, rolling back:', error.message)
      return Cluster.rollbackTransaction(transaction)
    })
  )
  
  if (result === 'Success') {
    yield* Cluster.commitTransaction(transaction)
    console.log('Transaction committed')
  }
  
  return 'Transaction completed'
})

Effect.runSync(program)
```

### 分布式任务队列

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 分布式任务队列
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'queue-cluster',
    nodeId: 'queue-1',
    address: 'localhost:3000'
  })
  
  // 添加任务
  for (let i = 0; i < 5; i++) {
    yield* Cluster.enqueueTask('task-queue', {
      id: i,
      data: `Task ${i}`
    })
    console.log(`Task ${i} enqueued`)
  }
  
  // 处理任务
  const processTasks = Effect.gen(function* () {
    while (true) {
      const task = yield* Cluster.dequeueTask('task-queue')
      console.log(`Processing task ${task.id}: ${task.data}`)
      yield* Effect.sleep(500)
    }
  })
  
  // 启动任务处理器
  yield* Effect.fork(processTasks)
  
  // 运行一段时间
  yield* Effect.sleep('5s')
  
  return 'Task queue completed'
})

Effect.runSync(program)
```
