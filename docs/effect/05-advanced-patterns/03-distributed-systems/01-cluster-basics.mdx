# 集群基础

> **版本**：v1.0  
> **最后更新**：2026-04-20

## 📚 概念介绍

Effect 提供了强大的分布式系统支持，通过 `@effect/cluster` 包提供集群管理、节点发现、节点间通信等功能。

## 🎯 @effect/cluster - 集群管理包

### 安装与设置

```js
// 安装依赖
// npm install @effect/cluster
// npm install @effect/cluster-node

// 基本导入
import { Effect } from 'effect'
import { Cluster, ClusterNode } from '@effect/cluster'
```

### 基本集群设置

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 创建集群
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'my-cluster',
    nodeId: 'node-1',
    address: 'localhost:3000'
  })
  
  console.log('Cluster created:', cluster.name)
  return cluster
})

Effect.runSync(program)
```

```js
Effect.runSync(program)
```

## 🎯 故障检测与恢复

### 心跳检测

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 心跳检测
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'my-cluster',
    nodeId: 'node-1',
    address: 'localhost:3000'
  })
  
  // 启动心跳检测

    interval: '5s',
    timeout: '15s'

  const status = yield* Cluster.getNodeStatus('node-2')
  console.log('Node status:', status)
  
  return status
})

Effect.runSync(program)
```

```js
Effect.runSync(program)
```

## 📝 实际应用示例

### 集群管理

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 集群管理
const program = Effect.gen(function* () {
  // 创建集群
  const cluster = yield* Cluster.make({
    name: 'production-cluster',
    nodeId: 'master-1',
    address: 'localhost:3000',
    role: 'master'
  })
  
  // 注册工作节点
  const workers = [
    { id: 'worker-1', address: 'localhost:3001' },
    { id: 'worker-2', address: 'localhost:3002' },
    { id: 'worker-3', address: 'localhost:3003' }
  ]
  
  for (const worker of workers) {
    yield* Cluster.registerNode({
      id: worker.id,
      address: worker.address,
      metadata: { role: 'worker' }
    })
    console.log(`Registered ${worker.id}`)
  }
  
  // 获取集群状态
  const status = yield* Cluster.getClusterStatus
  console.log('Cluster status:', status)
  
  // 获取所有节点
  const nodes = yield* Cluster.getNodes
  console.log('Total nodes:', nodes.length)
  
  return { cluster, nodes, status }
})

Effect.runSync(program)
```

### 节点监控

```js
import { Effect } from 'effect'
import { Cluster } from '@effect/cluster'

// 节点监控
const program = Effect.gen(function* () {
  const cluster = yield* Cluster.make({
    name: 'monitoring-cluster',
    nodeId: 'monitor-1',
    address: 'localhost:3000'
  })
  
  // 启动监控
  const monitor = Effect.gen(function* () {
    while (true) {
      const nodes = yield* Cluster.getNodes
      
      console.log('\n=== Node Status ===')
      for (const node of nodes) {
        const status = yield* Cluster.getNodeStatus(node.id)
        console.log(`${node.id}: ${status}`)
      }
      
      yield* Effect.sleep('10s')
    }
  })
  
  // 在后台运行监控
  yield* Effect.fork(monitor)
  
  // 模拟运行一段时间
  yield* Effect.sleep('30s')
  
  return 'Monitoring completed'
})

Effect.runSync(program)
```
