> **版本**：v1.0

## 📚 概念介绍

Effect 提供了强大的性能分析工具，帮助开发者测量性能、识别瓶颈、分析内存和 CPU 使用情况。

## 🎯 性能测量

### 基本性能测量

```js
import { Effect } from 'effect'

// 基本性能测量
const program = Effect.gen(function* () {
  const start = Date.now()
  
  // 执行操作
  yield* Effect.gen(function* () {
    yield* Effect.sleep(1000)
    console.log('Operation completed')
  })
  
  const end = Date.now()
  const duration = end - start
  console.log(`Operation took ${duration}ms`)
  
  return duration
})

Effect.runSync(program)
```

### Effect 性能测量

```js
import { Effect } from 'effect'

// 使用 Effect 的性能测量
const program = Effect.gen(function* () {
  const result = yield* Effect.gen(function* () {
    yield* Effect.sleep(500)
    return 'Operation result'
  }).pipe(
    Effect.timed
  )
  
  console.log(`Operation took ${result.duration}ms`)
  console.log(`Result: ${result.value}`)
  
  return result
})

Effect.runSync(program)
```

## 💡 瓶颈识别

### 操作跟踪

```js
import { Effect } from 'effect'

// 操作跟踪
const program = Effect.gen(function* () {
  const operation1 = Effect.gen(function* () {
    yield* Effect.sleep(300)
    console.log('Operation 1 completed')
    return 'Result 1'
  }).pipe(
    Effect.withSpan('operation-1')
  )
  
  const operation2 = Effect.gen(function* () {
    yield* Effect.sleep(500)
    console.log('Operation 2 completed')
    return 'Result 2'
  }).pipe(
    Effect.withSpan('operation-2')
  )
  
  const operation3 = Effect.gen(function* () {
    yield* Effect.sleep(200)
    console.log('Operation 3 completed')
    return 'Result 3'
  }).pipe(
    Effect.withSpan('operation-3')
  )
  
  const result = yield* Effect.all([operation1, operation2, operation3], {
    concurrency: 'unbounded'
  })
  
  console.log('All operations completed:', result)
  return result
})

Effect.runSync(program)
```

## 🚀 内存分析

### 内存使用测量

```js
import { Effect } from 'effect'

// 内存使用测量
const program = Effect.gen(function* () {
  const initialMemory = process.memoryUsage().heapUsed
  console.log(`Initial memory: ${(initialMemory / 1024 / 1024).toFixed(2)}MB`)
  
  // 执行内存密集型操作
  const data = yield* Effect.gen(function* () {
    const largeArray = Array.from({ length: 1000000 }, (_, i) => i)
    yield* Effect.sleep(100)
    return largeArray
  })
  
  const finalMemory = process.memoryUsage().heapUsed
  console.log(`Final memory: ${(finalMemory / 1024 / 1024).toFixed(2)}MB`)
  console.log(`Memory used: ${((finalMemory - initialMemory) / 1024 / 1024).toFixed(2)}MB`)
  
  return data.length
})

Effect.runSync(program)
```

### 内存泄漏检测

```js
import { Effect } from 'effect'

// 内存泄漏检测
const program = Effect.gen(function* () {
  const memorySnapshots = []
  
  // 多次执行操作并记录内存
  for (let i = 0; i < 10; i++) {
    yield* Effect.gen(function* () {
      const data = Array.from({ length: 100000 }, (_, j) => j)
      yield* Effect.sleep(50)
    })
    
    const memory = process.memoryUsage().heapUsed
    memorySnapshots.push(memory)
    console.log(`Iteration ${i + 1}: ${(memory / 1024 / 1024).toFixed(2)}MB`)
  }
  
  // 分析内存增长
  const growth = memorySnapshots[9] - memorySnapshots[0]
  console.log(`Memory growth: ${(growth / 1024 / 1024).toFixed(2)}MB`)
  
  if (growth > 10 * 1024 * 1024) {
    console.warn('Potential memory leak detected!')
  }
  
  return memorySnapshots
})

Effect.runSync(program)
```

## 🎯 CPU 分析

### CPU 使用测量

```js
import { Effect } from 'effect'

// CPU 使用测量
const program = Effect.gen(function* () {
  const startCpu = process.cpuUsage()
  
  // 执行 CPU 密集型操作
  const result = yield* Effect.gen(function* () {
    let sum = 0
    for (let i = 0; i < 10000000; i++) {
      sum += Math.sqrt(i)
    }
    return sum
  })
  
  const endCpu = process.cpuUsage(startCpu)
  console.log(`CPU user time: ${(endCpu.user / 1000).toFixed(2)}ms`)
  console.log(`CPU system time: ${(endCpu.system / 1000).toFixed(2)}ms`)
  console.log(`Total CPU time: ${((endCpu.user + endCpu.system) / 1000).toFixed(2)}ms`)
  
  return result
})

Effect.runSync(program)
```

### 性能监控

```js
import { Effect } from 'effect'

// 性能监控
const program = Effect.gen(function* () {
  const monitor = Effect.gen(function* () {
    const startTime = Date.now()
    const startCpu = process.cpuUsage()
    const startMemory = process.memoryUsage().heapUsed
    
    // 执行操作
    const result = yield* Effect.gen(function* () {
      yield* Effect.sleep(1000)
      return 'Operation completed'
    })
    
    const endTime = Date.now()
    const endCpu = process.cpuUsage(startCpu)
    const endMemory = process.memoryUsage().heapUsed
    
    const metrics = {
      duration: endTime - startTime,
      cpuUser: endCpu.user / 1000,
      cpuSystem: endCpu.system / 1000,
      memoryUsed: (endMemory - startMemory) / 1024 / 1024
    }
    
    console.log('Performance metrics:', metrics)
    return { result, metrics }
  })
  
  const { result, metrics } = yield* monitor
  return { result, metrics }
})

Effect.runSync(program)
```

## 📝 实际应用示例

### 完整的性能分析

```js
import { Effect } from 'effect'

// 完整的性能分析
const program = Effect.gen(function* () {
  const analyzePerformance = Effect.gen(function* () {
    // 初始状态
    const startMemory = process.memoryUsage().heapUsed
    const startTime = Date.now()
    const startCpu = process.cpuUsage()
    
    console.log('=== Starting Performance Analysis ===')
    console.log(`Initial memory: ${(startMemory / 1024 / 1024).toFixed(2)}MB`)
    
    // 执行操作
    const result = yield* Effect.gen(function* () {
      // 模拟数据处理
      const data = yield* Effect.gen(function* () {
        const largeArray = Array.from({ length: 500000 }, (_, i) => ({
          id: i,
          value: Math.random()
        }))
        yield* Effect.sleep(500)
        return largeArray
      }).pipe(
        Effect.withSpan('data-generation')
      )
      
      // 模拟计算
      const processed = yield* Effect.gen(function* () {
        const sum = data.reduce((acc, item) => acc + item.value, 0)
        const avg = sum / data.length
        yield* Effect.sleep(300)
        return { sum, avg, count: data.length }
      }).pipe(
        Effect.withSpan('data-processing')
      )
      
      // 模拟输出
      yield* Effect.gen(function* () {
        console.log(`Processed ${processed.count} items`)
        console.log(`Average: ${processed.avg.toFixed(4)}`)
        yield* Effect.sleep(100)
      }).pipe(
        Effect.withSpan('data-output')
      )
      
      return processed
    })
    
    // 最终状态
    const endMemory = process.memoryUsage().heapUsed
    const endTime = Date.now()
    const endCpu = process.cpuUsage(startCpu)
    
    // 性能指标
    const metrics = {
      duration: endTime - startTime,
      cpuUser: endCpu.user / 1000,
      cpuSystem: endCpu.system / 1000,
      memoryUsed: (endMemory - startMemory) / 1024 / 1024,
      peakMemory: process.memoryUsage().heapUsed / 1024 / 1024
    }
    
    console.log('\n=== Performance Results ===')
    console.log(`Duration: ${metrics.duration}ms`)
    console.log(`CPU (user): ${metrics.cpuUser.toFixed(2)}ms`)
    console.log(`CPU (system): ${metrics.cpuSystem.toFixed(2)}ms`)
    console.log(`Memory used: ${metrics.memoryUsed.toFixed(2)}MB`)
    console.log(`Peak memory: ${metrics.peakMemory.toFixed(2)}MB`)
    
    return { result, metrics }
  })
  
  const { result, metrics } = yield* analyzePerformance
  return { result, metrics }
})

Effect.runSync(program)
```

### 性能基准测试

```js
import { Effect } from 'effect'

// 性能基准测试
const program = Effect.gen(function* () {
  const benchmark = (name, fn) => Effect.gen(function* () {
    console.log(`\n=== Benchmark: ${name} ===`)
    
    const iterations = 100
    const results = []
    
    for (let i = 0; i < iterations; i++) {
      const start = Date.now()
      yield* fn
      const end = Date.now()
      results.push(end - start)
    }
    
    const avg = results.reduce((a, b) => a + b, 0) / results.length
    const min = Math.min(...results)
    const max = Math.max(...results)
    const median = results.sort((a, b) => a - b)[Math.floor(results.length / 2)]
    
    console.log(`Average: ${avg.toFixed(2)}ms`)
    console.log(`Min: ${min}ms`)
    console.log(`Max: ${max}ms`)
    console.log(`Median: ${median}ms`)
    
    return { avg, min, max, median }
  })
  
  // 运行基准测试
  const result1 = yield* benchmark('Effect.sleep', Effect.sleep(100))
  const result2 = yield* benchmark('Array operations', Effect.gen(function* () {
    const arr = Array.from({ length: 1000 }, (_, i) => i)
    arr.map(x => x * 2).filter(x => x % 4 === 0)
  }))
  const result3 = yield* benchmark('Math operations', Effect.gen(function* () {
    let sum = 0
    for (let i = 0; i < 10000; i++) {
      sum += Math.sqrt(i)
    }
  }))
  
  return { result1, result2, result3 }
})

Effect.runSync(program)
```
