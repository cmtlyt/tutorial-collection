# 基础类型系统

> **版本**: v1.0  
> **最后更新**: 2026-04-22

## 📚 概念介绍

TypeScript 的核心价值在于其类型系统。通过类型，编译器能够在代码运行之前就捕获潜在的错误。本节将介绍 TypeScript 提供的所有基础类型，帮助你理解每种类型的使用场景和区别。

## 🎯 原始类型

TypeScript 中的原始类型与 JavaScript 的原始值一一对应：

### string、number、boolean

这三种是最常用的基础类型：

```ts
const username: string = "Alice";
const age: number = 28;
const isActive: boolean = true;

// number 类型支持各种数字表示
const decimal: number = 42;
const hex: number = 0xff;
const binary: number = 0b1010;
const octal: number = 0o744;
const float: number = 3.14;
const negative: number = -100;
const infinity: number = Infinity;
const notANumber: number = NaN;
```

### symbol 与 bigint

```ts
// symbol：创建全局唯一的标识符
const uniqueId: symbol = Symbol("id");
const anotherId: symbol = Symbol("id");
console.log(uniqueId === anotherId); // false

// unique symbol：可以作为常量使用的特殊 symbol 类型
const FILE_KEY: unique symbol = Symbol("file_key");

// bigint：表示任意精度的整数（ES2020+）
const largeNumber: bigint = 9007199254740991n;
const anotherBig: bigint = BigInt("123456789012345678901234567890");
console.log(largeNumber + 1n); // 9007199254740992n
```

:::caution
`bigint` 不能与 `number` 混合运算，需要显式转换。
:::

## 📦 数组与元组

### 数组类型

数组有两种类型声明方式：

```ts
// 方式一：类型 + 方括号
const numbers: number[] = [1, 2, 3, 4, 5];
const names: string[] = ["Alice", "Bob", "Charlie"];

// 方式二：泛型语法
const scores: Array<number> = [90, 85, 92];

// 只读数组：禁止修改数组内容
const frozenList: readonly number[] = [1, 2, 3];
// frozenList.push(4);  // 编译错误：readonly 数组没有 push 方法
```

### 元组（Tuple）

元组是固定长度、每个位置有确定类型的数组：

```ts
// 基本元组：[名字, 年龄]
const person: [string, number] = ["Alice", 28];
const name = person[0]; // string 类型
const personAge = person[1]; // number 类型

// 带可选元素的元组
const coordinate: [number, number, number?] = [10, 20];

// 带标签的元组（提高可读性）
const userInfo: [name: string, age: number, email: string] = [
  "Bob",
  25,
  "bob@example.com",
];

// 只读元组
const point: readonly [number, number] = [10, 20];
// point[0] = 30;  // 编译错误

// 元组在函数返回值中的应用
function useState<T>(initial: T): [T, (value: T) => void] {
  let state = initial;
  const setter = (value: T) => {
    state = value;
  };
  return [state, setter];
}

const [count, setCount] = useState(0);
console.log(count); // 0
```

## 🏷️ 枚举（Enum）

### 数字枚举

```ts
// 默认从 0 开始递增
enum Direction {
  Up,    // 0
  Down,  // 1
  Left,  // 2
  Right, // 3
}

const move: Direction = Direction.Up;
console.log(move); // 0

// 自定义起始值
enum HttpStatus {
  Ok = 200,
  Created = 201,
  BadRequest = 400,
  NotFound = 404,
  InternalServerError = 500,
}

function handleResponse(status: HttpStatus): string {
  switch (status) {
    case HttpStatus.Ok:
      return "Success";
    case HttpStatus.NotFound:
      return "Resource not found";
    case HttpStatus.InternalServerError:
      return "Server error";
    default:
      return "Unknown status";
  }
}

console.log(handleResponse(HttpStatus.Ok)); // "Success"
```

### 字符串枚举

```ts
enum LogLevel {
  Debug = "DEBUG",
  Info = "INFO",
  Warn = "WARN",
  Error = "ERROR",
}

function log(level: LogLevel, message: string): void {
  console.log(`[${level}] ${message}`);
}

log(LogLevel.Info, "Application started");
// 输出: [INFO] Application started
```

### 常量枚举

常量枚举在编译时会被完全内联，不会生成额外的 JavaScript 代码：

```ts
const enum Color {
  Red = "RED",
  Green = "GREEN",
  Blue = "BLUE",
}

const selectedColor = Color.Red;
// 编译后直接变成: const selectedColor = "RED"
```

:::tip
在大多数现代项目中，推荐使用 `as const` 对象或联合类型来替代枚举，因为它们更轻量且更符合 JavaScript 的习惯：

```ts
// 推荐：使用 as const 替代枚举
const Direction = {
  Up: "UP",
  Down: "DOWN",
  Left: "LEFT",
  Right: "RIGHT",
} as const;

type Direction = (typeof Direction)[keyof typeof Direction];
// "UP" | "DOWN" | "LEFT" | "RIGHT"
```
:::

## 🔍 any、unknown、never、void

这四种特殊类型在 TypeScript 中各有不同的用途，理解它们的区别非常重要。

### any：任意类型（逃生舱）

`any` 会关闭类型检查，应尽量避免使用：

```ts
let anything: any = 42;
anything = "hello";
anything = true;
anything = { name: "test" };

// any 可以赋值给任何类型，也可以接受任何类型
// 这意味着 TypeScript 不会进行任何检查
anything.nonExistentMethod(); // 不报错，但运行时会崩溃
```

### unknown：安全的任意类型

`unknown` 是 `any` 的类型安全替代品。它可以接受任何值，但使用前必须进行类型检查：

```ts
let value: unknown = "hello";
value = 42;
value = { name: "Alice" };

// 不能直接使用 unknown 类型的值
// value.toFixed(2);  // 编译错误

// 必须先进行类型检查（类型收窄）
if (typeof value === "number") {
  console.log(value.toFixed(2)); // 正常工作
}

if (typeof value === "string") {
  console.log(value.toUpperCase()); // 正常工作
}
```

### never：永远不会发生的类型

`never` 表示永远不可能出现的值：

```ts
// 永远不会正常返回的函数
function throwError(message: string): never {
  throw new Error(message);
}

// 无限循环
function infiniteLoop(): never {
  while (true) {
    // 永远不会结束
  }
}

// never 在穷尽性检查中非常有用
type Shape = "circle" | "square" | "triangle";

function getArea(shape: Shape): number {
  switch (shape) {
    case "circle":
      return Math.PI * 10 * 10;
    case "square":
      return 10 * 10;
    case "triangle":
      return (10 * 8) / 2;
    default:
      // 如果所有 case 都处理了，shape 类型就是 never
      const exhaustiveCheck: never = shape;
      return exhaustiveCheck;
  }
}
```

### void：没有返回值

`void` 表示函数没有返回值：

```ts
function logMessage(message: string): void {
  console.log(message);
  // 没有 return 语句，或者 return 后面不跟值
}

// void 和 undefined 的区别
let undefinedValue: undefined = undefined;
let voidValue: void = undefined; // void 可以接受 undefined

// 回调函数声明为 void 返回值时，实际返回值会被忽略
const numbers = [1, 2, 3];
// forEach 的回调类型是 (value: number) => void
// 即使 push 返回了一个 number，也不影响
const result: number[] = [];
numbers.forEach((num) => result.push(num));
```

### 四种类型的对比

```
类型       可赋任意值   使用前需检查   适用场景
───────────────────────────────────────────────
any        ✅          ❌            临时迁移、第三方库无类型
unknown    ✅          ✅            接收不确定类型的值
never      ❌          -             不可能到达的代码、穷尽检查
void       ❌          -             无返回值的函数
```

## ✋ 类型断言与非空断言

### 类型断言（Type Assertion）

当你比编译器更了解一个值的类型时，可以使用类型断言：

```ts
// as 语法（推荐）
const inputElement = document.getElementById("myInput") as HTMLInputElement;
inputElement.value = "Hello";

// 尖括号语法（在 JSX 中不可用）
const canvas = <HTMLCanvasElement>document.getElementById("myCanvas");

// 断言为更具体的类型
const data: unknown = JSON.parse('{"name": "Alice", "age": 28}');
const user = data as { name: string; age: number };
console.log(user.name); // "Alice"
```

:::caution
类型断言不会进行运行时检查，如果断言错误，可能导致运行时错误。仅在你确信类型正确时使用。
:::

### 非空断言（Non-null Assertion）

使用 `!` 后缀告诉编译器某个值不是 `null` 或 `undefined`：

```ts
function getElement(id: string): HTMLElement {
  const element = document.getElementById(id);
  // element 的类型是 HTMLElement | null
  // 使用 ! 断言它不是 null
  return element!;
}

// 更安全的替代方案：显式检查
function getElementSafe(id: string): HTMLElement {
  const element = document.getElementById(id);
  if (element === null) {
    throw new Error(`Element with id "${id}" not found`);
  }
  return element; // 这里 TypeScript 自动收窄为 HTMLElement
}
```

## 📝 实际应用示例

### 用户注册表单验证

```ts
enum UserRole {
  Admin = "ADMIN",
  Editor = "EDITOR",
  Viewer = "VIEWER",
}

interface RegistrationForm {
  username: string;
  email: string;
  age: number;
  role: UserRole;
  acceptedTerms: boolean;
}

type ValidationResult = [isValid: boolean, errors: string[]];

function validateRegistration(form: RegistrationForm): ValidationResult {
  const errors: string[] = [];

  if (form.username.length < 3) {
    errors.push("Username must be at least 3 characters");
  }

  if (!form.email.includes("@")) {
    errors.push("Invalid email format");
  }

  if (form.age < 0 || form.age > 150) {
    errors.push("Age must be between 0 and 150");
  }

  if (!form.acceptedTerms) {
    errors.push("You must accept the terms");
  }

  return [errors.length === 0, errors];
}

const form: RegistrationForm = {
  username: "Al",
  email: "invalid-email",
  age: 28,
  role: UserRole.Viewer,
  acceptedTerms: false,
};

const [isValid, errors] = validateRegistration(form);
console.log(isValid); // false
console.log(errors);
// ["Username must be at least 3 characters", "Invalid email format", "You must accept the terms"]
```
