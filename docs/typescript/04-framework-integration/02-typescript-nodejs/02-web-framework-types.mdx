# Web 框架类型

> **版本**: v1.0  
> **最后更新**: 2026-04-22

## 📚 概念介绍

Node.js 生态中有多个主流 Web 框架，它们对 TypeScript 的支持程度各不相同。本节将介绍四个代表性框架的类型用法：**Express**（最广泛使用）、**Koa**（轻量中间件架构）、**Fastify**（高性能、Schema 驱动）以及 **NestJS**（装饰器 + TypeScript 深度集成）。

## 🎯 Express 类型

### 安装与配置

```bash
npm install express
npm install -D @types/express typescript @types/node
```

### Request、Response 与 Middleware 类型

```ts
import express, { Request, Response, NextFunction } from 'express';

const app = express();
app.use(express.json());

// 基本路由 — Request 和 Response 的类型
app.get('/api/hello', (req: Request, res: Response) => {
  const name = req.query.name as string | undefined;
  res.json({ message: `Hello, ${name ?? 'World'}!` });
});

// 中间件类型
function loggerMiddleware(req: Request, res: Response, next: NextFunction): void {
  console.log(`${req.method} ${req.url} - ${new Date().toISOString()}`);
  next();
}

app.use(loggerMiddleware);
```

### 泛型 Request 参数

Express 的 `Request` 类型支持泛型参数来定义路由参数、响应体、请求体和查询参数的类型：

```ts
import { Request, Response } from 'express';

// Request<Params, ResBody, ReqBody, Query>
interface UserParams {
  id: string;
}

interface CreateUserBody {
  name: string;
  email: string;
  age?: number;
}

interface UserResponse {
  id: string;
  name: string;
  email: string;
}

// 路由参数类型
app.get('/api/users/:id', (req: Request<UserParams>, res: Response<UserResponse>) => {
  const userId: string = req.params.id;
  res.json({ id: userId, name: 'Alice', email: 'alice@example.com' });
});

// 请求体类型
app.post(
  '/api/users',
  (req: Request<{}, UserResponse, CreateUserBody>, res: Response<UserResponse>) => {
    const { name, email } = req.body;
    const newUser: UserResponse = { id: '1', name, email };
    res.status(201).json(newUser);
  }
);

// 查询参数类型
interface UserQuery {
  page?: string;
  limit?: string;
  sort?: 'name' | 'email' | 'createdAt';
}

app.get(
  '/api/users',
  (req: Request<{}, UserResponse[], {}, UserQuery>, res: Response<UserResponse[]>) => {
    const page = parseInt(req.query.page ?? '1', 10);
    const limit = parseInt(req.query.limit ?? '10', 10);
    console.log(`Page: ${page}, Limit: ${limit}, Sort: ${req.query.sort}`);
    res.json([]);
  }
);
```

### 错误处理中间件

```ts
import { Request, Response, NextFunction } from 'express';

// 自定义错误类
class AppError extends Error {
  constructor(
    public statusCode: number,
    message: string
  ) {
    super(message);
    this.name = 'AppError';
  }
}

// 错误处理中间件（必须有 4 个参数）
function errorHandler(
  err: Error,
  req: Request,
  res: Response,
  next: NextFunction
): void {
  if (err instanceof AppError) {
    res.status(err.statusCode).json({
      error: err.message,
      statusCode: err.statusCode,
    });
    return;
  }

  console.error('Unhandled error:', err);
  res.status(500).json({ error: 'Internal Server Error' });
}

app.use(errorHandler);
```

### 扩展 Request 类型

```ts
// 通过声明合并扩展 Request，添加自定义属性
declare global {
  namespace Express {
    interface Request {
      userId?: string;
      role?: 'admin' | 'user';
    }
  }
}

// 认证中间件 — 向 req 注入自定义属性
function authMiddleware(req: Request, res: Response, next: NextFunction): void {
  const token = req.headers.authorization?.replace('Bearer ', '');

  if (!token) {
    res.status(401).json({ error: 'Unauthorized' });
    return;
  }

  // 模拟 token 解析
  req.userId = 'user-123';
  req.role = 'admin';
  next();
}

// 后续中间件可以安全地访问自定义属性
app.get('/api/profile', authMiddleware, (req: Request, res: Response) => {
  res.json({ userId: req.userId, role: req.role });
});

app.listen(3000, () => {
  console.log('Express server running on port 3000');
});
```

## 🎯 Koa 类型

### 安装与配置

```bash
npm install koa @koa/router koa-bodyparser
npm install -D @types/koa @types/koa__router @types/koa-bodyparser typescript
```

### 基本使用

```ts
import Koa, { Context, Next } from 'koa';
import Router from '@koa/router';
import bodyParser from 'koa-bodyparser';

const app = new Koa();
const router = new Router();

app.use(bodyParser());

// 中间件 — Koa 使用 Context 对象
async function loggerMiddleware(ctx: Context, next: Next): Promise<void> {
  const start = Date.now();
  await next();
  const duration = Date.now() - start;
  console.log(`${ctx.method} ${ctx.url} - ${duration}ms`);
}

app.use(loggerMiddleware);

// 路由
router.get('/api/hello', (ctx: Context) => {
  const name = ctx.query.name as string | undefined;
  ctx.body = { message: `Hello, ${name ?? 'World'}!` };
});

// 带请求体的路由
interface CreatePostBody {
  title: string;
  content: string;
}

router.post('/api/posts', (ctx: Context) => {
  const body = ctx.request.body as CreatePostBody;
  ctx.status = 201;
  ctx.body = { id: 1, title: body.title, content: body.content };
});

app.use(router.routes());
app.use(router.allowedMethods());

app.listen(3000, () => {
  console.log('Koa server running on port 3000');
});
```

### 自定义 Context State

```ts
import Koa from 'koa';

// 定义 State 类型
interface AppState {
  userId: string;
  role: 'admin' | 'user';
}

// 创建带类型的 Koa 实例
const app = new Koa<AppState>();

// 中间件可以访问类型化的 state
app.use(async (ctx, next) => {
  ctx.state.userId = 'user-123';
  ctx.state.role = 'admin';
  await next();
});

app.use(async (ctx) => {
  // ctx.state 拥有完整类型
  ctx.body = {
    userId: ctx.state.userId,
    role: ctx.state.role,
  };
});
```

## 🎯 Fastify 类型

Fastify 对 TypeScript 提供了一流的原生支持，无需额外安装 `@types` 包。它通过 JSON Schema 驱动来实现请求/响应的类型推导。

### 安装与配置

```bash
npm install fastify
npm install -D typescript @types/node
```

### 基本使用

```ts
import Fastify, { FastifyRequest, FastifyReply } from 'fastify';

const server = Fastify({ logger: true });

// 路由 — 通过泛型定义请求参数类型
server.get<{
  Querystring: { name?: string };
}>('/api/hello', async (request, reply) => {
  const name = request.query.name ?? 'World';
  return { message: `Hello, ${name}!` };
});

// 带路由参数和请求体的路由
interface CreateUserBody {
  name: string;
  email: string;
}

interface UserParams {
  id: string;
}

server.post<{
  Body: CreateUserBody;
}>('/api/users', async (request, reply) => {
  const { name, email } = request.body;
  reply.status(201);
  return { id: '1', name, email };
});

server.get<{
  Params: UserParams;
}>('/api/users/:id', async (request) => {
  const { id } = request.params;
  return { id, name: 'Alice', email: 'alice@example.com' };
});
```

### Schema 驱动类型推断

Fastify 的核心优势是通过 JSON Schema 同时实现运行时验证和类型推导：

```ts
import Fastify from 'fastify';

const server = Fastify();

// 定义 JSON Schema
const createUserSchema = {
  body: {
    type: 'object',
    required: ['name', 'email'],
    properties: {
      name: { type: 'string', minLength: 1 },
      email: { type: 'string', format: 'email' },
      age: { type: 'number', minimum: 0 },
    },
  },
  response: {
    201: {
      type: 'object',
      properties: {
        id: { type: 'string' },
        name: { type: 'string' },
        email: { type: 'string' },
      },
    },
  },
} as const;

// 配合 TypeBox 或手动定义接口来获得完整类型
interface CreateUserRoute {
  Body: {
    name: string;
    email: string;
    age?: number;
  };
  Reply: {
    id: string;
    name: string;
    email: string;
  };
}

server.post<CreateUserRoute>(
  '/api/users',
  { schema: createUserSchema },
  async (request, reply) => {
    const { name, email } = request.body;
    reply.status(201);
    return { id: crypto.randomUUID(), name, email };
  }
);

server.listen({ port: 3000 }, (err) => {
  if (err) {
    server.log.error(err);
    process.exit(1);
  }
});
```

### Fastify 插件类型

```ts
import Fastify, { FastifyInstance, FastifyPluginAsync } from 'fastify';
import fp from 'fastify-plugin';

// 声明插件添加的装饰器
declare module 'fastify' {
  interface FastifyInstance {
    config: {
      dbUrl: string;
      port: number;
    };
  }
}

// 定义插件
const configPlugin: FastifyPluginAsync = async (fastify: FastifyInstance) => {
  fastify.decorate('config', {
    dbUrl: process.env.DATABASE_URL ?? 'postgresql://localhost:5432/mydb',
    port: parseInt(process.env.PORT ?? '3000', 10),
  });
};

// 使用 fp 包装确保装饰器在外层可用
const configPluginWrapped = fp(configPlugin);

const server = Fastify();

server.register(configPluginWrapped);

server.get('/api/config', async (request, reply) => {
  // fastify.config 拥有完整类型
  return { port: server.config.port };
});
```

## 🎯 NestJS

NestJS 是 TypeScript 原生框架，深度使用装饰器和依赖注入，无需额外的类型定义。

### 安装与配置

```bash
# 使用 NestJS CLI 创建项目
npm install -g @nestjs/cli
nest new my-project
```

### Controller 类型

```ts
import {
  Controller,
  Get,
  Post,
  Put,
  Delete,
  Param,
  Body,
  Query,
  HttpCode,
  HttpStatus,
  ParseIntPipe,
} from '@nestjs/common';

// DTO（Data Transfer Object）定义
class CreateUserDto {
  name!: string;
  email!: string;
  age?: number;
}

class UpdateUserDto {
  name?: string;
  email?: string;
  age?: number;
}

class UserResponseDto {
  id!: number;
  name!: string;
  email!: string;
}

class PaginationQueryDto {
  page?: number;
  limit?: number;
}

@Controller('users')
class UsersController {
  @Get()
  findAll(@Query() query: PaginationQueryDto): UserResponseDto[] {
    console.log(`Page: ${query.page}, Limit: ${query.limit}`);
    return [{ id: 1, name: 'Alice', email: 'alice@example.com' }];
  }

  @Get(':id')
  findOne(@Param('id', ParseIntPipe) id: number): UserResponseDto {
    return { id, name: 'Alice', email: 'alice@example.com' };
  }

  @Post()
  @HttpCode(HttpStatus.CREATED)
  create(@Body() createUserDto: CreateUserDto): UserResponseDto {
    return { id: 1, name: createUserDto.name, email: createUserDto.email };
  }

  @Put(':id')
  update(
    @Param('id', ParseIntPipe) id: number,
    @Body() updateUserDto: UpdateUserDto
  ): UserResponseDto {
    return { id, name: updateUserDto.name ?? 'Alice', email: updateUserDto.email ?? '' };
  }

  @Delete(':id')
  @HttpCode(HttpStatus.NO_CONTENT)
  remove(@Param('id', ParseIntPipe) id: number): void {
    console.log(`Deleted user ${id}`);
  }
}
```

### Service 与依赖注入

```ts
import { Injectable, NotFoundException } from '@nestjs/common';

interface User {
  id: number;
  name: string;
  email: string;
}

@Injectable()
class UsersService {
  private users: User[] = [
    { id: 1, name: 'Alice', email: 'alice@example.com' },
    { id: 2, name: 'Bob', email: 'bob@example.com' },
  ];

  findAll(): User[] {
    return this.users;
  }

  findOne(id: number): User {
    const user = this.users.find((u) => u.id === id);
    if (!user) {
      throw new NotFoundException(`User with id ${id} not found`);
    }
    return user;
  }

  create(data: { name: string; email: string }): User {
    const newUser: User = {
      id: this.users.length + 1,
      name: data.name,
      email: data.email,
    };
    this.users.push(newUser);
    return newUser;
  }

  update(id: number, data: Partial<Omit<User, 'id'>>): User {
    const user = this.findOne(id);
    Object.assign(user, data);
    return user;
  }

  remove(id: number): void {
    const index = this.users.findIndex((u) => u.id === id);
    if (index === -1) {
      throw new NotFoundException(`User with id ${id} not found`);
    }
    this.users.splice(index, 1);
  }
}
```

### Guard 与 Interceptor 类型

```ts
import {
  Injectable,
  CanActivate,
  ExecutionContext,
  NestInterceptor,
  CallHandler,
} from '@nestjs/common';
import { Observable, map } from 'rxjs';

// Guard — 控制路由访问权限
@Injectable()
class AuthGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const token = request.headers.authorization?.replace('Bearer ', '');
    return token !== undefined && token.length > 0;
  }
}

// Interceptor — 统一包装响应格式
interface ApiResponse<T> {
  data: T;
  timestamp: string;
  statusCode: number;
}

@Injectable()
class TransformInterceptor<T> implements NestInterceptor<T, ApiResponse<T>> {
  intercept(
    context: ExecutionContext,
    next: CallHandler<T>
  ): Observable<ApiResponse<T>> {
    const statusCode = context.switchToHttp().getResponse().statusCode;

    return next.handle().pipe(
      map((data) => ({
        data,
        timestamp: new Date().toISOString(),
        statusCode,
      }))
    );
  }
}
```

## 📝 实际应用示例

### 综合示例：Express 类型安全的 REST API

```ts
import express, { Request, Response, NextFunction, Router } from 'express';

// 通用类型定义
interface ApiResponse<T> {
  success: boolean;
  data: T;
  message?: string;
}

interface PaginatedResponse<T> extends ApiResponse<T[]> {
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
  };
}

// 实体类型
interface Task {
  id: string;
  title: string;
  description: string;
  status: 'pending' | 'in_progress' | 'done';
  priority: 'low' | 'medium' | 'high';
  createdAt: string;
}

type CreateTaskInput = Pick<Task, 'title' | 'description' | 'priority'>;
type UpdateTaskInput = Partial<Pick<Task, 'title' | 'description' | 'status' | 'priority'>>;

// 类型安全的路由工厂
function createTaskRouter(): Router {
  const router = Router();
  const tasks: Task[] = [];
  let nextId = 1;

  // GET /tasks?page=1&limit=10&status=pending
  router.get(
    '/',
    (
      req: Request<{}, PaginatedResponse<Task>, {}, { page?: string; limit?: string; status?: Task['status'] }>,
      res: Response<PaginatedResponse<Task>>
    ) => {
      const page = parseInt(req.query.page ?? '1', 10);
      const limit = parseInt(req.query.limit ?? '10', 10);
      const statusFilter = req.query.status;

      let filtered = tasks;
      if (statusFilter) {
        filtered = tasks.filter((t) => t.status === statusFilter);
      }

      const total = filtered.length;
      const paginated = filtered.slice((page - 1) * limit, page * limit);

      res.json({
        success: true,
        data: paginated,
        pagination: {
          page,
          limit,
          total,
          totalPages: Math.ceil(total / limit),
        },
      });
    }
  );

  // POST /tasks
  router.post(
    '/',
    (
      req: Request<{}, ApiResponse<Task>, CreateTaskInput>,
      res: Response<ApiResponse<Task>>
    ) => {
      const { title, description, priority } = req.body;

      const task: Task = {
        id: String(nextId++),
        title,
        description,
        status: 'pending',
        priority,
        createdAt: new Date().toISOString(),
      };

      tasks.push(task);
      res.status(201).json({ success: true, data: task });
    }
  );

  // PUT /tasks/:id
  router.put(
    '/:id',
    (
      req: Request<{ id: string }, ApiResponse<Task>, UpdateTaskInput>,
      res: Response<ApiResponse<Task> | ApiResponse<null>>
    ) => {
      const task = tasks.find((t) => t.id === req.params.id);

      if (!task) {
        res.status(404).json({ success: false, data: null, message: 'Task not found' });
        return;
      }

      if (req.body.title !== undefined) task.title = req.body.title;
      if (req.body.description !== undefined) task.description = req.body.description;
      if (req.body.status !== undefined) task.status = req.body.status;
      if (req.body.priority !== undefined) task.priority = req.body.priority;

      res.json({ success: true, data: task });
    }
  );

  return router;
}

// 组装应用
const app = express();
app.use(express.json());
app.use('/api/tasks', createTaskRouter());

app.listen(3000, () => {
  console.log('Server running on port 3000');
});
```
