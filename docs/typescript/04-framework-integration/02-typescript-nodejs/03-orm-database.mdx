# ORM 与数据库

> **版本**: v1.0  
> **最后更新**: 2026-04-22

## 📚 概念介绍

在 Node.js + TypeScript 项目中，选择类型安全的 ORM 能够极大提升数据库操作的可靠性。本节将介绍四种主流方案：**Prisma**（Schema 驱动、自动生成类型）、**Drizzle ORM**（SQL-like API、轻量级）、**TypeORM**（装饰器驱动、Active Record / Data Mapper 模式），以及如何构建**类型安全的原生 SQL 查询**。

## 🎯 Prisma

Prisma 通过 Schema 文件定义数据模型，自动生成完全类型化的客户端代码，是目前 TypeScript 生态中类型安全做得最好的 ORM 之一。

### 安装与初始化

```bash
npm install prisma @prisma/client
npx prisma init
```

### Schema 定义

```prisma
// prisma/schema.prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id        Int      @id @default(autoincrement())
  email     String   @unique
  name      String
  role      Role     @default(USER)
  posts     Post[]
  profile   Profile?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

model Post {
  id        Int      @id @default(autoincrement())
  title     String
  content   String?
  published Boolean  @default(false)
  author    User     @relation(fields: [authorId], references: [id])
  authorId  Int
  tags      Tag[]
  createdAt DateTime @default(now())
}

model Profile {
  id     Int    @id @default(autoincrement())
  bio    String
  avatar String?
  user   User   @relation(fields: [userId], references: [id])
  userId Int    @unique
}

model Tag {
  id    Int    @id @default(autoincrement())
  name  String @unique
  posts Post[]
}

enum Role {
  USER
  ADMIN
  MODERATOR
}
```

运行 `npx prisma generate` 后，Prisma 会自动生成完全类型化的客户端。

### CRUD 操作

```ts
import { PrismaClient, Role } from '@prisma/client';

const prisma = new PrismaClient();

// 创建 — 所有字段都有类型提示
async function createUser() {
  const user = await prisma.user.create({
    data: {
      email: 'alice@example.com',
      name: 'Alice',
      role: Role.ADMIN,
      profile: {
        create: {
          bio: 'Full-stack developer',
        },
      },
    },
    // include 决定返回值中包含哪些关联
    include: {
      profile: true,
    },
  });

  // user 的类型自动包含 profile
  console.log(user.name);           // string
  console.log(user.profile?.bio);   // string | undefined
  return user;
}

// 查询 — select 精确控制返回字段
async function getUsers() {
  const users = await prisma.user.findMany({
    where: {
      role: Role.USER,
      email: { contains: '@example.com' },
    },
    select: {
      id: true,
      name: true,
      email: true,
      _count: {
        select: { posts: true },
      },
    },
    orderBy: { createdAt: 'desc' },
    take: 10,
    skip: 0,
  });

  // 返回类型精确匹配 select 中的字段
  for (const user of users) {
    console.log(`${user.name}: ${user._count.posts} posts`);
  }

  return users;
}

// 更新
async function updateUser(id: number) {
  const user = await prisma.user.update({
    where: { id },
    data: {
      name: 'Alice Updated',
      posts: {
        create: {
          title: 'New Post',
          content: 'Post content',
        },
      },
    },
    include: { posts: true },
  });

  return user;
}

// 删除
async function deleteUser(id: number) {
  await prisma.user.delete({
    where: { id },
  });
}

// 事务
async function transferPost(postId: number, newAuthorId: number) {
  const [post, author] = await prisma.$transaction([
    prisma.post.update({
      where: { id: postId },
      data: { authorId: newAuthorId },
    }),
    prisma.user.findUniqueOrThrow({
      where: { id: newAuthorId },
    }),
  ]);

  console.log(`Post "${post.title}" transferred to ${author.name}`);
  return { post, author };
}
```

### Prisma 生成的类型

```ts
import type { Prisma } from '@prisma/client';

// 使用 Prisma 生成的输入类型
type UserCreateInput = Prisma.UserCreateInput;
type UserWhereInput = Prisma.UserWhereInput;
type UserOrderByInput = Prisma.UserOrderByWithRelationInput;

// 使用 Prisma 的 validator 创建可复用的查询片段
const userWithPosts = Prisma.validator<Prisma.UserDefaultArgs>()({
  include: { posts: true, profile: true },
});

// 从查询片段推导返回类型
type UserWithPosts = Prisma.UserGetPayload<typeof userWithPosts>;

// 在服务层使用
async function getUserWithPosts(id: number): Promise<UserWithPosts | null> {
  return prisma.user.findUnique({
    where: { id },
    ...userWithPosts,
  });
}
```

## 🎯 Drizzle ORM

Drizzle ORM 提供类似 SQL 的 API，类型推导完全基于 Schema 定义，打包体积很小。

### 安装与配置

```bash
npm install drizzle-orm pg
npm install -D drizzle-kit @types/pg
```

### Schema 定义

```ts
// src/db/schema.ts
import { pgTable, serial, text, boolean, integer, timestamp, pgEnum } from 'drizzle-orm/pg-core';
import { relations } from 'drizzle-orm';

// 枚举
export const roleEnum = pgEnum('role', ['user', 'admin', 'moderator']);

// 用户表
export const users = pgTable('users', {
  id: serial('id').primaryKey(),
  email: text('email').notNull().unique(),
  name: text('name').notNull(),
  role: roleEnum('role').default('user').notNull(),
  createdAt: timestamp('created_at').defaultNow().notNull(),
});

// 文章表
export const posts = pgTable('posts', {
  id: serial('id').primaryKey(),
  title: text('title').notNull(),
  content: text('content'),
  published: boolean('published').default(false).notNull(),
  authorId: integer('author_id')
    .references(() => users.id)
    .notNull(),
  createdAt: timestamp('created_at').defaultNow().notNull(),
});

// 关系定义
export const usersRelations = relations(users, ({ many }) => ({
  posts: many(posts),
}));

export const postsRelations = relations(posts, ({ one }) => ({
  author: one(users, {
    fields: [posts.authorId],
    references: [users.id],
  }),
}));
```

### CRUD 操作

```ts
import { drizzle } from 'drizzle-orm/node-postgres';
import { eq, and, like, desc, count } from 'drizzle-orm';
import pg from 'pg';
import * as schema from './schema';

const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL });
const db = drizzle(pool, { schema });

// 类型从 Schema 自动推导
type User = typeof schema.users.$inferSelect;       // 查询结果类型
type NewUser = typeof schema.users.$inferInsert;     // 插入数据类型

// 插入
async function createUser(data: NewUser): Promise<User> {
  const [user] = await db
    .insert(schema.users)
    .values(data)
    .returning();

  return user;
}

// 查询 — SQL-like API
async function getUsers(role?: 'user' | 'admin' | 'moderator') {
  const conditions = role ? eq(schema.users.role, role) : undefined;

  const result = await db
    .select({
      id: schema.users.id,
      name: schema.users.name,
      email: schema.users.email,
      postCount: count(schema.posts.id),
    })
    .from(schema.users)
    .leftJoin(schema.posts, eq(schema.users.id, schema.posts.authorId))
    .where(conditions)
    .groupBy(schema.users.id)
    .orderBy(desc(schema.users.createdAt))
    .limit(10);

  // result 的类型是 { id: number; name: string; email: string; postCount: number }[]
  return result;
}

// 关联查询（使用 relations API）
async function getUserWithPosts(userId: number) {
  const user = await db.query.users.findFirst({
    where: eq(schema.users.id, userId),
    with: {
      posts: {
        where: eq(schema.posts.published, true),
        orderBy: [desc(schema.posts.createdAt)],
      },
    },
  });

  // user 类型自动包含 posts 数组
  return user;
}

// 更新
async function updateUser(id: number, data: Partial<NewUser>): Promise<User> {
  const [user] = await db
    .update(schema.users)
    .set(data)
    .where(eq(schema.users.id, id))
    .returning();

  return user;
}

// 删除
async function deleteUser(id: number): Promise<void> {
  await db.delete(schema.users).where(eq(schema.users.id, id));
}

// 事务
async function createUserWithPost(
  userData: NewUser,
  postData: { title: string; content: string }
) {
  return db.transaction(async (tx) => {
    const [user] = await tx
      .insert(schema.users)
      .values(userData)
      .returning();

    const [post] = await tx
      .insert(schema.posts)
      .values({
        title: postData.title,
        content: postData.content,
        authorId: user.id,
      })
      .returning();

    return { user, post };
  });
}
```

## 🎯 TypeORM

TypeORM 使用装饰器定义实体，支持 Active Record 和 Data Mapper 两种模式。

### 安装与配置

```bash
npm install typeorm pg reflect-metadata
npm install -D @types/node typescript
```

`tsconfig.json` 中需要启用装饰器：

```json
{
  "compilerOptions": {
    "experimentalDecorators": true,
    "emitDecoratorMetadata": true
  }
}
```

### 实体定义

```ts
// src/entities/User.ts
import {
  Entity,
  PrimaryGeneratedColumn,
  Column,
  CreateDateColumn,
  UpdateDateColumn,
  OneToMany,
  OneToOne,
} from 'typeorm';

// 使用 const enum 确保 TypeORM 可以正确处理
enum UserRole {
  USER = 'user',
  ADMIN = 'admin',
  MODERATOR = 'moderator',
}

@Entity('users')
class User {
  @PrimaryGeneratedColumn()
  id!: number;

  @Column({ unique: true })
  email!: string;

  @Column()
  name!: string;

  @Column({ type: 'enum', enum: UserRole, default: UserRole.USER })
  role!: UserRole;

  @OneToMany(() => Post, (post) => post.author)
  posts!: Post[];

  @CreateDateColumn()
  createdAt!: Date;

  @UpdateDateColumn()
  updatedAt!: Date;
}

@Entity('posts')
class Post {
  @PrimaryGeneratedColumn()
  id!: number;

  @Column()
  title!: string;

  @Column({ type: 'text', nullable: true })
  content!: string | null;

  @Column({ default: false })
  published!: boolean;

  @Column()
  authorId!: number;

  @ManyToOne(() => User, (user) => user.posts)
  author!: User;

  @CreateDateColumn()
  createdAt!: Date;
}

// 需要在文件顶部导入
import { ManyToOne } from 'typeorm';
```

### Repository 模式（Data Mapper）

```ts
import { DataSource, Repository } from 'typeorm';

// 初始化数据源
const dataSource = new DataSource({
  type: 'postgres',
  url: process.env.DATABASE_URL,
  entities: [User, Post],
  synchronize: false,
});

// 初始化连接
await dataSource.initialize();

// 获取类型化的 Repository
const userRepo: Repository<User> = dataSource.getRepository(User);
const postRepo: Repository<Post> = dataSource.getRepository(Post);

// 查询
async function findUsers(): Promise<User[]> {
  return userRepo.find({
    where: { role: UserRole.USER },
    relations: { posts: true },
    order: { createdAt: 'DESC' },
    take: 10,
  });
}

// 创建
async function createUser(data: { name: string; email: string }): Promise<User> {
  const user = userRepo.create({
    name: data.name,
    email: data.email,
    role: UserRole.USER,
  });

  return userRepo.save(user);
}

// QueryBuilder — 更灵活的查询
async function searchUsers(keyword: string): Promise<User[]> {
  return userRepo
    .createQueryBuilder('user')
    .leftJoinAndSelect('user.posts', 'post')
    .where('user.name ILIKE :keyword', { keyword: `%${keyword}%` })
    .orWhere('user.email ILIKE :keyword', { keyword: `%${keyword}%` })
    .orderBy('user.createdAt', 'DESC')
    .getMany();
}
```

## 🎯 类型安全的 SQL 查询

当 ORM 的抽象不满足需求时，可以构建类型安全的原生 SQL 查询方案。

### 简单的类型安全查询构建器

```ts
import pg from 'pg';

const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL });

// 类型安全的查询函数
async function query<T extends Record<string, unknown>>(
  sql: string,
  params: unknown[] = []
): Promise<T[]> {
  const result = await pool.query<T>(sql, params);
  return result.rows;
}

// 单条查询
async function queryOne<T extends Record<string, unknown>>(
  sql: string,
  params: unknown[] = []
): Promise<T | null> {
  const rows = await query<T>(sql, params);
  return rows[0] ?? null;
}

// 使用
interface UserRow {
  id: number;
  name: string;
  email: string;
  post_count: number;
}

async function getUsersWithPostCount(): Promise<UserRow[]> {
  return query<UserRow>(
    `SELECT u.id, u.name, u.email, COUNT(p.id)::int AS post_count
     FROM users u
     LEFT JOIN posts p ON p.author_id = u.id
     GROUP BY u.id
     ORDER BY post_count DESC
     LIMIT $1`,
    [10]
  );
}
```

### 标签模板的类型安全 SQL

```ts
// 使用 tagged template literal 构建安全的 SQL
class SqlQuery {
  constructor(
    public readonly text: string,
    public readonly values: unknown[]
  ) {}
}

function sql(strings: TemplateStringsArray, ...values: unknown[]): SqlQuery {
  const parameterizedParts: string[] = [];
  const params: unknown[] = [];

  for (let i = 0; i < strings.length; i++) {
    parameterizedParts.push(strings[i]);
    if (i < values.length) {
      params.push(values[i]);
      parameterizedParts.push(`${params.length}`);
    }
  }

  return new SqlQuery(parameterizedParts.join(''), params);
}

// 类型安全的执行器
async function execute<T extends Record<string, unknown>>(
  query: SqlQuery
): Promise<T[]> {
  const result = await pool.query<T>(query.text, query.values);
  return result.rows;
}

// 使用 — SQL 注入安全 + 类型安全
async function findUserByEmail(email: string) {
  interface Result {
    id: number;
    name: string;
    email: string;
  }

  const users = await execute<Result>(
    sql`SELECT id, name, email FROM users WHERE email = ${email}`
  );

  return users[0] ?? null;
}

async function findPublishedPosts(authorId: number, limit: number) {
  interface PostResult {
    id: number;
    title: string;
    content: string | null;
    author_name: string;
  }

  return execute<PostResult>(
    sql`SELECT p.id, p.title, p.content, u.name AS author_name
        FROM posts p
        JOIN users u ON u.id = p.author_id
        WHERE p.published = true AND p.author_id = ${authorId}
        ORDER BY p.created_at DESC
        LIMIT ${limit}`
  );
}
```

## 📝 实际应用示例

### 综合示例：Prisma + Express 类型安全 API

```ts
import express, { Request, Response, Router } from 'express';
import { PrismaClient, Prisma, Role } from '@prisma/client';

const prisma = new PrismaClient();
const app = express();
app.use(express.json());

// 通用响应类型
interface ApiResult<T> {
  success: boolean;
  data: T;
  message?: string;
}

// 用户查询参数
interface UserQueryParams {
  page?: string;
  limit?: string;
  role?: Role;
  search?: string;
}

// 用户创建请求体
interface CreateUserInput {
  name: string;
  email: string;
  role?: Role;
  bio?: string;
}

// 定义返回类型模板
const userWithPostCount = Prisma.validator<Prisma.UserDefaultArgs>()({
  select: {
    id: true,
    name: true,
    email: true,
    role: true,
    createdAt: true,
    _count: { select: { posts: true } },
  },
});

type UserWithPostCount = Prisma.UserGetPayload<typeof userWithPostCount>;

function createUserRouter(): Router {
  const router = Router();

  // GET /api/users
  router.get(
    '/',
    async (
      req: Request<{}, ApiResult<UserWithPostCount[]>, {}, UserQueryParams>,
      res: Response<ApiResult<UserWithPostCount[]>>
    ) => {
      const page = parseInt(req.query.page ?? '1', 10);
      const limit = parseInt(req.query.limit ?? '10', 10);

      const where: Prisma.UserWhereInput = {};
      if (req.query.role) where.role = req.query.role;
      if (req.query.search) {
        where.OR = [
          { name: { contains: req.query.search, mode: 'insensitive' } },
          { email: { contains: req.query.search, mode: 'insensitive' } },
        ];
      }

      const users = await prisma.user.findMany({
        ...userWithPostCount,
        where,
        orderBy: { createdAt: 'desc' },
        skip: (page - 1) * limit,
        take: limit,
      });

      res.json({ success: true, data: users });
    }
  );

  // POST /api/users
  router.post(
    '/',
    async (
      req: Request<{}, ApiResult<UserWithPostCount>, CreateUserInput>,
      res: Response<ApiResult<UserWithPostCount>>
    ) => {
      const { name, email, role, bio } = req.body;

      const user = await prisma.user.create({
        data: {
          name,
          email,
          role: role ?? Role.USER,
          profile: bio ? { create: { bio } } : undefined,
        },
        ...userWithPostCount,
      });

      res.status(201).json({ success: true, data: user });
    }
  );

  // GET /api/users/:id
  router.get(
    '/:id',
    async (req: Request<{ id: string }>, res: Response) => {
      const user = await prisma.user.findUnique({
        where: { id: parseInt(req.params.id, 10) },
        include: {
          profile: true,
          posts: {
            where: { published: true },
            orderBy: { createdAt: 'desc' },
            take: 5,
          },
        },
      });

      if (!user) {
        res.status(404).json({ success: false, data: null, message: 'User not found' });
        return;
      }

      res.json({ success: true, data: user });
    }
  );

  return router;
}

app.use('/api/users', createUserRouter());

app.listen(3000, () => {
  console.log('Server running on port 3000');
});
```
