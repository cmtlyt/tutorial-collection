### 基础聊天组件

```js
import { useState } from 'react'

function ChatComponent() {
  const [messages, setMessages] = useState([
    { role: 'assistant', content: '你好！我是 AI 助手，有什么可以帮助你的吗？' }
  ])
  const [input, setInput] = useState('')
  const [isLoading, setIsLoading] = useState(false)
  
  const sendMessage = async () => {
    if (!input.trim()) return
    
    const userMessage = { role: 'user', content: input }
    setMessages(prev => [...prev, userMessage])
    setInput('')
    setIsLoading(true)
    
    try {
      const response = await fetch('https://api.openai.com/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer YOUR_API_KEY'
        },
        body: JSON.stringify({
          model: 'gpt-3.5-turbo',
          messages: [...messages, userMessage],
          max_tokens: 500
        })
      })
      
      const data = await response.json()
      const assistantMessage = {
        role: 'assistant',
        content: data.choices[0].message.content
      }
      
      setMessages(prev => [...prev, assistantMessage])
    } catch (error) {
      console.error('Error:', error)
      setMessages(prev => [...prev, {
        role: 'assistant',
        content: '抱歉，发生了错误。请稍后重试。'
      }])
    }
    
    setIsLoading(false)
  }
  
  return (
    <div style={{ maxWidth: '800px', margin: '0 auto', padding: '20px' }}>
      <h1>AI 聊天助手</h1>
      
      <div style={{ 
        height: '400px', 
        overflowY: 'auto',
        border: '1px solid #ddd',
        borderRadius: '8px',
        padding: '16px',
        marginBottom: '16px',
        background: '#f9f9f9'
      }}>
        {messages.map((message, index) => (
          <div 
            key={index}
            style={{ 
              marginBottom: '12px',
              display: 'flex',
              justifyContent: message.role === 'user' ? 'flex-end' : 'flex-start'
            }}
          >
            <div style={{
              maxWidth: '70%',
              padding: '12px 16px',
              borderRadius: '12px',
              background: message.role === 'user' ? '#007bff' : '#fff',
              color: message.role === 'user' ? '#fff' : '#333',
              boxShadow: '0 1px 2px rgba(0,0,0,0.1)'
            }}>
              {message.content}
            </div>
          </div>
        ))}
        
        {isLoading && (
          <div style={{ display: 'flex', justifyContent: 'flex-start' }}>
            <div style={{
              padding: '12px 16px',
              borderRadius: '12px',
              background: '#fff',
              color: '#666'
            }}>
              正在思考...
            </div>
          </div>
        )}
      </div>
      
      <div style={{ display: 'flex', gap: '8px' }}>
        <input
          type="text"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && sendMessage()}
          placeholder="输入你的消息..."
          disabled={isLoading}
          style={{
            flex: 1,
            padding: '12px',
            border: '1px solid #ddd',
            borderRadius: '8px',
            fontSize: '16px'
          }}
        />
        <button
          onClick={sendMessage}
          disabled={isLoading || !input.trim()}
          style={{
            padding: '12px 24px',
            background: isLoading ? '#ccc' : '#007bff',
            color: '#fff',
            border: 'none',
            borderRadius: '8px',
            cursor: isLoading ? 'not-allowed' : 'pointer',
            fontSize: '16px'
          }}
        >
          发送
        </button>
      </div>
    </div>
  )
}

export default ChatComponent
```

```js
import { useState } from 'react'

function StreamingChat() {
  const [messages, setMessages] = useState([])
  const [input, setInput] = useState('')
  const [isLoading, setIsLoading] = useState(false)
  const [currentResponse, setCurrentResponse] = useState('')
  
  const sendStreamingMessage = async () => {
    if (!input.trim()) return
    
    const userMessage = { role: 'user', content: input }
    setMessages(prev => [...prev, userMessage])
    setInput('')
    setIsLoading(true)
    setCurrentResponse('')
    
    try {
      const response = await fetch('https://api.openai.com/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer YOUR_API_KEY'
        },
        body: JSON.stringify({
          model: 'gpt-3.5-turbo',
          messages: [...messages, userMessage],
          stream: true
        })
      })
      
      const reader = response.body.getReader()
      const decoder = new TextDecoder()
      let fullResponse = ''
      
      while (true) {
        const { done, value } = await reader.read()
        
        if (done) break
        
        const chunk = decoder.decode(value)
        const lines = chunk.split('\n').filter(line => line.trim())
        
        for (const line of lines) {
          if (line.startsWith('data: ')) {
            const data = line.slice(6)
            
            if (data === '[DONE]') continue
            
            try {
              const parsed = JSON.parse(data)
              const content = parsed.choices[0]?.delta?.content || ''
              
              if (content) {
                fullResponse += content
                setCurrentResponse(fullResponse)
              }
            } catch (e) {
              console.error('Error parsing:', e)
            }
          }
        }
      }
      
      setMessages(prev => [...prev, {
        role: 'assistant',
        content: fullResponse
      }])
    } catch (error) {
      console.error('Error:', error)
      setMessages(prev => [...prev, {
        role: 'assistant',
        content: '抱歉，发生了错误。'
      }])
    }
    
    setIsLoading(false)
    setCurrentResponse('')
  }
  
  return (
    <div style={{ maxWidth: '800px', margin: '0 auto', padding: '20px' }}>
      <h1>流式 AI 聊天</h1>
      
      <div style={{ 
        height: '400px', 
        overflowY: 'auto',
        border: '1px solid #ddd',
        borderRadius: '8px',
        padding: '16px',
        marginBottom: '16px',
        background: '#f9f9f9'
      }}>
        {messages.map((message, index) => (
          <div 
            key={index}
            style={{ 
              marginBottom: '12px',
              display: 'flex',
              justifyContent: message.role === 'user' ? 'flex-end' : 'flex-start'
            }}
          >
            <div style={{
              maxWidth: '70%',
              padding: '12px 16px',
              borderRadius: '12px',
              background: message.role === 'user' ? '#007bff' : '#fff',
              color: message.role === 'user' ? '#fff' : '#333',
              boxShadow: '0 1px 2px rgba(0,0,0,0.1)'
            }}>
              {message.content}
            </div>
          </div>
        ))}
        
        {isLoading && currentResponse && (
          <div style={{ display: 'flex', justifyContent: 'flex-start' }}>
            <div style={{
              maxWidth: '70%',
              padding: '12px 16px',
              borderRadius: '12px',
              background: '#fff',
              color: '#333',
              boxShadow: '0 1px 2px rgba(0,0,0,0.1)'
            }}>
              {currentResponse}
              <span style={{ animation: 'blink 1s infinite' }}>▋</span>
            </div>
          </div>
        )}
      </div>
      
      <div style={{ display: 'flex', gap: '8px' }}>
        <input
          type="text"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && sendStreamingMessage()}
          placeholder="输入你的消息..."
          disabled={isLoading}
          style={{
            flex: 1,
            padding: '12px',
            border: '1px solid #ddd',
            borderRadius: '8px',
            fontSize: '16px'
          }}
        />
        <button
          onClick={sendStreamingMessage}
          disabled={isLoading || !input.trim()}
          style={{
            padding: '12px 24px',
            background: isLoading ? '#ccc' : '#007bff',
            color: '#fff',
            border: 'none',
            borderRadius: '8px',
            cursor: isLoading ? 'not-allowed' : 'pointer',
            fontSize: '16px'
          }}
        >
          发送
        </button>
      </div>
      
      <style jsx>{`
        @keyframes blink {
          0%, 50% { opacity: 1; }
          51%, 100% { opacity: 0; }
        }
      `}</style>
    </div>
  )
}
```

## 🎯 智能表单

### AI 表单助手

```js
import { useState } from 'react'

function AIFormAssistant() {
  const [formData, setFormData] = useState({
    name: '',
    email: '',
    message: ''
  })
  const [suggestions, setSuggestions] = useState({})
  const [isAnalyzing, setIsAnalyzing] = useState(false)
  
  const analyzeField = async (fieldName, value) => {
    if (!value.trim()) {
      setSuggestions(prev => ({ ...prev, [fieldName]: null }))
      return
    }
    
    setIsAnalyzing(true)
    
    try {
      const response = await fetch('https://api.openai.com/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer YOUR_API_KEY'
        },
        body: JSON.stringify({
          model: 'gpt-3.5-turbo',
          messages: [
            {
              role: 'system',
              content: '你是一个表单助手，分析用户输入并提供改进建议。'
            },
            {
              role: 'user',
              content: `分析这个${fieldName}字段的输入：${value}`
            }
          ],
          max_tokens: 100
        })
      })
      
      const data = await response.json()
      const suggestion = data.choices[0].message.content
      
      setSuggestions(prev => ({ ...prev, [fieldName]: suggestion }))
    } catch (error) {
      console.error('Error:', error)
    }
    
    setIsAnalyzing(false)
  }
  
  const handleChange = (fieldName, value) => {
    setFormData(prev => ({ ...prev, [fieldName]: value }))
    
    if (fieldName === 'message') {
      analyzeField(fieldName, value)
    }
  }
  
  const handleSubmit = async (e) => {
    e.preventDefault()
    
    const response = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer YOUR_API_KEY'
      },
      body: JSON.stringify({
        model: 'gpt-3.5-turbo',
        messages: [
          {
            role: 'system',
            content: '你是一个表单验证助手，检查表单数据并提供反馈。'
          },
          {
            role: 'user',
            content: `验证这个表单数据：${JSON.stringify(formData)}`
          }
        ],
        max_tokens: 200
      })
    })
    
    const data = await response.json()
    const validation = data.choices[0].message.content
    
    alert(`AI 验证结果：\n${validation}`)
  }
  
  return (
    <div style={{ maxWidth: '600px', margin: '0 auto', padding: '20px' }}>
      <h1>AI 表单助手</h1>
      
      <form onSubmit={handleSubmit} style={{ display: 'flex', flexDirection: 'column', gap: '20px' }}>
        <div>
          <label style={{ display: 'block', marginBottom: '8px', fontWeight: 'bold' }}>
            姓名
            <input
              type="text"
              value={formData.name}
              onChange={(e) => handleChange('name', e.target.value)}
              required
              style={{
                width: '100%',
                padding: '10px',
                border: '1px solid #ddd',
                borderRadius: '4px',
                fontSize: '16px'
              }}
            />
          </label>
        </div>
        
        <div>
          <label style={{ display: 'block', marginBottom: '8px', fontWeight: 'bold' }}>
            邮箱
            <input
              type="email"
              value={formData.email}
              onChange={(e) => handleChange('email', e.target.value)}
              required
              style={{
                width: '100%',
                padding: '10px',
                border: '1px solid #ddd',
                borderRadius: '4px',
                fontSize: '16px'
              }}
            />
          </label>
        </div>
        
        <div>
          <label style={{ display: 'block', marginBottom: '8px', fontWeight: 'bold' }}>
            消息
            <textarea
              value={formData.message}
              onChange={(e) => handleChange('message', e.target.value)}
              required
              rows={6}
              style={{
                width: '100%',
                padding: '10px',
                border: '1px solid #ddd',
                borderRadius: '4px',
                fontSize: '16px',
                resize: 'vertical'
              }}
            />
          </label>
          
          {suggestions.message && (
            <div style={{
              marginTop: '8px',
              padding: '12px',
              background: '#e3f2fd',
              border: '1px solid #2196f3',
              borderRadius: '4px',
              fontSize: '14px'
            }}>
              <strong>AI 建议：</strong> {suggestions.message}
            </div>
          )}
          
          {isAnalyzing && (
            <div style={{ marginTop: '8px', color: '#666', fontSize: '14px' }}>
              AI 正在分析...
            </div>
          )}
        </div>
        
        <button
          type="submit"
          style={{
            padding: '12px 24px',
            background: '#007bff',
            color: '#fff',
            border: 'none',
            borderRadius: '4px',
            cursor: 'pointer',
            fontSize: '16px',
            fontWeight: 'bold'
          }}
        >
          提交并验证
        </button>
      </form>
    </div>
  )
}

export default AIFormAssistant
```

```js
import { useState } from 'react'

function ContentGenerator() {
  const [prompt, setPrompt] = useState('')
  const [contentType, setContentType] = useState('article')
  const [generatedContent, setGeneratedContent] = useState('')
  const [isGenerating, setIsGenerating] = useState(false)
  
  const contentTypes = [
    { value: 'article', label: '文章' },
    { value: 'summary', label: '摘要' },
    { value: 'translation', label: '翻译' },
    { value: 'code', label: '代码' },
    { value: 'email', label: '邮件' }
  ]
  
  const generateContent = async () => {
    if (!prompt.trim()) return
    
    setIsGenerating(true)
    setGeneratedContent('')
    
    const systemPrompt = {
      article: '你是一个专业的内容写手，根据主题写一篇详细的文章。',
      summary: '你是一个专业的编辑，对给定的内容进行简洁的摘要。',

        method: 'POST',

          'Content-Type': 'application/json',
          'Authorization': 'Bearer YOUR_API_KEY'

        body: JSON.stringify({

          messages: [
            {
              role: 'system',
              content: systemPrompt[contentType]
            },

              role: 'user',
              content: prompt
            }

          temperature: 0.7
        })

    } catch (error) {
      console.error('Error:', error)
      setGeneratedContent('生成失败，请稍后重试。')
    }
    
    setIsGenerating(false)
  }

    navigator.clipboard.writeText(generatedContent)
    alert('已复制到剪贴板')
  }
  
  return (
    <div style={{ maxWidth: '900px', margin: '0 auto', padding: '20px' }}>
      <h1>AI 内容生成器</h1>
      
      <div style={{ marginBottom: '20px' }}>
        <label style={{ display: 'block', marginBottom: '8px', fontWeight: 'bold' }}>
          内容类型：
          <select
            value={contentType}
            onChange={(e) => setContentType(e.target.value)}
            style={{
              marginLeft: '8px',
              padding: '8px',
              border: '1px solid #ddd',
              borderRadius: '4px',
              fontSize: '16px'
            }}
          >
            {contentTypes.map(type => (
              <option key={type.value} value={type.value}>
                {type.label}
              </option>
            ))}
          </select>
        </label>
      </div>
      
      <div style={{ marginBottom: '20px' }}>
        <label style={{ display: 'block', marginBottom: '8px', fontWeight: 'bold' }}>
          输入提示：
          <textarea
            value={prompt}
            onChange={(e) => setPrompt(e.target.value)}
            placeholder="描述你想要生成的内容..."
            rows={4}
            style={{
              width: '100%',
              padding: '12px',
              border: '1px solid #ddd',
              borderRadius: '4px',
              fontSize: '16px',
              resize: 'vertical'
            }}
          />
        </label>
      </div>
      
      <button
        onClick={generateContent}
        disabled={isGenerating || !prompt.trim()}
        style={{
          padding: '12px 24px',
          background: isGenerating ? '#ccc' : '#007bff',
          color: '#fff',
          border: 'none',
          borderRadius: '4px',
          cursor: isGenerating ? 'not-allowed' : 'pointer',
          fontSize: '16px',
          marginBottom: '20px'
        }}
      >
        {isGenerating ? '生成中...' : '生成内容'}
      </button>
      
      {generatedContent && (
        <div>
          <div style={{ 
            display: 'flex', 
            justifyContent: 'space-between', 
            alignItems: 'center',
            marginBottom: '12px' 
          }}>
            <h3>生成结果：</h3>
            <button
              onClick={copyToClipboard}
              style={{
                padding: '8px 16px',
                background: '#28a745',
                color: '#fff',
                border: 'none',
                borderRadius: '4px',
                cursor: 'pointer'
              }}
            >
              复制
            </button>
          </div>
          
          <div style={{
            padding: '20px',
            background: '#f9f9f9',
            border: '1px solid #ddd',
            borderRadius: '8px',
            whiteSpace: 'pre-wrap',
            lineHeight: '1.8'
          }}>
            {generatedContent}
          </div>
        </div>
      )}
    </div>
  )
}

export default ContentGenerator
```
