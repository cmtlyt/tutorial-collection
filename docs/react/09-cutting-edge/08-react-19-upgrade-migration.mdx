### 版本检查

```js
// 检查当前 React 版本
import React from 'react'
import ReactDOM from 'react-dom'

console.log('React 版本:', React.version)
console.log('ReactDOM 版本:', ReactDOM.version)

// 检查依赖兼容性
const packageJson = {
  dependencies: {
    'react': '^18.3.1',
    'react-dom': '^18.3.1',
    'react-router-dom': '^6.22.0',
    '@testing-library/react': '^14.2.0'
  }
}

function DependencyChecker() {
  const [dependencies, setDependencies] = useState(packageJson.dependencies)
  const [compatibility, setCompatibility] = useState({})
  
  const checkCompatibility = () => {
    const results = {}
    
    Object.entries(dependencies).forEach(([name, version]) => {
      if (name.startsWith('react')) {
        results[name] = {
          current: version,
          recommended: '^19.0.0',
          status: version.startsWith('^18') ? '需要升级' : '兼容'
        }
      } else {
        results[name] = {
          current: version,
          recommended: version,
          status: '需要检查兼容性'
        }
      }
    })
    
    setCompatibility(results)
  }
  
  return (
    <div style={{ padding: '20px', maxWidth: '800px', margin: '0 auto' }}>
      <h1>依赖兼容性检查</h1>
      
      <button 
        onClick={checkCompatibility}
        style={{ padding: '12px 24px', background: '#007bff', color: '#fff', border: 'none', borderRadius: '8px', cursor: 'pointer' }}
      >
        检查兼容性
      </button>
      
      {Object.keys(compatibility).length > 0 && (
        <div style={{ marginTop: '20px' }}>
          <h3>检查结果：</h3>
          {Object.entries(compatibility).map(([name, info]) => (
            <div key={name} style={{ 
              padding: '12px', 
              marginBottom: '8px',
              background: info.status === '需要升级' ? '#fee' : '#efe',
              border: `1px solid ${info.status === '需要升级' ? '#fcc' : '#cfc'}`,
              borderRadius: '4px'
            }}>
              <strong>{name}</strong>
              <br />
              当前版本：{info.current}
              <br />
              推荐版本：{info.recommended}
              <br />
              状态：<strong style={{ color: info.status === '需要升级' ? 'red' : 'green' }}>
                {info.status}
              </strong>
            </div>
          ))}
        </div>
      )}
    </div>
  )
}
```

```js
function UpgradeChecklist() {
  const [checklist, setChecklist] = useState([
    { id: 1, task: '备份当前代码', completed: false, critical: true },
    { id: 2, task: '更新 React 到 18.3', completed: false, critical: true },
    { id: 3, task: '运行测试套件', completed: false, critical: true },
    { id: 4, task: '检查废弃警告', completed: false, critical: true },
    { id: 5, task: '更新第三方依赖', completed: false, critical: false },
    { id: 6, task: '更新构建工具', completed: false, critical: false },
```

## 🎯 破坏性变更

React 19 移除了一些废弃的 API，需要相应地更新代码。

### 移除的 API

```js
// React 19 中移除的 API 示例

// ❌ 已移除：PropTypes
import PropTypes from 'prop-types'

function OldComponent({ name, age }) {
  return <div>{name} - {age}</div>
}

OldComponent.propTypes = {
  name: PropTypes.string.isRequired,
  age: PropTypes.number
}

// ✅ 使用 TypeScript 或其他类型检查
function NewComponent({ name, age }) {
  if (!name) {
    throw new Error('name is required')
  }
  
  return <div>{name} - {age}</div>
}

// ❌ 已移除：defaultProps
function OldButton({ text, color }) {
  return <button style={{ background: color }}>{text}</button>
}

OldButton.defaultProps = {
  text: 'Click me',
  color: 'blue'
}

// ✅ 使用默认参数
function NewButton({ text = 'Click me', color = 'blue' }) {
  return <button style={{ background: color }}>{text}</button>
}

// ❌ 已移除：String Refs
class OldComponent extends React.Component {
  handleClick = () => {
    this.refs.inputRef.focus()
  }
  
  render() {
    return (
      <div>
        <input ref="inputRef" />
        <button onClick={this.handleClick}>Focus</button>
      </div>
    )
  }
}

// ✅ 使用 useRef 或 createRef
function NewComponent() {
  const inputRef = React.useRef(null)
  
  const handleClick = () => {
    inputRef.current.focus()
  }
  
  return (
    <div>
      <input ref={inputRef} />
      <button onClick={handleClick}>Focus</button>
    </div>
  )
}

// ❌ 已移除：UNSAFE_ 生命周期方法
class OldLifecycle extends React.Component {
  UNSAFE_componentWillMount() {
    console.log('即将挂载')
  }
  
  UNSAFE_componentWillReceiveProps(nextProps) {
    console.log('即将接收新 props')
  }
  
  UNSAFE_componentWillUpdate(nextProps, nextState) {
    console.log('即将更新')
  }
  
  render() {
    return <div>旧生命周期</div>
  }
}

// ✅ 使用新的生命周期方法
class NewLifecycle extends React.Component {
  componentDidMount() {
    console.log('已挂载')
  }
  
  componentDidUpdate(prevProps) {
    console.log('已更新', prevProps, this.props)
  }
  
  render() {
    return <div>新生命周期</div>
  }
}
```

```js
function MigrationDiagnostics() {
  const [issues, setIssues] = useState([])
  const [isScanning, setIsScanning] = useState(false)
  
  const commonIssues = [
    {
      id: 1,
      category: 'API 变更',
      severity: 'high',
      title: '使用已移除的 PropTypes',
      description: 'PropTypes 在 React 19 中已被移除',
      solution: '使用 TypeScript 或其他类型检查工具',
      code: 'OldComponent.propTypes = { name: PropTypes.string }'
    },
    {
      id: 2,
      category: 'API 变更',
      severity: 'high',
      title: '使用 defaultProps',
      description: 'defaultProps 在 React 19 中已被移除',
      solution: '使用函数默认参数',
      code: 'function Component({ prop = defaultValue }) {}'
    },
    {
      id: 3,
      category: 'API 变更',
      severity: 'medium',
      title: '使用 String Refs',
      description: 'String refs 在 React 19 中已被移除',
      solution: '使用 useRef 或 createRef',
      code: 'const ref = useRef(null)'
    },
    {
      id: 4,
      category: '生命周期',
      severity: 'high',
      title: '使用 UNSAFE_ 生命周期',
      description: 'UNSAFE_ 生命周期方法已被移除',
      solution: '使用新的生命周期方法',
      code: 'useEffect(() => {}, [dependencies])'
    },
    {
      id: 5,
      category: '类型系统',
      severity: 'medium',
      title: 'TypeScript 类型错误',
      description: 'React 19 的 TypeScript 类型有变化',
      solution: '更新 @types/react 到最新版本',
      code: 'npm install @types/react@latest'
    }
  ]
  
  const scanForIssues = async () => {
    setIsScanning(true)
    
    await new Promise(resolve => setTimeout(resolve, 2000))
    
    setIssues(commonIssues)
    setIsScanning(false)
  }
  
  const getSeverityColor = (severity) => {
    switch (severity) {
      case 'high': return '#dc3545'
      case 'medium': return '#ffc107'
      case 'low': return '#28a745'
      default: return '#6c757d'
    }
  }
  
  return (
    <div style={{ padding: '20px', maxWidth: '1000px', margin: '0 auto' }}>
      <h1>迁移问题诊断</h1>
      
      <div style={{ marginBottom: '24px' }}>
        <button 
          onClick={scanForIssues}
          disabled={isScanning}
          style={{
            padding: '12px 24px',
            background: isScanning ? '#ccc' : '#007bff',
            color: '#fff',
            border: 'none',
            borderRadius: '8px',
            cursor: isScanning ? 'not-allowed' : 'pointer',
            fontSize: '16px'
          }}
        >
          {isScanning ? '扫描中...' : '开始扫描'}
        </button>
      </div>
      
      {issues.length > 0 && (
        <div>
          <h3>发现的问题 ({issues.length})</h3>
          
          {issues.map(issue => (
            <div 
              key={issue.id}
              style={{ 
                marginBottom: '16px',
                padding: '16px',
                border: `1px solid ${getSeverityColor(issue.severity)}`,
                borderRadius: '8px',
                background: '#fff'
              }}
            >
              <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start', marginBottom: '12px' }}>
                <div>
                  <span style={{ 
                    padding: '4px 8px',
                    background: getSeverityColor(issue.severity),
                    color: '#fff',
                    borderRadius: '4px',
                    fontSize: '12px',
                    marginRight: '8px'
                  }}>
                    {issue.severity.toUpperCase()}
                  </span>
                  <span style={{ fontSize: '14px', color: '#666' }}>
                    {issue.category}
                  </span>
                </div>
              </div>
              
              <h4 style={{ margin: '0 0 8px 0' }}>{issue.title}</h4>
              <p style={{ margin: '0 0 12px 0', color: '#666' }}>
                {issue.description}
              </p>
              
              <div style={{ marginBottom: '12px' }}>
                <strong>解决方案：</strong>
                <p style={{ margin: '4px 0 0 0', color: '#28a745' }}>
                  {issue.solution}
                </p>
              </div>
              
              <div>
                <strong>代码示例：</strong>
                <pre style={{ 
                  margin: '8px 0 0 0',
                  padding: '12px',
                  background: '#f5f5f5',
                  borderRadius: '4px',
                  overflow: 'auto',
                  fontSize: '14px'
                }}>
                  {issue.code}
                </pre>
              </div>
            </div>
          ))}
        </div>
      )}
      
      {issues.length === 0 && !isScanning && (
        <div style={{ 
          padding: '20px', 
          background: '#d4edda', 
          border: '1px solid #c3e6cb',
          borderRadius: '8px',
          color: '#155724',
          textAlign: 'center'
        }}>
          ✅ 没有发现迁移问题
        </div>
      )}
    </div>
  )
}
```

```js
    reactVersion: '18.3.1',
    backupCreated: false,
    testsPassed: false,
    codemodsRun: false
  })
  
  const steps = [
    { id: 1, title: '项目准备', description: '准备项目信息和备份' },
    { id: 2, title: '依赖更新', description: '更新 React 和相关依赖' },

    { id: 4, title: '测试验证', description: '运行测试确保功能正常' },
    { id: 5, title: '完成', description: '迁移完成' }
  ]
  
  const handleNext = () => {
    if (currentStep < steps.length) {
      setCurrentStep(currentStep + 1)
    }
  }
  
  const handlePrevious = () => {
    if (currentStep > 1) {
      setCurrentStep(currentStep - 1)
    }
  }
  
  const renderStepContent = () => {
    switch (currentStep) {
      case 1:
        return (
          <div>
            <h3>项目准备</h3>
            <div style={{ marginBottom: '16px' }}>
              <label style={{ display: 'block', marginBottom: '8px', fontWeight: 'bold' }}>
                项目路径：
                <input
                  type="text"
                  value={migrationData.projectPath}
                  onChange={(e) => setMigrationData({ ...migrationData, projectPath: e.target.value })}
                  placeholder="/path/to/your/project"
                  style={{
                    width: '100%',
                    padding: '10px',
                    border: '1px solid #ddd',
                    borderRadius: '4px',
                    marginTop: '4px'
                  }}
                />
              </label>
            </div>
            
            <div style={{ marginBottom: '16px' }}>
              <label style={{ display: 'block', marginBottom: '8px', fontWeight: 'bold' }}>
                当前 React 版本：
                <select
                  value={migrationData.reactVersion}
                  onChange={(e) => setMigrationData({ ...migrationData, reactVersion: e.target.value })}
                  style={{
                    width: '100%',
                    padding: '10px',
                    border: '1px solid #ddd',
                    borderRadius: '4px',
                    marginTop: '4px'

                >
                  <option value="18.3.1">18.3.1</option>

              </label>
            </div>
            
            <div style={{

              background: '#fff3cd', 
              border: '1px solid #ffeeba',
              borderRadius: '4px',
              color: '#856404'

              <strong>⚠️ 重要提示：</strong>
              <ul style={{ margin: '8px 0 0 0', paddingLeft: '20px' }}>
                <li>确保已创建项目备份</li>
                <li>建议在分支上进行迁移</li>
                <li>运行所有测试确保当前状态正常</li>
              </ul>
            </div>
          </div>
        )
      
      case 2:
        return (
          <div>
            <h3>依赖更新</h3>
            
            <div style={{ marginBottom: '20px' }}>
              <h4>需要更新的包：</h4>
              <div style={{ 
                padding: '16px', 
                background: '#f9f9f9',
                borderRadius: '4px'
              }}>
                <pre style={{ margin: 0 }}>
{`"dependencies": {
  "react": "^19.0.0",
  "react-dom": "^19.0.0"
},
"devDependencies": {
  "@types/react": "^19.0.0",
  "@types/react-dom": "^19.0.0",
  "@testing-library/react": "^16.0.0"
}`}
                </pre>
              </div>
            </div>
            
            <div style={{ marginBottom: '20px' }}>
              <h4>运行命令：</h4>
              <div style={{ 
                padding: '12px', 
                background: '#e3f2fd',
                borderRadius: '4px',
                fontFamily: 'monospace'
              }}>
                npm install react@19 react-dom@19 --save
              </div>
            </div>
            
            <div style={{ 
              padding: '16px',

              borderRadius: '4px',
              color: '#155724'

                <li>检查 package.json 中的版本</li>
                <li>运行 npm install 安装依赖</li>
                <li>查看是否有冲突或警告</li>
              </ul>

          </div>

      case 3:
        return (

      <div style={{ marginBottom: '24px' }}>
        {steps.map((step, index) => (
          <div 
            key={step.id}
            style={{ 
              padding: '12px',
              marginBottom: '8px',
              borderRadius: '4px',
              background: index + 1 === currentStep ? '#e3f2fd' : '#f9f9f9',
              border: index + 1 === currentStep ? '2px solid #007bff' : 'none'
            }}
          >
            <strong>{step.id}. {step.title}</strong>
            {index + 1 === currentStep && (
              <p style={{ margin: '8px 0 0 0', color: '#666' }}>

        ))}
      </div>

        padding: '24px', 
        border: '1px solid #ddd', 
        borderRadius: '8px',
        marginBottom: '24px'
      }}>
        {renderStepContent()}
      </div>
      
      <div style={{ display: 'flex', justifyContent: 'space-between' }}>
        <button
          onClick={handlePrevious}
          disabled={currentStep === 1}
          style={{
            padding: '12px 24px',
            background: currentStep === 1 ? '#ccc' : '#6c757d',
            color: '#fff',
            border: 'none',
            borderRadius: '8px',
            cursor: currentStep === 1 ? 'not-allowed' : 'pointer',
            fontSize: '16px'
          }}
        >
          上一步
        </button>
        
        {currentStep < steps.length ? (
          <button
            onClick={handleNext}
            style={{
              padding: '12px 24px',
              background: '#007bff',
              color: '#fff',
              border: 'none',
              borderRadius: '8px',
              cursor: 'pointer',
              fontSize: '16px'
            }}
          >
            下一步
          </button>
        ) : (
          <button
            onClick={() => setCurrentStep(1)}
            style={{
              padding: '12px 24px',
              background: '#28a745',
              color: '#fff',
              border: 'none',
              borderRadius: '8px',
              cursor: 'pointer',
              fontSize: '16px'
            }}
          >
            重新开始
          </button>
        )}
      </div>
    </div>
  )
}
```
