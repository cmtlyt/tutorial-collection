## 🎯 Remix 框架

Remix 是一个现代化的全栈 React 框架，专注于 Web 标准和渐进增强。

### 基础路由和数据加载

```js
import { json, useLoaderData } from '@remix-run/react'

export async function loader() {
  const response = await fetch('https://api.example.com/posts')
  const posts = await response.json()
  return json(posts)
}

export default function Posts() {
  const posts = useLoaderData()
  
  return (
    <div style={{ padding: '20px' }}>
      <h1>博客文章</h1>
      <ul style={{ listStyle: 'none', padding: 0 }}>
        {posts.map(post => (
          <li key={post.id} style={{ marginBottom: '16px', padding: '16px', border: '1px solid #ddd', borderRadius: '4px' }}>
            <h2>{post.title}</h2>
            <p>{post.excerpt}</p>
          </li>
        ))}
      </ul>
    </div>
  )
}
```

```js
import { ref, computed } from 'vue'

export default {
  setup() {
    const count = ref(0)
```

### React 与 Vue 对比

```js
import { useState, useMemo } from 'react'

function ReactTodoApp() {
  const [todos, setTodos] = useState([
    { id: 1, text: '学习 React', completed: false },
    { id: 2, text: '学习 Vue', completed: false }
  ])
  const [newTodo, setNewTodo] = useState('')
  
  const addTodo = () => {
    if (newTodo.trim()) {
      setTodos([...todos, {
        id: Date.now(),
        text: newTodo,
        completed: false
      }])
      setNewTodo('')
    }
  }
  
  const toggleTodo = (id) => {
    setTodos(todos.map(todo =>
      todo.id === id ? { ...todo, completed: !todo.completed } : todo
    ))
  }
  
  const completedCount = todos.filter(t => t.completed).length
  
  return (
    <div style={{ padding: '20px', maxWidth: '500px' }}>
      <h2>React Todo 应用</h2>
      
      <div style={{ marginBottom: '16px' }}>
        <input
          type="text"
          value={newTodo}
          onChange={(e) => setNewTodo(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && addTodo()}
          placeholder="添加新任务..."
          style={{ 
            padding: '8px', 
            width: 'calc(100% - 80px)',
            marginRight: '8px'
          }}
        />
        <button onClick={addTodo}>添加</button>
      </div>
      
      <ul style={{ listStyle: 'none', padding: 0 }}>
        {todos.map(todo => (
          <li key={todo.id} style={{ 
            padding: '8px', 
            marginBottom: '8px',
            background: '#f5f5f5',
            borderRadius: '4px',
            display: 'flex',
            alignItems: 'center'
          }}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={() => toggleTodo(todo.id)}
              style={{ marginRight: '8px' }}
            />
            <span style={{ 
              textDecoration: todo.completed ? 'line-through' : 'none',
              color: todo.completed ? '#999' : '#333'
            }}>
              {todo.text}
            </span>
          </li>
        ))}
      </ul>
      
      <p style={{ marginTop: '16px', color: '#666' }}>
        已完成：{completedCount} / {todos.length}
      </p>
    </div>
  )
}

export default ReactTodoApp
```

```js
import { useState } from 'react'

function ComponentBuilder() {
  const [components, setComponents] = useState([])
  const [selectedComponent, setSelectedComponent] = useState(null)
  
  const addComponent = (type) => {
    const newComponent = {
      id: Date.now(),
      type,
      props: getDefaultProps(type)
    }
    setComponents([...components, newComponent])
  }
  
  const updateComponent = (id, props) => {
    setComponents(components.map(comp =>
      comp.id === id ? { ...comp, props } : comp
    ))
  }
  
  const getDefaultProps = (type) => {
    switch (type) {
      case 'button':
        return { text: '按钮', color: '#007bff' }
      case 'input':
        return { placeholder: '输入框' }
      case 'text':
        return { content: '文本内容' }
      default:
        return {}
    }
  }
  
  const renderComponent = (component) => {
    const { type, props } = component
    
    switch (type) {
      case 'button':
        return (
          <button
            key={component.id}
            style={{
              padding: '10px 20px',
              background: props.color,
              color: '#fff',
              border: 'none',
              borderRadius: '4px',
              cursor: 'pointer'
            }}
          >
            {props.text}
          </button>
        )
      case 'input':
        return (
          <input
            key={component.id}
            type="text"
            placeholder={props.placeholder}
            style={{
              padding: '8px',
              border: '1px solid #ddd',
              borderRadius: '4px',
              width: '200px'
            }}
          />
        )
      case 'text':
        return (
          <p key={component.id} style={{ padding: '8px' }}>
            {props.content}
          </p>
        )
      default:
        return null
    }
  }
  
  return (
    <div style={{ display: 'flex', height: '100vh' }}>
      <div style={{ 
        width: '250px', 
        padding: '20px', 
        background: '#f5f5f5',
        borderRight: '1px solid #ddd'
      }}>
        <h3>组件库</h3>

          <button 
            onClick={() => addComponent('button')}

            + 按钮

          <button 
            onClick={() => addComponent('input')}
            style={{ padding: '8px' }}
          >
            + 输入框
```

```js
function ComponentNode({ name, children, depth = 0 }) {
  const [isExpanded, setIsExpanded] = useState(true)

    <div style={{ marginLeft: `${depth * 20}px` }}>
      <div 
        style={{ 
          padding: '4px 8px',
          cursor: children && children.length > 0 ? 'pointer' : 'default',
          display: 'flex',

          gap: '8px'
        }}
```
