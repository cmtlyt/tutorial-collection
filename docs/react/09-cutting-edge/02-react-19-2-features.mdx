# React 19.2 新特性

> **版本**：v1.0

## 📚 概念介绍

React 19.2 是 React 19 的一个重要更新版本，引入了许多令人兴奋的新特性和性能优化。这些特性主要集中在性能追踪、状态管理、服务端渲染改进等方面，为开发者提供了更强大的工具来构建高性能的 React 应用。

## 🎯 Activity 组件

`Activity` 组件是 React 19.2 中最重要的新特性之一，它解决了隐藏 UI 时保持状态的核心问题。`Activity` 组件允许你隐藏子组件的 UI 和内部状态，同时保持状态不丢失，这对于性能优化和用户体验提升具有重要意义。

### 基础用法

```js
import { useState } from 'react'

function SidebarContent() {
  const [expandedSections, setExpandedSections] = useState([1])
  const [draftText, setDraftText] = useState('')
  
  const toggleSection = (sectionId) => {
    setExpandedSections(prev => 
      prev.includes(sectionId) 
        ? prev.filter(id => id !== sectionId)
        : [...prev, sectionId]
    )
  }
  
  return (
    <div style={{ padding: '20px', background: '#f5f5f5' }}>
      <h3>侧边栏内容</h3>
      
      <div style={{ marginBottom: '16px' }}>
        <button onClick={() => toggleSection(1)}>
          {expandedSections.includes(1) ? '折叠' : '展开'} 第一部分
        </button>
        {expandedSections.includes(1) && (
          <div style={{ padding: '10px', marginTop: '8px', background: '#fff' }}>
            <p>这是第一部分的内容</p>
          </div>
        )}
      </div>
      
      <div style={{ marginBottom: '16px' }}>
        <button onClick={() => toggleSection(2)}>
          {expandedSections.includes(2) ? '折叠' : '展开'} 第二部分
        </button>
        {expandedSections.includes(2) && (
          <div style={{ padding: '10px', marginTop: '8px', background: '#fff' }}>
            <p>这是第二部分的内容</p>
          </div>
        )}
      </div>
      
      <div>
        <label>
          草稿内容：
          <textarea 
            value={draftText}
            onChange={(e) => setDraftText(e.target.value)}
            placeholder="输入内容..."
            style={{ width: '100%', minHeight: '80px', marginTop: '8px' }}
          />
        </label>
      </div>
    </div>
  )
}

function App() {
  const [showSidebar, setShowSidebar] = useState(true)
  
  return (
    <div style={{ display: 'flex', height: '100vh' }}>
      {showSidebar && (
        <aside style={{ width: '300px', borderRight: '1px solid #ddd' }}>
          <SidebarContent />
        </aside>
      )}
      
      <main style={{ flex: 1, padding: '20px' }}>
        <h1>主内容区域</h1>

          style={{ padding: '10px 20px' }}
        >
          {showSidebar ? '隐藏侧边栏' : '显示侧边栏'}
        </button>
      </main>
    </div>
  )
}
```

```js
import { useState, useEffect } from 'react'
```

## 🎯 useEffectEvent Hook

`useEffectEvent` 是 React 19.2 引入的新 Hook，用于分离 Effect 中的事件逻辑，避免因事件处理函数变化导致 Effect 重新运行。

### 基础用法

```js
import { useState, useEffect } from 'react'

function ChatRoom({ roomId }) {
  const [messages, setMessages] = useState([])
  const [isOnline, setIsOnline] = useState(false)
  
  const onConnect = useEffectEvent(() => {
    console.log(`已连接到房间 ${roomId}`)
    setIsOnline(true)
  })
  
  const onMessage = useEffectEvent((message) => {
    console.log(`收到消息：${message}`)
    setMessages(prev => [...prev, message])
  })
  
  useEffect(() => {
    const connection = createChatConnection(roomId)
    
    connection.on('connect', onConnect)
    connection.on('message', onMessage)
    
    connection.connect()
    
    return () => {
      connection.disconnect()
      setIsOnline(false)
    }
  }, [roomId])
  
  return (
    <div style={{ padding: '20px' }}>
      <h2>聊天室：{roomId}</h2>
      
      <div style={{ 
        marginBottom: '16px', 
        padding: '8px',
        background: isOnline ? '#d4edda' : '#f8d7da',
        borderRadius: '4px'
      }}>
        状态：{isOnline ? '在线' : '离线'}
      </div>
      
      <div style={{ 
        height: '300px', 
        overflowY: 'auto',
        border: '1px solid #ddd',
        padding: '10px',
        marginBottom: '16px'
      }}>
        {messages.length === 0 ? (
          <p style={{ color: '#999' }}>暂无消息</p>
        ) : (
          messages.map((msg, index) => (
            <div key={index} style={{ marginBottom: '8px' }}>
              <strong>{msg.sender}：</strong>{msg.text}
            </div>
          ))
        )}
      </div>
      
      <input 
        type="text" 
        placeholder="输入消息..."
        style={{ 
          width: 'calc(100% - 80px)', 
          padding: '8px',
          marginRight: '8px'
        }} 
      />
      <button style={{ padding: '8px 16px' }}>发送</button>
    </div>
  )
}

function createChatConnection(roomId) {
  let listeners = {}
  
  return {
    connect() {
      setTimeout(() => {
        if (listeners.connect) {
          listeners.connect()
        }
        
        setTimeout(() => {
          if (listeners.message) {
            listeners.message({ sender: '系统', text: `欢迎来到房间 ${roomId}！` })
          }
        }, 500)
      }, 1000)
    },
    
    disconnect() {
      listeners = {}
    },
    
    on(event, callback) {
      listeners[event] = callback
    }
  }
}

export default ChatRoom
```

## 🎯 Performance Tracks

Performance Tracks 是 React 19.2 引入的性能追踪功能，它为 Chrome DevTools 提供了自定义的性能轨道，帮助开发者更好地分析和优化 React 应用的性能。

### Scheduler 轨道

Scheduler 轨道显示了 React 调度器的工作情况，包括不同优先级的任务执行。

```js
import { useState, useTransition, startTransition } from 'react'

function HeavyComputation({ iterations }) {
  const [result, setResult] = useState(null)
  const [isComputing, setIsComputing] = useState(false)
  
  const compute = () => {
    setIsComputing(true)
    
    startTransition(() => {
      const start = performance.now()
      let sum = 0
      
      for (let i = 0; i < iterations; i++) {
        sum += Math.sqrt(i) * Math.sin(i)
      }
      
      const end = performance.now()
      
      setResult({
        sum: sum.toFixed(2),
        time: (end - start).toFixed(2)
      })
      setIsComputing(false)
    })
  }
  
  return (
    <div style={{ padding: '16px', border: '1px solid #ddd', margin: '8px 0' }}>
      <h3>重计算任务（{iterations} 次迭代）</h3>
      
      <button 
        onClick={compute}
        disabled={isComputing}
        style={{ padding: '8px 16px' }}
      >
        {isComputing ? '计算中...' : '开始计算'}
      </button>
      
      {result && (
        <div style={{ marginTop: '16px' }}>
          <p>结果：{result.sum}</p>
          <p>耗时：{result.time}ms</p>
        </div>
      )}
    </div>
  )
}

function SchedulerDemo() {
  const [showHeavyTasks, setShowHeavyTasks] = useState(false)
  
  return (
    <div style={{ padding: '20px' }}>
      <h1>Performance Tracks - Scheduler 轨道演示</h1>
      
      <p>

        然后点击下面的按钮执行不同优先级的任务。

      </p>
```

```js
    }, 500)
    
    return () => clearTimeout(timer)
  }, [value, onUpdate])
  
  const startTime = performance.now()
  
  for (let i = 0; i < 10000; i++) {
    Math.sqrt(i) * Math.sin(i)
  }
  
  const endTime = performance.now()

  )
})
```

## 🎯 部分预渲染

部分预渲染（Partial Pre-rendering）允许你预渲染应用的静态部分，然后动态恢复渲染动态内容，这对于提升首屏加载性能非常有用。

## 🎯 Web Streams 支持

React 19.2 增强了对 Web Streams 的支持，特别是在 Node.js 环境中，提供了更好的流式渲染能力。

### 流式数据渲染

```js
import { useState, useEffect } from 'react'

function createDataStream(messageCount, delay) {
  return new ReadableStream({
    async start(controller) {
      for (let i = 1; i <= messageCount; i++) {
        await new Promise(resolve => setTimeout(resolve, delay))
        
        const message = {
          id: i,
          text: `消息 ${i}`,
          timestamp: new Date().toISOString()
        }
        
        controller.enqueue(JSON.stringify(message) + '\n')
      }
      
      controller.close()
    }
  })
}

function StreamReader({ stream }) {
  const [messages, setMessages] = useState([])
  const [isComplete, setIsComplete] = useState(false)
  
  useEffect(() => {
    let cancelled = false
    
    const readStream = async () => {
      const reader = stream.getReader()
      const decoder = new TextDecoder()
      
      try {
        while (true) {
          const { done, value } = await reader.read()
          
          if (done || cancelled) {
            setIsComplete(true)
            break
          }
          
          const text = decoder.decode(value, { stream: true })
          const lines = text.split('\n').filter(line => line.trim())
          
          lines.forEach(line => {
            try {
              const message = JSON.parse(line)
              setMessages(prev => [...prev, message])
            } catch (e) {
              console.error('解析消息失败:', e)
            }
          })
        }
      } catch (error) {
        console.error('读取流失败:', error)
      }
    }
    
    readStream()
    
    return () => {
      cancelled = true
    }
  }, [stream])
  
  return (
    <div>
      <div style={{ marginBottom: '16px' }}>
        <strong>状态：</strong>
        {isComplete ? '完成' : '接收中...'}
        <strong>（已接收 {messages.length} 条消息）</strong>
      </div>
      
      <div style={{ 
        height: '300px', 
        overflowY: 'auto',
        border: '1px solid #ddd',
        padding: '10px',
        background: '#f9f9f9'
      }}>
        {messages.length === 0 ? (
          <p style={{ color: '#999' }}>等待消息...</p>
        ) : (
          messages.map(message => (
            <div key={message.id} style={{ 
              padding: '8px', 
              marginBottom: '8px',
              background: '#fff',
              borderRadius: '4px'
            }}>
              <strong>{message.text}</strong>
              <div style={{ fontSize: '12px', color: '#666' }}>
                {new Date(message.timestamp).toLocaleTimeString()}
              </div>
            </div>
          ))
        )}
      </div>
    </div>
  )
}

function WebStreamsDemo() {
  const [stream, setStream] = useState(null)
  const [messageCount, setMessageCount] = useState(10)
  const [delay, setDelay] = useState(500)
  
  const startStream = () => {
    const newStream = createDataStream(parseInt(messageCount), parseInt(delay))
    setStream(newStream)
  }
  
  return (
    <div style={{ padding: '20px' }}>
      <h1>Web Streams 支持演示</h1>
      
      <div style={{ marginBottom: '20px', padding: '16px', background: '#f0f0f0' }}>
        <h3>流配置</h3>
        <div style={{ marginBottom: '16px' }}>
          <label>
            消息数量：
            <input 
              type="number" 
              value={messageCount}
              onChange={(e) => setMessageCount(e.target.value)}
              min="1"
              max="100"
              style={{ marginLeft: '8px', padding: '4px' }}
            />
          </label>
        </div>
        
        <div style={{ marginBottom: '16px' }}>
          <label>
            延迟（毫秒）：
            <input 
              type="number" 
              value={delay}
              onChange={(e) => setDelay(e.target.value)}
              min="100"
              max="5000"
              style={{ marginLeft: '8px', padding: '4px' }}
            />
          </label>
        </div>
        
        <button 
          onClick={startStream}
          style={{ padding: '10px 20px' }}
        >
          开始流式传输
        </button>
      </div>
      
      {stream && (
        <div style={{ padding: '16px', border: '1px solid #ddd', borderRadius: '8px' }}>
          <h3>流式数据</h3>
          <StreamReader stream={stream} />
        </div>
      )}
    </div>
  )
}
```
