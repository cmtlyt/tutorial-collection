# Activity 组件

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

Activity 组件是 React 19.2 引入的新特性，用于优化长列表和频繁更新的性能。它提供了一种更高效的方式来处理大量数据渲染，减少不必要的重新计算和渲染。本章节将详细介绍 Activity 组件的使用方法和最佳实践。

## 🎯 Activity 组件基础

### 基本用法

```jsx
import React from 'react';

function ActivityComponent() {
  const [items, setItems] = React.useState(
    Array.from({ length: 1000 }, (_, i) => ({
      id: i,
      name: `Item ${i + 1}`,
      value: Math.random()
    }))
  );
  
  const [filter, setFilter] = React.useState('');
  
  // 使用 Activity 组件优化列表渲染
  const renderItem = (item) => (
    <div key={item.id} style={{ padding: '8px', borderBottom: '1px solid #eee' }}>
      <h3>{item.name}</h3>
      <p>Value: {item.value.toFixed(2)}</p>
    </div>
  );
  
  const filteredItems = items.filter(item =>
    item.name.toLowerCase().includes(filter.toLowerCase())
  );
  
  return (
    <div>
      <input
        type="text"
        value={filter}
        onChange={(e) => setFilter(e.target.value)}
        placeholder="Filter items..."
        style={{ width: '100%', padding: '8px', marginBottom: '16px' }}
      />
      
      <div style={{ maxHeight: '500px', overflowY: 'auto' }}>
        {filteredItems.map(renderItem)}
      </div>
    </div>
  );
}

export default ActivityComponent;
```

```jsx
import React from 'react';

function OptimizedList() {
  const [items, setItems] = React.useState(
    Array.from({ length: 5000 }, (_, i) => ({
      id: i,
      name: `Item ${i + 1}`,
      value: Math.random(),
      category: ['A', 'B', 'C'][i % 3]
    }))
  );
  
  const [sortBy, setSortBy] = React.useState('name');
  const [sortOrder, setSortOrder] = React.useState('asc');
  
  // 排序逻辑
  const sortedItems = React.useMemo(() => {
    return [...items].sort((a, b) => {
      const comparison = a[sortBy] > b[sortBy] ? 1 : -1;
      return sortOrder === 'asc' ? comparison : -comparison;
    });
  }, [items, sortBy, sortOrder]);
  
  // 使用 Activity 组件渲染
  return (
    <div>
      <div style={{ marginBottom: '16px' }}>
        <label>Sort by:</label>
        <select value={sortBy} onChange={(e) => setSortBy(e.target.value)}>
          <option value="name">Name</option>
          <option value="value">Value</option>
          <option value="category">Category</option>
        </select>
        
        <button onClick={() => setSortOrder(sortOrder === 'asc' ? 'desc' : 'asc')}>
          {sortOrder === 'asc' ? '↑' : '↓'}
        </button>
      </div>
      
      <div style={{ maxHeight: '600px', overflowY: 'auto' }}>

          <div key={item.id} style={{ padding: '8px', borderBottom: '1px solid #eee' }}>

            <p>Value: {item.value.toFixed(2)}</p>
            <p>Category: {item.category}</p>
          </div>
        ))}
      </div>
    </div>
  );
}
```

## 🎯 性能优化场景

### 频繁更新的列表

```jsx
import React from 'react';

function FrequentlyUpdatedList() {
  const [items, setItems] = React.useState(
    Array.from({ length: 100 }, (_, i) => ({
      id: i,
      name: `Item ${i + 1}`,
      value: Math.random(),
      lastUpdated: Date.now()
    }))
  );
  
  // 定期更新数据
  React.useEffect(() => {
    const interval = setInterval(() => {
      setItems(prevItems => {
        const randomIndex = Math.floor(Math.random() * prevItems.length);
        return prevItems.map((item, index) =>
          index === randomIndex
            ? { ...item, value: Math.random(), lastUpdated: Date.now() }
            : item
        );
      });
    }, 1000);
    
    return () => clearInterval(interval);
  }, []);
  
  return (
    <div>
      <h1>Frequently Updated List</h1>
      <div style={{ maxHeight: '500px', overflowY: 'auto' }}>
        {items.map(item => (
          <div key={item.id} style={{ padding: '8px', borderBottom: '1px solid #eee' }}>
            <h3>{item.name}</h3>
            <p>Value: {item.value.toFixed(2)}</p>
            <small>Last updated: {new Date(item.lastUpdated).toLocaleTimeString()}</small>
          </div>
        ))}
      </div>
    </div>
  );
}

export default FrequentlyUpdatedList;
```

### 复杂计算优化

```jsx
import React from 'react';

function ComplexCalculations() {
  const [numbers, setNumbers] = React.useState(
    Array.from({ length: 1000 }, (_, i) => i)
  );
  
  const [operation, setOperation] = React.useState('square');
  
  // 使用 useMemo 缓存计算结果
  const results = React.useMemo(() => {
    console.log('Calculating results...');
    return numbers.map(num => {
      switch (operation) {
        case 'square':
          return num * num;
        case 'cube':
          return num * num * num;
        case 'sqrt':
          return Math.sqrt(num);
        default:
          return num;
      }
    });
  }, [numbers, operation]);
  
  return (
    <div>
      <h1>Complex Calculations</h1>
      
      <div style={{ marginBottom: '16px' }}>
        <label>Operation:</label>
        <select value={operation} onChange={(e) => setOperation(e.target.value)}>
          <option value="square">Square (x²)</option>
          <option value="cube">Cube (x³)</option>
          <option value="sqrt">Square Root (√x)</option>
        </select>
      </div>
      
      <div style={{ maxHeight: '500px', overflowY: 'auto' }}>
        {results.slice(0, 100).map((result, index) => (
          <div key={index} style={{ padding: '4px', borderBottom: '1px solid #eee' }}>
            {index} → {result.toFixed(2)}
          </div>
        ))}
      </div>
    </div>
  );
}

export default ComplexCalculations;
```

### Activity + 虚拟化

```jsx
import React from 'react';

function VirtualizedActivityList() {
  const [items, setItems] = React.useState(
    Array.from({ length: 10000 }, (_, i) => ({
      id: i,
      name: `Item ${i + 1}`,
      value: Math.random()
    }))
  );
  
  const [scrollTop, setScrollTop] = React.useState(0);
  const itemHeight = 50;
  const containerHeight = 500;
  
  const visibleStart = Math.floor(scrollTop / itemHeight);
  const visibleEnd = Math.min(
    visibleStart + Math.ceil(containerHeight / itemHeight) + 1,
    items.length
  );
  
  const visibleItems = items.slice(visibleStart, visibleEnd);
  
  return (
    <div>
      <h1>Virtualized Activity List</h1>
      
      <div
        style={{
          height: containerHeight,
          overflowY: 'auto',
          border: '1px solid #ddd'
        }}
        onScroll={(e) => setScrollTop(e.target.scrollTop)}
      >
        <div style={{ height: items.length * itemHeight, position: 'relative' }}>
          {visibleItems.map((item, index) => (
            <div
              key={item.id}
              style={{
                position: 'absolute',
                top: (visibleStart + index) * itemHeight,
                height: itemHeight,
                padding: '8px',
                borderBottom: '1px solid #eee'
              }}
            >
              <h3>{item.name}</h3>
              <p>Value: {item.value.toFixed(2)}</p>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

export default VirtualizedActivityList;
```

## 📝 实际应用示例

### 综合示例：优化的数据表格

```jsx
import React from 'react';

const TableCell = React.memo(function TableCell({ value }) {
  return <td style={{ padding: '8px', border: '1px solid #ddd' }}>{value}</td>;
});

const TableRow = React.memo(function TableRow({ row, columns, onRowClick }) {
  return (
    <tr
      onClick={() => onRowClick(row)}
      style={{ cursor: 'pointer' }}
    >
      {columns.map(column => (
        <TableCell key={column.key} value={row[column.key]} />
      ))}
    </tr>
  );
});

function OptimizedDataTable({ data, columns }) {
  const [sortBy, setSortBy] = React.useState(null);
  const [sortOrder, setSortOrder] = React.useState('asc');
  const [selectedRow, setSelectedRow] = React.useState(null);
  const [filter, setFilter] = React.useState('');
  
  // 过滤数据
  const filteredData = React.useMemo(() => {
    if (!filter) return data;
    
    return data.filter(row =>
      Object.values(row).some(value =>
        String(value).toLowerCase().includes(filter.toLowerCase())
      )
    );
  }, [data, filter]);
  
  // 排序数据
  const sortedData = React.useMemo(() => {
    if (!sortBy) return filteredData;
    
    return [...filteredData].sort((a, b) => {
      const comparison = a[sortBy] > b[sortBy] ? 1 : -1;
      return sortOrder === 'asc' ? comparison : -comparison;
    });
  }, [filteredData, sortBy, sortOrder]);
  
  // 处理排序
  const handleSort = React.useCallback((column) => {
    if (sortBy === column) {
      setSortOrder(sortOrder === 'asc' ? 'desc' : 'asc');
    } else {
      setSortBy(column);
      setSortOrder('asc');
    }
  }, [sortBy, sortOrder]);
  
  // 处理行点击
  const handleRowClick = React.useCallback((row) => {
    setSelectedRow(row);
  }, []);
  
  return (
    <div style={{ padding: '20px' }}>
      <h1>Optimized Data Table</h1>
      
      <div style={{ marginBottom: '16px' }}>
        <input
          type="text"
          value={filter}
          onChange={(e) => setFilter(e.target.value)}
          placeholder="Filter data..."
          style={{ width: '300px', padding: '8px' }}
        />
      </div>
      
      <table style={{ width: '100%', borderCollapse: 'collapse' }}>
        <thead>
          <tr>
            {columns.map(column => (
              <th
                key={column.key}
                onClick={() => handleSort(column.key)}
                style={{
                  padding: '8px',
                  border: '1px solid #ddd',
                  cursor: 'pointer',
                  background: sortBy === column.key ? '#f0f0f0' : 'white'
                }}
              >
                {column.label}
                {sortBy === column.key && (sortOrder === 'asc' ? ' ↑' : ' ↓')}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {sortedData.map(row => (
            <TableRow
              key={row.id}
              row={row}
              columns={columns}
              onRowClick={handleRowClick}
              style={{
                background: selectedRow?.id === row.id ? '#e6f7ff' : 'white'
              }}
            />
          ))}
        </tbody>
      </table>
      
      {selectedRow && (
        <div style={{ marginTop: '20px', padding: '16px', background: '#f0f0f0' }}>
          <h3>Selected Row:</h3>
          <pre>{JSON.stringify(selectedRow, null, 2)}</pre>
        </div>
      )}
    </div>
  );
}

function App() {
  const data = React.useMemo(() => {
    return Array.from({ length: 1000 }, (_, i) => ({
      id: i,
      name: `User ${i + 1}`,
      email: `user${i + 1}@example.com`,
      age: 20 + (i % 50),
      score: Math.random() * 100
    }));
  }, []);
  
  const columns = React.useMemo(() => [
    { key: 'id', label: 'ID' },
    { key: 'name', label: 'Name' },
    { key: 'email', label: 'Email' },
    { key: 'age', label: 'Age' },
    { key: 'score', label: 'Score' }
  ], []);
  
  return (
    <OptimizedDataTable data={data} columns={columns} />
  );
}

export default App;
```
