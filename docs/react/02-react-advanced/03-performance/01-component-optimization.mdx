# 组件优化

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

组件优化是提升 React 应用性能的关键。通过合理使用 React.memo、useMemo、useCallback 等工具，可以避免不必要的重新渲染，提高应用性能。本章节将详细介绍组件优化的各种技巧。

## 🎯 React.memo

### 基本用法

```jsx
import React from 'react';

// 普通组件
function RegularComponent({ name }) {
  console.log('RegularComponent rendered');
  return <div>Hello, {name}!</div>;
}

// 使用 React.memo 优化
const MemoizedComponent = React.memo(function MemoizedComponent({ name }) {
  console.log('MemoizedComponent rendered');
  return <div>Hello, {name}!</div>;
});

function App() {
  const [count, setCount] = React.useState(0);
  const [name, setName] = React.useState('Alice');
  
  return (
    <div>
      <h1>Count: {count}</h1>
      <button onClick={() => setCount(count + 1)}>Increment</button>
      
      <h2>Regular Component:</h2>
      <RegularComponent name={name} />
      
      <h2>Memoized Component:</h2>
      <MemoizedComponent name={name} />
      
      <button onClick={() => setName(name === 'Alice' ? 'Bob' : 'Alice')}>
        Change Name
      </button>
    </div>
  );
}

export default App;
```

### 自定义比较函数

```jsx
import React from 'react';

const MemoizedComponent = React.memo(function MemoizedComponent({ user, settings }) {
  console.log('MemoizedComponent rendered');
  return (
    <div>
      <h2>{user.name}</h2>
      <p>Email: {user.email}</p>
      <p>Theme: {settings.theme}</p>
    </div>
  );
}, function arePropsEqual(prevProps, nextProps) {
  // 只在 user.id 或 settings.theme 变化时重新渲染
  return (
    prevProps.user.id === nextProps.user.id &&
    prevProps.settings.theme === nextProps.settings.theme
  );
});

function App() {
  const [user, setUser] = React.useState({
    id: 1,
    name: 'Alice',
    email: 'alice@example.com'
  });
  
  const [settings, setSettings] = React.useState({
    theme: 'light',
    language: 'en'
  });
  
  return (
    <div>
      <MemoizedComponent user={user} settings={settings} />
      
      <button onClick={() => setUser({ ...user, name: user.name === 'Alice' ? 'Bob' : 'Alice' })}>
        Change Name
      </button>
      
      <button onClick={() => setSettings({ ...settings, language: settings.language === 'en' ? 'zh' : 'en' })}>
        Change Language
      </button>
      
      <button onClick={() => setSettings({ ...settings, theme: settings.theme === 'light' ? 'dark' : 'light' })}>
        Change Theme
      </button>
    </div>
  );
}

export default App;
```

## 🎯 useMemo

### 缓存计算结果

```jsx
import React from 'react';

function ExpensiveCalculation({ numbers }) {
  // 使用 useMemo 缓存计算结果
  const sum = React.useMemo(() => {
    console.log('Calculating sum...');
    return numbers.reduce((acc, num) => acc + num, 0);
  }, [numbers]);
  
  const average = React.useMemo(() => {
    console.log('Calculating average...');
    return numbers.length > 0 ? sum / numbers.length : 0;
  }, [sum, numbers.length]);
  
  return (
    <div>
      <h2>Statistics</h2>
      <p>Sum: {sum}</p>
      <p>Average: {average}</p>
      <p>Count: {numbers.length}</p>
    </div>
  );
}

function App() {
  const [count, setCount] = React.useState(0);
  const [numbers, setNumbers] = React.useState([1, 2, 3, 4, 5]);
  
  return (
    <div>
      <h1>Count: {count}</h1>
      <button onClick={() => setCount(count + 1)}>Increment</button>
      
      <ExpensiveCalculation numbers={numbers} />
      
      <button onClick={() => setNumbers([...numbers, numbers.length + 1])}>
        Add Number
      </button>
    </div>
  );
}

export default App;
```

### 缓存对象和数组

```jsx
import React from 'react';

function UserCard({ user }) {
  // 缓存样式对象
  const cardStyle = React.useMemo(() => ({
    padding: '16px',
    margin: '8px',
    border: '1px solid #ddd',
    borderRadius: '8px',
    backgroundColor: user.isActive ? '#f0f8ff' : '#fff'
  }), [user.isActive]);
  
  // 缓存头像 URL
  const avatarUrl = React.useMemo(() => {
    return user.avatar || `https://api.dicebear.com/7.x/avataaars/svg?seed=${user.id}`;
  }, [user.avatar, user.id]);
  
  return (
    <div style={cardStyle}>
      <img src={avatarUrl} alt={user.name} style={{ width: '64px', height: '64px' }} />
      <h3>{user.name}</h3>
      <p>{user.email}</p>
    </div>
  );
}

function App() {
  const [users, setUsers] = React.useState([
    { id: 1, name: 'Alice', email: 'alice@example.com', isActive: true },
    { id: 2, name: 'Bob', email: 'bob@example.com', isActive: false }
  ]);
  
  return (
    <div>
      {users.map(user => (
        <UserCard key={user.id} user={user} />
      ))}
    </div>
  );
}

export default App;
```

## 🎯 useCallback

### 缓存函数

```jsx
import React from 'react';

function Button({ onClick, children }) {
  console.log('Button rendered');
  return <button onClick={onClick}>{children}</button>;
}

function App() {
  const [count, setCount] = React.useState(0);
  
  // 不使用 useCallback - 每次渲染都创建新函数
  const handleClickWithoutCallback = () => {
    setCount(count + 1);
  };
  
  // 使用 useCallback - 只在 count 变化时创建新函数
  const handleClickWithCallback = React.useCallback(() => {
    setCount(count + 1);
  }, [count]);
  
  // 使用 useCallback 和函数式更新 - 不依赖 count
  const handleClickFunctional = React.useCallback(() => {
    setCount(prev => prev + 1);
  }, []);
  
  return (
    <div>
      <h1>Count: {count}</h1>
      
      <Button onClick={handleClickWithoutCallback}>
        Without Callback
      </Button>
      
      <Button onClick={handleClickWithCallback}>
        With Callback
      </Button>
      
      <Button onClick={handleClickFunctional}>
        Functional Update
      </Button>
    </div>
  );
}

export default App;
```

### 与 React.memo 配合使用

```jsx
import React from 'react';

const MemoizedButton = React.memo(function MemoizedButton({ onClick, children }) {
  console.log('MemoizedButton rendered');
  return <button onClick={onClick}>{children}</button>;
});

function App() {
  const [count, setCount] = React.useState(0);
  const [otherState, setOtherState] = React.useState(false);
  
  // 使用 useCallback 避免不必要的重新渲染
  const handleClick = React.useCallback(() => {
    setCount(prev => prev + 1);
  }, []);
  
  return (
    <div>
      <h1>Count: {count}</h1>
      <MemoizedButton onClick={handleClick}>Increment</MemoizedButton>
      
      <button onClick={() => setOtherState(!otherState)}>
        Toggle Other State
      </button>
    </div>
  );
}

export default App;
```

## 🎯 列表优化

### 虚拟化长列表

```jsx
import React from 'react';

function VirtualizedList({ items, itemHeight, containerHeight }) {
  const [scrollTop, setScrollTop] = React.useState(0);
  
  const containerRef = React.useRef(null);
  
  const visibleStart = Math.floor(scrollTop / itemHeight);
  const visibleEnd = Math.min(
    visibleStart + Math.ceil(containerHeight / itemHeight) + 1,
    items.length
  );
  
  const visibleItems = items.slice(visibleStart, visibleEnd);
  
  const handleScroll = (e) => {
    setScrollTop(e.target.scrollTop);
  };
  
  return (
    <div
      ref={containerRef}
      style={{
        height: containerHeight,
        overflowY: 'auto',
        border: '1px solid #ddd'
      }}
      onScroll={handleScroll}
    >
      <div style={{ height: items.length * itemHeight, position: 'relative' }}>
        {visibleItems.map((item, index) => (
          <div
            key={item.id}
            style={{
              position: 'absolute',
              top: (visibleStart + index) * itemHeight,
              height: itemHeight,
              padding: '8px',
              borderBottom: '1px solid #eee'
            }}
          >
            {item.name}
          </div>
        ))}
      </div>
    </div>
  );
}

function App() {
  const items = React.useMemo(() => {
    return Array.from({ length: 10000 }, (_, i) => ({
      id: i,
      name: `Item ${i + 1}`
    }));
  }, []);
  
  return (
    <div>
      <h1>Virtualized List</h1>
      <VirtualizedList
        items={items}
        itemHeight={50}
        containerHeight={500}
      />
    </div>
  );
}

export default App;
```

## 🎯 代码分割

### React.lazy 和 Suspense

```jsx
import React, { lazy, Suspense } from 'react';

// 懒加载组件
const HeavyComponent = lazy(() => import('./HeavyComponent'));
const AnotherComponent = lazy(() => import('./AnotherComponent'));

function App() {
  const [showHeavy, setShowHeavy] = React.useState(false);
  const [showAnother, setShowAnother] = React.useState(false);
  
  return (
    <div>
      <h1>Code Splitting Example</h1>
      
      <button onClick={() => setShowHeavy(!showHeavy)}>
        Toggle Heavy Component
      </button>
      
      <button onClick={() => setShowAnother(!showAnother)}>
        Toggle Another Component
      </button>
      
      <Suspense fallback={<div>Loading component...</div>}>
        {showHeavy && <HeavyComponent />}
        {showAnother && <AnotherComponent />}
      </Suspense>
    </div>
  );
}

export default App;
```

## 📝 实际应用示例

### 综合示例：优化的列表应用

```jsx
import React, { lazy, Suspense, useCallback, useMemo } from 'react';

// 懒加载组件
const DetailView = lazy(() => import('./DetailView'));

const ListItem = React.memo(function ListItem({ item, onSelect }) {
  console.log(`ListItem ${item.id} rendered`);
  
  return (
    <div
      style={{
        padding: '12px',
        margin: '8px 0',
        border: '1px solid #ddd',
        borderRadius: '4px',
        cursor: 'pointer'
      }}
      onClick={() => onSelect(item)}
    >
      <h3>{item.name}</h3>
      <p>{item.description}</p>
    </div>
  );
});

function App() {
  const [selectedItem, setSelectedItem] = React.useState(null);
  const [filter, setFilter] = React.useState('');
  
  // 生成大量数据
  const items = useMemo(() => {
    return Array.from({ length: 1000 }, (_, i) => ({
      id: i,
      name: `Item ${i + 1}`,
      description: `Description for item ${i + 1}`,
      category: i % 3 === 0 ? 'A' : i % 3 === 1 ? 'B' : 'C'
    }));
  }, []);
  
  // 过滤数据
  const filteredItems = useMemo(() => {
    return items.filter(item =>
      item.name.toLowerCase().includes(filter.toLowerCase()) ||
      item.description.toLowerCase().includes(filter.toLowerCase())
    );
  }, [items, filter]);
  
  // 处理选择
  const handleSelectItem = useCallback((item) => {
    setSelectedItem(item);
  }, []);
  
  // 处理过滤
  const handleFilterChange = useCallback((e) => {
    setFilter(e.target.value);
  }, []);
  
  return (
    <div style={{ padding: '20px' }}>
      <h1>Optimized List Application</h1>
      
      <input
        type="text"
        placeholder="Filter items..."
        value={filter}
        onChange={handleFilterChange}
        style={{
          width: '100%',
          padding: '10px',
          marginBottom: '20px',
          fontSize: '16px'
        }}
      />
      
      <div style={{ display: 'flex', gap: '20px' }}>
        <div style={{ flex: 1 }}>
          <h2>Items ({filteredItems.length})</h2>
          <div style={{ maxHeight: '600px', overflowY: 'auto' }}>
            {filteredItems.map(item => (
              <ListItem
                key={item.id}
                item={item}
                onSelect={handleSelectItem}
              />
            ))}
          </div>
        </div>
        
        <div style={{ flex: 1 }}>
          <Suspense fallback={<div>Loading details...</div>}>
            {selectedItem && <DetailView item={selectedItem} />}
          </Suspense>
        </div>
      </div>
    </div>
  );
}

export default App;
```
