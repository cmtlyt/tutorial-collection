# React Router 基础

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

React Router 是 React 生态中最流行的路由库，用于在单页应用中管理导航和页面切换。它提供了声明式的路由配置、嵌套路由、动态路由等功能。本章节将详细介绍 React Router 的基础用法。

## 🎯 基础路由配置

### 安装和设置

```jsx
import { BrowserRouter, Routes, Route, Link } from 'react-router-dom';

// 主应用组件
function App() {
  return (
    <BrowserRouter>
      <div>
        <nav>
          <ul>
            <li>
              <Link to="/">Home</Link>
            </li>
            <li>
              <Link to="/about">About</Link>
            </li>
            <li>
              <Link to="/contact">Contact</Link>
            </li>
          </ul>
        </nav>
        
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/about" element={<About />} />
          <Route path="/contact" element={<Contact />} />
        </Routes>
      </div>
    </BrowserRouter>
  );
}

// 页面组件
function Home() {
  return <h1>Home Page</h1>;
}

function About() {
  return <h1>About Page</h1>;
}

function Contact() {
  return <h1>Contact Page</h1>;
}

export default App;
```

### 404 页面处理

```jsx
import { BrowserRouter, Routes, Route, Link } from 'react-router-dom';

function App() {
  return (
    <BrowserRouter>
      <div>
        <nav>
          <ul>
            <li><Link to="/">Home</Link></li>
            <li><Link to="/about">About</Link></li>
            <li><Link to="/contact">Contact</Link></li>

        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/about" element={<About />} />
          <Route path="/contact" element={<Contact />} />

        </Routes>
      </div>

  );

  return (
    <div>
      <h1>404 - Page Not Found</h1>
      <p>The page you are looking for does not exist.</p>
      <Link to="/">Go back to Home</Link>
    </div>
  );
}

export default App;
```

## 🎯 动态路由

### 路径参数

```jsx
import { BrowserRouter, Routes, Route, Link, useParams } from 'react-router-dom';

function App() {
  return (
    <BrowserRouter>
      <div>
        <nav>
          <ul>
            <li><Link to="/">Home</Link></li>
            <li><Link to="/users/1">User 1</Link></li>
            <li><Link to="/users/2">User 2</Link></li>
            <li><Link to="/users/3">User 3</Link></li>
          </ul>
        </nav>
        
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/users/:userId" element={<UserDetail />} />
        </Routes>
      </div>
    </BrowserRouter>
  );
}

function UserDetail() {
  const { userId } = useParams();
  
  return (
    <div>
      <h1>User Detail</h1>
      <p>User ID: {userId}</p>
    </div>
  );
}

export default App;
```

### 多个路径参数

```jsx
import { BrowserRouter, Routes, Route, Link, useParams } from 'react-router-dom';

        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/posts/:postId/comments/:commentId" element={<CommentDetail />} />

  );
}

function CommentDetail() {

  return (
    <div>
      <h1>Comment Detail</h1>
      <p>Post ID: {postId}</p>
      <p>Comment ID: {commentId}</p>
    </div>
  );
}

export default App;
```

## 🎯 编程式导航

### useNavigate Hook

```jsx
import { BrowserRouter, Routes, Route, Link, useNavigate } from 'react-router-dom';

function App() {
  return (
    <BrowserRouter>
      <div>
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/about" element={<About />} />
          <Route path="/contact" element={<Contact />} />
        </Routes>
      </div>
    </BrowserRouter>
  );
}

function Home() {
  const navigate = useNavigate();
  
  return (
    <div>
      <h1>Home Page</h1>
      <button onClick={() => navigate('/about')}>Go to About</button>
      <button onClick={() => navigate('/contact')}>Go to Contact</button>
      <button onClick={() => navigate(-1)}>Go Back</button>
    </div>
  );
}

function About() {
  const navigate = useNavigate();
  
  return (
    <div>
      <h1>About Page</h1>
      <button onClick={() => navigate('/')}>Go to Home</button>
    </div>
  );
}

function Contact() {
  const navigate = useNavigate();
  
  return (
    <div>
      <h1>Contact Page</h1>
      <button onClick={() => navigate('/')}>Go to Home</button>
    </div>
  );
}

export default App;
```

### 导航带参数

```jsx
import { BrowserRouter, Routes, Route, useNavigate } from 'react-router-dom';

function App() {
  return (
    <BrowserRouter>
      <div>
        <Routes>
          <Route path="/" element={<UserList />} />
          <Route path="/users/:userId" element={<UserDetail />} />
        </Routes>
      </div>
    </BrowserRouter>
  );
}

function UserList() {
  const navigate = useNavigate();
  
  const users = [
    { id: 1, name: 'Alice' },
    { id: 2, name: 'Bob' },
    { id: 3, name: 'Charlie' }
  ];
  
  return (
    <div>
      <h1>User List</h1>
      <ul>
        {users.map(user => (
          <li key={user.id}>
            <span>{user.name}</span>
            <button onClick={() => navigate(`/users/${user.id}`)}>
              View Details
            </button>
          </li>
        ))}
      </ul>
    </div>
  );
}

function UserDetail() {
  const { userId } = useParams();
  const navigate = useNavigate();
  
  return (
    <div>
      <h1>User Detail</h1>
      <p>User ID: {userId}</p>
      <button onClick={() => navigate('/')}>Back to List</button>
    </div>
  );
}

export default App;
```

## 🎯 路由守卫

### 认证守卫

```jsx
import { BrowserRouter, Routes, Route, Navigate, useNavigate } from 'react-router-dom';

// 模拟认证状态
function useAuth() {
  const [isAuthenticated, setIsAuthenticated] = React.useState(false);
  
  const login = () => setIsAuthenticated(true);
  const logout = () => setIsAuthenticated(false);
  
  return { isAuthenticated, login, logout };
}

// 私有路由组件
function PrivateRoute({ children }) {
  const { isAuthenticated } = useAuth();
  
  if (!isAuthenticated) {
    return <Navigate to="/login" />;
  }
  
  return children;
}

function App() {
  const { isAuthenticated, login, logout } = useAuth();
  
  return (
    <BrowserRouter>
      <div>
        <nav>
          <ul>
            <li><Link to="/">Home</Link></li>
            <li><Link to="/dashboard">Dashboard</Link></li>
            {isAuthenticated ? (
              <li><button onClick={logout}>Logout</button></li>
            ) : (
              <li><Link to="/login">Login</Link></li>
            )}
          </ul>
        </nav>
        
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/login" element={<Login onLogin={login} />} />
          <Route
            path="/dashboard"
            element={
              <PrivateRoute>
                <Dashboard />
              </PrivateRoute>
            }
          />
        </Routes>
      </div>
    </BrowserRouter>
  );
}

function Login({ onLogin }) {
  const navigate = useNavigate();
  
  const handleLogin = () => {
    onLogin();
    navigate('/dashboard');
  };
  
  return (
    <div>
      <h1>Login Page</h1>
      <button onClick={handleLogin}>Login</button>
    </div>
  );
}

function Dashboard() {
  return <h1>Dashboard (Protected)</h1>;
}

function Home() {
  return <h1>Home Page</h1>;
}

export default App;
```

## 📝 实际应用示例

### 综合示例：博客应用

```jsx
import { BrowserRouter, Routes, Route, Link, Outlet, useParams, useNavigate, useLocation } from 'react-router-dom';

// 模拟数据
const posts = [
  { id: 1, title: 'React Router Basics', content: 'Learn about React Router basics...' },
  { id: 2, title: 'Advanced Routing', content: 'Explore advanced routing features...' },
  { id: 3, title: 'Route Guards', content: 'Implement authentication and authorization...' }
];

function App() {
  return (
    <BrowserRouter>
      <div className="app">
        <Routes>
          <Route path="/" element={<Layout />}>
            <Route index element={<Home />} />
            <Route path="posts" element={<PostsLayout />}>
              <Route index element={<PostsList />} />
              <Route path=":postId" element={<PostDetail />} />
            </Route>
            <Route path="about" element={<About />} />
            <Route path="*" element={<NotFound />} />
          </Route>
        </Routes>
      </div>
    </BrowserRouter>
  );
}

function Layout() {
  const location = useLocation();
  
  return (
    <div className="layout">
      <header>
        <nav>
          <ul>
            <li>
              <Link to="/" className={location.pathname === '/' ? 'active' : ''}>
                Home
              </Link>
            </li>
            <li>
              <Link to="/posts" className={location.pathname.startsWith('/posts') ? 'active' : ''}>
                Posts
              </Link>
            </li>
            <li>
              <Link to="/about" className={location.pathname === '/about' ? 'active' : ''}>
                About
              </Link>
            </li>
          </ul>
        </nav>
      </header>
      
      <main>
        <Outlet />
      </main>
      
      <footer>
        <p>&copy; 2026 My Blog</p>
      </footer>
    </div>
  );
}

function Home() {
  const navigate = useNavigate();
  
  return (
    <div className="home">
      <h1>Welcome to My Blog</h1>
      <p>Explore our latest posts and articles.</p>
      <button onClick={() => navigate('/posts')}>View All Posts</button>
    </div>
  );
}

function PostsLayout() {
  return (
    <div className="posts-layout">
      <h1>Blog Posts</h1>
      <Outlet />
    </div>
  );
}

function PostsList() {
  return (
    <div className="posts-list">
      <ul>
        {posts.map(post => (
          <li key={post.id}>
            <Link to={`/posts/${post.id}`}>
              <h3>{post.title}</h3>
            </Link>
            <p>{post.content.substring(0, 100)}...</p>
          </li>
        ))}
      </ul>
    </div>
  );
}

function PostDetail() {
  const { postId } = useParams();
  const navigate = useNavigate();
  const post = posts.find(p => p.id === parseInt(postId));
  
  if (!post) {
    return <div>Post not found</div>;
  }
  
  return (
    <div className="post-detail">
      <button onClick={() => navigate('/posts')}>← Back to Posts</button>
      <h1>{post.title}</h1>
      <p>{post.content}</p>
    </div>
  );
}

function About() {
  return (
    <div className="about">
      <h1>About Us</h1>
      <p>We are a team of passionate developers sharing knowledge.</p>
    </div>
  );
}

function NotFound() {
  const navigate = useNavigate();
  
  return (
    <div className="not-found">
      <h1>404 - Page Not Found</h1>
      <p>The page you are looking for does not exist.</p>
      <button onClick={() => navigate('/')}>Go to Home</button>
    </div>
  );
}

export default App;
```
