# React 19.2 Performance Tracks

> **版本**: v1.0  
> **最后更新**: 2026-04-22

## 📚 概念介绍

React 19.2 引入了强大的 Performance Tracks 功能，与 Chrome DevTools Performance 面板深度集成。这个功能提供了前所未有的 React 应用性能可视化能力，包括 Scheduler 轨道、Components 轨道和 Server 轨道，帮助开发者深入理解应用的性能特征。

## 🎯 Performance Tracks 核心功能

### 1. Scheduler 轨道

Scheduler 轨道显示 React 的调度器工作情况，包括不同优先级的任务执行。

```jsx
// src/components/PerformanceScheduler.jsx
import { useState, useTransition, useDeferredValue } from 'react';

function PerformanceScheduler() {
  const [items, setItems] = useState([]);
  const [isPending, startTransition] = useTransition();
  const deferredItems = useDeferredValue(items);

  const addItems = useCallback(() => {
    const newItems = Array.from({ length: 1000 }, (_, i) => ({
      id: Date.now() + i,
      name: `Item ${items.length + i}`,
      value: Math.random() * 100
    }));

    // 使用 startTransition 标记非紧急更新
    startTransition(() => {
      setItems(prev => [...prev, ...newItems]);
    });
  }, [items.length]);

  const handleSearch = useCallback((query) => {
    // 搜索操作使用低优先级
    startTransition(() => {
      const filtered = items.filter(item => 
        item.name.toLowerCase().includes(query.toLowerCase())
      );
      setItems(filtered);
    });
  }, [items]);

  return (
    <div>
      <div style={{ marginBottom: '20px' }}>
        <button 
          onClick={addItems}
          disabled={isPending}
          style={{
            padding: '10px 20px',
            backgroundColor: '#1976d2',
            color: 'white',
            border: 'none',
            borderRadius: '4px',
            cursor: isPending ? 'not-allowed' : 'pointer',
            opacity: isPending ? 0.6 : 1
          }}
        >
          {isPending ? 'Adding items...' : 'Add 1000 Items'}
        </button>

        <input
          type="text"
          placeholder="Search items..."
          onChange={(e) => handleSearch(e.target.value)}
          style={{
            marginLeft: '10px',
            padding: '8px',
            fontSize: '16px'
          }}
        />
      </div>

      <div style={{ maxHeight: '400px', overflow: 'auto' }}>
        {deferredItems.map(item => (
          <div 
            key={item.id}
            style={{
              padding: '10px',
              borderBottom: '1px solid #eee',
              display: 'flex',
              justifyContent: 'space-between'
            }}
          >
            <span>{item.name}</span>
            <span>{item.value.toFixed(2)}</span>
          </div>
        ))}
      </div>

      {isPending && (
        <div style={{ 
          marginTop: '10px', 
          color: '#666',
          fontStyle: 'italic'
        }}>
          Processing updates...
        </div>
      )}
    </div>
  );
}

export default PerformanceScheduler;
```

### 2. Components 轨道

Components 轨道显示组件的渲染性能，包括渲染时间、Effect 执行时间等。

```jsx
// src/components/PerformanceProfiler.jsx
import { Profiler } from 'react';

function onRenderCallback(
  id,
  phase,
  actualDuration,
  baseDuration,
  startTime,
  commitTime
) {
  // 发送到分析服务
  if (typeof window !== 'undefined' && window.gtag) {
    window.gtag('event', 'component_render', {
      event_category: 'Performance',
      event_label: id,
      value: Math.round(actualDuration),
      custom_map: {
        phase,
        base_duration: Math.round(baseDuration),
        start_time: Math.round(startTime),
        commit_time: Math.round(commitTime)
      }
    });
  }

  // 记录到控制台（开发模式）
  if (process.env.NODE_ENV === 'development') {
    console.log(`${id} ${phase}:`, {
      actualDuration: `${actualDuration.toFixed(2)}ms`,
      baseDuration: `${baseDuration.toFixed(2)}ms`,
      startTime: `${startTime.toFixed(2)}ms`,
      commitTime: `${commitTime.toFixed(2)}ms`
    });
  }
}

export function PerformanceProfiler({ children, id }) {
  return (
    <Profiler id={id} onRender={onRenderCallback}>
      {children}
    </Profiler>
  );
}

// 使用示例
export function OptimizedComponent() {
  return (
    <PerformanceProfiler id="OptimizedComponent">
      <div>
        {/* 组件内容 */}
      </div>
    </PerformanceProfiler>
  );
}
```

```jsx
// src/hooks/useOptimizedEffect.jsx
import { useState, useEffect, useEffectEvent } from 'react';

function useOptimizedEffect(callback, deps) {
  const handleEvent = useEffectEvent(callback);

  useEffect(() => {
    handleEvent();
  }, deps);
}

// 使用示例：避免因事件处理函数变化导致 Effect 重新运行
export function ChatRoom({ roomId }) {
  const [messages, setMessages] = useState([]);

  const onMessage = useEffectEvent((msg) => {
    setMessages(prev => [...prev, msg]);
  });

  useEffect(() => {
    const connection = createChatConnection(roomId);
    connection.on('message', onMessage);
    connection.connect();

    return () => {
      connection.disconnect();
    };
  }, [roomId, onMessage]);

  return (
    <div>
      <h2>Chat Room: {roomId}</h2>
      <div style={{ maxHeight: '400px', overflow: 'auto' }}>
        {messages.map((msg, index) => (
          <div key={index} style={{ marginBottom: '10px' }}>
            <strong>{msg.user}:</strong> {msg.text}
          </div>
        ))}
      </div>
    </div>
  );
}

function createChatConnection(roomId) {
  return {

      console.log(`Connected to room ${roomId}`);
    },
    disconnect() {
      console.log(`Disconnected from room ${roomId}`);
    },

      console.log(`Listening for ${event} events`);
```

```jsx
// src/utils/performanceTracker.js
export class PerformanceTracker {
  constructor() {
    this.metrics = new Map();
    this.isEnabled = process.env.NODE_ENV === 'development';
  }

  // 标记组件渲染开始
  markComponentRender(componentName) {
    if (!this.isEnabled) return;

    const markName = `react-component-${componentName}-start`;
    performance.mark(markName);
  }

  // 标记组件渲染结束
  measureComponentRender(componentName) {
    if (!this.isEnabled) return;

    const startMark = `react-component-${componentName}-start`;
    const endMark = `react-component-${componentName}-end`;
    const measureName = `react-component-${componentName}`;

    performance.mark(endMark);
    performance.measure(measureName, startMark, endMark);

    const measure = performance.getEntriesByName(measureName)[0];
    
    this.metrics.set(componentName, {
      duration: measure.duration,
      startTime: measure.startTime,
      endTime: measure.endTime
    });

    // 清理 marks
    performance.clearMarks(startMark);
    performance.clearMarks(endMark);
  }

  // 标记调度器任务
  markSchedulerTask(taskName, priority) {
    if (!this.isEnabled) return;

    const markName = `react-scheduler-${taskName}-${priority}`;
    performance.mark(markName);
  }

  // 测量调度器任务
  measureSchedulerTask(taskName, priority) {
    if (!this.isEnabled) return;

    const startMark = `react-scheduler-${taskName}-${priority}-start`;
    const endMark = `react-scheduler-${taskName}-${priority}-end`;
    const measureName = `react-scheduler-${taskName}-${priority}`;

    performance.mark(endMark);
    performance.measure(measureName, startMark, endMark);

    const measure = performance.getEntriesByName(measureName)[0];
    
    return {
      taskName,
      priority,
      duration: measure.duration
    };
  }

  // 获取所有指标
  getMetrics() {
    return Object.fromEntries(this.metrics);
  }

  // 清除所有指标
  clearMetrics() {
    this.metrics.clear();
    performance.clearMarks();
    performance.clearMeasures();
  }

  // 导出性能报告
  exportReport() {
    const metrics = this.getMetrics();
    const report = {
      timestamp: Date.now(),
      metrics,
      summary: this.generateSummary(metrics)
    };

    return JSON.stringify(report, null, 2);
  }

  // 生成摘要
  generateSummary(metrics) {
    const entries = Object.entries(metrics);
    
    if (entries.length === 0) {
      return { totalComponents: 0, averageRenderTime: 0 };
    }

    const totalDuration = entries.reduce((sum, [, value]) => sum + value.duration, 0);
    const averageRenderTime = totalDuration / entries.length;
    
    const slowestComponent = entries.reduce((slowest, [name, value]) => 
      value.duration > slowest.duration ? { name, duration: value.duration } : slowest,
      { name: '', duration: 0 }
    );

    return {
      totalComponents: entries.length,
      totalDuration,
      averageRenderTime,
      slowestComponent
    };
  }
}

export const performanceTracker = new PerformanceTracker();
```

```jsx
// src/utils/devToolsIntegration.js
export class DevToolsIntegration {
  constructor() {
    this.isRecording = false;
    this.performanceData = [];
  }

  // 开始性能录制
  startRecording() {
    if (this.isRecording) return;

    this.isRecording = true;
    this.performanceData = [];

    // 清除之前的性能数据
    performance.clearMarks();
    performance.clearMeasures();

    // 开始录制
    if (window.performance && window.performance.mark) {
      performance.mark('recording-start');
    }
  }

  // 停止性能录制
  stopRecording() {
    if (!this.isRecording) return;

    this.isRecording = false;

    // 标记录制结束
    if (window.performance && window.performance.mark) {
      performance.mark('recording-end');
      performance.measure('recording-duration', 'recording-start', 'recording-end');
    }

    // 收集性能数据
    this.collectPerformanceData();

    return this.getReport();
  }

  // 收集性能数据
  collectPerformanceData() {
    // 收集所有 measures
    const measures = performance.getEntriesByType('measure');
    
    // 收集所有 marks
    const marks = performance.getEntriesByType('mark');
    
    // 收集所有资源加载
    const resources = performance.getEntriesByType('resource');
    
    // 收集所有导航
    const navigation = performance.getEntriesByType('navigation')[0];

    this.performanceData = {
      measures,
      marks,
      resources,
      navigation
    };
  }

  // 获取性能报告
  getReport() {
    if (this.performanceData.length === 0) {
      return null;
    }

    const { measures, marks, resources, navigation } = this.performanceData;

    return {
      timestamp: Date.now(),
      summary: {
        totalMeasures: measures.length,
        totalMarks: marks.length,
        totalResources: resources.length,
        pageLoadTime: navigation?.loadEventEnd - navigation?.navigationStart,
        domContentLoadedTime: navigation?.domContentLoadedEventEnd - navigation?.domContentLoadedEventStart
      },
      details: {
        measures: measures.map(m => ({
          name: m.name,
          duration: m.duration,
          startTime: m.startTime
        })),
        resources: resources.map(r => ({
          name: r.name,
          duration: r.duration,
          size: r.transferSize,
          type: r.initiatorType
        }))
      }
    };
  }

  // 导出为 Chrome DevTools 格式
  exportForDevTools() {
    const report = this.getReport();
    
    if (!report) return null;

    return {
      traceEvents: this.generateTraceEvents(report)
    };
  }

  // 生成 trace events
  generateTraceEvents(report) {
    const events = [];

    // 添加导航事件
    if (report.details.navigation) {
      events.push({
        name: 'Navigation',
        ph: 'X',
        ts: report.details.navigation.startTime * 1000,
        dur: report.details.navigation.duration * 1000,
        pid: 1,
        tid: 1
      });
    }

    // 添加资源加载事件
    report.details.resources.forEach(resource => {
      events.push({
        name: resource.name,
        ph: 'X',
        ts: resource.startTime * 1000,
        dur: resource.duration * 1000,
        pid: 1,
        tid: 1,
        args: {
          size: resource.size,
          type: resource.type
        }
      });
    });

    // 添加自定义测量事件
    report.details.measures.forEach(measure => {
      events.push({
        name: measure.name,
        ph: 'X',
        ts: measure.startTime * 1000,
        dur: measure.duration * 1000,
        pid: 1,
        tid: 1
      });
    });

    return events;
  }
}

export const devToolsIntegration = new DevToolsIntegration();
```

```jsx
// src/components/PerformanceOptimizedList.jsx
import { useState, useMemo, useCallback } from 'react';

function PerformanceOptimizedList({ items }) {
  const [filter, setFilter] = useState('');
  const [sortBy, setSortBy] = useState('name');

  // 使用 useMemo 缓存过滤和排序结果
  const filteredAndSortedItems = useMemo(() => {
    let result = [...items];

    // 过滤
    if (filter) {
      result = result.filter(item =>
        item.name.toLowerCase().includes(filter.toLowerCase())
      );
    }

    // 排序
    result.sort((a, b) => {
      if (sortBy === 'name') {
        return a.name.localeCompare(b.name);
      } else if (sortBy === 'value') {
        return a.value - b.value;
      }
      return 0;
    });

    return result;
  }, [items, filter, sortBy]);

  // 使用 useCallback 避免不必要的函数重新创建
  const handleFilterChange = useCallback((e) => {
    setFilter(e.target.value);
  }, []);

  const handleSortChange = useCallback((e) => {
    setSortBy(e.target.value);
  }, []);

  return (
    <div>
      <div style={{ marginBottom: '20px' }}>
        <input
          type="text"
          placeholder="Filter items..."
          value={filter}
          onChange={handleFilterChange}
          style={{ padding: '8px', fontSize: '16px' }}
        />

        <select
          value={sortBy}
          onChange={handleSortChange}
          style={{ marginLeft: '10px', padding: '8px' }}
        >
          <option value="name">Sort by Name</option>
          <option value="value">Sort by Value</option>
        </select>
      </div>

      <ul style={{ margin: 0, padding: 0 }}>
        {filteredAndSortedItems.map(item => (
          <li 
            key={item.id}
            style={{
              padding: '10px',
              borderBottom: '1px solid #eee',
              display: 'flex',
              justifyContent: 'space-between'
            }}
          >
            <span>{item.name}</span>
            <span>{item.value.toFixed(2)}</span>
          </li>
        ))}
      </ul>
    </div>
  );
}

export default PerformanceOptimizedList;
```

## 🎯 重点提示

### Performance Tracks 优势
- **可视化**: 直观显示性能数据
- **实时**: 实时监控应用性能
- **详细**: 提供详细的性能指标
- **集成**: 与 Chrome DevTools 无缝集成

### 使用要点
- **开发模式**: 仅在开发和 profiling 构建中启用
- **性能影响**: profiling 会增加额外开销
- **数据分析**: 结合其他工具进行综合分析
- **持续优化**: 建立持续优化的流程

### React 19.2 新特性
- **Activity 组件**: 优化隐藏和显示组件的性能
- **useEffectEvent**: 分离 Effect 中的事件逻辑
- **性能追踪**: 更强大的性能分析能力
- **Chrome DevTools 集成**: 与 DevTools 深度集成
