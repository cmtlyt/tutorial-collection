# XSS 防护详解

> **版本**: v1.0  
> **最后更新**: 2026-04-22

## 📚 概念介绍

XSS（跨站脚本攻击）是一种常见的安全漏洞，攻击者通过在网页中注入恶意脚本来窃取用户数据或执行未授权操作。React 19 提供了内置的 XSS 防护机制，但开发者仍需了解额外的防护措施来确保应用安全。

## 🎯 React 内置防护

### 1. 自动转义

React 默认会转义 JSX 中的内容，这是最基本的 XSS 防护：

```jsx
// React 自动转义，安全
function SafeComponent({ userInput }) {
  return <div>{userInput}</div>;
}

// 即使输入包含恶意脚本
const maliciousInput = '<script>alert("XSS")</script>';
<SafeComponent userInput={maliciousInput} />;
// 输出: &lt;script&gt;alert("XSS")&lt;/script&gt;
```

### 2. dangerouslySetInnerHTML 谨慎使用

当需要渲染 HTML 时，必须先清理内容：

```jsx
import DOMPurify from 'dompurify';

function SafeHTMLRenderer({ htmlContent }) {

`;

  return <SafeHTMLRenderer htmlContent={userContent} />;
}

export default App;
```

```jsx
export function generateCSPNonce() {
  const array = new Uint8Array(16);
  window.crypto.getRandomValues(array);
  return Array.from(array, byte => byte.toString(16).padStart(2, '0')).join('');
}
```

### 2. URL 验证和清理

```jsx
// src/utils/urlSanitizer.js
export function sanitizeURL(url) {
  // 基本验证
  if (!url || typeof url !== 'string') {
    return '#';
  }

  // 移除危险协议
  const dangerousProtocols = ['javascript:', 'data:', 'vbscript:', 'file:'];
  const lowercasedURL = url.toLowerCase();

  for (const protocol of dangerousProtocols) {
    if (lowercasedURL.startsWith(protocol)) {
      return '#';
    }
  }

  // 验证 URL 格式
  try {
    const parsedURL = new URL(url, window.location.origin);
    
    // 只允许 http 和 https
    if (!['http:', 'https:'].includes(parsedURL.protocol)) {
      return '#';
    }

    // 验证域名
    const allowedDomains = [
      'example.com',
      'www.example.com',
      'cdn.example.com'
    ];

    if (!allowedDomains.includes(parsedURL.hostname)) {
      return '#';
    }

    return parsedURL.href;
  } catch {
    return '#';
  }
}

// 使用示例
function SafeLink({ href, children, ...props }) {
  const safeHref = sanitizeURL(href);

  return (
    <a 
      href={safeHref} 
      rel="noopener noreferrer"
      {...props}
    >
      {children}
    </a>
  );
}

function SafeImage({ src, alt, ...props }) {
  const safeSrc = sanitizeURL(src);

  return (
    <img 
      src={safeSrc} 
      alt={alt}
      {...props}
    />
  );
}

export { SafeLink, SafeImage };
```

```jsx
// src/utils/inputSanitizer.js
import DOMPurify from 'dompurify';

// 清理文本输入
export function sanitizeTextInput(input) {
  if (typeof input !== 'string') {
    return '';
  }

  // 移除 HTML 标签
  return input.replace(/<[^>]*>/g, '');
}

// 清理富文本输入
export function sanitizeRichText(input) {
  if (typeof input !== 'string') {
    return '';
  }

  return DOMPurify.sanitize(input, {
    ALLOWED_TAGS: [
      'p', 'br', 'strong', 'em', 'u', 'strike',
      'h1', 'h2', 'h3', 'h4', 'h5', 'h6',
      'ul', 'ol', 'li',
      'a', 'blockquote', 'code', 'pre'
    ],
    ALLOWED_ATTR: ['href', 'target', 'rel', 'class'],
    ALLOW_DATA_ATTR: false
  });
}

// 验证和清理 URL
export function validateAndSanitizeURL(url) {
  if (typeof url !== 'string') {
    return null;
  }

  try {
    const parsed = new URL(url);
    
    // 只允许特定协议
    if (!['http:', 'https:'].includes(parsed.protocol)) {
      return null;
    }

    // 移除危险字符
    const sanitized = parsed.href
      .replace(/[<>"'{}|\\^`[\]]/g, '');

    return sanitized;
  } catch {
    return null;
  }
}

// 验证和清理 CSS
export function sanitizeCSS(css) {
  if (typeof css !== 'string') {
    return '';
  }

  // 移除危险的表达式
  const dangerousPatterns = [
    /expression\s*\(/gi,
    /javascript\s*:/gi,
    /behavior\s*:/gi,
    /moz-binding\s*:/gi
  ];

  let sanitized = css;
  for (const pattern of dangerousPatterns) {
    sanitized = sanitized.replace(pattern, '');
  }

  return sanitized;
}
```

```jsx
    username: '',
    email: '',
    message: ''
  });
  const [errors, setErrors] = useState({});

  const handleChange = useCallback((field, value) => {
    setFormData(prev => ({
      ...prev,
      [field]: value
    }));

    // 清除字段错误

      ...prev,
      [field]: ''
    }));
  }, []);

  const validateField = useCallback((field, value) => {
    switch (field) {
      case 'username':
        if (!value.trim()) {
          return 'Username is required';
        }
        if (value.length < 3) {
          return 'Username must be at least 3 characters';
        }
        if (value.length > 20) {
          return 'Username must be less than 20 characters';
        }
        if (!/^[a-zA-Z0-9_]+$/.test(value)) {
          return 'Username can only contain letters, numbers, and underscores';
        }
        break;

      case 'email':
        if (!value.trim()) {
          return 'Email is required';
        }
        if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
          return 'Invalid email format';
        }
        break;

      case 'message':
        if (!value.trim()) {
          return 'Message is required';
        }
        if (value.length > 1000) {
          return 'Message must be less than 1000 characters';
        }
        break;

      default:
        break;
    }

    return '';

            borderRadius: '4px'
          }}
        />
        {errors.username && (
          <span style={{ color: '#e74c3c', fontSize: '12px' }}>
            {errors.username}
          </span>
        )}
      </div>

      <div style={{ marginBottom: '20px' }}>
        <label 
          htmlFor="email"
          style={{ display: 'block', marginBottom: '5px', fontWeight: 'bold' }}
        >
          Email
        </label>
        <input
          id="email"
          type="email"

          style={{
```

```jsx
import { useState, useCallback, useEffect } from 'react';
import DOMPurify from 'dompurify';

function SecureRichTextEditor({ initialValue = '', onSave }) {
  const [content, setContent] = useState(initialValue);
  const [isPreview, setIsPreview] = useState(false);

  const sanitizeContent = useCallback((html) => {
    return DOMPurify.sanitize(html, {
      ALLOWED_TAGS: [
        'p', 'br', 'strong', 'em', 'u', 'strike', 'sub', 'sup',
        'h1', 'h2', 'h3', 'h4', 'h5', 'h6',
        'ul', 'ol', 'li',
        'a', 'blockquote', 'code', 'pre',
        'table', 'thead', 'tbody', 'tr', 'th', 'td'
      ],
      ALLOWED_ATTR: ['href', 'target', 'rel', 'class', 'style'],
      ALLOW_DATA_ATTR: false,
      ADD_ATTR: ['target'],
      FORBID_TAGS: ['script', 'style', 'iframe', 'object', 'embed'],
      FORBID_ATTR: ['onerror', 'onload', 'onclick']
    });
  }, []);

  const handleContentChange = useCallback((e) => {
    const newContent = e.target.value;
    setContent(newContent);
  }, []);

    const sanitizedContent = sanitizeContent(content);
    onSave(sanitizedContent);

    setIsPreview(prev => !prev);
```

## 🎨 CSRF 防护

### CSRF Token 实现

```jsx
// src/utils/csrf.js
export function getCSRFToken() {
  // 从 cookie 中获取 CSRF token
  const cookies = document.cookie.split(';');
  for (const cookie of cookies) {
    const [name, value] = cookie.trim().split('=');
    if (name === 'csrf_token') {
      return decodeURIComponent(value);
    }
  }
  return null;
}

export function setCSRFToken(token) {
  // 设置 CSRF token 到 cookie
  document.cookie = `csrf_token=${encodeURIComponent(token)}; path=/; secure; samesite=strict`;
}

// 创建带有 CSRF token 的请求
export async function fetchWithCSRF(url, options = {}) {
  const csrfToken = getCSRFToken();

  const headers = {
    'Content-Type': 'application/json',
    'X-CSRF-Token': csrfToken,
    ...options.headers
  };

  const response = await fetch(url, {
    ...options,
    credentials: 'include',
    headers
  });

  // 检查响应中的新 CSRF token
  const newCSRFToken = response.headers.get('X-CSRF-Token');
  if (newCSRFToken) {
    setCSRFToken(newCSRFToken);
  }

  return response;
}

// 使用示例
function SecureForm() {
  const handleSubmit = async (formData) => {
    try {
      const response = await fetchWithCSRF('/api/submit', {
        method: 'POST',
        body: JSON.stringify(formData)
      });

      if (!response.ok) {
        throw new Error('Request failed');
      }

      const data = await response.json();
      console.log('Success:', data);
    } catch (error) {
      console.error('Error:', error);
    }
  };

  return (
    <form onSubmit={(e) => {
      e.preventDefault();
      const formData = new FormData(e.target);
      handleSubmit(Object.fromEntries(formData));
    }}>
      {/* 表单字段 */}
    </form>
  );
}

export { SecureForm };
```

### 实时安全监控

```jsx
// src/utils/securityMonitor.js
export class SecurityMonitor {
  constructor() {
    this.violations = [];
    this.isEnabled = process.env.NODE_ENV === 'production';
  }

  // 检测 XSS 攻击尝试
  detectXSSAttempt(input) {
    const xssPatterns = [
      /<script[^>]*>.*?<\/script>/gi,
      /javascript:/gi,
      /on\w+\s*=/gi,
      /<iframe[^>]*>/gi,
      /<object[^>]*>/gi,
      /<embed[^>]*>/gi,
      /<link[^>]*>/gi,
      /<meta[^>]*>/gi,
      /<style[^>]*>.*?<\/style>/gi,
      /eval\s*\(/gi,
      /expression\s*\(/gi
    ];

    for (const pattern of xssPatterns) {
      if (pattern.test(input)) {
        this.logViolation('XSS', {
          input,
          pattern: pattern.source
        });
        return true;
      }
    }

    return false;
  }

  // 检测可疑的 URL
  detectSuspiciousURL(url) {
    const suspiciousPatterns = [
      /javascript:/gi,
      /data:/gi,
      /vbscript:/gi,
      /file:/gi,
      /about:/gi,
      /chrome:/gi
    ];

    for (const pattern of suspiciousPatterns) {
      if (pattern.test(url)) {
        this.logViolation('Suspicious URL', {
          url,
          pattern: pattern.source
        });
        return true;
      }
    }

    return false;
  }

  // 记录安全违规
  logViolation(type, details) {
    const violation = {
      type,
      details,
      timestamp: Date.now(),
      url: window.location.href,
      userAgent: navigator.userAgent
    };

    this.violations.push(violation);

    // 发送到安全监控服务
    this.reportViolation(violation);

    // 在开发模式下显示警告
    if (process.env.NODE_ENV === 'development') {
      console.warn(`Security violation detected: ${type}`, details);
    }
  }

  // 报告违规到服务器
  async reportViolation(violation) {
    if (!this.isEnabled) return;

    try {
      await fetch('/api/security/violations', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(violation)
      });
    } catch (error) {
      console.error('Failed to report security violation:', error);
    }
  }

  // 获取所有违规记录
  getViolations() {
    return this.violations;
  }

  // 清除违规记录
  clearViolations() {
    this.violations = [];
  }
}

export const securityMonitor = new SecurityMonitor();
```

## 🔗 相关概念

## 🎯 重点提示

### XSS 防护原则
- **永远不要信任用户输入**: 对所有输入进行验证和清理
- **使用 React 自动转义**: 依赖 React 的内置防护
- **谨慎使用 dangerouslySetInnerHTML**: 必须先清理 HTML 内容
- **配置严格的 CSP**: 限制可以加载的资源

### React 19 安全特性
- **自动转义**: React 19 继续提供强大的自动转义功能
- **改进的类型安全**: TypeScript 支持更安全的类型检查
- **Server Actions**: 内置的安全机制防止 CSRF 攻击
- **更好的错误处理**: 改进的错误边界和错误报告

### 安全最佳实践
- **深度防御**: 使用多层防护策略
- **定期审计**: 定期进行安全审计和渗透测试
- **保持更新**: 及时更新依赖和安全补丁
- **安全监控**: 建立实时的安全监控系统
