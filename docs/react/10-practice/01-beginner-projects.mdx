## 🎯 项目一：Todo List 应用（使用 React 19 Actions）

### 项目概述

这是一个使用 React 19 Actions 构建的 Todo List 应用，展示了如何利用新的 Actions 系统简化表单处理和状态管理。

### 核心特性

- 使用 React 19 Actions 处理表单提交
- 集成 useActionState 管理表单状态
- 使用 useOptimistic 实现乐观更新
- 完整的 TypeScript 类型支持
- 响应式设计

### 完整代码实现

```tsx
// types.ts
export interface Todo {
  id: string;
  text: string;
  completed: boolean;
  createdAt: Date;
}

export interface TodoState {
  todos: Todo[];
  error: string | null;
  isPending: boolean;
}

// actions.ts
'use server';

import { revalidatePath } from 'next/cache';
import { Todo } from './types';

let todos: Todo[] = [
  { id: '1', text: '学习 React 19', completed: false, createdAt: new Date() },
  { id: '2', text: '实践 Actions 系统', completed: false, createdAt: new Date() },
];

export async function addTodo(formData: FormData): Promise<TodoState> {
  const text = formData.get('text') as string;
  
  if (!text || text.trim().length === 0) {
    return {
      todos,
      error: '请输入有效的任务内容',
      isPending: false,
    };
  }

  const newTodo: Todo = {
    id: Date.now().toString(),
    text: text.trim(),
    completed: false,
    createdAt: new Date(),
  };

  todos = [...todos, newTodo];
  revalidatePath('/');
  
  return {
    todos,
    error: null,
    isPending: false,
  };
}

export async function toggleTodo(formData: FormData): Promise<TodoState> {
  const id = formData.get('id') as string;
  
  todos = todos.map(todo =>
    todo.id === id ? { ...todo, completed: !todo.completed } : todo
  );
  
  revalidatePath('/');
  
  return {
    todos,
    error: null,
    isPending: false,
  };
}

export async function deleteTodo(formData: FormData): Promise<TodoState> {
  const id = formData.get('id') as string;
  
  todos = todos.filter(todo => todo.id !== id);
  revalidatePath('/');
  
  return {
    todos,
    error: null,
    isPending: false,
  };
}

// TodoForm.tsx
'use client';

import { useActionState } from 'react';

interface TodoFormProps {
  addTodoAction: (formData: FormData) => Promise<any>;
}

export function TodoForm({ addTodoAction }: TodoFormProps) {
  const [state, formAction, isPending] = useActionState(addTodoAction, {
    todos: [],
    error: null,
    isPending: false,
  });

  return (
    <form action={formAction} className="todo-form">
      <input
        type="text"
        name="text"
        placeholder="添加新任务..."
        disabled={isPending}
        className="todo-input"
      />
      <button
        type="submit"
        disabled={isPending}
        className="todo-button"
      >
        {isPending ? '添加中...' : '添加'}
      </button>
      {state.error && (
        <p className="error-message">{state.error}</p>
      )}
    </form>
  );
}

// TodoList.tsx
'use client';

import { useOptimistic } from 'react';
import { Todo } from './types';

interface TodoListProps {
  todos: Todo[];
  toggleTodoAction: (formData: FormData) => Promise<any>;
  deleteTodoAction: (formData: FormData) => Promise<any>;
}

export function TodoList({ todos, toggleTodoAction, deleteTodoAction }: TodoListProps) {
  const [optimisticTodos, addOptimisticTodo] = useOptimistic(
    todos,
    (state, newTodo: Todo) => [...state, newTodo]
  );

  return (
    <ul className="todo-list">
      {optimisticTodos.map(todo => (
        <li key={todo.id} className={`todo-item ${todo.completed ? 'completed' : ''}`}>
          <form action={toggleTodoAction}>
            <input type="hidden" name="id" value={todo.id} />
            <button type="submit" className="todo-checkbox">
              {todo.completed ? '✓' : '○'}
            </button>
          </form>
          <span className="todo-text">{todo.text}</span>
          <form action={deleteTodoAction}>
            <input type="hidden" name="id" value={todo.id} />
            <button type="submit" className="todo-delete">
              删除
            </button>
          </form>
        </li>
      ))}
    </ul>
  );
}
```

## 🎯 项目二：天气预报应用

### 项目概述

使用 React 19 构建的天气预报应用，展示数据获取、状态管理和错误处理的最佳实践。

### 完整代码实现

```tsx
// types.ts
export interface WeatherResponse {
  current: {
    temp_c: number;
    condition: {
      text: string;
    };
    humidity: number;
    wind_kph: number;
  };
  location: {
    name: string;
  };
}

// weather-service.ts
export async function getWeather(city: string): Promise<WeatherData> {
  // 使用 Open-Meteo 免费天气 API（无需 API Key）
  // 首先通过地理编码 API 获取城市坐标
  const geoResponse = await fetch(
    `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(city)}&count=1&language=zh&format=json`
  );

  if (!geoResponse.ok) {
    throw new Error('无法获取城市信息');
  }

  const geoData = await geoResponse.json();
  
  if (!geoData.results || geoData.results.length === 0) {
    throw new Error('找不到该城市');
  }

  const { latitude, longitude, name, country } = geoData.results[0];

  // 获取天气数据
  const weatherResponse = await fetch(
    `https://api.open-meteo.com/v1/forecast?latitude=${latitude}&longitude=${longitude}&current=temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m&timezone=auto`
  );

  if (!weatherResponse.ok) {
    throw new Error('无法获取天气数据');
  }

  const weatherData = await weatherResponse.json();

  // 将天气代码转换为描述
  const weatherCondition = getWeatherCondition(weatherData.current.weather_code);

  return {
    location: `${name}, ${country}`,
    temperature: weatherData.current.temperature_2m,
    condition: weatherCondition,
    humidity: weatherData.current.relative_humidity_2m,
```

```tsx
'use client';

import { useState, useMemo } from 'react';
import { ProductList } from './ProductList';
import { ShoppingCart } from './ShoppingCart';
import { products } from './data';
import { Product, CartItem } from './types';

export function ShopApp() {
  const [cart, setCart] = useState<CartItem[]>([]);

  const addToCart = (item: CartItem) => {
    setCart(prevCart => {
      const existingItem = prevCart.find(i => i.id === item.id);
      if (existingItem) {
        return prevCart.map(i =>
          i.id === item.id ? { ...i, quantity: i.quantity + 1 } : i
        );
      }
      return [...prevCart, item];
    });
  };

  const removeFromCart = (id: string) => {
    setCart(prevCart => prevCart.filter(item => item.id !== id));
  };

  const updateQuantity = (id: string, quantity: number) => {
    if (quantity < 1) return;
    setCart(prevCart =>
      prevCart.map(item =>
        item.id === id ? { ...item, quantity } : item
      )
    );
  };

  const cartItemCount = useMemo(() => 
    cart.reduce((sum, item) => sum + item.quantity, 0),
    [cart]
  );

  return (
    <div className="shop-app">
      <header className="shop-header">
        <h1>在线商城</h1>
        <div className="cart-indicator">
          🛒 {cartItemCount}
        </div>
      </header>
      <main className="shop-main">
        <ProductList products={products} onAddToCart={addToCart} />
        <ShoppingCart
          items={cart}
          onRemove={removeFromCart}
          onUpdateQuantity={updateQuantity}
        />
      </main>
    </div>
  );
}
```
