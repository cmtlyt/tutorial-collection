'use server';

import { revalidatePath } from 'next/cache';
import { FormData } from './types';

export async function registerUser(formData: FormData): Promise<{
  success: boolean;
  errors?: Record<string, string>;
}> {
  // 验证表单数据
  const errors: Record<string, string> = {};
  
  if (!formData.username || formData.username.length < 3) {
    errors.username = '用户名至少需要3个字符';
  }
  
  if (!formData.email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email)) {
    errors.email = '请输入有效的邮箱地址';
  }
  
  if (!formData.password || formData.password.length < 6) {
    errors.password = '密码至少需要6个字符';
  }
  
  if (formData.password !== formData.confirmPassword) {
    errors.confirmPassword = '两次输入的密码不一致';
  }
  
  if (Object.keys(errors).length > 0) {
    return { success: false, errors };
  }
  
  // 模拟数据库操作
  await new Promise(resolve => setTimeout(resolve, 1000));
  
  // 保存用户数据
  console.log('注册用户:', formData);
  
  revalidatePath('/');
  
  return { success: true };
}

  const initialState: FormState = {
    data: {
      username: '',
      email: '',
      password: '',
      confirmPassword: '',
      bio: '',
      interests: [],
    },
    errors: {},
    isSubmitting: false,
    isSuccess: false,

  const [state, formAction] = useActionState(registerUser, initialState);
  const [formData, setFormData] = useState<FormData>(initialState.data);

  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: value }));
  };

    setFormData(prev => ({
      ...prev,
      interests: prev.interests.includes(interest)
        ? prev.interests.filter(i => i !== interest)

    }));

  if (state.isSuccess) {

      <div className="success-message">
        <h2>🎉 注册成功！</h2>
        <p>欢迎加入我们！</p>
      </div>
    );
  }

  return (
    <form action={formAction} className="register-form">
      <div className="form-group">
        <label htmlFor="username">用户名</label>
        <input
          type="text"
          id="username"
          name="username"
          value={formData.username}
          onChange={handleInputChange}
          required
        />

          <span className="error">{state.errors.username}</span>

// types.ts

// ActivityTabs.tsx
'use client';

import { useState } from 'react';
import { Activity } from 'react';
import { Tab } from './types';

interface ActivityTabsProps {
  tabs: Tab[];
}

export function ActivityTabs({ tabs }: ActivityTabsProps) {
  const [activeTab, setActiveTab] = useState<string | null>(tabs[0]?.id || null);
  const [hiddenTabs, setHiddenTabs] = useState<string[]>([]);

  const handleTabClick = (tabId: string) => {
    if (hiddenTabs.includes(tabId)) {
      // 显示隐藏的标签页
      setHiddenTabs(prev => prev.filter(id => id !== tabId));
    }
    setActiveTab(tabId);
  };

  const handleTabClose = (tabId: string) => {
    setHiddenTabs(prev => [...prev, tabId]);
    if (activeTab === tabId) {
      const remainingTabs = tabs.filter(tab => tab.id !== tabId && !hiddenTabs.includes(tab.id));
      setActiveTab(remainingTabs[0]?.id || null);
    }
  };

  return (
    <div className="activity-tabs">
      <div className="tabs-header">
        {tabs.map(tab => (
          <div
            key={tab.id}
            className={`tab-header ${activeTab === tab.id ? 'active' : ''} ${hiddenTabs.includes(tab.id) ? 'hidden' : ''}`}
            onClick={() => handleTabClick(tab.id)}
          >
            <span>{tab.title}</span>
            <button
              className="close-button"
              onClick={(e) => {
                e.stopPropagation();
                handleTabClose(tab.id);
              }}
            >
              ×
            </button>
          </div>
        ))}
      </div>

      <div className="tabs-content">
        {tabs.map(tab => (
          <Activity
            key={tab.id}
            mode={hiddenTabs.includes(tab.id) ? 'hidden' : 'visible'}
          >
            <TabContent tab={tab} isActive={activeTab === tab.id} />
          </Activity>
        ))}
      </div>
    </div>
  );
}

function TabContent({ tab, isActive }: { tab: Tab; isActive: boolean }) {
  const [inputValue, setInputValue] = useState('');
  const [items, setItems] = useState<string[]>(['项目 1', '项目 2', '项目 3']);

  const addItem = () => {
    if (inputValue.trim()) {
      setItems([...items, inputValue.trim()]);
      setInputValue('');
    }
  };

  return (
    <div className={`tab-content ${isActive ? 'active' : ''}`}>
      <h2>{tab.title}</h2>
      <p>{tab.content}</p>
      
      <div className="interactive-demo">
        <h3>状态保持演示</h3>
        <p>这个输入框和列表的状态会在标签页切换时保持不变。</p>
        
        <div className="input-group">
          <input
            type="text"
            value={inputValue}
            onChange={(e) => setInputValue(e.target.value)}
            placeholder="添加新项目..."
          />
          <button onClick={addItem}>添加</button>
        </div>
        
        <ul className="items-list">
          {items.map((item, index) => (
            <li key={index}>{item}</li>
          ))}
        </ul>
      </div>
    </div>
  );
}

'use client';

import { useState } from 'react';
import { Activity } from 'react';

export function SidebarActivity() {
  const [isSidebarOpen, setIsSidebarOpen] = useState(true);
  const [searchTerm, setSearchTerm] = useState('');
  const [selectedItems, setSelectedItems] = useState<string[]>([]);

  const items = [
    '文档管理',
    '用户设置',
    '数据分析',
    '系统监控',
    '报告生成',
  ];

  const filteredItems = items.filter(item =>
    item.toLowerCase().includes(searchTerm.toLowerCase())
  );

  const toggleItem = (item: string) => {
    setSelectedItems(prev =>
      prev.includes(item)
        ? prev.filter(i => i !== item)
        : [...prev, item]
    );
  };

  return (
    <Activity mode={isSidebarOpen ? 'visible' : 'hidden'}>
      <aside className="sidebar">
        <div className="sidebar-header">
          <h2>侧边栏</h2>
          <button
            className="toggle-button"
            onClick={() => setIsSidebarOpen(!isSidebarOpen)}
          >
            {isSidebarOpen ? '◀' : '▶'}
          </button>
        </div>

        <div className="sidebar-content">
          <input
            type="text"
            value={searchTerm}
            onChange={(e) => setSearchTerm(e.target.value)}
            placeholder="搜索..."
            className="search-input"
          />

import { useState, useEffect } from 'react';
import { products } from './data';
import { Product } from './types';

export function DynamicProductList() {
  const [loadedProducts, setLoadedProducts] = useState<Product[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // 模拟动态加载
    const timer = setTimeout(() => {
      setLoadedProducts(products);
      setLoading(false);
    }, 1000);

    return () => clearTimeout(timer);
  }, []);

  return (
    <section className="dynamic-products">
      <h2>动态产品列表</h2>
      
      {loading ? (
        <div className="loading-state">
          <p>加载产品中...</p>

        <div className="products-grid">
