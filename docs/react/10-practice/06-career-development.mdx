export interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {

  size?: 'sm' | 'md' | 'lg';
  loading?: boolean;
}

export const Button = forwardRef<HTMLButtonElement, ButtonProps>(
  ({ className, variant = 'primary', size = 'md', loading, disabled, children, ...props }, ref) => {
    return (
      <button
        ref={ref}
        disabled={disabled || loading}
        className={cn(
          'button',
          `button-${variant}`,
          `button-${size}`,
          loading && 'button-loading',
          className
        )}
        {...props}
      >
        {loading && <span className="button-spinner" />}
        <span className="button-content">{children}</span>
      </button>
    );
  }
);

Button.displayName = 'Button';

1. **实现一个带防抖的搜索组件**
   ```tsx
   import { useState, useEffect, useCallback } from 'react';
   
   function SearchComponent() {
     const [query, setQuery] = useState('');
     const [results, setResults] = useState([]);
     const [loading, setLoading] = useState(false);
     
     // 防抖函数
     const debounce = (func: Function, delay: number) => {
       let timeoutId: NodeJS.Timeout;
       return (...args: any[]) => {
         clearTimeout(timeoutId);
         timeoutId = setTimeout(() => func(...args), delay);
       };
     };
     
     // 搜索函数
     const search = useCallback(async (q: string) => {
       if (!q.trim()) {
         setResults([]);
         return;
       }
       
       setLoading(true);
       try {
         const response = await fetch(`/api/search?q=${encodeURIComponent(q)}`);
         const data = await response.json();
         setResults(data);
       } catch (error) {
         console.error('Search failed:', error);
       } finally {
         setLoading(false);
       }
     }, []);
     
     // 防抖搜索
     const debouncedSearch = useCallback(
       debounce((q: string) => search(q), 300),
       [search]
     );
     
     useEffect(() => {
       debouncedSearch(query);
     }, [query, debouncedSearch]);
     
     return (
       <div>
         <input
           type="text"
           value={query}
           onChange={(e) => setQuery(e.target.value)}
           placeholder="搜索..."
         />
         {loading && <p>搜索中...</p>}
         <ul>
           {results.map((item: any) => (
             <li key={item.id}>{item.name}</li>
           ))}
         </ul>
       </div>
     );
   }
   ```

2. **实现一个自定义 Hook**
   ```tsx
   // useLocalStorage.ts
   import { useState, useEffect } from 'react';
   
   function useLocalStorage<T>(key: string, initialValue: T) {
     // 获取初始值
     const [storedValue, setStoredValue] = useState<T>(() => {
       try {
         const item = window.localStorage.getItem(key);
         return item ? JSON.parse(item) : initialValue;
       } catch (error) {
         console.error('Error reading localStorage:', error);
         return initialValue;
       }
     });
     
     // 更新 localStorage
     useEffect(() => {
       try {
         window.localStorage.setItem(key, JSON.stringify(storedValue));
       } catch (error) {
         console.error('Error setting localStorage:', error);
       }
     }, [key, storedValue]);
     
     return [storedValue, setStoredValue] as const;
   }
   
   // 使用示例
   function App() {
     const [count, setCount] = useLocalStorage('count', 0);
     
     return (
       <div>
         <p>Count: {count}</p>
         <button onClick={() => setCount(count + 1)}>Increment</button>
       </div>
     );
   }
   ```

// React 组件实现示例
import { useEditor, EditorContent } from '@tiptap/react';
import Collaboration from '@tiptap/extension-collaboration';
import { WebsocketProvider } from 'y-websocket';

function CollaborativeEditor() {
  const [provider, setProvider] = useState<WebsocketProvider | null>(null);
  
  useEffect(() => {
    // 初始化 WebSocket 连接
    const wsProvider = new WebsocketProvider(
      'ws://localhost:1234',
      'room-1',
      doc
    );
    
    setProvider(wsProvider);
    
    return () => {
      wsProvider.destroy();
    };
  }, []);
  
  const editor = useEditor({
    extensions: [
      Collaboration.configure({
        document: doc,
      }),
    ],
    content: '',
  });
  
  if (!editor || !provider) {
    return <div>Loading...</div>;
  }
  
  return (
    <div className="editor">
      <div className="toolbar">
        {/* 编辑器工具栏 */}
      </div>
      <EditorContent editor={editor} />
      <div className="collaborators">
        {/* 在线协作者列表 */}
        {provider.awareness.getStates().map((state, index) => (
          <div key={index} className="collaborator">
            {state.user?.name}
          </div>
        ))}
      </div>
    </div>
  );
}

### 薪资谈判

#### 谈判准备

**市场调研：**
- 了解目标岗位的市场薪资范围
- 调研目标公司的薪资水平
- 考虑城市、行业、公司规模等因素

**自我评估：**
- 评估自己的技能和经验
- 列出自己的核心优势
- 准备具体的成果和数据

#### 谈判策略

**谈判技巧：**

1. **不要先透露期望薪资**
   - 让公司先给出范围
   - 基于市场价值进行谈判

2. **强调价值而非需求**
   - ❌ "我需要 20K 因为房贷压力大"
   - ✅ "基于我的技能和经验，我认为 20K 是合理的"

3. **准备谈判筹码**
   - 展示项目成果
   - 提供市场数据
   - 说明你的独特价值

4. **考虑整体薪酬包**
   - 基本工资
   - 年终奖
   - 股票期权
   - 福利待遇
   - 工作时间和远程办公

**谈判话术示例：**

```markdown
# 薪资谈判话术

## 公司给出报价后

"感谢您的 offer。我对这个职位很感兴趣，但薪资方面我希望能再讨论一下。"

## 提出期望薪资

"基于我的技能和经验，以及市场调研的结果，我期望的薪资范围是 X-Y。我相信我能为团队带来相应的价值。"

## 强调价值

"在过去的项目中，我主导了技术重构，将性能提升了 40%。我相信我的经验能够帮助团队解决类似的问题。"

## 考虑整体薪酬包

"除了基本工资，我也很关注股票期权和年终奖。能否详细介绍一下这方面的政策？"

## 谈判不顺利时

"我理解公司的预算限制。能否在其他方面提供一些灵活性，比如更多的股票期权或更好的福利？"
```
