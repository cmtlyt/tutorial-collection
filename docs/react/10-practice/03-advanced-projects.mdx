```tsx
import { useState, useEffect, useRef } from 'react';

  const [sharedState, setSharedState] = useState<SharedState>({
    user: null,
    theme: 'light',
    language: 'zh',
  });
  const [activeApp, setActiveApp] = useState<string | null>(null);
  const messageChannelRef = useRef<BroadcastChannel | null>(null);

  useEffect(() => {
    // 初始化消息通道
    messageChannelRef.current = new BroadcastChannel('micro-apps-channel');
    
    messageChannelRef.current.onmessage = (event: MessageEvent<AppMessage>) => {
      const { type, payload, source } = event.data;
      
      switch (type) {
        case 'STATE_UPDATE':
          setSharedState(prev => ({ ...prev, ...payload }));
          break;
        case 'NAVIGATION':
          setActiveApp(payload.app);
          break;
        default:
          console.log('Unknown message type:', type);
      }
    };

    return () => {
      messageChannelRef.current?.close();
    };
  }, []);

  const broadcastMessage = (message: AppMessage) => {
    messageChannelRef.current?.postMessage(message);
  };

  const handleUserLogin = (user: User) => {
    const newState = { ...sharedState, user };
    setSharedState(newState);
    broadcastMessage({
      type: 'STATE_UPDATE',
      payload: { user },
      source: 'host',
    });
  };

  const handleThemeToggle = () => {
    const newTheme = sharedState.theme === 'light' ? 'dark' : 'light';
    const newState = { ...sharedState, theme: newTheme };
    setSharedState(newState);
    broadcastMessage({
      type: 'STATE_UPDATE',
      payload: { theme: newTheme },
      source: 'host',
    });
  };

  const loadMicroApp = async (app: MicroApp) => {
    try {
      // 动态加载微应用
      await loadScript(app.entry);
      setActiveApp(app.name);
    } catch (error) {
      console.error(`Failed to load micro-app ${app.name}:`, error);
    }
  };

  return (
    <div className={`micro-frontend-app ${sharedState.theme}`}>
      <Header 
        user={sharedState.user}
        theme={sharedState.theme}
        onThemeToggle={handleThemeToggle}
        onUserLogin={handleUserLogin}
      />
      
      <main className="main-content">
        <Sidebar 
          apps={microApps}
          activeApp={activeApp}
          onAppSelect={(app) => loadMicroApp(app)}
        />
        
        <div className="content-area">

            <div id="dashboard-container" className="micro-app-container" />
          )}
          {activeApp === 'settings' && (
            <div id="settings-container" className="micro-app-container" />
          )}
          {!activeApp && (
            <WelcomeScreen onUserLogin={handleUserLogin} />
          )}
        </div>
      </main>
    </div>
  );
}

function loadScript(src: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const script = document.createElement('script');
    script.src = src;
    script.async = true;
    script.onload = () => resolve();
    script.onerror = reject;
    document.head.appendChild(script);
  });
}
```

```tsx
// types.ts
export interface Document {
  id: string;
  title: string;
  content: string;
  createdBy: string;
  createdAt: Date;
  updatedAt: Date;
  collaborators: Collaborator[];
}

export interface Collaborator {
  id: string;
  name: string;
  color: string;
  cursor: Cursor | null;
}

export interface Cursor {
  position: number;
  selection: number;
}

export interface Operation {
  type: 'insert' | 'delete' | 'retain';
  content?: string;
  length?: number;
  userId: string;
  timestamp: Date;
}

export interface WebSocketMessage {
  type: 'DOCUMENT_UPDATE' | 'CURSOR_MOVE' | 'USER_JOIN' | 'USER_LEAVE';
  payload: any;
  userId: string;
}

// DocumentEditor.tsx
'use client';

import { useState, useEffect, useRef, useCallback } from 'react';
import { Document, Collaborator, Operation, WebSocketMessage } from './types';

interface DocumentEditorProps {
  document: Document;
  currentUser: { id: string; name: string };
  onDocumentUpdate: (document: Document) => void;
}

export function DocumentEditor({ document, currentUser, onDocumentUpdate }: DocumentEditorProps) {
  const [content, setContent] = useState(document.content);
  const [collaborators, setCollaborators] = useState<Collaborator[]>(document.collaborators);
  const [isConnected, setIsConnected] = useState(false);
  const wsRef = useRef<WebSocket | null>(null);
  const textareaRef = useRef<HTMLTextAreaElement>(null);

  // 建立 WebSocket 连接
  useEffect(() => {
    const ws = new WebSocket('ws://localhost:8080/collaboration');
    
    ws.onopen = () => {
      setIsConnected(true);
      console.log('WebSocket connected');
      
      // 发送加入文档消息
      ws.send(JSON.stringify({
        type: 'JOIN_DOCUMENT',
        payload: {
          documentId: document.id,
          userId: currentUser.id,
          userName: currentUser.name,
        },
      }));
    };
    
    ws.onmessage = (event) => {
      const message: WebSocketMessage = JSON.parse(event.data);
      
      switch (message.type) {
        case 'DOCUMENT_UPDATE':
          handleRemoteUpdate(message.payload);
          break;
        case 'CURSOR_MOVE':
          handleCursorMove(message.payload);
          break;
        case 'USER_JOIN':
          handleUserJoin(message.payload);
          break;
        case 'USER_LEAVE':
          handleUserLeave(message.payload);
          break;
      }
    };
    
    ws.onerror = (error) => {
      console.error('WebSocket error:', error);
      setIsConnected(false);
    };
    
    ws.onclose = () => {
      setIsConnected(false);
      console.log('WebSocket disconnected');
    };
    
    wsRef.current = ws;
    
    return () => {
      ws.close();
    };
  }, [document.id, currentUser]);

  const handleRemoteUpdate = useCallback((payload: { operation: Operation; newContent: string }) => {
    const { operation, newContent } = payload;
    
    // 如果不是当前用户的操作，则应用远程更新
    if (operation.userId !== currentUser.id) {
      setContent(newContent);
      onDocumentUpdate({
        ...document,
        content: newContent,
        updatedAt: new Date(),
      });
    }
  }, [currentUser.id, document, onDocumentUpdate]);

  const handleCursorMove = useCallback((payload: { userId: string; cursor: Cursor }) => {
    setCollaborators(prev =>
      prev.map(collaborator =>
        collaborator.id === payload.userId
          ? { ...collaborator, cursor: payload.cursor }
          : collaborator
      )
    );
  }, []);

  const handleUserJoin = useCallback((payload: { user: Collaborator }) => {
    setCollaborators(prev => {
      const exists = prev.some(c => c.id === payload.user.id);
      if (exists) return prev;
      return [...prev, payload.user];
    });
  }, []);

  const handleUserLeave = useCallback((payload: { userId: string }) => {
    setCollaborators(prev => prev.filter(c => c.id !== payload.userId));
  }, []);

  const handleContentChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    const newContent = e.target.value;
    const cursorPosition = e.target.selectionStart;
    
    setContent(newContent);
    
    // 发送更新到服务器
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      const operation: Operation = {
        type: 'insert',
        content: newContent,
        userId: currentUser.id,
        timestamp: new Date(),
      };
      
      wsRef.current.send(JSON.stringify({
        type: 'DOCUMENT_UPDATE',
        payload: {
          documentId: document.id,
          operation,
          newContent,
        },
      }));
    }
    
    onDocumentUpdate({
      ...document,
      content: newContent,
      updatedAt: new Date(),
    });
  };

  const handleCursorChange = useCallback(() => {
    if (!textareaRef.current) return;
    
    const cursor: Cursor = {
      position: textareaRef.current.selectionStart,
      selection: textareaRef.current.selectionEnd,
    };
    
    // 发送光标位置
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify({
        type: 'CURSOR_MOVE',
        payload: {
          documentId: document.id,
          cursor,
        },
      }));
    }
  }, [document.id]);

  return (
    <div className="document-editor">
      <div className="editor-header">
        <div className="connection-status">
          <span className={`status-indicator ${isConnected ? 'connected' : 'disconnected'}`} />
          <span>{isConnected ? '已连接' : '未连接'}</span>
        </div>
        
        <div className="collaborators">
          {collaborators.map(collaborator => (
            <div
              key={collaborator.id}
              className="collaborator-badge"
              style={{ backgroundColor: collaborator.color }}
              title={collaborator.name}
            >
              {collaborator.name.charAt(0)}
            </div>
          ))}
        </div>
        
        <div className="document-info">
          <span className="last-updated">
            最后更新: {document.updatedAt.toLocaleTimeString('zh-CN')}
          </span>
        </div>
      </div>

      <div className="editor-content">
        <textarea
          ref={textareaRef}
          value={content}
          onChange={handleContentChange}
          onSelect={handleCursorChange}
          onKeyUp={handleCursorChange}
          className="editor-textarea"
          placeholder="开始编辑文档..."
          spellCheck={false}
        />
        
        {collaborators.map(collaborator => (
          collaborator.cursor && collaborator.id !== currentUser.id && (
            <RemoteCursor
              key={collaborator.id}
              cursor={collaborator.cursor}
              color={collaborator.color}
              name={collaborator.name}
            />
          )
        ))}
      </div>
    </div>
  );
}

function RemoteCursor({ cursor, color, name }: {
  cursor: Cursor;
  color: string;
  name: string;
}) {
  const [position, setPosition] = useState({ top: 0, left: 0 });

  useEffect(() => {
    // 计算光标位置（简化版本）
    const textarea = document.querySelector('.editor-textarea') as HTMLTextAreaElement;
    if (textarea) {
      const text = textarea.value.substring(0, cursor.position);
      const lines = text.split('\n');
      const lineHeight = 24; // 假设行高
      const charWidth = 14; // 假设字符宽度
      
      setPosition({
        top: lines.length * lineHeight - lineHeight,
        left: lines[lines.length - 1].length * charWidth,
      });
    }
  }, [cursor.position]);

  return (
    <div
      className="remote-cursor"
      style={{
        top: `${position.top}px`,
        left: `${position.left}px`,
        borderLeftColor: color,
      }}
    >
      <div className="cursor-flag" style={{ backgroundColor: color }}>
        {name}
      </div>
    </div>
  );
}
```

```tsx
// ChatInterface.tsx
'use client';

import { useState, useRef, useEffect, useCallback } from 'react';
import { Message, Conversation, AIResponse, Attachment } from './types';

interface ChatInterfaceProps {
  conversation: Conversation;
  onSendMessage: (message: Message) => Promise<void>;
  onUpdateConversation: (conversation: Conversation) => void;
}

export function ChatInterface({ conversation, onSendMessage, onUpdateConversation }: ChatInterfaceProps) {
  const [inputText, setInputText] = useState('');
  const [isTyping, setIsTyping] = useState(false);
  const [attachments, setAttachments] = useState<Attachment[]>([]);
  const messagesEndRef = useRef<HTMLDivElement>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const scrollToBottom = useCallback(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, []);

  useEffect(() => {
    scrollToBottom();
  }, [conversation.messages, scrollToBottom]);

  const handleSendMessage = async () => {
    if (!inputText.trim() && attachments.length === 0) return;

    const userMessage: Message = {
      id: Date.now().toString(),
      role: 'user',
      content: inputText,
      timestamp: new Date(),
      attachments: attachments.length > 0 ? attachments : undefined,
    };

    setInputText('');
    setAttachments([]);
    setIsTyping(true);

    // 创建助手消息占位符
    const assistantMessage: Message = {
      id: (Date.now() + 1).toString(),
      role: 'assistant',
      content: '',
      timestamp: new Date(),
    };

    const updatedMessages = [...conversation.messages, userMessage, assistantMessage];
    onUpdateConversation({
      ...conversation,
      messages: updatedMessages,
      updatedAt: new Date(),
    });

    try {
      // 调用 AI API（模拟）
      const response = await fetchAIResponse(userMessage, conversation.messages);
      
      // 流式更新响应
      await streamResponse(response, assistantMessage.id);
    } catch (error) {
      console.error('Failed to get AI response:', error);
      
      // 更新为错误消息
      const errorMessage: Message = {
        ...assistantMessage,
        content: '抱歉，我遇到了一些问题。请稍后再试。',
      };
      
      onUpdateConversation({
        ...conversation,
        messages: [...updatedMessages.slice(0, -1), errorMessage],
        updatedAt: new Date(),
      });
    } finally {
      setIsTyping(false);
```
