## 📚 概念介绍

函数式编程是一种编程范式，它将计算视为数学函数的求值，并避免使用可变状态和副作用。React 深受函数式编程思想的影响，理解这些概念对于编写高质量的 React 应用至关重要。

## 🎯 高阶函数

### 基本概念

高阶函数是接受函数作为参数或返回函数的函数：

```js
// 接受函数作为参数
function withLogging(fn) {
  return function(...args) {
    console.log(`Calling ${fn.name} with:`, args);
    const result = fn(...args);
    console.log(`Result:`, result);
    return result;
  };
}

// 返回函数
function createMultiplier(multiplier) {
  return function(x) {
    return x * multiplier;
  };
}

// 使用示例
const add = (a, b) => a + b;
const loggedAdd = withLogging(add);

console.log(loggedAdd(2, 3)); 
// Calling add with: [2, 3]
// Result: 5
// 5

const double = createMultiplier(2);
const triple = createMultiplier(3);

console.log(double(5)); // 10
console.log(triple(5)); // 15
```

### 常用高阶函数

```js
// map - 转换数组
const numbers = [1, 2, 3, 4, 5];
const doubled = numbers.map(x => x * 2);
console.log(doubled); // [2, 4, 6, 8, 10]

const users = [
  { name: 'Alice', age: 25 },
  { name: 'Bob', age: 30 },
  { name: 'Charlie', age: 35 }
];
const names = users.map(user => user.name);
console.log(names); // ['Alice', 'Bob', 'Charlie']

// filter - 过滤数组
const evenNumbers = numbers.filter(x => x % 2 === 0);
console.log(evenNumbers); // [2, 4]

const adults = users.filter(user => user.age >= 30);
console.log(adults); // [{ name: 'Bob', age: 30 }, { name: 'Charlie', age: 35 }]

// reduce - 归约数组
const sum = numbers.reduce((acc, x) => acc + x, 0);
console.log(sum); // 15

const max = numbers.reduce((acc, x) => Math.max(acc, x), -Infinity);
console.log(max); // 5

// 组合使用
const totalAge = users
  .filter(user => user.age >= 30)
  .reduce((acc, user) => acc + user.age, 0);
console.log(totalAge); // 65

// find - 查找元素
const bob = users.find(user => user.name === 'Bob');
console.log(bob); // { name: 'Bob', age: 30 }

// some - 检查是否有满足条件的元素
const hasAdult = users.some(user => user.age >= 30);
console.log(hasAdult); // true

// every - 检查是否所有元素都满足条件
const allAdults = users.every(user => user.age >= 18);
console.log(allAdults); // true

// forEach - 遍历数组（无返回值）
users.forEach(user => {
  console.log(`${user.name} is ${user.age} years old`);
});

// sort - 排序数组
const sortedUsers = [...users].sort((a, b) => a.age - b.age);
console.log(sortedUsers);
// [
//   { name: 'Alice', age: 25 },
//   { name: 'Bob', age: 30 },
//   { name: 'Charlie', age: 35 }
// ]
```

### React 中的应用

```js
import React, { useState } from 'react';

function TodoList({ todos }) {
  // 使用高阶函数处理数据
  const activeTodos = todos.filter(todo => !todo.completed);
  const completedTodos = todos.filter(todo => todo.completed);
  
  return (
    <div>
      <h2>Todo List</h2>

        {activeTodos.map(todo => (

        ))}
      </ul>
      
      <h3>Completed ({completedTodos.length})</h3>

        {completedTodos.map(todo => (
          <li key={todo.id}>
            <del>{todo.text}</del>
          </li>
        ))}
      </ul>
    </div>
  );
}

// 使用示例
const todos = [
  { id: 1, text: 'Learn React', completed: true },
  { id: 2, text: 'Build a project', completed: false },
  { id: 3, text: 'Write documentation', completed: false }
];

function App() {
  return <TodoList todos={todos} />;
}

export default App;
```

## 🎯 纯函数与副作用

### 纯函数

纯函数是相同的输入总是产生相同的输出，且没有任何副作用的函数：

```js
// 纯函数示例
function add(a, b) {
  return a + b;
}

function square(x) {
  return x * x;
}

function greet(name) {
  return `Hello, ${name}!`;
}

// 验证纯函数
console.log(add(2, 3)); // 5
console.log(add(2, 3)); // 5 (相同输入，相同输出)

console.log(square(4)); // 16
console.log(square(4)); // 16

console.log(greet('Alice')); // Hello, Alice!
console.log(greet('Alice')); // Hello, Alice!
```

### 非纯函数（有副作用）

```js
// 非纯函数 - 依赖外部状态
let counter = 0;

function increment() {
  counter += 1;
  return counter;
}

console.log(increment()); // 1
console.log(increment()); // 2 (相同调用，不同输出)

// 非纯函数 - 修改参数
function appendToArray(array, item) {
  array.push(item);
  return array;
}

const arr = [1, 2, 3];
appendToArray(arr, 4);
console.log(arr); // [1, 2, 3, 4] (原数组被修改)

// 非纯函数 - 依赖外部函数
function getCurrentTime() {
  return new Date().toISOString();
}

console.log(getCurrentTime()); // 2026-04-21T12:00:00.000Z
console.log(getCurrentTime()); // 2026-04-21T12:00:01.000Z (不同输出)
```

### 重构为纯函数

```js
// 重构 increment 为纯函数
function incrementPure(counter) {
  return counter + 1;
}

let counter = 0;
counter = incrementPure(counter);
console.log(counter); // 1
counter = incrementPure(counter);
console.log(counter); // 2

// 重构 appendToArray 为纯函数
function appendToArrayPure(array, item) {
  return [...array, item];
}

const arr = [1, 2, 3];
const newArr = appendToArrayPure(arr, 4);
console.log(arr); // [1, 2, 3] (原数组未修改)
console.log(newArr); // [1, 2, 3, 4]

// 重构 getCurrentTime 为纯函数
function formatTime(date) {
  return date.toISOString();
}

const now = new Date();
console.log(formatTime(now)); // 2026-04-21T12:00:00.000Z
console.log(formatTime(now)); // 2026-04-21T12:00:00.000Z (相同输出)
```

### React 中的应用

```js

      <p>Subtotal: ${subtotal.toFixed(2)}</p>
      <p>Tax: ${tax.toFixed(2)}</p>
      <p>Total: ${total.toFixed(2)}</p>
    </div>
  );
}

// 使用示例
const items = [
  { id: 1, name: 'Laptop', price: 999.99, quantity: 1 },
  { id: 2, name: 'Mouse', price: 29.99, quantity: 2 },

];

function App() {
  return (

    </div>
  );
}

export default App;
```

## 🎯 不可变数据

### 基本概念

不可变数据是指数据创建后不能被修改，任何修改都会创建新的数据：

```js
// 对象不可变性
const user = {
  name: 'Alice',
  age: 25,
  email: 'alice@example.com'
};

// ❌ 错误：直接修改对象
// user.age = 26;

// ✅ 正确：创建新对象
const updatedUser = {
  ...user,
  age: 26
};

console.log(user); // { name: 'Alice', age: 25, email: 'alice@example.com' }
console.log(updatedUser); // { name: 'Alice', age: 26, email: 'alice@example.com' }

// 嵌套对象
const userWithAddress = {
  name: 'Alice',
  age: 25,
  address: {
    city: 'Beijing',
    country: 'China'
  }
};

// 更新嵌套属性
const updatedAddress = {
  ...userWithAddress,
  address: {
    ...userWithAddress.address,
    city: 'Shanghai'
  }
};

console.log(updatedAddress);
// {
//   name: 'Alice',
//   age: 25,
//   address: { city: 'Shanghai', country: 'China' }
// }

// 数组不可变性
const numbers = [1, 2, 3, 4, 5];

// ❌ 错误：直接修改数组
// numbers.push(6);
// numbers[0] = 10;

// ✅ 正确：创建新数组
const withAdded = [...numbers, 6];
const withRemoved = numbers.filter(x => x !== 3);
const withUpdated = numbers.map(x => x === 3 ? 30 : x);
const withSorted = [...numbers].sort((a, b) => b - a);

console.log(numbers); // [1, 2, 3, 4, 5]
console.log(withAdded); // [1, 2, 3, 4, 5, 6]
console.log(withRemoved); // [1, 2, 4, 5]
console.log(withUpdated); // [1, 2, 30, 4, 5]
console.log(withSorted); // [5, 4, 3, 2, 1]
```

### React 中的应用

```js
import React, { useState } from 'react';

function TodoApp() {
  const [todos, setTodos] = useState([
    { id: 1, text: 'Learn React', completed: false },
    { id: 2, text: 'Build a project', completed: false },
    { id: 3, text: 'Write documentation', completed: false }
  ]);
  
  const [input, setInput] = useState('');
  
  // 添加 todo（不可变更新）
  const addTodo = (text) => {
    const newTodo = {
      id: Date.now(),
      text,
      completed: false
    };
    setTodos([...todos, newTodo]);
  };
  
  // 切换完成状态（不可变更新）
  const toggleTodo = (id) => {
    setTodos(todos.map(todo => 
      todo.id === id ? { ...todo, completed: !todo.completed } : todo
    ));
  };
  
  // 删除 todo（不可变更新）
  const deleteTodo = (id) => {
    setTodos(todos.filter(todo => todo.id !== id));
  };
  
  // 清除已完成的 todo（不可变更新）
  const clearCompleted = () => {
    setTodos(todos.filter(todo => !todo.completed));
  };
  
  const handleSubmit = (e) => {
    e.preventDefault();
    if (input.trim()) {
      addTodo(input.trim());
      setInput('');
    }
  };
  
  const activeCount = todos.filter(todo => !todo.completed).length;
  const completedCount = todos.filter(todo => todo.completed).length;
  
  return (
    <div>
      <h1>Todo App</h1>
      
      <form onSubmit={handleSubmit}>
        <input
          type="text"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          placeholder="Add a new todo"
        />
        <button type="submit">Add</button>
      </form>
      
      <ul>
        {todos.map(todo => (
          <li key={todo.id}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={() => toggleTodo(todo.id)}
            />
            <span style={{ textDecoration: todo.completed ? 'line-through' : 'none' }}>
              {todo.text}
            </span>
            <button onClick={() => deleteTodo(todo.id)}>Delete</button>
          </li>
        ))}
      </ul>
      
      <p>
        Active: {activeCount} | Completed: {completedCount}
      </p>
      
      {completedCount > 0 && (
        <button onClick={clearCompleted}>
          Clear Completed ({completedCount})
        </button>
      )}
    </div>
  );
}

export default TodoApp;
```

## 🎯 数组方法（map, filter, reduce）

### 详解 map

```js
// map - 转换数组的每个元素
const numbers = [1, 2, 3, 4, 5];

// 基本用法
const doubled = numbers.map(x => x * 2);
console.log(doubled); // [2, 4, 6, 8, 10]

// 对象数组
const users = [
  { id: 1, name: 'Alice', age: 25 },
  { id: 2, name: 'Bob', age: 30 },
  { id: 3, name: 'Charlie', age: 35 }
];

// 提取属性
const names = users.map(user => user.name);
console.log(names); // ['Alice', 'Bob', 'Charlie']

// 转换对象
const userSummaries = users.map(user => ({
  id: user.id,
  displayName: `${user.name} (${user.age})`
}));
console.log(userSummaries);
// [
//   { id: 1, displayName: 'Alice (25)' },
//   { id: 2, displayName: 'Bob (30)' },
//   { id: 3, displayName: 'Charlie (35)' }
// ]

// 索引作为第二个参数
const withIndex = numbers.map((num, index) => `${index}: ${num}`);
console.log(withIndex); // ['0: 1', '1: 2', '2: 3', '3: 4', '4: 5']
```

### 详解 reduce

```js
// reduce - 归约数组为单个值
const numbers = [1, 2, 3, 4, 5];

// 基本用法 - 求和
const sum = numbers.reduce((acc, x) => acc + x, 0);
console.log(sum); // 15

// 求积
const product = numbers.reduce((acc, x) => acc * x, 1);
console.log(product); // 120

// 查找最大值
const max = numbers.reduce((acc, x) => Math.max(acc, x), -Infinity);
console.log(max); // 5

// 查找最小值
const min = numbers.reduce((acc, x) => Math.min(acc, x), Infinity);
console.log(min); // 1

// 对象数组 - 分组
const users = [
  { name: 'Alice', age: 25, department: 'Engineering' },
  { name: 'Bob', age: 30, department: 'Marketing' },
  { name: 'Charlie', age: 35, department: 'Engineering' },
  { name: 'Diana', age: 28, department: 'Marketing' }
];

const usersByDepartment = users.reduce((acc, user) => {
  const dept = user.department;
  if (!acc[dept]) {
    acc[dept] = [];
  }
  acc[dept].push(user);
  return acc;
}, {});

console.log(usersByDepartment);
// {
//   Engineering: [
//     { name: 'Alice', age: 25, department: 'Engineering' },
//     { name: 'Charlie', age: 35, department: 'Engineering' }
//   ],
//   Marketing: [
//     { name: 'Bob', age: 30, department: 'Marketing' },
//     { name: 'Diana', age: 28, department: 'Marketing' }
//   ]
// }

// 计算平均年龄
const averageAge = users.reduce((acc, user, index, array) => {
  acc.sum += user.age;
  if (index === array.length - 1) {
    return acc.sum / array.length;
  }
  return acc;
}, { sum: 0 });

console.log(averageAge); // 29.5

// 转换为对象映射
const userMap = users.reduce((acc, user) => {
  acc[user.name] = user;
  return acc;
}, {});

console.log(userMap);
// {
//   Alice: { name: 'Alice', age: 25, department: 'Engineering' },

// }
```

### React 中的应用

```js
import React, { useState } from 'react';

function ProductFilter({ products }) {
  const [filter, setFilter] = useState('all');
  const [sortBy, setSortBy] = useState('name');
  
  // 过滤产品
  const filteredProducts = products.filter(product => {
    if (filter === 'all') return true;
    if (filter === 'in-stock') return product.inStock;
    if (filter === 'out-of-stock') return !product.inStock;
    return product.category === filter;
  });
  
  // 排序产品
  const sortedProducts = [...filteredProducts].sort((a, b) => {
    if (sortBy === 'name') return a.name.localeCompare(b.name);
    if (sortBy === 'price') return a.price - b.price;
    if (sortBy === 'price-desc') return b.price - a.price;
    return 0;
  });
  
  // 计算统计信息
  const stats = products.reduce((acc, product) => {
    acc.total += product.price;
    acc.inStockCount += product.inStock ? 1 : 0;
    acc.outOfStockCount += !product.inStock ? 1 : 0;
    return acc;
  }, { total: 0, inStockCount: 0, outOfStockCount: 0 });
  
  // 获取所有类别
  const categories = [
    'all',
    ...new Set(products.map(product => product.category))
  ];
  
  return (
    <div>
      <h1>Product Filter</h1>
      
      <div className="filters">
        <label>
          Filter:
          <select value={filter} onChange={(e) => setFilter(e.target.value)}>
            {categories.map(category => (
              <option key={category} value={category}>
                {category === 'all' ? 'All Products' : category}
              </option>
            ))}
            <option value="in-stock">In Stock</option>
            <option value="out-of-stock">Out of Stock</option>
          </select>
        </label>
        
        <label>
          Sort by:
          <select value={sortBy} onChange={(e) => setSortBy(e.target.value)}>
            <option value="name">Name</option>
            <option value="price">Price (Low to High)</option>
            <option value="price-desc">Price (High to Low)</option>
          </select>
        </label>
      </div>
      
      <div className="stats">
        <p>Total Products: {products.length}</p>
        <p>In Stock: {stats.inStockCount}</p>
        <p>Out of Stock: {stats.outOfStockCount}</p>
        <p>Total Value: ${stats.total.toFixed(2)}</p>
      </div>
      
      <div className="products">
        {sortedProducts.length === 0 ? (
          <p>No products found</p>
        ) : (
          <ul>
            {sortedProducts.map(product => (
              <li key={product.id}>
                <h3>{product.name}</h3>
```

```js
// utils.js - 工具函数
export const formatCurrency = (amount) => {
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD'
  }).format(amount);
};

export const calculateSubtotal = (items) => {
  return items.reduce((sum, item) => sum + item.price * item.quantity, 0);
};

export const calculateTax = (subtotal, taxRate) => {
  return subtotal * taxRate;
};

export const calculateTotal = (subtotal, tax) => {
  return subtotal + tax;
};

export const applyDiscount = (total, discount) => {
  return total * (1 - discount / 100);
};

// cart.js - 购物车逻辑
export const addToCart = (cart, product) => {
  const existingItem = cart.find(item => item.id === product.id);
  
  if (existingItem) {
    return cart.map(item =>
      item.id === product.id
        ? { ...item, quantity: item.quantity + 1 }
        : item
    );
  }
  
  return [...cart, { ...product, quantity: 1 }];
};

export const removeFromCart = (cart, productId) => {
  return cart.filter(item => item.id !== productId);
};

export const updateQuantity = (cart, productId, quantity) => {
  if (quantity <= 0) {
    return removeFromCart(cart, productId);
  }
  
  return cart.map(item =>
    item.id === productId ? { ...item, quantity } : item
  );
};

export const clearCart = () => [];

export const getCartCount = (cart) => {
  return cart.reduce((sum, item) => sum + item.quantity, 0);
};

export const getCartTotal = (cart) => {
  return cart.reduce((sum, item) => sum + item.price * item.quantity, 0);
};

// Cart.jsx - 购物车组件
import React, { useState } from 'react';
import {
  formatCurrency,
  calculateSubtotal,
  calculateTax,
  calculateTotal,
  applyDiscount
} from './utils';
import {
  addToCart,
  removeFromCart,
  updateQuantity,
  clearCart,
  getCartCount,
  getCartTotal
} from './cart';

function Cart({ products, taxRate = 0.08, discount = 0 }) {
  const [cart, setCart] = useState([]);
  
  const handleAddToCart = (product) => {
    setCart(addToCart(cart, product));
  };
  
  const handleRemoveFromCart = (productId) => {
    setCart(removeFromCart(cart, productId));
  };
  
  const handleUpdateQuantity = (productId, quantity) => {
    setCart(updateQuantity(cart, productId, quantity));
  };
  
  const handleClearCart = () => {
    setCart(clearCart());
  };
  
  const subtotal = getCartTotal(cart);
  const tax = calculateTax(subtotal, taxRate);
  const totalBeforeDiscount = calculateTotal(subtotal, tax);
  const total = applyDiscount(totalBeforeDiscount, discount);
  
  const cartCount = getCartCount(cart);
  
  return (
    <div className="cart">
      <h1>Shopping Cart</h1>
      
      <div className="cart-summary">
        <p>Items: {cartCount}</p>
        <p>Subtotal: {formatCurrency(subtotal)}</p>
        <p>Tax ({(taxRate * 100).toFixed(0)}%): {formatCurrency(tax)}</p>
        {discount > 0 && (
          <p>Discount ({discount}%): -{formatCurrency(totalBeforeDiscount - total)}</p>
        )}
        <h3>Total: {formatCurrency(total)}</h3>
      </div>
      
      <div className="products">
        <h2>Products</h2>
        {products.map(product => (
          <div key={product.id} className="product">
            <h3>{product.name}</h3>
            <p>Price: {formatCurrency(product.price)}</p>
            <button onClick={() => handleAddToCart(product)}>
              Add to Cart
            </button>
          </div>
        ))}
      </div>
      
      {cart.length > 0 && (
        <div className="cart-items">
          <h2>Cart Items ({cartCount})</h2>
          <ul>
            {cart.map(item => (
              <li key={item.id}>
                <h4>{item.name}</h4>
                <p>Price: {formatCurrency(item.price)}</p>
                <div className="quantity">
                  <button
                    onClick={() => handleUpdateQuantity(item.id, item.quantity - 1)}
                  >
                    -
                  </button>
                  <span>{item.quantity}</span>
                  <button
                    onClick={() => handleUpdateQuantity(item.id, item.quantity + 1)}
                  >
                    +
                  </button>
                </div>
                <p>Subtotal: {formatCurrency(item.price * item.quantity)}</p>
                <button onClick={() => handleRemoveFromCart(item.id)}>
                  Remove
                </button>
              </li>
            ))}
          </ul>
          <button onClick={handleClearCart}>Clear Cart</button>
        </div>
      )}
    </div>
  );
}

// 使用示例
const products = [
  { id: 1, name: 'Laptop', price: 999.99 },
  { id: 2, name: 'Mouse', price: 29.99 },
  { id: 3, name: 'Keyboard', price: 79.99 },
  { id: 4, name: 'Monitor', price: 299.99 }
];

function App() {
  return (
    <Cart 
      products={products} 
      taxRate={0.08}
      discount={10}
    />
  );
}

export default App;
```

- **函数组合**：将多个函数组合成一个新函数
- **递归**：函数调用自身来解决问题
