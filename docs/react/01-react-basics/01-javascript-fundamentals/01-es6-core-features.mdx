## 📚 概念介绍

ES6+（ECMAScript 2015 及后续版本）是现代 JavaScript 开发的基础。掌握这些核心特性对于学习 React 至关重要，因为 React 大量使用了 ES6+ 的语法特性。本章节将详细介绍 React 开发中最常用的 ES6+ 特性。

## 🎯 箭头函数与函数表达式

### 基本语法

箭头函数提供了一种更简洁的函数书写方式：

```js
// 传统函数表达式
const add = function(a, b) {
  return a + b;
};

// 箭头函数
const add = (a, b) => {
  return a + b;
};

// 简化形式（单行返回）
const add = (a, b) => a + b;

// 单参数可省略括号
const square = x => x * x;

// 无参数需要空括号
const greet = () => 'Hello, World!';

// 使用示例
console.log(add(2, 3)); // 5
console.log(square(4)); // 16
console.log(greet()); // Hello, World!
```

### this 绑定

箭头函数不绑定自己的 `this`，而是继承外层作用域的 `this`：

```js
// 传统函数中的 this
const person = {
  name: 'Alice',
  hobbies: ['reading', 'coding'],
  
  printHobbies: function() {
    this.hobbies.forEach(function(hobby) {
      console.log(`${this.name} likes ${hobby}`); // this.name 是 undefined
    });
  }
};

person.printHobbies(); // undefined likes reading, undefined likes coding

// 使用箭头函数
const person2 = {
  name: 'Bob',
  hobbies: ['reading', 'coding'],
  
  printHobbies: function() {
    this.hobbies.forEach(hobby => {
      console.log(`${this.name} likes ${hobby}`); // this.name 是 'Bob'
    });
  }
};

person2.printHobbies(); // Bob likes reading, Bob likes coding
```

## 🎯 Promise 与 Async/Await

### Promise 基础

```js
// 创建 Promise
const fetchData = () => {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      const success = Math.random() > 0.5;
      if (success) {
        resolve({ data: 'Success!' });
      } else {
        reject(new Error('Failed!'));
      }
    }, 1000);
  });
};

// 使用 Promise
fetchData()
  .then(result => {
    console.log(result.data); // Success!
  })
  .catch(error => {
    console.error(error.message); // Failed!
  })
  .finally(() => {
    console.log('Request completed');
  });

// Promise 链
fetchData()
  .then(result => {
    console.log('First then:', result.data);
    return fetchData();
  })
  .then(result => {
    console.log('Second then:', result.data);
  })
  .catch(error => {
    console.error('Error:', error.message);
  });

// Promise.all - 并行执行
const promises = [
  fetchData(),
  fetchData(),
  fetchData()
];

Promise.all(promises)
  .then(results => {
    console.log('All completed:', results);
  })
  .catch(error => {
    console.error('One failed:', error.message);
  });

// Promise.race - 返回最先完成的
Promise.race([fetchData(), fetchData()])
  .then(result => {
    console.log('First to complete:', result);
  });
```

### Async/Await

```js
// async/await 语法
async function fetchDataAsync() {
  try {
    const result = await fetchData();
    console.log('Data:', result.data);
    return result;
  } catch (error) {
    console.error('Error:', error.message);
    throw error;
  }
}

// 并行执行
async function fetchMultipleData() {
  try {
    const [result1, result2, result3] = await Promise.all([
      fetchData(),
      fetchData(),
      fetchData()
    ]);
    console.log('All results:', [result1, result2, result3]);
  } catch (error) {
    console.error('Error:', error.message);
  }
}

// 顺序执行
async function fetchSequentialData() {
  try {
    const result1 = await fetchData();
    console.log('First:', result1.data);
    
    const result2 = await fetchData();
    console.log('Second:', result2.data);
    
    const result3 = await fetchData();
    console.log('Third:', result3.data);
  } catch (error) {
    console.error('Error:', error.message);
  }
}

// 执行
fetchDataAsync();
fetchMultipleData();
fetchSequentialData();
```

## 🎯 Class 与继承

### Class 基本语法

```js
// 定义类
class Person {
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }
  
  // 实例方法
  greet() {
    return `Hello, my name is ${this.name}`;
  }
  
  // Getter
  get info() {
    return `${this.name} is ${this.age} years old`;
  }
  
  // Setter
  set age(newAge) {
    if (newAge > 0 && newAge < 150) {
      this._age = newAge;
    } else {
      console.log('Invalid age');
    }
  }
  
  get age() {
    return this._age;
  }
  
  // 静态方法
  static createAnonymous() {
    return new Person('Anonymous', 0);
  }
}

// 使用类
const person = new Person('Alice', 25);
console.log(person.greet()); // Hello, my name is Alice
console.log(person.info); // Alice is 25 years old
console.log(Person.createAnonymous()); // Person { name: 'Anonymous', age: 0 }
```

```js
const BASE_URL = 'https://api.example.com';

export async function fetchUsers() {
  const response = await fetch(`${BASE_URL}/users`);
  if (!response.ok) {
    throw new Error('Failed to fetch users');
  }
  return response.json();
}

export async function createUser(userData) {
  const response = await fetch(`${BASE_URL}/users`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(userData)
  });
  if (!response.ok) {
    throw new Error('Failed to create user');
  }
  return response.json();
}

export async function updateUser(userId, userData) {
  const response = await fetch(`${BASE_URL}/users/${userId}`, {
    method: 'PUT',
    headers: {
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(userData)
  });
  if (!response.ok) {
    throw new Error('Failed to update user');
  }
  return response.json();
}

export async function deleteUser(userId) {
  const response = await fetch(`${BASE_URL}/users/${userId}`, {

export function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}

// UserList.jsx - 用户列表组件
import React, { useState, useEffect } from 'react';
import { fetchUsers, deleteUser } from './api';
import { formatDate } from './utils';

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  useEffect(() => {
    loadUsers();
  }, []);
  
  const loadUsers = async () => {
    try {
      setLoading(true);
      setError(null);
      const data = await fetchUsers();
      setUsers(data);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };
  
  const handleDelete = async (userId) => {
    if (!confirm('Are you sure you want to delete this user?')) {
      return;
    }
    
    try {
      await deleteUser(userId);
      setUsers(users.filter(user => user.id !== userId));
    } catch (err) {
      alert(`Failed to delete user: ${err.message}`);
    }
  };
  
  if (loading) return <p>Loading users...</p>;
  if (error) return <p>Error: {error}</p>;
  
  return (
    <div>
      <h2>User List ({users.length})</h2>
      {users.length === 0 ? (
        <p>No users found</p>
      ) : (
        <ul>
          {users.map(user => (
            <li key={user.id}>
              <h3>{user.name}</h3>
              <p>Email: {user.email}</p>
              <p>Age: {user.age}</p>
              <p>Created: {formatDate(user.createdAt)}</p>
              <button onClick={() => handleDelete(user.id)}>
                Delete
              </button>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}

export default UserList;

      handleUpdateUser(userData);
    } else {
      handleCreateUser(userData);
    }
  };
  
  return (
    <div>
      <h1>User Management System</h1>
      
      {!showForm ? (
        <div>
          <button onClick={() => {
            setEditingUser(null);
            setShowForm(true);
          }}>
            Add New User
          </button>
          <UserList />
        </div>
      ) : (
        <div>
          <h2>{editingUser ? 'Edit User' : 'Add New User'}</h2>
          <UserForm 
            onSubmit={handleSubmit}
            initialData={editingUser || {}}
          />
          <button onClick={() => setShowForm(false)}>
            Cancel
          </button>
        </div>
      )}
    </div>
  );
}

export default App;
```

## 🔗 相关概念

- **高阶函数**：接受函数作为参数或返回函数的函数
- **纯函数**：相同的输入总是产生相同的输出，无副作用
- **不可变数据**：数据创建后不能被修改，只能创建新数据
- **数组方法**：map、filter、reduce 等函数式编程方法
