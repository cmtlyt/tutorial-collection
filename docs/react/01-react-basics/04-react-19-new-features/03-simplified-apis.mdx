# 简化的 API

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

React 19 简化了许多 API，使开发更加直观和高效。这些简化包括 ref 作为 prop、Context 作为 provider、ref 清理函数等。本章节将详细介绍这些简化的 API。

## 🎯 ref 作为 prop（无需 forwardRef）

### 传统方式（React 18 及之前）

```jsx
// ❌ 旧方式 - 需要使用 forwardRef
const MyInput = React.forwardRef((props, ref) => {
  return <input ref={ref} {...props} />;
});

function App() {
  const inputRef = React.useRef(null);
  
  const handleClick = () => {
    inputRef.current.focus();
  };
  
  return (
    <div>
      <MyInput ref={inputRef} placeholder="Type something..." />
      <button onClick={handleClick}>Focus Input</button>
    </div>
  );
}

export default App;
```

### React 19 新方式

```jsx
// ✅ 新方式 - 直接传递 ref 作为 prop
function MyInput({ ref, ...props }) {
  return <input ref={ref} {...props} />;
}

function App() {
  const inputRef = React.useRef(null);
  
  const handleClick = () => {
    inputRef.current.focus();
  };
  
  return (
    <div>
      <MyInput ref={inputRef} placeholder="Type something..." />
      <button onClick={handleClick}>Focus Input</button>
    </div>
  );
}

export default App;
```

### 复杂组件示例

```jsx
// 自定义组件接收 ref
function CustomInput({ ref, type = 'text', ...props }) {
  return (
    <div className="input-wrapper">
      <input
        ref={ref}
        type={type}
        className="custom-input"
        {...props}
      />
    </div>
  );
}

function Form() {
  const inputRef = React.useRef(null);
  const passwordRef = React.useRef(null);
  
  const handleSubmit = (e) => {
    e.preventDefault();
    console.log('Username:', inputRef.current.value);
    console.log('Password:', passwordRef.current.value);
  };
  
  return (
    <form onSubmit={handleSubmit}>
      <CustomInput
        ref={inputRef}
        name="username"
        placeholder="Username"
        required
      />
      
      <CustomInput
        ref={passwordRef}
        type="password"
        name="password"
        placeholder="Password"
        required
      />
      
      <button type="submit">Submit</button>
    </form>
  );
}

export default Form;
```

## 🎯 Context 作为 provider（无需 Context.Provider）

### 传统方式（React 18 及之前）

```jsx
// ❌ 旧方式 - 需要使用 Context.Provider
const ThemeContext = React.createContext();

function App() {
  return (
    <ThemeContext.Provider value="dark">
      <ChildComponent />
    </ThemeContext.Provider>
  );
}

function ChildComponent() {
  const theme = React.useContext(ThemeContext);
  return <div>Theme: {theme}</div>;
}

export default App;
```

### React 19 新方式

```jsx
// ✅ 新方式 - 直接使用 Context 作为 provider
const ThemeContext = React.createContext();

function App() {
  return (
    <ThemeContext value="dark">
      <ChildComponent />
    </ThemeContext>
  );
}

function ChildComponent() {
  const theme = React.useContext(ThemeContext);
  return <div>Theme: {theme}</div>;
}

export default App;
```

### 多个 Context 示例

```jsx
const ThemeContext = React.createContext();
const UserContext = React.createContext();
const LanguageContext = React.createContext();

function App() {
  const [theme, setTheme] = React.useState('light');
  const [user, setUser] = React.useState(null);
  const [language, setLanguage] = React.useState('en');
  
  return (
    <ThemeContext value={{ theme, setTheme }}>
      <UserContext value={{ user, setUser }}>
        <LanguageContext value={{ language, setLanguage }}>
          <MainContent />
        </LanguageContext>
      </UserContext>
    </ThemeContext>
  );
}

function MainContent() {
  const { theme } = React.useContext(ThemeContext);
  const { user } = React.useContext(UserContext);
  const { language } = React.useContext(LanguageContext);
  
  return (
    <div style={{ 
      background: theme === 'dark' ? '#333' : '#fff',
      color: theme === 'dark' ? '#fff' : '#000'
    }}>
      <h1>Theme: {theme}</h1>
      <p>User: {user ? user.name : 'Guest'}</p>
      <p>Language: {language}</p>
    </div>
  );
}

export default App;
```

## 🎯 ref 清理函数

### 基本用法

```jsx
import React, { useRef, useEffect } from 'react';

function Timer() {
  const intervalRef = useRef(null);
  
  useEffect(() => {
    // 设置定时器
    intervalRef.current = setInterval(() => {
      console.log('Tick');
    }, 1000);
    
    // 清理函数
    return () => {
      if (intervalRef.current) {
        clearInterval(intervalRef.current);
      }
    };
  }, []);
  
  return <div>Timer running...</div>;
}

export default Timer;
```

### 事件监听器清理

```jsx
import React, { useRef, useEffect } from 'react';

function WindowSize() {
  const [size, setSize] = React.useState({ width: 0, height: 0 });
  
  useEffect(() => {
    const handleResize = () => {
      setSize({
        width: window.innerWidth,
        height: window.innerHeight
      });
    };
    
    window.addEventListener('resize', handleResize);
    handleResize(); // 初始化
    
    // 清理函数
    return () => {
      window.removeEventListener('resize', handleResize);
    };
  }, []);
  
  return (
    <div>
      <p>Width: {size.width}px</p>
      <p>Height: {size.height}px</p>
    </div>
  );
}

export default WindowSize;
```

### WebSocket 清理

```jsx
import React, { useRef, useEffect, useState } from 'react';

function WebSocketComponent() {
  const [messages, setMessages] = useState([]);
  const wsRef = useRef(null);
  
  useEffect(() => {
    // 创建 WebSocket 连接
    wsRef.current = new WebSocket('ws://localhost:8080');
    
    wsRef.current.onopen = () => {
      console.log('WebSocket connected');
    };
    
    wsRef.current.onmessage = (event) => {
      const message = JSON.parse(event.data);
      setMessages(prev => [...prev, message]);
    };
    
    wsRef.current.onerror = (error) => {
      console.error('WebSocket error:', error);
    };
    
    // 清理函数
    return () => {
      if (wsRef.current) {
        wsRef.current.close();
      }
    };
  }, []);
  
  return (
    <div>
      <h2>Messages</h2>
      <ul>
        {messages.map((msg, index) => (
          <li key={index}>{JSON.stringify(msg)}</li>
        ))}
      </ul>
    </div>
  );
}

export default WebSocketComponent;
```

## 📝 实际应用示例

### 综合示例：模态框组件

```jsx
import React, { useRef, useEffect, createContext, useContext } from 'react';

// 使用 Context 作为 provider
const ModalContext = createContext();

function ModalProvider({ children }) {
  const [isModalOpen, setIsModalOpen] = React.useState(false);
  const modalRef = useRef(null);
  
  const openModal = () => setIsModalOpen(true);
  const closeModal = () => setIsModalOpen(false);
  
  // 点击外部关闭
  useEffect(() => {
    const handleClickOutside = (event) => {
      if (modalRef.current && !modalRef.current.contains(event.target)) {
        closeModal();
      }
    };
    
    if (isModalOpen) {
      document.addEventListener('mousedown', handleClickOutside);
    }
    
    return () => {
      document.removeEventListener('mousedown', handleClickOutside);
    };
  }, [isModalOpen]);
  
  // ESC 键关闭
  useEffect(() => {
    const handleEscape = (event) => {
      if (event.key === 'Escape') {
        closeModal();
      }
    };
    
    if (isModalOpen) {
      document.addEventListener('keydown', handleEscape);
    }
    
    return () => {
      document.removeEventListener('keydown', handleEscape);
    };
  }, [isModalOpen]);
  
  return (
    <ModalContext value={{ isModalOpen, openModal, closeModal, modalRef }}>
      {children}
    </ModalContext>
  );
}

// 自定义组件接收 ref
function ModalContent({ ref, children }) {
  const { isModalOpen, closeModal } = useContext(ModalContext);
  
  if (!isModalOpen) return null;
  
  return (
    <div className="modal-overlay">
      <div ref={ref} className="modal">
        <button className="modal-close" onClick={closeModal}>
          ×
        </button>
        {children}
      </div>
    </div>
  );
}

function ModalTrigger({ children }) {
  const { openModal } = useContext(ModalContext);
  
  return React.cloneElement(children, {
    onClick: () => {
      openModal();
      if (children.props.onClick) {
        children.props.onClick();
      }
    }
  });
}

function App() {
  return (
    <ModalProvider>
      <div>
        <h1>Modal Example</h1>
        
        <ModalTrigger>
          <button>Open Modal</button>
        </ModalTrigger>
        
        <ModalContent>
          <h2>Modal Title</h2>
          <p>This is the modal content.</p>
          <p>You can click outside or press ESC to close.</p>
        </ModalContent>
      </div>
    </ModalProvider>
  );
}

export default App;
```
