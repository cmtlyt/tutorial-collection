# React 19 数据获取

> **版本**：v1.0

## 📚 概念介绍

React 19 引入了新的数据获取特性，简化了服务器组件和客户端组件之间的数据传递。本章节将详细介绍 React 19 的数据获取新特性，包括 Server Components、Suspense for Data Fetching 等。

## 🎯 use Hook

### 基础用法

```jsx
import React, { use } from 'react';

// 模拟异步数据获取
function fetchUserData(userId) {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve({
        id: userId,
        name: `User ${userId}`,
        email: `user${userId}@example.com`
      });
    }, 1000);
  });
}

// 数据获取函数
function getUserData(userId) {
  return fetchUserData(userId);
}

function UserProfile({ userId }) {
  // 使用 use Hook 获取数据
  const user = use(getUserData(userId));
  
  return (
    <div style={{ padding: '24px' }}>
      <h1>User Profile</h1>
      <div style={{ padding: '16px', border: '1px solid #ddd', borderRadius: '4px' }}>
        <p><strong>Name:</strong> {user.name}</p>
        <p><strong>Email:</strong> {user.email}</p>
        <p><strong>ID:</strong> {user.id}</p>
      </div>
    </div>
  );
}

export default UserProfile;
```

### 与 Suspense 配合

```jsx
import React, { use, Suspense } from 'react';

function fetchPosts() {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve(Array.from({ length: 5 }, (_, i) => ({
        id: i + 1,
        title: `Post ${i + 1}`,
        body: `This is the content of post ${i + 1}`
      })));
    }, 1500);
  });
}

function getPosts() {
  return fetchPosts();
}

function PostsList() {
  const posts = use(getPosts());
  
  return (
    <div>
      <h2>Posts List</h2>
      <ul>
        {posts.map(post => (
          <li key={post.id} style={{ marginBottom: '12px', padding: '12px', border: '1px solid #ddd', borderRadius: '4px' }}>
            <h3 style={{ margin: '0 0 8px 0' }}>{post.title}</h3>
            <p style={{ margin: 0, color: '#666' }}>{post.body}</p>
          </li>
        ))}
      </ul>
    </div>
  );
}

function App() {
  return (
    <div style={{ padding: '24px' }}>
      <h1>React 19 Data Fetching</h1>
      
      <Suspense fallback={<div>Loading posts...</div>}>
        <PostsList />
      </Suspense>
    </div>
  );
}

export default App;
```

## 🎯 并行数据获取

### 多个数据源

```jsx
import React, { use, Suspense } from 'react';

function fetchUsers() {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve(Array.from({ length: 3 }, (_, i) => ({
        id: i + 1,
        name: `User ${i + 1}`,
        email: `user${i + 1}@example.com`
      })));
    }, 1000);
  });
}

function fetchPosts() {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve(Array.from({ length: 5 }, (_, i) => ({
        id: i + 1,
        title: `Post ${i + 1}`,
        userId: (i % 3) + 1
      })));
    }, 1200);
  });
}

function fetchComments() {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve(Array.from({ length: 3 }, (_, i) => ({
        id: i + 1,
        text: `Comment ${i + 1}`,
        postId: i + 1
      })));
    }, 800);
  });
}

function Dashboard() {
  // 并行获取多个数据源
  const users = use(fetchUsers());
  const posts = use(fetchPosts());
  const comments = use(fetchComments());
  
  return (
    <div>
      <h1>Dashboard</h1>
      
      <div style={{ marginBottom: '24px' }}>
        <h2>Users ({users.length})</h2>
        <ul>
          {users.map(user => (
            <li key={user.id}>{user.name} - {user.email}</li>
          ))}
        </ul>
      </div>
      
      <div style={{ marginBottom: '24px' }}>
        <h2>Posts ({posts.length})</h2>
        <ul>
          {posts.map(post => (
            <li key={post.id}>{post.title} (by User {post.userId})</li>
          ))}
        </ul>
      </div>
      
      <div>
        <h2>Comments ({comments.length})</h2>
        <ul>
          {comments.map(comment => (
            <li key={comment.id}>{comment.text} (on Post {comment.postId})</li>
          ))}
        </ul>
      </div>
    </div>
  );
}

function App() {
  return (
    <div style={{ padding: '24px' }}>
      <Suspense fallback={<div>Loading dashboard...</div>}>
        <Dashboard />
      </Suspense>
    </div>
  );
}

export default App;
```

## 🎯 错误处理

### Error Boundary

```jsx
import React, { use, Suspense } from 'react';

class ErrorBoundary extends React.Component {
  constructor(props) {
    super(props);
    this.state = { hasError: false, error: null };
  }
  
  static getDerivedStateFromError(error) {
    return { hasError: true, error };
  }
  
  render() {
    if (this.state.hasError) {
      return (
        <div style={{ padding: '24px', background: '#fff2f0', border: '1px solid #ffccc7', borderRadius: '4px' }}>
          <h2>Something went wrong!</h2>
          <p>{this.state.error?.message}</p>
          <button
            onClick={() => this.setState({ hasError: false, error: null })}
            style={{
              padding: '8px 16px',
              background: '#1890ff',
              color: 'white',
              border: 'none',
              borderRadius: '4px',
              cursor: 'pointer',
              marginTop: '16px'
            }}
          >
            Try again
          </button>
        </div>
      );
    }
    
    return this.props.children;
  }
}

function fetchWithError(shouldFail) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (shouldFail) {
        reject(new Error('Failed to fetch data'));
      } else {
        resolve({ data: 'Success!' });
      }
    }, 1000);
  });
}

function DataComponent({ shouldFail }) {
  const result = use(fetchWithError(shouldFail));
  
  return (
    <div>
      <h2>Data Result</h2>
      <p>{result.data}</p>
    </div>
  );
}

function App() {
  const [shouldFail, setShouldFail] = React.useState(false);
  
  return (
    <div style={{ padding: '24px' }}>
      <h1>Error Handling</h1>
      
      <div style={{ marginBottom: '16px' }}>
        <label>
          <input
            type="checkbox"
            checked={shouldFail}
            onChange={(e) => setShouldFail(e.target.checked)}
            style={{ marginRight: '8px' }}
          />
          Simulate error
        </label>
      </div>
      
      <ErrorBoundary key={shouldFail}>
        <Suspense fallback={<div>Loading...</div>}>
          <DataComponent shouldFail={shouldFail} />
        </Suspense>
      </ErrorBoundary>
    </div>
  );
}

export default App;
```

## 🎯 缓存与重新验证

### 数据缓存

```jsx
import React, { use, Suspense, useState } from 'react';

// 简单的缓存实现
const cache = new Map();

function cachedFetch(key, fetcher) {
  if (cache.has(key)) {
    return Promise.resolve(cache.get(key));
  }
  
  return fetcher().then(data => {
    cache.set(key, data);
    return data;
  });
}

function fetchProducts() {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve(Array.from({ length: 10 }, (_, i) => ({
        id: i + 1,
        name: `Product ${i + 1}`,
        price: (Math.random() * 100).toFixed(2),
        inStock: Math.random() > 0.3
      })));
    }, 1000);
  });
}

function getProducts() {
  return cachedFetch('products', fetchProducts);
}

function ProductsList() {
  const products = use(getProducts());
  
  return (
    <div>
      <h2>Products ({products.length})</h2>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(200px, 1fr))', gap: '16px' }}>
        {products.map(product => (
          <div
            key={product.id}
            style={{
              padding: '16px',
              border: '1px solid #ddd',
              borderRadius: '4px',
              background: product.inStock ? 'white' : '#f5f5f5'
            }}
          >
            <h3 style={{ margin: '0 0 8px 0' }}>{product.name}</h3>
            <p style={{ margin: '0 0 8px 0', color: '#666' }}>
              ${product.price}
            </p>
            <span style={{
              padding: '4px 8px',
              background: product.inStock ? '#52c41a' : '#ff4d4f',
              color: 'white',
              borderRadius: '4px',
              fontSize: '12px'
            }}>
              {product.inStock ? 'In Stock' : 'Out of Stock'}
            </span>
          </div>
        ))}
      </div>
    </div>
  );
}

function App() {
  const [key, setKey] = useState(0);
  
  const handleRefresh = () => {
    // 清除缓存并重新获取
    cache.delete('products');
    setKey(prev => prev + 1);
  };
  
  return (
    <div style={{ padding: '24px' }}>
      <h1>Products</h1>
      
      <button
        onClick={handleRefresh}
        style={{
          padding: '8px 16px',
          background: '#1890ff',
          color: 'white',
          border: 'none',
          borderRadius: '4px',
          cursor: 'pointer',
          marginBottom: '16px'
        }}
      >
        Refresh
      </button>
      
      <Suspense fallback={<div>Loading products...</div>} key={key}>
        <ProductsList />
      </Suspense>
    </div>
  );
}

export default App;
```

## 📝 实际应用示例

### 综合示例：博客应用

```jsx
import React, { use, Suspense, useState } from 'react';

// 数据获取函数
function fetchBlogPosts() {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve(Array.from({ length: 5 }, (_, i) => ({
        id: i + 1,
        title: `Blog Post ${i + 1}`,
        excerpt: `This is an excerpt for blog post ${i + 1}`,
        author: `Author ${i + 1}`,
        date: new Date(Date.now() - i * 86400000).toLocaleDateString(),
        content: `This is the full content of blog post ${i + 1}. It contains detailed information about the topic.`
      })));
    }, 1000);
  });
}

function fetchPostDetail(postId) {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve({
        id: postId,
        title: `Blog Post ${postId}`,
        content: `This is the full content of blog post ${postId}. It contains detailed information about the topic.`,
        author: `Author ${postId}`,
        date: new Date().toLocaleDateString(),
        tags: ['React', 'JavaScript', 'Web Development']
      });
    }, 800);
  });
}

function fetchComments(postId) {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve(Array.from({ length: 3 }, (_, i) => ({
        id: i + 1,
        author: `Commenter ${i + 1}`,
        text: `This is comment ${i + 1} on post ${postId}`,
        date: new Date().toLocaleDateString()
      })));
    }, 600);
  });
}

// 组件
function BlogList() {
  const posts = use(fetchBlogPosts());
  
  return (
    <div>
      <h2>Recent Posts</h2>
      <ul style={{ listStyle: 'none', padding: 0 }}>
        {posts.map(post => (
          <li
            key={post.id}
            style={{
              padding: '16px',
              marginBottom: '16px',
              border: '1px solid #ddd',
              borderRadius: '4px',
              cursor: 'pointer',
              transition: 'box-shadow 0.2s'
            }}
          >
            <h3 style={{ margin: '0 0 8px 0' }}>{post.title}</h3>
            <p style={{ margin: '0 0 8px 0', color: '#666' }}>{post.excerpt}</p>
            <small style={{ color: '#999' }}>
              By {post.author} • {post.date}
            </small>
          </li>
        ))}
      </ul>
    </div>
  );
}

function PostDetail({ postId }) {
  const post = use(fetchPostDetail(postId));
  const comments = use(fetchComments(postId));
  
  return (
    <div>
      <article style={{ marginBottom: '32px' }}>
        <h1>{post.title}</h1>
        <div style={{ marginBottom: '16px', color: '#666' }}>
          By {post.author} • {post.date}
        </div>
        <div style={{ marginBottom: '16px' }}>
          {post.tags.map(tag => (
            <span
              key={tag}
              style={{
                display: 'inline-block',
                padding: '4px 8px',
                background: '#e6f7ff',
                color: '#1890ff',
                borderRadius: '4px',
                fontSize: '12px',
                marginRight: '8px'
              }}
            >
              {tag}
            </span>
          ))}
        </div>
        <p style={{ lineHeight: '1.8' }}>{post.content}</p>
      </article>
      
      <section>
        <h2>Comments ({comments.length})</h2>
        <ul style={{ listStyle: 'none', padding: 0 }}>
          {comments.map(comment => (
            <li
              key={comment.id}
              style={{
                padding: '12px',
                marginBottom: '12px',
                background: '#f5f5f5',
                borderRadius: '4px'
              }}
            >
              <strong>{comment.author}</strong>
              <small style={{ display: 'block', color: '#999', marginBottom: '4px' }}>
                {comment.date}
              </small>
              <p style={{ margin: 0 }}>{comment.text}</p>
            </li>
          ))}
        </ul>
      </section>
    </div>
  );
}

function BlogApp() {
  const [selectedPostId, setSelectedPostId] = useState(null);
  
  return (
    <div style={{ padding: '24px', maxWidth: '1000px', margin: '0 auto' }}>
      <header style={{ marginBottom: '32px', paddingBottom: '16px', borderBottom: '2px solid #ddd' }}>
        <h1 style={{ margin: 0 }}>My Blog</h1>
        <p style={{ margin: '8px 0 0 0', color: '#666' }}>
          Exploring React 19 Data Fetching
        </p>
      </header>
      
      <Suspense fallback={<div>Loading...</div>}>
        {selectedPostId ? (
          <div>
            <button
              onClick={() => setSelectedPostId(null)}
              style={{
                padding: '8px 16px',
                background: '#1890ff',
                color: 'white',
                border: 'none',
                borderRadius: '4px',
                cursor: 'pointer',
                marginBottom: '16px'
              }}
            >
              ← Back to Posts
            </button>
            <PostDetail postId={selectedPostId} />
          </div>
        ) : (
          <BlogList />
        )}
      </Suspense>
    </div>
  );
}

export default BlogApp;
```
