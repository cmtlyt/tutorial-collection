# React Query

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

React Query（现称 TanStack Query）是一个强大的数据获取和状态管理库，它简化了服务器状态的管理，提供了缓存、重试、轮询等功能。React Query 让开发者能够专注于业务逻辑，而不是手动管理数据获取和状态。

## 🎯 基础用法

### 安装和设置

```jsx
import React from 'react';
import { QueryClient, QueryClientProvider, useQuery } from '@tanstack/react-query';

// 创建 QueryClient
const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      refetchOnWindowFocus: false,
      retry: 1,
      staleTime: 5 * 60 * 1000, // 5 分钟
    },
  },
});

// 模拟 API 调用
async function fetchUsers() {
  const response = await fetch('https://jsonplaceholder.typicode.com/users');
  if (!response.ok) {
    throw new Error('Network response was not ok');
  }
  return response.json();
}

function UsersList() {
  // 使用 useQuery 获取数据
  const { data, isLoading, error, isError } = useQuery({
    queryKey: ['users'],
    queryFn: fetchUsers,
  });
  
  if (isLoading) {
    return <div>Loading...</div>;
  }
  
  if (isError) {
    return <div>Error: {error.message}</div>;
  }
  
  return (
    <div>
      <h2>Users List</h2>
      <ul>
        {data.map(user => (
          <li key={user.id}>
            <strong>{user.name}</strong> - {user.email}
          </li>
        ))}
      </ul>
    </div>
  );
}

function App() {
  return (
    <QueryClientProvider client={queryClient}>
      <div style={{ padding: '24px' }}>
        <h1>React Query Example</h1>
        <UsersList />
      </div>
    </QueryClientProvider>
  );
}

export default App;
```

## 🎯 数据修改

### 使用 useMutation

```jsx
import React from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

async function fetchTodos() {
  const response = await fetch('https://jsonplaceholder.typicode.com/todos?_limit=5');
  if (!response.ok) {
    throw new Error('Network response was not ok');
  }
  return response.json();
}

async function createTodo(newTodo) {
  const response = await fetch('https://jsonplaceholder.typicode.com/todos', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(newTodo),
  });
  if (!response.ok) {
    throw new Error('Network response was not ok');
  }
  return response.json();
}

async function deleteTodo(id) {
  const response = await fetch(`https://jsonplaceholder.typicode.com/todos/${id}`, {
    method: 'DELETE',
  });
  if (!response.ok) {
    throw new Error('Network response was not ok');
  }
  return id;
}

function TodoApp() {
  const queryClient = useQueryClient();
  
  // 获取 todos
  const { data: todos, isLoading, error } = useQuery({
    queryKey: ['todos'],
    queryFn: fetchTodos,
  });
  
  // 创建 todo mutation
  const createMutation = useMutation({
    mutationFn: createTodo,
    onSuccess: (newTodo) => {
      // 成功后更新缓存
      queryClient.setQueryData(['todos'], (old) => [...old, newTodo]);
    },
  });
  
  // 删除 todo mutation
  const deleteMutation = useMutation({
    mutationFn: deleteTodo,
    onSuccess: (deletedId) => {
      // 成功后更新缓存
      queryClient.setQueryData(['todos'], (old) => 
        old.filter(todo => todo.id !== deletedId)
      );
    },
  });
  
  const [newTodoTitle, setNewTodoTitle] = React.useState('');
  
  const handleAddTodo = (e) => {
    e.preventDefault();
    if (!newTodoTitle.trim()) return;
    
    createMutation.mutate({
      title: newTodoTitle,
      completed: false,
      userId: 1,
    });
    
    setNewTodoTitle('');
  };
  
  const handleDeleteTodo = (id) => {
    deleteMutation.mutate(id);
  };
  
  if (isLoading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;
  
  return (
    <div style={{ padding: '24px', maxWidth: '600px', margin: '0 auto' }}>
      <h1>Todo App</h1>
      
      {/* 添加 todo 表单 */}
      <form onSubmit={handleAddTodo} style={{ marginBottom: '24px' }}>
        <div style={{ display: 'flex', gap: '8px' }}>
          <input
            type="text"
            value={newTodoTitle}
            onChange={(e) => setNewTodoTitle(e.target.value)}
            placeholder="Enter new todo"
            style={{
              flex: 1,
              padding: '8px',
              border: '1px solid #ddd',
              borderRadius: '4px'
            }}
          />
          <button
            type="submit"
            disabled={createMutation.isLoading}
            style={{
              padding: '8px 16px',
              background: '#1890ff',
              color: 'white',
              border: 'none',
              borderRadius: '4px',
              cursor: createMutation.isLoading ? 'not-allowed' : 'pointer'
            }}
          >
            {createMutation.isLoading ? 'Adding...' : 'Add'}
          </button>
        </div>
      </form>
      
      {/* Todo 列表 */}
      <ul style={{ listStyle: 'none', padding: 0 }}>
        {todos.map(todo => (
          <li
            key={todo.id}
            style={{
              padding: '12px',
              marginBottom: '8px',
              border: '1px solid #ddd',
              borderRadius: '4px',
              display: 'flex',
              justifyContent: 'space-between',
              alignItems: 'center'
            }}
          >
            <span style={{ textDecoration: todo.completed ? 'line-through' : 'none' }}>
              {todo.title}
            </span>
            <button
              onClick={() => handleDeleteTodo(todo.id)}
              disabled={deleteMutation.isLoading}
              style={{
                padding: '4px 8px',
                background: '#ff4d4f',
                color: 'white',
                border: 'none',
                borderRadius: '4px',
                cursor: deleteMutation.isLoading ? 'not-allowed' : 'pointer'
              }}
            >
              Delete
            </button>
          </li>
        ))}
      </ul>
    </div>
  );
}

export default TodoApp;
```

## 🎯 缓存管理

### 缓存配置

```jsx
import React from 'react';
import { useQuery } from '@tanstack/react-query';

async function fetchPosts() {
  const response = await fetch('https://jsonplaceholder.typicode.com/posts?_limit=10');
  if (!response.ok) {
    throw new Error('Network response was not ok');
  }
  return response.json();
}

function PostsList() {
  const { data, isLoading, error, refetch } = useQuery({
    queryKey: ['posts'],
    queryFn: fetchPosts,
    // 缓存配置
    staleTime: 5 * 60 * 1000, // 数据在 5 分钟内被认为是新鲜的
    cacheTime: 10 * 60 * 1000, // 缓存保留 10 分钟
    refetchOnWindowFocus: false, // 不在窗口聚焦时重新获取
    refetchOnReconnect: true, // 重新连接时重新获取
    refetchInterval: 60 * 1000, // 每 60 秒自动重新获取
    retry: 3, // 失败时重试 3 次
    retryDelay: (attemptIndex) => Math.min(1000 * 2 ** attemptIndex, 30000), // 指数退避
  });
  
  if (isLoading) return <div>Loading posts...</div>;
  if (error) return <div>Error: {error.message}</div>;
  
  return (
    <div style={{ padding: '24px' }}>
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '16px' }}>
        <h1>Posts List</h1>
        <button
          onClick={() => refetch()}
          style={{
            padding: '8px 16px',
            background: '#1890ff',
            color: 'white',
            border: 'none',
            borderRadius: '4px',
            cursor: 'pointer'
          }}
        >
          Refresh
        </button>
      </div>
      
      <ul style={{ listStyle: 'none', padding: 0 }}>
        {data.map(post => (
          <li
            key={post.id}
            style={{
              padding: '16px',
              marginBottom: '12px',
              border: '1px solid #ddd',
              borderRadius: '4px'
            }}
          >
            <h3 style={{ margin: '0 0 8px 0' }}>{post.title}</h3>
            <p style={{ margin: 0, color: '#666' }}>{post.body}</p>
          </li>
        ))}
      </ul>
    </div>
  );
}

export default PostsList;
```

### 手动缓存管理

```jsx
import React from 'react';
import { useQuery, useQueryClient } from '@tanstack/react-query';

async function fetchComments() {
  const response = await fetch('https://jsonplaceholder.typicode.com/comments?_limit=5');
  if (!response.ok) {
    throw new Error('Network response was not ok');
  }
  return response.json();
}

function CommentsManager() {
  const queryClient = useQueryClient();
  
  const { data: comments } = useQuery({
    queryKey: ['comments'],
    queryFn: fetchComments,
  });
  
  const handleAddComment = () => {
    // 手动添加新评论到缓存
    const newComment = {
      id: Date.now(),
      name: 'New User',
      email: 'newuser@example.com',
      body: 'This is a new comment added manually',
    };
    
    queryClient.setQueryData(['comments'], (old) => [...old, newComment]);
  };
  
  const handleClearCache = () => {
    // 清除特定查询的缓存
    queryClient.removeQueries(['comments']);
  };
  
  const handleInvalidateCache = () => {
    // 使缓存失效，触发重新获取
    queryClient.invalidateQueries(['comments']);
  };
  
  const handleResetCache = () => {
    // 重置缓存到初始状态
    queryClient.resetQueries(['comments']);
  };
  
  return (
    <div style={{ padding: '24px' }}>
      <h1>Comments Manager</h1>
      
      <div style={{ marginBottom: '16px' }}>
        <button
          onClick={handleAddComment}
          style={{
            padding: '8px 16px',
            background: '#52c41a',
            color: 'white',
            border: 'none',
            borderRadius: '4px',
            cursor: 'pointer',
            marginRight: '8px'
          }}
        >
          Add Comment
        </button>
        
        <button
          onClick={handleInvalidateCache}
          style={{
            padding: '8px 16px',
            background: '#1890ff',
            color: 'white',
            border: 'none',
            borderRadius: '4px',
            cursor: 'pointer',
            marginRight: '8px'
          }}
        >
          Invalidate Cache
        </button>
        
        <button
          onClick={handleResetCache}
          style={{
            padding: '8px 16px',
            background: '#faad14',
            color: 'white',
            border: 'none',
            borderRadius: '4px',
            cursor: 'pointer',
            marginRight: '8px'
          }}
        >
          Reset Cache
        </button>
        
        <button
          onClick={handleClearCache}
          style={{
            padding: '8px 16px',
            background: '#ff4d4f',
            color: 'white',
            border: 'none',
            borderRadius: '4px',
            cursor: 'pointer'
          }}
        >
          Clear Cache
        </button>
      </div>
      
      <ul style={{ listStyle: 'none', padding: 0 }}>
        {comments?.map(comment => (
          <li
            key={comment.id}
            style={{
              padding: '12px',
              marginBottom: '8px',
              border: '1px solid #ddd',
              borderRadius: '4px'
            }}
          >
            <strong>{comment.name}</strong> ({comment.email})
            <p style={{ margin: '8px 0 0 0', color: '#666' }}>{comment.body}</p>
          </li>
        ))}
      </ul>
    </div>
  );
}

export default CommentsManager;
```

## 📝 实际应用示例

### 综合示例：博客应用

```jsx
import React from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

async function fetchPosts() {
  const response = await fetch('https://jsonplaceholder.typicode.com/posts?_limit=5');
  if (!response.ok) throw new Error('Failed to fetch posts');
  return response.json();
}

async function fetchPost(id) {
  const response = await fetch(`https://jsonplaceholder.typicode.com/posts/${id}`);
  if (!response.ok) throw new Error('Failed to fetch post');
  return response.json();
}

async function fetchComments(postId) {
  const response = await fetch(`https://jsonplaceholder.typicode.com/posts/${postId}/comments`);
  if (!response.ok) throw new Error('Failed to fetch comments');
  return response.json();
}

async function createPost(post) {
  const response = await fetch('https://jsonplaceholder.typicode.com/posts', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(post),
  });
  if (!response.ok) throw new Error('Failed to create post');
  return response.json();
}

function BlogApp() {
  const queryClient = useQueryClient();
  const [selectedPostId, setSelectedPostId] = React.useState(null);
  
  // 获取文章列表
  const { data: posts, isLoading: postsLoading } = useQuery({
    queryKey: ['posts'],
    queryFn: fetchPosts,
  });
  
  // 获取选中的文章
  const { data: selectedPost, isLoading: postLoading } = useQuery({
    queryKey: ['post', selectedPostId],
    queryFn: () => fetchPost(selectedPostId),
    enabled: !!selectedPostId,
  });
  
  // 获取评论
  const { data: comments, isLoading: commentsLoading } = useQuery({
    queryKey: ['comments', selectedPostId],
    queryFn: () => fetchComments(selectedPostId),
    enabled: !!selectedPostId,
  });
  
  // 创建文章
  const createMutation = useMutation({
    mutationFn: createPost,
    onSuccess: (newPost) => {
      queryClient.setQueryData(['posts'], (old) => [newPost, ...old]);
    },
  });
  
  const [newPostTitle, setNewPostTitle] = React.useState('');
  const [newPostBody, setNewPostBody] = React.useState('');
  
  const handleCreatePost = (e) => {
    e.preventDefault();
    createMutation.mutate({
      title: newPostTitle,
      body: newPostBody,
      userId: 1,
    });
    setNewPostTitle('');
    setNewPostBody('');
  };
  
  if (postsLoading) return <div>Loading posts...</div>;
  
  return (
    <div style={{ padding: '24px', maxWidth: '1200px', margin: '0 auto' }}>
      <h1>Blog App</h1>
      
      {/* 创建文章表单 */}
      <div style={{ marginBottom: '24px', padding: '16px', background: '#f5f5f5', borderRadius: '4px' }}>
        <h2>Create New Post</h2>
        <form onSubmit={handleCreatePost}>
          <div style={{ marginBottom: '12px' }}>
            <input
              type="text"
              value={newPostTitle}
              onChange={(e) => setNewPostTitle(e.target.value)}
              placeholder="Post title"
              required
              style={{
                width: '100%',
                padding: '8px',
                border: '1px solid #ddd',
                borderRadius: '4px'
              }}
            />
          </div>
          <div style={{ marginBottom: '12px' }}>
            <textarea
              value={newPostBody}
              onChange={(e) => setNewPostBody(e.target.value)}
              placeholder="Post content"
              required
              rows={4}
              style={{
                width: '100%',
                padding: '8px',
                border: '1px solid #ddd',
                borderRadius: '4px'
              }}
            />
          </div>
          <button
            type="submit"
            disabled={createMutation.isLoading}
            style={{
              padding: '8px 16px',
              background: '#1890ff',
              color: 'white',
              border: 'none',
              borderRadius: '4px',
              cursor: createMutation.isLoading ? 'not-allowed' : 'pointer'
            }}
          >
            {createMutation.isLoading ? 'Creating...' : 'Create Post'}
          </button>
        </form>
      </div>
      
      <div style={{ display: 'flex', gap: '24px' }}>
        {/* 文章列表 */}
        <div style={{ flex: 1 }}>
          <h2>Posts ({posts.length})</h2>
          <ul style={{ listStyle: 'none', padding: 0 }}>
            {posts.map(post => (
              <li
                key={post.id}
                onClick={() => setSelectedPostId(post.id)}
                style={{
                  padding: '12px',
                  marginBottom: '8px',
                  border: selectedPostId === post.id ? '2px solid #1890ff' : '1px solid #ddd',
                  borderRadius: '4px',
                  cursor: 'pointer',
                  background: selectedPostId === post.id ? '#e6f7ff' : 'white'
                }}
              >
                <h3 style={{ margin: '0 0 4px 0' }}>{post.title}</h3>
                <p style={{ margin: 0, fontSize: '14px', color: '#666' }}>
                  {post.body.substring(0, 100)}...
                </p>
              </li>
            ))}
          </ul>
        </div>
        
        {/* 文章详情 */}
        <div style={{ flex: 2 }}>
          {selectedPostId ? (
            postLoading ? (
              <div>Loading post...</div>
            ) : selectedPost ? (
              <div>
                <h2>{selectedPost.title}</h2>
                <p style={{ lineHeight: '1.6' }}>{selectedPost.body}</p>
                
                {commentsLoading ? (
                  <div>Loading comments...</div>
                ) : (
                  <div style={{ marginTop: '24px' }}>
                    <h3>Comments ({comments.length})</h3>
                    <ul style={{ listStyle: 'none', padding: 0 }}>
                      {comments.map(comment => (
                        <li
                          key={comment.id}
                          style={{
                            padding: '12px',
                            marginBottom: '8px',
                            border: '1px solid #ddd',
                            borderRadius: '4px'
                          }}
                        >
                          <strong>{comment.name}</strong> ({comment.email})
                          <p style={{ margin: '8px 0 0 0', color: '#666' }}>{comment.body}</p>
                        </li>
                      ))}
                    </ul>
                  </div>
                )}
              </div>
            ) : null
          ) : (
            <div style={{ padding: '24px', textAlign: 'center', color: '#999' }}>
              Select a post to view details
            </div>
          )}
        </div>
      </div>
    </div>
  );
}

export default BlogApp;
```
