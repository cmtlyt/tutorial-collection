# React Hook Form

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

React Hook Form 是一个高性能的 React 表单库，它使用 Hooks 来管理表单状态，提供简洁的 API 和优秀的性能表现。相比传统的受控组件方式，React Hook Form 减少了不必要的重新渲染，特别适合处理复杂表单。

## 🎯 基础用法

### 简单表单

```jsx
import React from 'react';
import { useForm } from 'react-hook-form';

function CustomValidationForm() {
  const {
    register,
    handleSubmit,
    formState: { errors }
  } = useForm();
  
  // 自定义异步验证函数
  const checkUsernameAvailability = async (username) => {
    // 模拟 API 调用
    await new Promise(resolve => setTimeout(resolve, 1000));
    
    // 模拟已存在的用户名
    const existingUsernames = ['admin', 'user', 'test'];
    
    if (existingUsernames.includes(username)) {
      return false;
    }
    return true;
  };
  
  const onSubmit = async (data) => {
    console.log('Form data:', data);
    alert('Form submitted successfully!');
  };
  
  return (
    <div style={{ padding: '24px', maxWidth: '600px', margin: '0 auto' }}>
      <h2>Custom Validation</h2>
      <form onSubmit={handleSubmit(onSubmit)}>
        <div style={{ marginBottom: '16px' }}>
          <label style={{ display: 'block', marginBottom: '8px' }}>

            {...register('username', {
              required: 'Username is required',
              validate: {
                minLength: (value) => value.length >= 3 || 'Username must be at least 3 characters',
                maxLength: (value) => value.length <= 20 || 'Username must not exceed 20 characters',
                pattern: (value) => /^[a-zA-Z0-9_]+$/.test(value) || 'Username can only contain letters, numbers, and underscores',

                  return isAvailable || 'Username is already taken';
                }

              width: '100%',
              padding: '8px',
              border: errors.username ? '1px solid red' : '1px solid #ccc',
```

### 动态表单

```jsx
import React from 'react';
import { useForm, useFieldArray } from 'react-hook-form';

function DynamicForm() {
  const {
    register,
    control,
    handleSubmit,
    formState: { errors }
  } = useForm({
    defaultValues: {
      users: [{ firstName: '', lastName: '' }]
    }
  });
  
  const { fields, append, remove } = useFieldArray({
    control,
    name: 'users'
  });
  
  const onSubmit = (data) => {
    console.log('Form data:', data);
    alert('Form submitted successfully!');
  };
  
  return (
    <div style={{ padding: '24px', maxWidth: '800px', margin: '0 auto' }}>
      <h2>Dynamic Form</h2>
      <form onSubmit={handleSubmit(onSubmit)}>
        {fields.map((field, index) => (
          <div
            key={field.id}
            style={{
              padding: '16px',
              marginBottom: '16px',
              border: '1px solid #ddd',
              borderRadius: '4px'
            }}
          >
            <h3>User {index + 1}</h3>
            
            <div style={{ marginBottom: '12px' }}>
              <label style={{ display: 'block', marginBottom: '8px' }}>
                First Name:
              </label>
              <input
                {...register(`users.${index}.firstName`, {
                  required: 'First name is required'
                })}
                style={{
                  width: '100%',
                  padding: '8px',
                  border: errors.users?.[index]?.firstName ? '1px solid red' : '1px solid #ccc',
                  borderRadius: '4px'
                }}
              />
              {errors.users?.[index]?.firstName && (
                <p style={{ color: 'red', fontSize: '12px', marginTop: '4px' }}>
                  {errors.users[index].firstName.message}
                </p>
              )}
            </div>
            
            <div style={{ marginBottom: '12px' }}>
              <label style={{ display: 'block', marginBottom: '8px' }}>
                Last Name:
              </label>
              <input
                {...register(`users.${index}.lastName`, {
                  required: 'Last name is required'
                })}
                style={{
                  width: '100%',
                  padding: '8px',
                  border: errors.users?.[index]?.lastName ? '1px solid red' : '1px solid #ccc',

                <p style={{ color: 'red', fontSize: '12px', marginTop: '4px' }}>
                  {errors.users[index].lastName.message}

              )}

                type="button"
                onClick={() => remove(index)}

                  padding: '8px 16px',
                  background: '#ff4d4f',
                  color: 'white',
                  border: 'none',
                  borderRadius: '4px',
                  cursor: 'pointer'
                }}
              >
                Remove User
              </button>
            )}
          </div>
        ))}
        
        <div style={{ display: 'flex', gap: '8px' }}>
          <button
            type="button"
            onClick={() => append({ firstName: '', lastName: '' })}
            style={{
              padding: '10px 20px',
              background: '#52c41a',
              color: 'white',
              border: 'none',
              borderRadius: '4px',
              cursor: 'pointer'
            }}
          >
            Add User
          </button>
          
          <button
            type="submit"
            style={{
              padding: '10px 20px',
              background: '#1890ff',
              color: 'white',
              border: 'none',
              borderRadius: '4px',
              cursor: 'pointer'
            }}
          >
            Submit
          </button>
        </div>
      </form>
    </div>
  );
}
```

```jsx
          />
        </Form.Item>
        
        <Form.Item
          label="Last Name"
          validateStatus={errors.lastName ? 'error' : ''}
          help={errors.lastName?.message}
        >
          <Controller
            name="lastName"
            control={control}
            rules={{
              required: 'Last name is required',
              pattern: {
                value: /^[a-zA-Z\s]+$/,
                message: 'Last name can only contain letters'
              }
            }}
            render={({ field }) => (
              <Input {...field} placeholder="Enter your last name" />
            )}
          />
        </Form.Item>
        
        <Form.Item
          label="Country"
          validateStatus={errors.country ? 'error' : ''}
          help={errors.country?.message}
        >
          <Controller
            name="country"
            control={control}
            rules={{ required: 'Country is required' }}
            render={({ field }) => (
              <Select {...field} placeholder="Select your country">
                <Option value="china">China</Option>
                <Option value="usa">United States</Option>
                <Option value="uk">United Kingdom</Option>
                <Option value="japan">Japan</Option>
                <Option value="other">Other</Option>
              </Select>
            )}
          />
        </Form.Item>
        
        {/* Terms and Conditions */}
        <Form.Item
          validateStatus={errors.agreeToTerms ? 'error' : ''}
          help={errors.agreeToTerms?.message}
        >
          <Controller
            name="agreeToTerms"
            control={control}
            rules={{
              required: 'You must agree to the terms and conditions'
            }}
            render={({ field }) => (
              <Checkbox {...field}>
                I agree to the terms and conditions
              </Checkbox>
            )}
          />
        </Form.Item>
        
        <Form.Item>
          <Button type="primary" htmlType="submit" block size="large">
            Register
          </Button>
        </Form.Item>
      </Form>
    </div>
  );
}

export default UserRegistrationForm;
```
