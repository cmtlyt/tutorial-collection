# React 19 表单特性

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

React 19 引入了许多新的表单特性，包括 Actions、useActionState、useFormStatus 等。这些新特性大大简化了表单处理，提供了更好的用户体验和开发体验。本章节将详细介绍 React 19 的表单新特性。

## 🎯 useActionState

### 基础用法

```jsx
import React, { useActionState } from 'react';

// Action 函数
function formAction(previousState, formData) {
  const username = formData.get('username');
  const email = formData.get('email');
  
  // 模拟验证
  if (!username || username.length < 3) {
    return {
      ...previousState,
      errors: {
        username: 'Username must be at least 3 characters'
      }
    };
  }
  
  if (!email || !email.includes('@')) {
    return {
      ...previousState,
      errors: {
        email: 'Invalid email address'
      }
    };
  }
  
  return {
    ...previousState,
    success: true,
    message: 'Form submitted successfully!',
    data: { username, email }
  };
}

function UseActionStateForm() {
  const [state, formAction, isPending] = useActionState(formAction, {
    success: false,
    message: '',
    errors: {},
    data: null
  });
  
  return (
    <div style={{ padding: '24px', maxWidth: '500px', margin: '0 auto' }}>
      <h2>useActionState</h2>
      
      {state.success && (
        <div style={{
          padding: '12px',
          marginBottom: '16px',
          background: '#f6ffed',
          border: '1px solid #b7eb8f',
          borderRadius: '4px',
          color: '#52c41a'
        }}>
          {state.message}
        </div>
      )}
      
      <form action={formAction}>
        <div style={{ marginBottom: '16px' }}>
          <label style={{ display: 'block', marginBottom: '8px' }}>
            Username:
          </label>
          <input
            type="text"
            name="username"
            defaultValue={state.data?.username}
            style={{
              width: '100%',
              padding: '8px',
              border: state.errors?.username ? '1px solid red' : '1px solid #ccc',
              borderRadius: '4px'
            }}
          />
          {state.errors?.username && (
            <p style={{ color: 'red', fontSize: '12px', marginTop: '4px' }}>
              {state.errors.username}
            </p>
          )}
        </div>
        
        <div style={{ marginBottom: '16px' }}>
          <label style={{ display: 'block', marginBottom: '8px' }}>
            Email:
          </label>
          <input
            type="email"
            name="email"
            defaultValue={state.data?.email}
            style={{
              width: '100%',
              padding: '8px',
              border: state.errors?.email ? '1px solid red' : '1px solid #ccc',
              borderRadius: '4px'
            }}
          />
          {state.errors?.email && (
            <p style={{ color: 'red', fontSize: '12px', marginTop: '4px' }}>
              {state.errors.email}
            </p>
          )}
        </div>
        
        <button
          type="submit"
          disabled={isPending}
          style={{
            padding: '10px 20px',
            background: isPending ? '#ccc' : '#1890ff',
            color: 'white',
            border: 'none',
            borderRadius: '4px',
            cursor: isPending ? 'not-allowed' : 'pointer'
          }}
        >
          {isPending ? 'Submitting...' : 'Submit'}
        </button>
      </form>
    </div>
  );
}

export default UseActionStateForm;
```

### 复杂状态管理

```jsx
import React, { useActionState } from 'react';

// 复杂 Action
function complexFormAction(previousState, formData) {
  const username = formData.get('username');
  const email = formData.get('email');
  const age = formData.get('age');
  const bio = formData.get('bio');
  
  const errors = {};
  
  // 验证
  if (!username || username.length < 3) {
    errors.username = 'Username must be at least 3 characters';
  }
  
  if (!email || !email.includes('@')) {
    errors.email = 'Invalid email address';
  }
  
  if (!age || parseInt(age) < 18) {
    errors.age = 'You must be at least 18 years old';
  }
  
  if (bio && bio.length > 500) {
    errors.bio = 'Bio must not exceed 500 characters';
  }
  
  // 如果有错误，返回错误状态
  if (Object.keys(errors).length > 0) {
    return {
      ...previousState,
      errors,
      success: false
    };
  }
  
  // 模拟 API 调用
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve({
        ...previousState,
        success: true,
        message: 'Form submitted successfully!',
        data: { username, email, age: parseInt(age), bio },
        errors: {}
      });
    }, 1000);
  });
}
```
