## 📚 概念介绍

除了 Material-UI 和 Ant Design，React 生态中还有许多优秀的 UI 组件库。本章节将介绍几个流行的替代方案，包括 Chakra UI、Mantine、Radix UI 等，帮助你根据项目需求选择最合适的组件库。

## 🎯 Chakra UI

### 基础使用

```jsx
import React from 'react';
import {
  ChakraProvider,
  Box,
  Button,
  Input,
  FormControl,
  FormLabel,
  Heading,
  Text,
  VStack,
  HStack,
  Card,
  CardBody,
  Stack,
  Divider,
  Badge,
  Avatar,
  Checkbox,
  Switch,
  Slider,
  useColorMode,
  useColorModeValue,
  IconButton
} from '@chakra-ui/react';
import { MoonIcon, SunIcon } from '@chakra-ui/icons';

function ChakraUIExample() {
  const { colorMode, toggleColorMode } = useColorMode();
  const bg = useColorModeValue('gray.50', 'gray.800');
  const color = useColorModeValue('gray.800', 'white');
  
  const [checked, setChecked] = React.useState(false);
  const [switchValue, setSwitchValue] = React.useState(false);
  const [sliderValue, setSliderValue] = React.useState(50);
  
  return (
    <ChakraProvider>
      <Box bg={bg} color={color} minH="100vh" p={6}>
        <VStack spacing={6} align="stretch">
          {/* Header */}
          <HStack justify="space-between">
            <Heading>Chakra UI Components</Heading>
            <IconButton
              icon={colorMode === 'light' ? <MoonIcon /> : <SunIcon />}
              onClick={toggleColorMode}
              aria-label="Toggle color mode"
            />
          </HStack>
          
          {/* Card */}
          <Card>
            <CardBody>
              <Heading size="md" mb={2}>Welcome to Chakra UI</Heading>
              <Text>
                This is a simple example using Chakra UI components.
              </Text>
            </CardBody>
          </Card>
          
          {/* Form Controls */}
          <Card>
            <CardBody>
              <Heading size="sm" mb={4}>Form Controls</Heading>
              <VStack spacing={4} align="stretch">
                <FormControl>
                  <FormLabel>Username</FormLabel>
                  <Input placeholder="Enter your username" />
                </FormControl>
                
                <HStack>
                  <Checkbox isChecked={checked} onChange={(e) => setChecked(e.target.checked)}>
                    Checkbox
                  </Checkbox>
                  
                  <Switch
                    isChecked={switchValue}
                    onChange={(e) => setSwitchValue(e.target.checked)}
                  />
                </HStack>
                
                <Box>
                  <Text mb={2}>Slider: {sliderValue}</Text>
                  <Slider
                    value={sliderValue}
                    onChange={(value) => setSliderValue(value)}
                  >
                    <Slider.Track>
                      <SliderFilledTrack />
                    </Slider.Track>
                    <SliderThumb />
                  </Slider>
                </Box>
              </VStack>
            </CardBody>
          </Card>
          
          {/* Badges */}
          <Card>
            <CardBody>
              <Heading size="sm" mb={4}>Badges</Heading>
              <HStack>
                <Badge>Default</Badge>
                <Badge colorScheme="green">Success</Badge>
                <Badge colorScheme="red">Error</Badge>
                <Badge colorScheme="blue">Info</Badge>
                <Badge colorScheme="purple">Purple</Badge>
              </HStack>
            </CardBody>
          </Card>
          
          {/* Avatars */}
          <Card>
            <CardBody>
              <Heading size="sm" mb={4}>Avatars</Heading>
              <HStack>
                <Avatar name="Alice" src="https://bit.ly/alice" />
                <Avatar name="Bob" src="https://bit.ly/bob" />
                <Avatar name="Charlie" src="https://bit.ly/charlie" />
                <Avatar name="Dan" />
              </HStack>
            </CardBody>
          </Card>
          
          {/* Buttons */}
          <Card>
            <CardBody>
              <Heading size="sm" mb={4}>Buttons</Heading>
              <HStack>
                <Button colorScheme="blue">Primary</Button>
                <Button colorScheme="green">Success</Button>
                <Button colorScheme="red">Danger</Button>
                <Button variant="outline">Outline</Button>
                <Button variant="ghost">Ghost</Button>
              </HStack>
            </CardBody>
          </Card>
        </VStack>
      </Box>
    </ChakraProvider>
  );
}

export default ChakraUIExample;
```

### 主题定制

```jsx
import React from 'react';
import {
  ChakraProvider,
  extendTheme,
  Box,
  Button,
  Heading,
  Text,
  Card,
  CardBody,
  VStack
} from '@chakra-ui/react';

// 自定义主题
const theme = extendTheme({
  colors: {
    brand: {
      50: '#e6f7ff',
      100: '#bae7ff',
      200: '#91d5ff',
      300: '#69c0ff',
      400: '#40a9ff',
      500: '#1890ff',
      600: '#096dd9',
      700: '#0050b3',
      800: '#003a8c',
      900: '#002766',
    },
  },
  fonts: {
    heading: '"Inter", sans-serif',
    body: '"Inter", sans-serif',
  },
  components: {
    Button: {
      baseStyle: {
        fontWeight: 'bold',
      },
      variants: {
        solid: (props) => ({
          bg: props.colorMode === 'dark' ? 'brand.300' : 'brand.500',
          color: 'white',
          _hover: {
            bg: props.colorMode === 'dark' ? 'brand.400' : 'brand.600',
          },
        }),
      },
    },
  },
});

function ThemedChakraApp() {
  return (
    <ChakraProvider theme={theme}>
      <Box p={6}>
        <VStack spacing={6} align="stretch">
          <Heading>Custom Chakra UI Theme</Heading>
          
          <Card>
            <CardBody>
              <Heading size="md" mb={2}>Themed Card</Heading>
              <Text>
                This card uses a custom theme with brand colors.
              </Text>
            </CardBody>
          </Card>
          
          <Button colorScheme="brand">Brand Button</Button>
        </VStack>
      </Box>
    </ChakraProvider>
  );
}

export default ThemedChakraApp;
```

## 🎯 Mantine

### 基础使用

```jsx
import React from 'react';
import {
  MantineProvider,
  Container,
  Title,
  Text,
  Card,
  Button,
  TextInput,
  PasswordInput,
  Checkbox,
  Switch,
  Slider,
  Badge,
  Avatar,
  Group,
  Stack,
  Paper,
  Divider
} from '@mantine/core';

function MantineExample() {
  const [checked, setChecked] = React.useState(false);
  const [switchValue, setSwitchValue] = React.useState(false);
  const [sliderValue, setSliderValue] = React.useState(50);
  
  return (
    <MantineProvider withGlobalStyles withNormalizeCSS>
      <Container size="md" py="xl">
        <Stack spacing="xl">
          <Title order={1}>Mantine Components</Title>
          
          <Card shadow="sm" padding="lg" radius="md" withBorder>
            <Title order={3}>Welcome to Mantine</Title>
            <Text mt="sm">
              This is a simple example using Mantine components.
            </Text>
          </Card>
          
          <Paper shadow="xs" p="md">
            <Title order={4}>Form Controls</Title>
            <Stack spacing="md" mt="md">
              <TextInput
                label="Username"
                placeholder="Enter your username"
              />
              
              <PasswordInput
                label="Password"
                placeholder="Enter your password"
              />
              
              <Group>
                <Checkbox
                  label="Checkbox"
                  checked={checked}
                  onChange={(event) => setChecked(event.currentTarget.checked)}
                />
                
                <Switch
                  label="Switch"
                  checked={switchValue}
                  onChange={(event) => setSwitchValue(event.currentTarget.checked)}
                />
              </Group>
              
              <div>
                <Text size="sm">Slider: {sliderValue}</Text>
                <Slider
                  value={sliderValue}
                  onChange={setSliderValue}
                  mt="xs"
                />
              </div>
            </Stack>
          </Paper>
          
          <Paper shadow="xs" p="md">
            <Title order={4}>Badges</Title>
            <Group mt="md">
              <Badge color="gray">Default</Badge>
              <Badge color="green">Success</Badge>
              <Badge color="red">Error</Badge>
              <Badge color="blue">Info</Badge>
              <Badge color="purple">Purple</Badge>
            </Group>
          </Paper>
          
          <Paper shadow="xs" p="md">
            <Title order={4}>Avatars</Title>

              <Avatar src="https://i.pravatar.cc/150?img=1" alt="Alice" radius="xl" />
              <Avatar src="https://i.pravatar.cc/150?img=2" alt="Bob" radius="xl" />

              <Button variant="subtle">Subtle</Button>

      </Container>
    </MantineProvider>
  );
}

export default MantineExample;
```

## 🎯 组件库对比

### 功能对比表格

| 特性 | Material-UI | Ant Design | Chakra UI | Mantine | Radix UI |
|------|-------------|------------|-----------|---------|----------|
| 设计风格 | Material Design | 企业级 | 简洁现代 | 简洁现代 | 无样式 |
| 组件数量 | 50+ | 60+ | 40+ | 100+ | 30+ |
| TypeScript | ✅ | ✅ | ✅ | ✅ | ✅ |
| 主题定制 | ✅ | ✅ | ✅ | ✅ | ✅ |
| 响应式 | ✅ | ✅ | ✅ | ✅ | ❌ |
| 无障碍 | ✅ | ✅ | ✅ | ✅ | ✅✅ |
| 包大小 | 较大 | 较大 | 中等 | 中等 | 小 |
| 学习曲线 | 中等 | 中等 | 低 | 低 | 中等 |

## 📝 实际应用示例

### 综合示例：多组件库混用

```jsx
import React, { useState } from 'react';
import { Button, Input, Card, Typography, Space } from 'antd';
import { ChakraProvider, Box, Text, VStack } from '@chakra-ui/react';
import { MantineProvider, Container, Title } from '@mantine/core';

function MixedLibrariesExample() {
  const [activeLibrary, setActiveLibrary] = useState('antd');
  
  const renderContent = () => {
    switch (activeLibrary) {
      case 'antd':
        return (
          <Card>
            <Title level={3}>Ant Design Component</Title>
            <Input placeholder="Ant Design Input" style={{ marginBottom: 16 }} />
            <Button type="primary">Ant Design Button</Button>
          </Card>
        );
      case 'chakra':
        return (
          <ChakraProvider>
            <Box p={4} borderWidth="1px" borderRadius="lg">
              <Title level={3}>Chakra UI Component</Title>
              <Input placeholder="Chakra UI Input" mb={4} />
              <Button colorScheme="blue">Chakra UI Button</Button>
            </Box>
          </ChakraProvider>
        );
      case 'mantine':
        return (
          <MantineProvider>
            <Paper shadow="xs" p="md">
              <Title order={3}>Mantine Component</Title>
              <TextInput placeholder="Mantine Input" mb={16} />
              <Button>Mantine Button</Button>
            </Paper>
          </MantineProvider>
        );
      default:
        return null;
    }
  };
  
  return (
    <div style={{ padding: '24px', maxWidth: '800px', margin: '0 auto' }}>
      <h1>Mixed UI Libraries Example</h1>
      
      <Space style={{ marginBottom: '24px' }}>
        <Button onClick={() => setActiveLibrary('antd')}>Ant Design</Button>
        <Button onClick={() => setActiveLibrary('chakra')}>Chakra UI</Button>
        <Button onClick={() => setActiveLibrary('mantine')}>Mantine</Button>
      </Space>
      
      {renderContent()}
    </div>
  );
}

export default MixedLibrariesExample;
```
