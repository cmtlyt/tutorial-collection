# React 19 样式特性

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

React 19 引入了一些新的样式特性和改进，使得样式管理更加灵活和高效。本章节将详细介绍 React 19 的样式新特性，包括新的样式属性、样式继承等。

## 🎯 样式属性

### 基础样式属性

```jsx
import React from 'react';

function StylePropsExample() {
  return (
    <div style={{ padding: '24px', maxWidth: '600px', margin: '0 auto' }}>
      <h1>React 19 Style Props</h1>
      
      {/* 使用 style 属性 */}
      <div
        style={{
          padding: '16px',
          background: 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)',
          borderRadius: '8px',
          color: 'white',
          marginBottom: '16px'
        }}
      >
        <h2 style={{ margin: '0 0 8px 0' }}>Styled Component</h2>
        <p style={{ margin: 0 }}>
          This component uses inline styles with React 19.
        </p>
      </div>
      
      {/* 动态样式 */}
      <div
        style={{
          padding: '16px',
          border: '2px solid #1890ff',
          borderRadius: '8px',
          background: '#e6f7ff',
          transition: 'all 0.3s'
        }}
      >
        <h3 style={{ margin: '0 0 8px 0', color: '#1890ff' }}>
          Dynamic Styling
        </h3>
        <p style={{ margin: 0, color: '#666' }}>
          Styles can be computed dynamically based on props or state.
        </p>
      </div>
    </div>
  );
}

export default StylePropsExample;
```

```jsx
import React from 'react';
```

## 🎯 样式继承

### 样式继承示例

```jsx
import React from 'react';

function StyleInheritanceExample() {
  const baseStyles = {
    padding: '12px',
    borderRadius: '4px',
    fontSize: '14px',
    fontWeight: '500'
  };
  
  const primaryButtonStyles = {
    ...baseStyles,
    background: '#1890ff',
    color: 'white',
    border: 'none',
    cursor: 'pointer'
  };
  
  const secondaryButtonStyles = {
    ...baseStyles,
    background: '#52c41a',
    color: 'white',
    border: 'none',
    cursor: 'pointer'
  };
  
  const dangerButtonStyles = {
    ...baseStyles,
    background: '#ff4d4f',
    color: 'white',
    border: 'none',
    cursor: 'pointer'
  };
  
  return (
    <div style={{ padding: '24px', maxWidth: '600px', margin: '0 auto' }}>
      <h1>Style Inheritance</h1>
      
      <div style={{ marginBottom: '16px' }}>
        <button style={primaryButtonStyles}>
          Primary Button
        </button>
      </div>
      
      <div style={{ marginBottom: '16px' }}>
        <button style={secondaryButtonStyles}>
          Secondary Button
        </button>
      </div>
      
      <div>
        <button style={dangerButtonStyles}>
          Danger Button
        </button>
      </div>
    </div>
  );
}

export default StyleInheritanceExample;
```

## 🎯 响应式样式

### 媒体查询样式

```jsx
import React, { useState } from 'react';

function ResponsiveStylesExample() {
  const [windowWidth, setWindowWidth] = useState(0);
  
  React.useEffect(() => {
    const handleResize = () => {
      setWindowWidth(window.innerWidth);
    };
    
    handleResize();
    window.addEventListener('resize', handleResize);
    
    return () => window.removeEventListener('resize', handleResize);
  }, []);
  
  const getResponsiveStyles = () => {
    if (windowWidth < 768) {
      return {
        padding: '16px',
        fontSize: '14px'
      };
    } else if (windowWidth < 1024) {
      return {
        padding: '24px',
        fontSize: '16px'
      };
    } else {
      return {
        padding: '32px',
        fontSize: '18px'
      };
    }
  };
  
  return (
    <div style={{ padding: '24px' }}>
      <h1>Responsive Styles</h1>
      <p style={{ marginBottom: '16px', color: '#666' }}>
        Current window width: {windowWidth}px
      </p>
      
      <div
        style={{
          ...getResponsiveStyles(),
          background: 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)',
          color: 'white',
          borderRadius: '8px',
          marginBottom: '16px'
        }}
      >
        <h2 style={{ margin: '0 0 8px 0' }}>
          Responsive Card
        </h2>
        <p style={{ margin: 0 }}>
          This card adjusts its padding and font size based on the window width.
        </p>
      </div>
      
      <div
        style={{
          display: 'grid',
          gridTemplateColumns: windowWidth < 768 ? '1fr' : 'repeat(3, 1fr)',
          gap: '16px'
        }}
      >
        {[1, 2, 3].map((item) => (
          <div
            key={item}
            style={{
              padding: '16px',
              background: 'white',
              border: '1px solid #ddd',
              borderRadius: '8px',
              textAlign: 'center'
            }}
          >
            <h3 style={{ margin: '0 0 8px 0' }}>
              Item {item}
            </h3>
            <p style={{ margin: 0, color: '#666' }}>
              Responsive grid item
            </p>
          </div>
        ))}
      </div>
    </div>
  );
}

export default ResponsiveStylesExample;
```

## 🎯 动画样式

### CSS 动画

```jsx
import React, { useState } from 'react';

function AnimationStylesExample() {
  const [isAnimating, setIsAnimating] = useState(false);
  
  const cardStyles = {
    padding: '24px',
    background: 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)',
    borderRadius: '12px',
    color: 'white',
    cursor: 'pointer',
    transition: 'all 0.3s ease',
    transform: isAnimating ? 'scale(1.05)' : 'scale(1)',
    boxShadow: isAnimating 
      ? '0 8px 16px rgba(0, 0, 0, 0.2)' 
      : '0 4px 8px rgba(0, 0, 0, 0.1)'
  };
  
  return (
    <div style={{ padding: '24px', maxWidth: '600px', margin: '0 auto' }}>
      <h1>Animation Styles</h1>
      
      <div
        style={cardStyles}
        onClick={() => setIsAnimating(!isAnimating)}
      >
        <h2 style={{ margin: '0 0 8px 0' }}>
          Interactive Card
        </h2>
        <p style={{ margin: 0 }}>
          Click me to see the animation!
        </p>
      </div>
      
      <div
        style={{
          marginTop: '24px',
          padding: '16px',
          background: '#f5f5f5',
          borderRadius: '8px',
          animation: 'fadeIn 0.5s ease-in'
        }}
      >
        <h3 style={{ margin: '0 0 8px 0' }}>
          Fade In Animation
        </h3>
        <p style={{ margin: 0, color: '#666' }}>
          This card fades in when the component mounts.
        </p>
      </div>
      
      <style>
        {`
          @keyframes fadeIn {
            from {
              opacity: 0;
              transform: translateY(10px);
            }
            to {
              opacity: 1;
              transform: translateY(0);
            }
          }
        `}
      </style>
    </div>
  );
}

export default AnimationStylesExample;
```

## 📝 实际应用示例

### 综合示例：主题切换器

```jsx
import React, { useState } from 'react';

function ThemeSwitcher() {
  const [darkMode, setDarkMode] = useState(false);
  
  const themes = {
    light: {
      background: '#ffffff',
      text: '#333333',
      cardBackground: '#f5f5f5',
      accent: '#1890ff',
      border: '#d9d9d9'
    },
    dark: {
      background: '#1f1f1f',
      text: '#ffffff',
      cardBackground: '#2f2f2f',
      accent: '#40a9ff',
      border: '#434343'
    }
  };
  
  const currentTheme = darkMode ? themes.dark : themes.light;
  
  const containerStyles = {
    minHeight: '100vh',
    background: currentTheme.background,
    color: currentTheme.text,
    padding: '24px',
    transition: 'all 0.3s ease'
  };
  
  const cardStyles = {
    background: currentTheme.cardBackground,
    border: `1px solid ${currentTheme.border}`,
    borderRadius: '12px',
    padding: '24px',
    marginBottom: '16px',
    transition: 'all 0.3s ease'
  };
  
  const buttonStyles = {
    padding: '12px 24px',
    background: currentTheme.accent,
    color: 'white',
    border: 'none',
    borderRadius: '8px',
    cursor: 'pointer',
    fontSize: '16px',
    fontWeight: '600',
    transition: 'all 0.2s ease'
  };
  
  const inputStyles = {
    width: '100%',
    padding: '12px',
    border: `1px solid ${currentTheme.border}`,
    borderRadius: '8px',
    background: currentTheme.background,
    color: currentTheme.text,
    fontSize: '16px',
    marginBottom: '16px',
    transition: 'all 0.3s ease'
  };
  
  const [formData, setFormData] = useState({
    name: '',
    email: '',
    message: ''
  });
  
  const handleChange = (e) => {
    setFormData({
      ...formData,
      [e.target.name]: e.target.value
    });
  };
  
  const handleSubmit = (e) => {
    e.preventDefault();
    console.log('Form submitted:', formData);
    alert('Form submitted successfully!');
    setFormData({ name: '', email: '', message: '' });
  };
  
  return (
    <div style={containerStyles}>
      <div style={{ maxWidth: '600px', margin: '0 auto' }}>
        {/* Header */}
        <div style={{ marginBottom: '24px', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
          <h1 style={{ margin: 0, fontSize: '32px' }}>
            Theme Switcher
          </h1>
          <button
            onClick={() => setDarkMode(!darkMode)}
            style={buttonStyles}
          >
            {darkMode ? '☀️ Light' : '🌙 Dark'}
          </button>
        </div>
        
        {/* Info Card */}
        <div style={cardStyles}>
          <h2 style={{ margin: '0 0 8px 0' }}>
            Current Theme: {darkMode ? 'Dark' : 'Light'}
          </h2>
          <p style={{ margin: 0, opacity: 0.8 }}>
            Click the button above to toggle between themes.
          </p>
        </div>
        
        {/* Features Card */}
        <div style={cardStyles}>
          <h2 style={{ margin: '0 0 16px 0' }}>
            Features
          </h2>
          <ul style={{ margin: 0, paddingLeft: '20px' }}>
            <li style={{ marginBottom: '8px' }}>
              Smooth theme transitions
            </li>
            <li style={{ marginBottom: '8px' }}>
              Persistent theme preference
            </li>
            <li style={{ marginBottom: '8px' }}>
              Accessible color contrast
            </li>
            <li>
              Responsive design
            </li>
          </ul>
        </div>
        
        {/* Contact Form */}
        <div style={cardStyles}>
          <h2 style={{ margin: '0 0 16px 0' }}>
            Contact Form
          </h2>
          <form onSubmit={handleSubmit}>
            <div>
              <label style={{ display: 'block', marginBottom: '8px', fontWeight: '600' }}>
                Name
              </label>
              <input
                type="text"
                name="name"
                value={formData.name}
                onChange={handleChange}
                required
                style={inputStyles}
                placeholder="Your name"
              />
            </div>
            
            <div>
              <label style={{ display: 'block', marginBottom: '8px', fontWeight: '600' }}>
                Email
              </label>
              <input
                type="email"
                name="email"
                value={formData.email}
                onChange={handleChange}
                required
                style={inputStyles}
                placeholder="your.email@example.com"
              />
            </div>
            
            <div>
              <label style={{ display: 'block', marginBottom: '8px', fontWeight: '600' }}>
                Message
              </label>
              <textarea
                name="message"
                value={formData.message}
                onChange={handleChange}
                required
                rows={4}
                style={{
                  ...inputStyles,
                  resize: 'none'
                }}
                placeholder="Your message"
              />
            </div>
            
            <button
              type="submit"
              style={{
                ...buttonStyles,
                width: '100%'
              }}
            >
              Submit
            </button>
          </form>
        </div>
      </div>
    </div>
  );
}

export default ThemeSwitcher;
```
