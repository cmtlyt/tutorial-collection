# CSS Modules

> **版本**：v1.0  
> **最后更新**：2026-04-21

## 📚 概念介绍

CSS Modules 是一种 CSS 文件的模块化方案，它通过自动生成唯一的类名来避免样式冲突。CSS Modules 让每个组件的样式都是独立的，提供了更好的可维护性和可预测性。

## 🎯 基础用法

### 创建 CSS Module

```css
/* UserCard.module.css */
.card {
  background: white;
  border: 1px solid #ddd;
```

```jsx
import React from 'react';
import styles from './UserCard.module.css';

function UserCard({ name, email, bio, avatar }) {
  return (
    <div className={styles.card}>
      <div className={styles.header}>
        <div className={styles.avatar}>
          {avatar || name.charAt(0)}
        </div>
        <div className={styles.userInfo}>
          <h3 className={styles.name}>{name}</h3>
          <p className={styles.email}>{email}</p>
        </div>
      </div>
      <p className={styles.bio}>{bio}</p>
      <button className={styles.button}>View Profile</button>
    </div>
  );
}

function App() {
  return (
    <div style={{ padding: '24px', maxWidth: '600px', margin: '0 auto' }}>
      <h1>CSS Modules Example</h1>
      <UserCard
        name="Alice Johnson"
        email="alice@example.com"
        bio="Software engineer passionate about building great user experiences."
        avatar="A"
      />
      <UserCard
        name="Bob Smith"
        email="bob@example.com"
        bio="Full-stack developer with expertise in Node.js and React."
        avatar="B"
      />
    </div>
  );
}

export default App;
```

```jsx
import React from 'react';
import styles from './Button.module.css';

function Button({ variant = 'primary', size = 'medium', disabled = false, children, onClick }) {
  const classes = [
    styles.button,
    styles[variant],
    styles[size],
    disabled ? styles.disabled : ''
  ].filter(Boolean).join(' ');
  
  return (
    <button
      className={classes}
      disabled={disabled}
      onClick={onClick}
    >
      {children}
    </button>
  );
}

function ButtonExample() {
  return (
    <div style={{ padding: '24px' }}>
      <h1>Dynamic CSS Modules</h1>
      
      <div style={{ marginBottom: '16px' }}>
        <h3>Primary Buttons</h3>
        <Button variant="primary" size="small">Small Primary</Button>
        <Button variant="primary">Medium Primary</Button>
        <Button variant="primary" size="large">Large Primary</Button>
        <Button variant="primary" disabled>Disabled</Button>
      </div>
      
      <div style={{ marginBottom: '16px' }}>
        <h3>Secondary Buttons</h3>
        <Button variant="secondary" size="small">Small Secondary</Button>
        <Button variant="secondary">Medium Secondary</Button>
        <Button variant="secondary" size="large">Large Secondary</Button>
      </div>
      
      <div>
        <h3>Danger Buttons</h3>
        <Button variant="danger" size="small">Small Danger</Button>
        <Button variant="danger">Medium Danger</Button>
        <Button variant="danger" size="large">Large Danger</Button>
      </div>
    </div>
  );
}

export default ButtonExample;
```

```jsx
import React from 'react';
import cardStyles from './Card.module.css';
import badgeStyles from './Badge.module.css';

function TaskCard({ title, status, description }) {
  const statusClass = {
    completed: badgeStyles.success,
    inProgress: badgeStyles.warning,
    failed: badgeStyles.error
  }[status] || '';
  
  return (
    <div className={cardStyles.card}>
      <h3 className={cardStyles.title}>{title}</h3>
      <p className={cardStyles.content}>{description}</p>
      <span className={`${badgeStyles.badge} ${statusClass}`}>
        {status}
      </span>
    </div>
  );
}

function TaskList() {
  const tasks = [
    {
      id: 1,
      title: 'Complete React Tutorial',
      status: 'completed',
      description: 'Finish all chapters of the React tutorial'
    },
    {
      id: 2,
      title: 'Build Portfolio',
      status: 'inProgress',
      description: 'Create a personal portfolio website'
    },
    {
      id: 3,
      title: 'Deploy Application',
      status: 'failed',
      description: 'Deploy the application to production'
    }
  ];
  
  return (
    <div style={{ padding: '24px', maxWidth: '600px', margin: '0 auto' }}>
      <h1>Task List</h1>
      {tasks.map(task => (
        <TaskCard key={task.id} {...task} />
      ))}
    </div>
  );
}

export default TaskList;
```
