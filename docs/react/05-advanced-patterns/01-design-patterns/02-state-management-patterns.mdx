## 📚 概念介绍

状态管理是 React 应用的核心，选择合适的状态管理策略对应用的可维护性和性能至关重要。本节将介绍状态提升、控制反转、观察者模式、单向数据流等经典状态管理模式，以及 React 19 中改进的 Context API、Actions 系统等新特性。

## 🎯 状态提升

### 基本概念

当多个组件需要共享状态时，将状态提升到它们最近的共同父组件中。

```js
import { useState } from 'react'

// 子组件 1: 温度输入（摄氏度）
function CelsiusInput({ celsius, onCelsiusChange }) {
  return (
    <div>
      <label>摄氏度: </label>
      <input
        type="number"
        value={celsius}
        onChange={(e) => onCelsiusChange(parseFloat(e.target.value) || 0)}
      />
    </div>
  )
}

// 子组件 2: 温度输入（华氏度）
function FahrenheitInput({ fahrenheit, onFahrenheitChange }) {
  return (
    <div>
      <label>华氏度: </label>
      <input
        type="number"
        value={fahrenheit}
        onChange={(e) => onFahrenheitChange(parseFloat(e.target.value) || 0)}
      />
    </div>
  )
}

// 温度显示组件
function TemperatureDisplay({ celsius }) {
  const boiling = celsius >= 100
  return (
    <div>
      <p>水会{boiling ? '沸腾' : '不会沸腾'}</p>
    </div>
  )
}

// 父组件: 状态提升
function TemperatureCalculator() {
  const [celsius, setCelsius] = useState(0)

  const handleCelsiusChange = (value) => {
    setCelsius(value)
  }

  const handleFahrenheitChange = (value) => {
    setCelsius((value - 32) * 5 / 9)
  }

  const fahrenheit = celsius * 9 / 5 + 32

  return (
    <div>
      <h1>温度转换器</h1>
      <CelsiusInput celsius={celsius} onCelsiusChange={handleCelsiusChange} />
      <FahrenheitInput 
        fahrenheit={fahrenheit} 
        onFahrenheitChange={handleFahrenheitChange} 
      />
      <TemperatureDisplay celsius={celsius} />
    </div>
  )
}

// 使用示例
function App() {
  return <TemperatureCalculator />
}

export default App
```

### React 19 优化版

使用 React 19 的 Actions 优化状态提升：

```js
import { useState, useActionState } from 'react'

// 温度转换 Action
function updateTemperature(prevState, formData) {
  const unit = formData.get('unit')
  const value = parseFloat(formData.get('value')) || 0

  if (unit === 'celsius') {
    return { celsius: value, fahrenheit: value * 9 / 5 + 32 }
  } else {
    return { celsius: (value - 32) * 5 / 9, fahrenheit: value }
  }
}

// 优化后的温度转换器
function TemperatureCalculator() {
  const [state, formAction] = useActionState(updateTemperature, {
    celsius: 0,
    fahrenheit: 32
  })

  return (
    <div>
      <h1>温度转换器</h1>
      <form action={formAction}>
        <div>
          <label>摄氏度: </label>
          <input 
            type="number" 
            name="value" 
            defaultValue={state.celsius}
          />
          <input type="hidden" name="unit" value="celsius" />
          <button type="submit">更新</button>
        </div>
        <div>
          <label>华氏度: </label>
          <span>{state.fahrenheit.toFixed(2)}</span>
        </div>
      </form>
      <p>水会{state.celsius >= 100 ? '沸腾' : '不会沸腾'}</p>
```

## 🎯 控制反转

### 基本概念

通过将控制权交给调用者，使组件更加灵活和可复用。

```js
import { useState } from 'react'

// 受控组件: 父组件完全控制状态
function ControlledInput({ value, onChange }) {
  return (
    <input
      type="text"
      value={value}
      onChange={(e) => onChange(e.target.value)}
      placeholder="受控输入"
    />
  )
}

// 非受控组件: 组件自己管理状态
function UncontrolledInput({ defaultValue, onSubmit }) {
  const [value, setValue] = useState(defaultValue || '')

  const handleSubmit = () => {
    onSubmit(value)
    setValue('')
  }

  return (
    <div>
      <input
        type="text"
        value={value}
        onChange={(e) => setValue(e.target.value)}
        placeholder="非受控输入"
      />
      <button onClick={handleSubmit}>提交</button>
    </div>
  )
}

// 混合组件: 支持受控和非受控模式
function FlexibleInput({ value, defaultValue, onChange, onSubmit }) {
  const isControlled = value !== undefined
  const [internalValue, setInternalValue] = useState(defaultValue || '')

  const currentValue = isControlled ? value : internalValue

  const handleChange = (newValue) => {
    if (isControlled) {
      onChange?.(newValue)
    } else {
      setInternalValue(newValue)
    }
  }

  const handleSubmit = () => {
    onSubmit?.(currentValue)
    if (!isControlled) {
      setInternalValue('')
    }
  }

  return (
    <div>
      <input
        type="text"
        value={currentValue}
        onChange={(e) => handleChange(e.target.value)}
        placeholder="灵活输入"
      />
      <button onClick={handleSubmit}>提交</button>
    </div>
  )
}

// 使用示例
function App() {
  const [controlledValue, setControlledValue] = useState('')

  const handleSubmit = (value) => {
    console.log('提交的值:', value)
  }

  return (
    <div>
      <h2>受控组件</h2>
      <ControlledInput 
        value={controlledValue} 
        onChange={setControlledValue} 
      />
      <p>值: {controlledValue}</p>

      <h2>非受控组件</h2>
      <UncontrolledInput onSubmit={handleSubmit} />

      <h2>灵活组件（受控）</h2>
      <FlexibleInput 
        value={controlledValue} 
        onChange={setControlledValue}
        onSubmit={handleSubmit}
      />

      <h2>灵活组件（非受控）</h2>
      <FlexibleInput 
        defaultValue="默认值"
        onSubmit={handleSubmit}
      />
    </div>
  )
}

export default App
```

## 🎯 观察者模式

### 基本概念

观察者模式允许对象订阅并接收其他对象状态变化的通知。

```js
import { useState, useEffect, createContext, useContext } from 'react'

// 简单的观察者实现
class Observable {
  constructor() {
    this.observers = []
  }

  subscribe(observer) {
    this.observers.push(observer)
    return () => {
      this.observers = this.observers.filter(obs => obs !== observer)
    }
  }

  notify(data) {
    this.observers.forEach(observer => observer(data))
  }
}

// 创建全局可观察对象
const userObservable = new Observable()

// 自定义 Hook: 订阅用户状态
function useUserSubscription() {
  const [user, setUser] = useState(null)

  useEffect(() => {
    const unsubscribe = userObservable.subscribe(setUser)
    return unsubscribe
  }, [])

  return user
}

// 用户管理器
function UserManager(observable) {
  const [currentUser, setCurrentUser] = useState(null)

  const login = (userData) => {
    setCurrentUser(userData)
    observable.notify(userData)
  }

  const logout = () => {
    setCurrentUser(null)
    observable.notify(null)
  }

  return { currentUser, login, logout }
}

// 使用全局用户管理器
const userManager = UserManager(userObservable)

// 订阅用户状态的组件
function UserProfile() {
  const user = useUserSubscription()

  if (!user) {
    return <p>未登录</p>
  }

  return (
    <div>
      <h2>用户信息</h2>
      <p>姓名: {user.name}</p>
      <p>邮箱: {user.email}</p>
    </div>
  )
}

// 另一个订阅用户状态的组件
function UserGreeting() {
  const user = useUserSubscription()

  if (!user) {
    return <p>欢迎访客</p>
  }

  return <p>欢迎, {user.name}!</p>
}

// 控制组件
function UserControls() {
  const handleLogin = () => {
    userManager.login({
      name: '张三',
      email: 'zhangsan@example.com'
    })
  }

  const handleLogout = () => {
    userManager.logout()
  }

  return (
    <div>
      <button onClick={handleLogin}>登录</button>
      <button onClick={handleLogout}>退出</button>
    </div>
  )
}

// 使用示例
function App() {
  return (
    <div>
      <h1>观察者模式示例</h1>
      <UserGreeting />
      <UserProfile />
      <UserControls />
    </div>
  )
}

export default App
```

### React 19 Context + useEffectEvent 优化

```js
import { createContext, useContext, useState, useEffect, useEffectEvent } from 'react'

// 创建 Context
const UserContext = createContext()

// User Provider
function UserProvider({ children }) {
  const [user, setUser] = useState(null)
  const [listeners, setListeners] = useState([])

  const subscribe = useEffectEvent((listener) => {
    setListeners(prev => [...prev, listener])
    return () => {
      setListeners(prev => prev.filter(l => l !== listener))
    }
  })

  const notify = useEffectEvent((userData) => {
    listeners.forEach(listener => listener(userData))
  })

  const login = (userData) => {
    setUser(userData)
    notify(userData)
  }

  const logout = () => {
    setUser(null)
    notify(null)
  }

  return (
    <UserContext value={{ user, login, logout, subscribe }}>
      {children}
    </UserContext>
  )
}

// 使用 Context
function useUser() {
  const context = useContext(UserContext)
  if (!context) {
    throw new Error('useUser must be used within UserProvider')
  }
  return context
}

// 订阅用户状态的组件
function UserProfile() {
  const { user, subscribe } = useUser()
  const [localUser, setLocalUser] = useState(user)

  useEffect(() => {
    const unsubscribe = subscribe((userData) => {
      setLocalUser(userData)
    })
    return unsubscribe
  }, [subscribe])

  if (!localUser) {
    return <p>未登录</p>
  }

  return (
    <div>
      <h2>用户信息</h2>
      <p>姓名: {localUser.name}</p>
      <p>邮箱: {localUser.email}</p>
    </div>
  )
}

// 使用示例
function App() {
  return (
    <UserProvider>
      <div>
        <h1>React 19 观察者模式</h1>
        <UserProfile />
        <UserControls />
      </div>
    </UserProvider>

}

  const handleLogout = () => {
    logout()
  }

  return (
    <div>
      <button onClick={handleLogin}>登录</button>
      <button onClick={handleLogout}>退出</button>
    </div>
  )
}
```

## 🎯 单向数据流

### 基本概念

数据只能从父组件流向子组件，通过回调函数实现子组件向父组件通信。

```js
import { useState } from 'react'

// 子组件
function TaskItem({ task, onToggle, onDelete }) {
  return (
    <div style={{ 
      padding: 8, 
      margin: 4, 
      backgroundColor: task.completed ? '#e0e0e0' : 'white',
      borderRadius: 4
    }}>
      <input
        type="checkbox"
        checked={task.completed}
        onChange={() => onToggle(task.id)}
      />
      <span style={{ 
        marginLeft: 8,
        textDecoration: task.completed ? 'line-through' : 'none'
      }}>
        {task.text}
      </span>
      <button 
        onClick={() => onDelete(task.id)}
        style={{ marginLeft: 16 }}
      >
        删除
      </button>
    </div>
  )
}

// 任务列表组件
function TaskList({ tasks, onToggle, onDelete }) {
  return (
    <div>
      {tasks.map(task => (
        <TaskItem
          key={task.id}
          task={task}
          onToggle={onToggle}
          onDelete={onDelete}
        />
      ))}
    </div>
  )
}

// 添加任务组件
function AddTask({ onAdd }) {
  const [text, setText] = useState('')

  const handleSubmit = (e) => {
    e.preventDefault()
    if (text.trim()) {
      onAdd(text)
      setText('')
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="text"
        value={text}
        onChange={(e) => setText(e.target.value)}
        placeholder="添加新任务"
      />
      <button type="submit">添加</button>
    </form>
  )
}

// 父组件
function TaskApp() {
  const [tasks, setTasks] = useState([
    { id: 1, text: '学习 React 19', completed: false },
    { id: 2, text: '编写文档', completed: true }
  ])

  const handleAddTask = (text) => {
    setTasks([
      ...tasks,
      { id: Date.now(), text, completed: false }
    ])
  }

  const handleToggleTask = (id) => {
    setTasks(tasks.map(task =>
      task.id === id ? { ...task, completed: !task.completed } : task
    ))
  }

  const handleDeleteTask = (id) => {
    setTasks(tasks.filter(task => task.id !== id))
  }

  return (
    <div style={{ maxWidth: 600, margin: '0 auto', padding: 20 }}>
      <h1>任务管理</h1>
      <AddTask onAdd={handleAddTask} />
      <TaskList 
        tasks={tasks}
        onToggle={handleToggleTask}
        onDelete={handleDeleteTask}
      />
    </div>
  )
}

// 使用示例
function App() {
  return <TaskApp />
}

export default App
```

## 🎯 React 19 Context 模式

### 基本概念

React 19 简化了 Context 的使用，可以直接使用 Context 作为 provider。

```js
import { createContext, useContext, useState } from 'react'

// 创建 Context
const ThemeContext = createContext()
const UserContext = createContext()

// Theme Provider (React 19: 直接使用 Context 作为 provider)
function ThemeProvider({ children, defaultTheme = 'light' }) {
  const [theme, setTheme] = useState(defaultTheme)

  return (
    <ThemeContext value={{ theme, setTheme }}>
      {children}
    </ThemeContext>
  )
}

// User Provider
function UserProvider({ children }) {
  const [user, setUser] = useState(null)

  const login = (userData) => setUser(userData)
  const logout = () => setUser(null)

  return (
    <UserContext value={{ user, login, logout }}>
      {children}
    </UserContext>
  )
}

// 使用 Context 的 Hooks
function useTheme() {
  const context = useContext(ThemeContext)
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider')
  }
  return context
}

function useUser() {
  const context = useContext(UserContext)
  if (!context) {
    throw new Error('useUser must be used within UserProvider')
  }
  return context
}

// 使用 Context 的组件
function ThemedComponent() {
  const { theme } = useTheme()
  const { user } = useUser()
```

### Context 性能优化

```js
import { createContext, useContext, useState, useMemo, memo } from 'react'

// 创建 Context
const DataContext = createContext()

// 数据 Provider
function DataProvider({ children }) {
  const [items, setItems] = useState([
    { id: 1, name: '项目 1', value: 100 },
    { id: 2, name: '项目 2', value: 200 }
  ])

  // 优化: 使用 useMemo 缓存计算结果
  const totalValue = useMemo(() => {
    return items.reduce((sum, item) => sum + item.value, 0)
  }, [items])

  // 优化: 使用 useMemo 缓存 Context 值
  const contextValue = useMemo(() => ({
    items,
    totalValue,
    addItem: (name, value) => {
      setItems(prev => [...prev, { id: Date.now(), name, value }])
    },
    removeItem: (id) => {
      setItems(prev => prev.filter(item => item.id !== id))
    }
  }), [items, totalValue])

  return (
    <DataContext value={contextValue}>
      {children}
    </DataContext>
  )
}

// 使用 Context
function useData() {
  const context = useContext(DataContext)
  if (!context) {
    throw new Error('useData must be used within DataProvider')
  }
  return context
}

// 优化: 使用 memo 避免不必要的重渲染
const ItemList = memo(function ItemList() {
  const { items, removeItem } = useData()

  return (
    <ul>
      {items.map(item => (
        <li key={item.id}>
          {item.name}: {item.value}
          <button onClick={() => removeItem(item.id)}>删除</button>
        </li>
      ))}
    </ul>
  )
})

const TotalValue = memo(function TotalValue() {
  const { totalValue } = useData()

  return <p>总价值: {totalValue}</p>
})

// 添加项目组件
function AddItem() {
  const { addItem } = useData()
  const [name, setName] = useState('')
  const [value, setValue] = useState('')

  const handleSubmit = (e) => {
    e.preventDefault()
    if (name && value) {
      addItem(name, parseInt(value))
      setName('')
      setValue('')
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="text"
        value={name}
        onChange={(e) => setName(e.target.value)}
        placeholder="名称"
      />
      <input
        type="number"
        value={value}
        onChange={(e) => setValue(e.target.value)}
        placeholder="价值"
      />
      <button type="submit">添加</button>
    </form>
  )
}

// 使用示例
function App() {
  return (
    <DataProvider>
      <div style={{ padding: 20 }}>
        <h1>Context 性能优化</h1>
        <AddItem />
        <ItemList />
        <TotalValue />
      </div>
    </DataProvider>
  )
}

export default App
```

### 综合示例：购物车系统

```js
import { createContext, useContext, useState, useMemo, memo, useOptimistic, useActionState } from 'react'

// 购物车 Context
const CartContext = createContext()

// 购物车 Provider
function CartProvider({ children }) {
  const [items, setItems] = useState([])
  const [discount, setDiscount] = useState(0)

  // 计算总价
  const total = useMemo(() => {
    const subtotal = items.reduce((sum, item) => sum + item.price * item.quantity, 0)
    return subtotal * (1 - discount / 100)
  }, [items, discount])

  // 添加商品
  const addItem = (product) => {
    setItems(prev => {
      const existing = prev.find(item => item.id === product.id)
      if (existing) {
        return prev.map(item =>
          item.id === product.id
            ? { ...item, quantity: item.quantity + 1 }
            : item
        )
      }
      return [...prev, { ...product, quantity: 1 }]
    })
  }

  // 移除商品
  const removeItem = (id) => {
    setItems(prev => prev.filter(item => item.id !== id))
  }

  // 更新数量
  const updateQuantity = (id, quantity) => {
    if (quantity <= 0) {
      removeItem(id)
    } else {
      setItems(prev =>
        prev.map(item =>
          item.id === id ? { ...item, quantity } : item
        )
      )
    }
  }

  // 应用折扣
  const applyDiscount = (code) => {
    const discounts = {
      'SAVE10': 10,
      'SAVE20': 20,
      'VIP30': 30
    }
    setDiscount(discounts[code] || 0)
  }

  const contextValue = useMemo(() => ({
    items,
    total,
    discount,
    addItem,
    removeItem,
    updateQuantity,
    applyDiscount
  }), [items, total, discount])

  return (
    <CartContext value={contextValue}>
      {children}
    </CartContext>
  )
}

// 使用购物车
function useCart() {
  const context = useContext(CartContext)

    throw new Error('useCart must be used within CartProvider')
  }
  return context

        {products.map(product => (
          <div 
            key={product.id} 
            style={{ 
              border: '1px solid #ccc', 
              borderRadius: 8, 
              padding: 16 
            }}
          >
            <h3>{product.name}</h3>
            <p>价格: ¥{product.price}</p>
            <button onClick={() => addItem(product)}>
              加入购物车
            </button>
          </div>
        ))}
      </div>
    </div>
  )
})

// 购物车项目
const CartItem = memo(function CartItem({ item }) {
  const { updateQuantity, removeItem } = useCart()

  return (
    <div 
      style={{ 
        display: 'flex', 
        justifyContent: 'space-between', 
        alignItems: 'center',

        borderBottom: '1px solid #eee'
      }}
    >
      <div>
        <strong>{item.name}</strong>
        <p>¥{item.price} x {item.quantity}</p>
      </div>
      <div>
        <button onClick={() => updateQuantity(item.id, item.quantity - 1)}>
          -
        </button>
        <span style={{ margin: '0 8px' }}>{item.quantity}</span>
        <button onClick={() => updateQuantity(item.id, item.quantity + 1)}>
          +
        </button>
        <button 
          onClick={() => removeItem(item.id)}
          style={{ marginLeft: 16, color: 'red' }}
        >
          删除
        </button>
      </div>
    </div>
  )
})

// 购物车组件
const ShoppingCart = memo(function ShoppingCart() {
  const { items, total, discount, applyDiscount } = useCart()

  async function checkoutAction(prevState, formData) {
    const discountCode = formData.get('discountCode')
    applyDiscount(discountCode)
    return { success: true }
  }

  const [checkoutState, checkoutAction, isPending] = useActionState(
    checkoutAction,
    null
  )

  if (items.length === 0) {
    return <p>购物车为空</p>
  }

  return (
    <div>
      <h2>购物车 ({items.length} 件商品)</h2>
      <div style={{ marginBottom: 16 }}>
        {items.map(item => (
          <CartItem key={item.id} item={item} />
        ))}
      </div>
      <form action={checkoutAction}>
        <input
          type="text"
          name="discountCode"
          placeholder="折扣代码 (SAVE10, SAVE20, VIP30)"
        />
        <button type="submit" disabled={isPending}>
          {isPending ? '应用中...' : '应用折扣'}
        </button>
      </form>
      {discount > 0 && (
        <p style={{ color: 'green' }}>
          已应用 {discount}% 折扣
        </p>
      )}
      <h3>总计: ¥{total.toFixed(2)}</h3>
    </div>
  )
})
```
