## 🎯 单元测试

### Jest 基础

Jest 是一个流行的 JavaScript 测试框架，提供了断言、mock、快照等功能。

```js
// math.test.js
describe('数学工具函数', () => {
  // 加法测试
  test('应该正确计算两个数的和', () => {
    expect(add(1, 2)).toBe(3)
    expect(add(-1, 1)).toBe(0)
    expect(add(0, 0)).toBe(0)
  })

  // 减法测试
  test('应该正确计算两个数的差', () => {
    expect(subtract(5, 3)).toBe(2)
    expect(subtract(3, 5)).toBe(-2)
    expect(subtract(0, 0)).toBe(0)
  })

  // 乘法测试
  test('应该正确计算两个数的积', () => {
    expect(multiply(2, 3)).toBe(6)
    expect(multiply(-2, 3)).toBe(-6)
    expect(multiply(0, 5)).toBe(0)
  })

  // 除法测试
  test('应该正确计算两个数的商', () => {
    expect(divide(6, 2)).toBe(3)
    expect(divide(5, 2)).toBe(2.5)
  })

  test('除数不能为零', () => {
    expect(() => divide(5, 0)).toThrow('除数不能为零')
  })
})

// math.js
export function add(a, b) {
  return a + b
}

export function subtract(a, b) {
  return a - b
}

export function multiply(a, b) {
  return a * b
}

export function divide(a, b) {
  if (b === 0) {
    throw new Error('除数不能为零')
  }
  return a / b
}
```

### React Testing Library

React Testing Library 提供了测试 React 组件的工具，强调测试用户行为而非实现细节。

```js
import { render, screen, fireEvent, waitFor } from '@testing-library/react'
import '@testing-library/jest-dom'
import Counter from './Counter'

describe('Counter 组件', () => {
  test('应该显示初始计数 0', () => {
    render(<Counter />)
    expect(screen.getByText('0')).toBeInTheDocument()
  })

  test('应该能够增加计数', () => {
    render(<Counter />)
    
    const incrementButton = screen.getByText('+')
    fireEvent.click(incrementButton)
    
    expect(screen.getByText('1')).toBeInTheDocument()
  })

  test('应该能够减少计数', () => {
    render(<Counter />)
    
    const decrementButton = screen.getByText('-')
    fireEvent.click(decrementButton)
    
    expect(screen.getByText('-1')).toBeInTheDocument()
  })

  test('应该能够重置计数', () => {
    render(<Counter />)
    
    const incrementButton = screen.getByText('+')
    const resetButton = screen.getByText('重置')
    
    fireEvent.click(incrementButton)
    fireEvent.click(incrementButton)
    fireEvent.click(resetButton)
    
    expect(screen.getByText('0')).toBeInTheDocument()
  })

  test('应该接受自定义初始值', () => {
    render(<Counter initialValue={10} />)
    expect(screen.getByText('10')).toBeInTheDocument()
  })
})

// Counter.jsx
import { useState } from 'react'

export default function Counter({ initialValue = 0 }) {
  const [count, setCount] = useState(initialValue)

  return (
    <div>
      <button onClick={() => setCount(count - 1)}>-</button>
      <span>{count}</span>
      <button onClick={() => setCount(count + 1)}>+</button>
      <button onClick={() => setCount(initialValue)}>重置</button>
    </div>
  )
}
```

### 测试自定义 Hooks

```js
import { renderHook, act } from '@testing-library/react'
import { useCounter } from './useCounter'

describe('useCounter Hook', () => {
  test('应该返回初始计数', () => {
    const { result } = renderHook(() => useCounter())
    expect(result.current.count).toBe(0)
  })

  test('应该接受自定义初始值', () => {
    const { result } = renderHook(() => useCounter(10))
    expect(result.current.count).toBe(10)
  })

  test('应该能够增加计数', () => {
    const { result } = renderHook(() => useCounter())
    
    act(() => {
      result.current.increment()
    })
    
    expect(result.current.count).toBe(1)
  })

  test('应该能够减少计数', () => {
    const { result } = renderHook(() => useCounter(10))
    
    act(() => {
      result.current.decrement()
    })
    
    expect(result.current.count).toBe(9)
  })

  test('应该能够重置计数', () => {
    const { result } = renderHook(() => useCounter(10))
    
    act(() => {
      result.current.increment()
      result.current.reset()
    })
    
    expect(result.current.count).toBe(10)
  })
})

// useCounter.js
import { useState } from 'react'

export function useCounter(initialValue = 0) {
  const [count, setCount] = useState(initialValue)

  const increment = () => setCount(count + 1)
  const decrement = () => setCount(count - 1)
  const reset = () => setCount(initialValue)

  return { count, increment, decrement, reset }
}
```

```js
import { render, screen, fireEvent, waitFor } from '@testing-library/react'
import '@testing-library/jest-dom'
import { useActionState } from 'react'

describe('React 19 Actions 测试', () => {
  test('应该处理表单提交', async () => {
    async function submitAction(prevState, formData) {
      const name = formData.get('name')
      const email = formData.get('email')
      
      // 模拟 API 调用
      await new Promise(resolve => setTimeout(resolve, 100))
      
      return { success: true, name, email }
    }

    function TestForm() {
      const [state, formAction, isPending] = useActionState(submitAction, null)

      return (
        <form action={formAction}>
          <input name="name" defaultValue="张三" />
          <input name="email" defaultValue="zhangsan@example.com" />
          <button type="submit" disabled={isPending}>
            {isPending ? '提交中...' : '提交'}
          </button>
          {state?.success && (
            <div>提交成功: {state.name} ({state.email})</div>
          )}
        </form>
      )
    }

    render(<TestForm />)
    
    const submitButton = screen.getByRole('button')
    fireEvent.click(submitButton)

    await waitFor(() => {
      expect(screen.getByText('提交成功: 张三 (zhangsan@example.com)')).toBeInTheDocument()
    })
  })

  test('应该处理错误情况', async () => {
    async function submitAction(prevState, formData) {
      await new Promise(resolve => setTimeout(resolve, 100))
      return { success: false, error: '提交失败' }
    }

    function TestForm() {
      const [state, formAction, isPending] = useActionState(submitAction, null)

      return (
        <form action={formAction}>
          <button type="submit" disabled={isPending}>
            {isPending ? '提交中...' : '提交'}
          </button>
          {state?.error && (
            <div style={{ color: 'red' }}>{state.error}</div>
          )}
        </form>
      )
    }

    render(<TestForm />)
    
    const submitButton = screen.getByRole('button')
    fireEvent.click(submitButton)

    await waitFor(() => {
      expect(screen.getByText('提交失败')).toBeInTheDocument()
    })
  })
})
```

### 测试 useEffectEvent

```js
import { render, screen } from '@testing-library/react'
import '@testing-library/jest-dom'
import { useEffectEvent, useEffect, useState } from 'react'

describe('useEffectEvent 测试', () => {
  test('应该避免因事件处理函数变化导致 Effect 重新运行', () => {
    const effectCallback = jest.fn()
    const eventCallback = jest.fn()

    function TestComponent({ value }) {
      const handleEvent = useEffectEvent(() => {
        eventCallback(value)
      })

      useEffect(() => {
        effectCallback()
        handleEvent()
      }, [])

      return <div>Value: {value}</div>
    }

    const { rerender } = render(<TestComponent value={1} />)
    
    expect(effectCallback).toHaveBeenCalledTimes(1)
    expect(eventCallback).toHaveBeenCalledTimes(1)

    rerender(<TestComponent value={2} />)
    
    // Effect 不应该重新运行
    expect(effectCallback).toHaveBeenCalledTimes(1)
    // 事件处理函数应该被调用并获取最新值
    expect(eventCallback).toHaveBeenCalledTimes(2)
    expect(eventCallback).toHaveBeenLastCalledWith(2)
  })
})
```

```js
import { useOptimistic, useActionState } from 'react'

describe('useOptimistic 测试', () => {
  test('应该立即显示乐观状态', async () => {
    async function addTodo(prevState, formData) {
      const text = formData.get('text')
      await new Promise(resolve => setTimeout(resolve, 100))
      return [...prevState, { id: Date.now(), text, completed: false }]

    function TodoApp() {

      const [state, formAction, isPending] = useActionState(addTodo, todos)

      const handleSubmit = (e) => {
        e.preventDefault()
        const text = e.target.elements.text.value
        addOptimisticTodo({ id: Date.now(), text, completed: false })
        formAction(new FormData(e.target))
      }

      return (
        <div>
          <form onSubmit={handleSubmit}>
            <input name="text" defaultValue="新任务" />
            <button type="submit" disabled={isPending}>
              {isPending ? '添加中...' : '添加'}
            </button>
          </form>
          <ul>
            {optimisticTodos.map(todo => (
              <li key={todo.id}>{todo.text}</li>
            ))}
          </ul>
        </div>
      )
    }

    render(<TodoApp />)
    
    const submitButton = screen.getByRole('button')
    fireEvent.click(submitButton)

    // 乐观状态应该立即显示
    expect(screen.getByText('新任务')).toBeInTheDocument()
```

## 🎯 集成测试

### 测试组件交互

```js
import { render, screen, fireEvent, waitFor } from '@testing-library/react'
import '@testing-library/jest-dom'
import TodoApp from './TodoApp'

describe('TodoApp 集成测试', () => {
  test('应该能够添加、完成和删除任务', () => {
    render(<TodoApp />)

    // 添加任务
    const input = screen.getByPlaceholderText('添加新任务')
    const addButton = screen.getByText('添加')

    fireEvent.change(input, { target: { value: '学习 React 19' } })
    fireEvent.click(addButton)

    expect(screen.getByText('学习 React 19')).toBeInTheDocument()

    // 完成任务
    const checkbox = screen.getByRole('checkbox')
    fireEvent.click(checkbox)

    expect(screen.getByText('学习 React 19')).toHaveStyle({
      textDecoration: 'line-through'
    })

    // 删除任务
    const deleteButton = screen.getByText('删除')
    fireEvent.click(deleteButton)

    expect(screen.queryByText('学习 React 19')).not.toBeInTheDocument()
  })

  test('应该能够过滤任务', () => {
    render(<TodoApp />)

    // 添加多个任务
    const input = screen.getByPlaceholderText('添加新任务')
    const addButton = screen.getByText('添加')

    fireEvent.change(input, { target: { value: '任务 1' } })
    fireEvent.click(addButton)

    fireEvent.change(input, { target: { value: '任务 2' } })
    fireEvent.click(addButton)

    // 完成第一个任务
    const checkboxes = screen.getAllByRole('checkbox')
    fireEvent.click(checkboxes[0])

    // 过滤已完成的任务
    const filterButton = screen.getByText('已完成')
    fireEvent.click(filterButton)

    expect(screen.getByText('任务 1')).toBeInTheDocument()
    expect(screen.queryByText('任务 2')).not.toBeInTheDocument()
  })
})

// TodoApp.jsx
import { useState } from 'react'

export default function TodoApp() {
  const [todos, setTodos] = useState([])
  const [filter, setFilter] = useState('all')
  const [input, setInput] = useState('')

  const addTodo = () => {
    if (input.trim()) {
      setTodos([...todos, { id: Date.now(), text: input, completed: false }])
      setInput('')
    }
  }

  const toggleTodo = (id) => {
    setTodos(todos.map(todo =>
      todo.id === id ? { ...todo, completed: !todo.completed } : todo
    ))
  }

  const deleteTodo = (id) => {
    setTodos(todos.filter(todo => todo.id !== id))
  }

  const filteredTodos = todos.filter(todo => {
    if (filter === 'active') return !todo.completed
    if (filter === 'completed') return todo.completed
    return true
  })

  return (
    <div>
      <input
        placeholder="添加新任务"
        value={input}
        onChange={(e) => setInput(e.target.value)}
      />
      <button onClick={addTodo}>添加</button>
      <div>
        <button onClick={() => setFilter('all')}>全部</button>
        <button onClick={() => setFilter('active')}>未完成</button>
        <button onClick={() => setFilter('completed')}>已完成</button>
      </div>
      <ul>
        {filteredTodos.map(todo => (
          <li key={todo.id}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={() => toggleTodo(todo.id)}
            />
            <span style={{ textDecoration: todo.completed ? 'line-through' : 'none' }}>
              {todo.text}
            </span>
            <button onClick={() => deleteTodo(todo.id)}>删除</button>
          </li>
        ))}
      </ul>
    </div>
  )
}
```

```js
// UserList.jsx
import { useState, useEffect } from 'react'

export default function UserList() {
  const [users, setUsers] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    fetch('/api/users')
      .then(response => {
        if (!response.ok) {
          throw new Error('加载失败')
        }
        return response.json()
      })
      .then(data => {
        setUsers(data)
        setLoading(false)
      })
      .catch(err => {
        setError(err.message)
        setLoading(false)
      })
  }, [])

  if (loading) return <div>加载中...</div>
  if (error) return <div>{error}</div>

  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>
          {user.name} ({user.email})
        </li>
      ))}
    </ul>
  )
}
```

## 🎯 端到端测试

### Cypress 基础

Cypress 是一个端到端测试框架，用于测试整个应用的流程。

```js
// cypress/e2e/todo-app.cy.js
describe('Todo App E2E 测试', () => {
  beforeEach(() => {
    cy.visit('/')
  })

  it('应该能够添加任务', () => {
    cy.get('input[placeholder="添加新任务"]').type('学习 React 19')
    cy.contains('添加').click()
    cy.contains('学习 React 19').should('be.visible')
  })

  it('应该能够完成任务', () => {
    cy.get('input[placeholder="添加新任务"]').type('学习 React 19')
    cy.contains('添加').click()
    cy.get('input[type="checkbox"]').first().click()
    cy.contains('学习 React 19').should('have.css', 'text-decoration', 'line-through')
  })

  it('应该能够删除任务', () => {
    cy.get('input[placeholder="添加新任务"]').type('学习 React 19')
    cy.contains('添加').click()
    cy.contains('删除').click()
    cy.contains('学习 React 19').should('not.exist')
  })

  it('应该能够过滤任务', () => {
    // 添加两个任务
    cy.get('input[placeholder="添加新任务"]').type('任务 1')
    cy.contains('添加').click()
    cy.get('input[placeholder="添加新任务"]').type('任务 2')
    cy.contains('添加').click()

    // 完成第一个任务
    cy.get('input[type="checkbox"]').first().click()

    // 过滤已完成的任务
    cy.contains('已完成').click()
    cy.contains('任务 1').should('be.visible')
    cy.contains('任务 2').should('not.exist')
  })
})
```

## 🔗 相关概念

- **单元测试**: 测试单个函数或组件
- **集成测试**: 测试多个组件的交互
- **端到端测试**: 测试完整的用户流程
- **Jest**: JavaScript 测试框架
- **React Testing Library**: React 组件测试工具
- **Cypress**: 端到端测试框架
- **React 19 Actions**: 测试服务端操作
- **React 19 useEffectEvent**: 测试事件处理
- **React 19 useOptimistic**: 测试乐观更新
