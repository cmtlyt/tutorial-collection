## 🎯 Error Boundaries

### 基本概念

Error Boundaries 是 React 组件，可以捕获子组件树中任何位置的 JavaScript 错误。

```js
import { Component } from 'react'

class ErrorBoundary extends Component {
  constructor(props) {
    super(props)
    this.state = { hasError: false, error: null }
  }

  static getDerivedStateFromError(error) {
    return { hasError: true, error }
  }

  componentDidCatch(error, errorInfo) {
    console.error('ErrorBoundary 捕获到错误:', error, errorInfo)
    
    // 可以在这里记录错误到错误追踪服务
    logErrorToService(error, errorInfo)
  }

  render() {
    if (this.state.hasError) {
      return (
        <div style={{ padding: 20, backgroundColor: '#ffebee', borderRadius: 8 }}>
          <h2>出错了！</h2>
          <p>抱歉，应用程序遇到了一个错误。</p>
          <details style={{ marginTop: 16 }}>
            <summary>错误详情</summary>
            <pre style={{ marginTop: 8, padding: 8, backgroundColor: '#f5f5f5' }}>
              {this.state.error?.toString()}
            </pre>
          </details>
          <button 
            onClick={() => window.location.reload()}
            style={{ marginTop: 16, padding: '8px 16px' }}
          >
            重新加载页面
          </button>
        </div>
      )
    }

    return this.props.children
  }
}

// 模拟错误记录服务
function logErrorToService(error, errorInfo) {
  console.log('记录错误到服务:', {
    message: error.message,
    stack: error.stack,
    componentStack: errorInfo.componentStack
  })
}

// 使用示例
function BuggyComponent() {
  throw new Error('这是一个测试错误')
}

function App() {
  return (
    <ErrorBoundary>
      <div>
        <h1>我的应用</h1>
        <BuggyComponent />
      </div>
    </ErrorBoundary>
  )
}

export default App
```

### Error Boundary 最佳实践

```js
import { Component } from 'react'

class ErrorBoundary extends Component {
  constructor(props) {
    super(props)
    this.state = { 
      hasError: false, 
      error: null,
      errorInfo: null
    }
  }

  static getDerivedStateFromError(error) {
    return { hasError: true }
  }

  componentDidCatch(error, errorInfo) {
    this.setState({
      error,
      errorInfo
    })

    // 记录错误到错误追踪服务
    this.logError(error, errorInfo)
  }

  logError(error, errorInfo) {
    const errorData = {
      message: error.message,
      stack: error.stack,
      componentStack: errorInfo.componentStack,
      timestamp: new Date().toISOString(),
      userAgent: navigator.userAgent,
      url: window.location.href
    }

    // 发送到错误追踪服务
    fetch('/api/errors', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(errorData)
    }).catch(err => {
      console.error('无法发送错误报告:', err)
    })
  }

  handleReset = () => {
    this.setState({ hasError: false, error: null, errorInfo: null })
  }

  render() {
    if (this.state.hasError) {
      return (
        <div style={{ 
          padding: 40, 
          textAlign: 'center',
          backgroundColor: '#fff3e0',
          borderRadius: 8,
          margin: 20
        }}>
          <h1 style={{ color: '#e65100' }}>出错了</h1>
          <p style={{ color: '#bf360c' }}>
            应用程序遇到了一个意外错误。我们已经记录了这个问题。
          </p>
          
          {this.state.error && (
            <details style={{ 
              margin: '20px auto', 
              maxWidth: 600, 
              textAlign: 'left',
              backgroundColor: 'white',
              padding: 16,
              borderRadius: 4,
              border: '1px solid #ffcc80'
            }}>
              <summary style={{ cursor: 'pointer', fontWeight: 'bold' }}>
                查看错误详情
              </summary>
              <pre style={{ 
                marginTop: 16, 
                padding: 12, 
                backgroundColor: '#f5f5f5',
                overflow: 'auto',
                fontSize: 12
              }}>
                {this.state.error.toString()}
                {this.state.errorInfo?.componentStack}
              </pre>
            </details>
          )}

          <div style={{ marginTop: 24 }}>
            <button 
              onClick={this.handleReset}
              style={{
                padding: '12px 24px',
                backgroundColor: '#ff9800',
                color: 'white',
                border: 'none',
                borderRadius: 4,
                cursor: 'pointer',
                fontSize: 16,
                marginRight: 12
              }}
            >
              重试
            </button>
            <button 
              onClick={() => window.location.reload()}
              style={{
                padding: '12px 24px',
                backgroundColor: '#757575',
                color: 'white',
                border: 'none',
                borderRadius: 4,
                cursor: 'pointer',
                fontSize: 16
              }}
            >
              重新加载页面
            </button>
          </div>
        </div>
      )
    }

    return this.props.children
  }
}

export default ErrorBoundary
```

## 🎯 React 19 错误处理增强

### onCaughtError 和 onUncaughtError

React 19 引入了 `onCaughtError` 和 `onUncaughtError` 根选项，用于捕获不同类型的错误。

```js
import { createRoot } from 'react-dom/client'
import App from './App'

const container = document.getElementById('root')
const root = createRoot(container, {
  onCaughtError: (error, errorInfo) => {
    console.log('Error Boundary 捕获的错误:', error, errorInfo)
    
    // 记录到错误追踪服务
    logError({
      type: 'caught',
      error: error.message,
      stack: error.stack,
      componentStack: errorInfo.componentStack
    })
  },
  onUncaughtError: (error, errorInfo) => {
    console.log('未捕获的错误:', error, errorInfo)
    
    // 记录到错误追踪服务
    logError({
      type: 'uncaught',
      error: error.message,
      stack: error.stack,
      componentStack: errorInfo.componentStack
    })
    
    // 显示友好的错误页面
    showErrorPage(error)
  }
})

root.render(<App />)

// 模拟错误记录函数
function logError(errorData) {
  console.log('记录错误:', errorData)
  // 实际应用中应该发送到错误追踪服务
}

function showErrorPage(error) {
  document.body.innerHTML = `
    <div style="padding: 40px; text-align: center;">
      <h1>出错了</h1>
      <p>${error.message}</p>
      <button onclick="location.reload()">重新加载</button>
    </div>
  `
}
```

## 🎯 全局错误捕获

### window.onerror

捕获全局的 JavaScript 错误。

```js
// 在应用初始化时设置全局错误处理
window.onerror = function(message, source, lineno, colno, error) {
  console.error('全局错误捕获:', {
    message,
    source,
    lineno,
    colno,
    error: error?.stack
  })

  // 记录到错误追踪服务
  logError({
    type: 'global',
    message,
    source,
    line: lineno,
    column: colno,
    stack: error?.stack
  })

  // 返回 true 阻止默认的错误处理
  return true
}

// 使用示例
function App() {
  const triggerError = () => {
    // 这会触发 window.onerror
    throw new Error('全局错误测试')
  }

  return (
    <div>
      <h1>全局错误捕获</h1>
      <button onClick={triggerError}>触发错误</button>
    </div>
  )
}

export default App
```

### window.addEventListener('error')

更灵活的全局错误捕获方式。

```js
// 添加全局错误监听器
window.addEventListener('error', (event) => {
  console.error('全局错误监听器:', event.error)

  logError({
    type: 'global-event',
    message: event.error?.message,
    filename: event.filename,
    lineno: event.lineno,
    colno: event.colno,
    stack: event.error?.stack
  })

  // 阻止错误继续传播
  event.preventDefault()
})

// 使用示例
function App() {
  const triggerError = () => {
    // 这会触发 error 事件
    throw new Error('全局错误事件测试')
  }

  return (
    <div>
      <h1>全局错误事件监听</h1>
      <button onClick={triggerError}>触发错误</button>
    </div>
  )
}

export default App
```

### window.addEventListener('unhandledrejection')

捕获未处理的 Promise 拒绝。

```js
// 添加未处理 Promise 拒绝监听器
window.addEventListener('unhandledrejection', (event) => {
  console.error('未处理的 Promise 拒绝:', event.reason)

  logError({
    type: 'unhandled-rejection',
    reason: event.reason?.message || String(event.reason),
    stack: event.reason?.stack
  })

  // 阻止默认的控制台错误输出
  event.preventDefault()
})

// 使用示例
function App() {
  const triggerPromiseError = () => {
    // 这会触发 unhandledrejection 事件
    Promise.reject(new Error('Promise 拒绝测试'))
  }

  return (
    <div>
      <h1>未处理的 Promise 拒绝</h1>
      <button onClick={triggerPromiseError}>触发 Promise 错误</button>
    </div>
  )
}

export default App
```

## 🎯 异步错误处理

### try-catch

使用 try-catch 处理同步和异步错误。

```js
import { useState } from 'react'

function AsyncComponent() {
  const [data, setData] = useState(null)
  const [error, setError] = useState(null)
  const [loading, setLoading] = useState(false)

  const fetchData = async () => {
    try {
      setLoading(true)
      setError(null)

      const response = await fetch('/api/data')
      
      if (!response.ok) {
        throw new Error(`HTTP 错误: ${response.status}`)
      }

      const result = await response.json()
      setData(result)
    } catch (err) {
      console.error('获取数据失败:', err)
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div>
      <button onClick={fetchData} disabled={loading}>
        {loading ? '加载中...' : '获取数据'}
      </button>
      
      {error && (
        <div style={{ color: 'red', marginTop: 16 }}>
          错误: {error}
        </div>
      )}
      
      {data && (
        <div style={{ marginTop: 16 }}>
          数据: {JSON.stringify(data)}
        </div>
      )}
    </div>
  )
}

export default AsyncComponent
```

### React 19 Actions 错误处理

```js
import { useActionState } from 'react'

function FormWithAction() {
  async function submitAction(prevState, formData) {
    try {
      const data = Object.fromEntries(formData)
      
      const response = await fetch('/api/submit', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      })

      if (!response.ok) {
        throw new Error(`提交失败: ${response.status}`)
      }

      const result = await response.json()
      return { success: true, data: result }
    } catch (error) {
      return { success: false, error: error.message }
    }
  }

  const [state, formAction, isPending] = useActionState(submitAction, null)

  return (
    <form action={formAction}>
      <input name="username" defaultValue="用户名" />
      <input name="email" defaultValue="email@example.com" />
      <button type="submit" disabled={isPending}>
        {isPending ? '提交中...' : '提交'}
      </button>
      
      {state?.success && (
        <div style={{ color: 'green', marginTop: 16 }}>
          提交成功！
        </div>
      )}
      
      {state?.error && (
        <div style={{ color: 'red', marginTop: 16 }}>
          错误: {state.error}
        </div>
      )}
    </form>
  )
}

export default FormWithAction
```

## 🎯 网络错误处理

### 请求重试机制

```js
import { useState } from 'react'

async function fetchWithRetry(url, options = {}, maxRetries = 3) {
  let lastError

  for (let i = 0; i < maxRetries; i++) {
    try {
      const response = await fetch(url, options)
      
      if (!response.ok) {
        throw new Error(`HTTP 错误: ${response.status}`)
      }

      return await response.json()
    } catch (error) {
      lastError = error
      
      // 如果是最后一次重试，直接抛出错误
      if (i === maxRetries - 1) {
        throw error
      }

      // 等待一段时间后重试（指数退避）
      await new Promise(resolve => 
        setTimeout(resolve, Math.pow(2, i) * 1000)
      )
    }
  }

  throw lastError
}

function DataFetcher() {
  const [data, setData] = useState(null)
  const [error, setError] = useState(null)
  const [loading, setLoading] = useState(false)

  const fetchData = async () => {
    try {
      setLoading(true)
      setError(null)

      const result = await fetchWithRetry('/api/data')
      setData(result)
    } catch (err) {
      console.error('获取数据失败:', err)
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div>
      <button onClick={fetchData} disabled={loading}>
        {loading ? '加载中...' : '获取数据'}
      </button>
      
      {error && (
        <div style={{ color: 'red', marginTop: 16 }}>
          错误: {error}
        </div>
      )}
      
      {data && (
        <div style={{ marginTop: 16 }}>
          数据: {JSON.stringify(data)}
        </div>
      )}
    </div>
  )
}

export default DataFetcher
```

### 网络状态检测

```js
import { useState, useEffect } from 'react'

function NetworkStatus() {
  const [isOnline, setIsOnline] = useState(navigator.onLine)

  useEffect(() => {
    const handleOnline = () => setIsOnline(true)
    const handleOffline = () => setIsOnline(false)

    window.addEventListener('online', handleOnline)
    window.addEventListener('offline', handleOffline)

    return () => {
      window.removeEventListener('online', handleOnline)
      window.removeEventListener('offline', handleOffline)
    }
  }, [])

  return (
    <div>
      <h1>网络状态</h1>
      <div style={{
        padding: 16,
        backgroundColor: isOnline ? '#e8f5e9' : '#ffebee',
        borderRadius: 8,
        color: isOnline ? '#2e7d32' : '#c62828'
      }}>
        {isOnline ? '✓ 在线' : '✗ 离线'}
      </div>
    </div>
  )
}

export default NetworkStatus
```

```js
  console.log('记录错误:', errorData)
```

## 🔗 相关概念

- **Error Boundaries**: 捕获组件树中的错误
- **onCaughtError**: React 19 捕获错误处理
- **onUncaughtError**: React 19 未捕获错误处理
- **onRecoverableError**: React 19 可恢复错误处理
- **全局错误捕获**: window.onerror 和事件监听器
- **异步错误处理**: try-catch 和 Promise 错误
- **网络错误处理**: 请求重试和网络状态检测
- **错误追踪**: 记录和上报错误
