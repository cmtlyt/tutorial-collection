## 🎯 组件边界划分

### 边界划分原则

```js
// 组件边界划分示例

// 1. 顶层布局 - Server Component

'use client';

import { useState } from 'react';

    setAdded(true);

      {added ? '已添加' : '添加到购物车'}
```

```js
// app/components/dashboard/NotificationList.js
'use client';

import { useState } from 'react';

export default function NotificationList({ notifications }) {
  const [readIds, setReadIds] = useState(new Set());
  
  function markAsRead(id) {
    setReadIds(new Set([...readIds, id]));
  }
  
  return (
    <ul>
      {notifications.map((notification) => (
        <li
          key={notification.id}
          style={{
            opacity: readIds.has(notification.id) ? 0.5 : 1,
          }}
        >
          <p>{notification.message}</p>
          <button onClick={() => markAsRead(notification.id)}>
            标记为已读
          </button>
        </li>
      ))}
    </ul>
  );
}
```

## 🎯 数据获取策略

### 并行数据获取

```js
// 并行获取多个数据源
async function getDashboardData() {
  const [users, products, orders, stats] = await Promise.all([
    fetch('https://api.example.com/users').then(r => r.json()),
    fetch('https://api.example.com/products').then(r => r.json()),
    fetch('https://api.example.com/orders').then(r => r.json()),
    fetch('https://api.example.com/stats').then(r => r.json()),
  ]);
  
  return { users, products, orders, stats };
}

export default async function DashboardPage() {
  const data = await getDashboardData();
  
  return (
    <div>
      <UserCount count={data.users.length} />
      <ProductCount count={data.products.length} />
      <OrderCount count={data.orders.length} />
      <Stats stats={data.stats} />
    </div>
  );
}
```

### 条件数据获取

```js
// 条件数据获取
export default async function UserPage({ params, searchParams }) {
  const userId = params.id;
  const tab = searchParams.tab || 'profile';
  
  // 基础数据总是获取
  const user = await getUser(userId);
  
  // 根据标签页获取不同数据
  let tabData;
  if (tab === 'profile') {
    tabData = await getUserProfile(userId);
  } else if (tab === 'posts') {
    tabData = await getUserPosts(userId);
  } else if (tab === 'comments') {
    tabData = await getUserComments(userId);
  }
  
  return (
    <div>
      <h1>{user.name}</h1>
      <TabContent tab={tab} data={tabData} />
    </div>
  );
}
```

### 缓存策略

```js
// 智能缓存策略
async function getCachedData(key, fetcher, options = {}) {
  const { revalidate = 60, tags = [] } = options;
  
  const res = await fetch(`https://api.example.com/${key}`, {
    next: { revalidate, tags },
  });
  
  if (!res.ok) {
    throw new Error(`Failed to fetch ${key}`);
  }
  
  return res.json();
}

// 使用缓存
async function getProducts() {
  return getCachedData('products', async () => {
    const res = await fetch('https://api.example.com/products');
    return res.json();
  }, { revalidate: 3600, tags: ['products'] });
}

async function getProduct(id) {
  return getCachedData(`products/${id}`, async () => {
    const res = await fetch(`https://api.example.com/products/${id}`);
    return res.json();
  }, { revalidate: 3600, tags: [`product-${id}`] });
}
```

### Client Components 状态

```js
// Client Components 管理交互状态
'use client';

import { useState } from 'react';

export default function ReviewForm({ productId }) {
  const [rating, setRating] = useState(5);
  const [comment, setComment] = useState('');
  const [submitting, setSubmitting] = useState(false);
  
  async function handleSubmit(e) {
    e.preventDefault();
    setSubmitting(true);
    
    await submitReview(productId, { rating, comment });
    
    setRating(5);
    setComment('');
    setSubmitting(false);
  }
  
  return (
    <form onSubmit={handleSubmit}>
      <div>
        <label>评分</label>
        <select value={rating} onChange={(e) => setRating(parseInt(e.target.value))}>
          {[1, 2, 3, 4, 5].map((value) => (
            <option key={value} value={value}>{value} 星</option>
          ))}
        </select>
      </div>
      
      <div>
        <label>评论</label>
        <textarea
          value={comment}
          onChange={(e) => setComment(e.target.value)}
          required
        />
      </div>
      
      <button type="submit" disabled={submitting}>
        {submitting ? '提交中...' : '提交评论'}
      </button>
    </form>
  );
}
```

## 🎯 交互处理

### 表单处理

```js
// Server Actions 处理表单
// app/actions.js
'use server';

import { revalidatePath } from 'next/cache';

export async function createComment(formData) {
  const productId = formData.get('productId');
  const comment = formData.get('comment');
  
  // 保存评论到数据库
  await db.comments.create({
    productId,
    comment,
  });
  
  // 重新验证页面
  revalidatePath(`/products/${productId}`);
}

// app/components/CommentForm.js
'use client';

import { createComment } from '../actions';

export default function CommentForm({ productId }) {
  return (
    <form action={createComment}>
      <input type="hidden" name="productId" value={productId} />
      
      <textarea
        name="comment"
        placeholder="写下你的评论..."
        required
      />
      
      <button type="submit">提交评论</button>
    </form>
  );
}
```

### 事件处理

```js
// Client Components 处理事件
'use client';

import { useState } from 'react';

export default function ProductCard({ product }) {
  const [liked, setLiked] = useState(false);
  const [cartCount, setCartCount] = useState(0);
  
  async function handleLike() {
    setLiked(!liked);
    await toggleLike(product.id);
  }
  
  async function handleAddToCart() {
    setCartCount(cartCount + 1);
    await addToCart(product.id);
  }
  
  return (
    <div>
      <h2>{product.name}</h2>
      <p>¥{product.price}</p>
      
      <div>
        <button onClick={handleLike}>
          {liked ? '❤️' : '🤍'}
        </button>
        
        <button onClick={handleAddToCart}>
          添加到购物车 ({cartCount})
        </button>
      </div>
    </div>
  );
}
```
