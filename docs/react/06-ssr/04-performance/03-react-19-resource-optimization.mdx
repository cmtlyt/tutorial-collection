# React 19 资源优化

> **版本**: v1.0  
> **最后更新**: 2026-04-21

## 📚 概念介绍

React 19 提供了新的资源优化 API，包括资源预加载 API（prefetchDNS、preconnect、preload、preinit）、异步脚本支持和样式表自动管理等。这些功能极大地简化了资源优化工作。本章节将详细介绍这些新特性。

## 🎯 资源预加载 API

### prefetchDNS

```js
// 使用 prefetchDNS 预解析 DNS
import { prefetchDNS } from 'react-dom';

export default function ResourceOptimizedPage() {
  // 预解析 DNS
  prefetchDNS('https://api.example.com');
  prefetchDNS('https://cdn.example.com');
  
  return (
    <div>
      <h1>资源优化页面</h1>
      {/* 页面内容 */}
    </div>
  );
}

// 在组件外部使用
prefetchDNS('https://fonts.googleapis.com');
prefetchDNS('https://fonts.gstatic.com');
```

### preconnect

```js
// 使用 preconnect 预连接
import { preconnect } from 'react-dom';

export default function PreconnectExample() {
  // 预连接到 API 服务器
  preconnect('https://api.example.com');
  
  // 预连接到 CDN
  preconnect('https://cdn.example.com', { crossOrigin: 'anonymous' });
  
  return (
    <div>
      <h1>预连接示例</h1>
    </div>
  );
}
```

### preload

```js
// 使用 preload 预加载资源
import { preload } from 'react-dom';

// 预加载图片
function preloadImage(src) {
  preload(src, { as: 'image' });
}

// 预加载字体
function preloadFont(href) {
  preload(href, {
    as: 'font',
    type: 'font/woff2',
    crossOrigin: 'anonymous',
  });
}

// 预加载样式
function preloadStyle(href) {
  preload(href, { as: 'style' });
}

export default function PreloadExample() {
  // 预加载关键资源
  preloadImage('/images/hero.jpg');
  preloadFont('/fonts/inter.woff2');
  preloadStyle('/styles/critical.css');
  
  return (
    <div>
      <h1>预加载示例</h1>
    </div>
  );
}
```

### preinit

```js
// 使用 preinit 预初始化资源
import { preinit } from 'react-dom';

// 预初始化样式表
function preinitStyle(href) {
  preinit(href, { as: 'style' });
}

// 预初始化脚本
function preinitScript(src, props = {}) {
  preinit(src, { as: 'script', ...props });
}

export default function PreinitExample() {
  // 预初始化样式表
  preinitStyle('/styles/main.css');
  
  // 预初始化脚本
  preinitScript('/scripts/analytics.js', {
    async: true,
    defer: true,
  });
  
  return (
    <div>
      <h1>预初始化示例</h1>
    </div>
  );
}
```

### 条件脚本加载

```js
// 条件加载脚本
'use client';

import { useState, useEffect } from 'react';
import { preinit } from 'react-dom';

export default function ConditionalScriptLoader({ shouldLoad }) {
  useEffect(() => {
    if (shouldLoad) {
      // 只在需要时加载脚本
      preinit('https://cdn.example.com/feature.js', {
        as: 'script',
        async: true,
      });
    }
  }, [shouldLoad]);
  
  return (
    <div>
      <h1>条件脚本加载</h1>
      <button onClick={() => setShouldLoad(true)}>
        加载功能脚本
      </button>
    </div>
  );
}
```

## 🎯 样式表自动管理

### 自动样式表管理

```js
// React 19 自动管理样式表
import { preinit } from 'react-dom';

export default function AutoStyleSheetExample() {
  // React 19 会自动去重样式表
  preinit('/styles/main.css', { as: 'style' });
  preinit('/styles/main.css', { as: 'style' }); // 不会重复加载
  
  return (
    <div>
      <h1>自动样式表管理</h1>
      <p>React 19 自动去重样式表</p>
    </div>
  );
}
```

### 样式表优先级

```js
// 使用 precedence 属性控制样式表优先级
import { preinit } from 'react-dom';

export default function StyleSheetPrecedenceExample() {
  // 高优先级样式表
  preinit('/styles/critical.css', {
    as: 'style',
    precedence: 'high',
  });
  
  // 中等优先级样式表
  preinit('/styles/main.css', {
    as: 'style',
    precedence: 'medium',
  });
  
  // 低优先级样式表
  preinit('/styles/deferred.css', {
    as: 'style',
    precedence: 'low',
  });
  
  return (
    <div>
      <h1>样式表优先级</h1>
    </div>
  );
}
```

### 动态样式表加载

```js
// 动态加载样式表
'use client';

import { useState, useEffect } from 'react';
import { preinit } from 'react-dom';

export default function DynamicStyleSheetLoader({ initialTheme = 'light' }) {
  const [theme, setTheme] = useState(initialTheme);
  
  useEffect(() => {
    // 根据主题动态加载样式表
    if (theme === 'dark') {
      preinit('/styles/dark.css', { as: 'style' });
    } else {
      preinit('/styles/light.css', { as: 'style' });
    }
  }, [theme]);
  
  return (
    <div>
      <h1>动态样式表加载</h1>
      <button onClick={() => setTheme('dark')}>深色主题</button>
      <button onClick={() => setTheme('light')}>浅色主题</button>
    </div>
  );
}
```

### 智能资源预加载

```js
// 智能预加载策略
'use client';

import { useState, useEffect } from 'react';
import { preload } from 'react-dom';

function HomePage() {
  return <div>首页内容</div>;
}

function AboutPage() {
  return <div>关于页面内容</div>;
}

function ContactPage() {
  return <div>联系页面内容</div>;
}

export default function SmartResourcePreloader() {
  const [currentPage, setCurrentPage] = useState('home');
  
  useEffect(() => {
    // 预加载可能访问的页面资源
    if (currentPage === 'home') {
      preload('/images/about-hero.jpg', { as: 'image' });
      preload('/styles/about.css', { as: 'style' });
    } else if (currentPage === 'about') {
      preload('/images/contact-hero.jpg', { as: 'image' });
      preload('/styles/contact.css', { as: 'style' });
    }
  }, [currentPage]);
  
  return (
    <div>
      <nav>
        <button onClick={() => setCurrentPage('home')}>首页</button>
        <button onClick={() => setCurrentPage('about')}>关于</button>
        <button onClick={() => setCurrentPage('contact')}>联系</button>
      </nav>
      
      {currentPage === 'home' && <HomePage />}
      {currentPage === 'about' && <AboutPage />}
      {currentPage === 'contact' && <ContactPage />}
    </div>
  );
}
```

```js
        {children}
      </body>
    </html>
  );
}

// app/page.js
import { Suspense } from 'react';
import { preload } from 'react-dom';

// 预加载下一页资源
preload('/images/products-hero.jpg', { as: 'image' });

async function getFeaturedProducts() {
  const res = await fetch('https://api.example.com/products/featured');
  return res.json();
}

async function FeaturedProducts() {
  const products = await getFeaturedProducts();
  
  return (
    <div>
      {products.map((product) => (
        <ProductCard key={product.id} product={product} />
      ))}
    </div>
  );
}

export default function HomePage() {
  return (
    <div>
      <HeroSection />
      
      <Suspense fallback={<ProductListSkeleton />}>
        <FeaturedProducts />
      </Suspense>
    </div>
  );
}
```
