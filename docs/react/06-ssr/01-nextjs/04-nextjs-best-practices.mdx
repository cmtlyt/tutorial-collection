```js
// app/lib/cache.js
export const cacheConfig = {
  // 静态内容：长时间缓存
  static: {
    revalidate: 86400, // 24 小时
    tags: ['static'],
  },
  
  // 动态内容：短时间缓存

    revalidate: 60, // 1 分钟
    tags: ['dynamic'],
  },
  
  // 实时内容：不缓存
  realtime: {
    revalidate: 0,
    tags: ['realtime'],
  },
};

// 使用示例

  const res = await fetch('https://api.example.com/static', {
    next: cacheConfig.static,
  });
  return res.json();
}

async function getDynamicData() {
  const res = await fetch('https://api.example.com/dynamic', {
    next: cacheConfig.dynamic,
  });
  return res.json();
}

async function getRealtimeData() {
  const res = await fetch('https://api.example.com/realtime', {
    cache: 'no-store',
  });

}
```

### API 安全

```js
// app/api/protected/route.js
import { NextResponse } from 'next/server';
import env from '../../lib/env';

export async function GET(request) {
  // 验证 API 密钥
  const apiKey = request.headers.get('x-api-key');
  
  if (apiKey !== env.apiSecretKey) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    );
  }
  
  // 验证请求来源
  const origin = request.headers.get('origin');
  const allowedOrigins = ['https://example.com', 'https://www.example.com'];
  
  if (!allowedOrigins.includes(origin)) {
    return NextResponse.json(
      { error: 'Forbidden' },
      { status: 403 }
    );
  }
  
  // 处理请求
  const data = await fetchProtectedData();
  
  return NextResponse.json(data);
}

async function fetchProtectedData() {
  // 获取受保护的数据
  return { message: 'Protected data' };
}
```

### Server Actions 安全

```js
// app/actions.js
'use server';

import { revalidatePath } from 'next/cache';
import { auth } from '../lib/auth';

// 验证用户权限
async function requireAuth() {
  const user = await auth();
  if (!user) {
    throw new Error('Unauthorized');
  }
  return user;
}

export async function createPost(formData) {
  // 验证用户身份
  const user = await requireAuth();
  
  // 验证输入
  const title = formData.get('title');
  const content = formData.get('content');
  
  if (!title || !content) {
    throw new Error('Title and content are required');
  }
  
  if (title.length > 200) {
    throw new Error('Title is too long');
  }
  
  // 创建文章
  const newPost = {
    id: Math.random().toString(36).substr(2, 9),
    title,
    content,
    authorId: user.id,
    createdAt: new Date().toISOString(),
  };
  
  revalidatePath('/blog');
  
  return newPost;
}
```

### 错误处理

```js
// app/lib/errors.js
export class AppError extends Error {
  constructor(message, statusCode = 500) {
    super(message);
    this.name = 'AppError';
    this.statusCode = statusCode;
  }
}

export class NotFoundError extends AppError {
  constructor(message = 'Resource not found') {
    super(message, 404);
    this.name = 'NotFoundError';
  }
}

export class ValidationError extends AppError {
  constructor(message = 'Validation failed') {
    super(message, 400);
    this.name = 'ValidationError';
  }
}

// 使用示例
// app/api/posts/route.js
import { NextResponse } from 'next/server';
import { ValidationError, NotFoundError } from '../../lib/errors';

export async function POST(request) {
  try {
    const body = await request.json();
    
    if (!body.title || !body.content) {
      throw new ValidationError('Title and content are required');
    }
    
    const post = await createPost(body);
    
    return NextResponse.json(post, { status: 201 });
  } catch (error) {
    if (error instanceof ValidationError) {
      return NextResponse.json(
        { error: error.message },
        { status: error.statusCode }
      );
    }
    
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    );
  }
}
```

### 日志记录

```js
// app/lib/logger.js
const isDevelopment = process.env.NODE_ENV === 'development';

export const logger = {
  info(message, meta = {}) {
    if (isDevelopment) {
      console.log(`[INFO] ${message}`, meta);
    }
    // 生产环境可以集成第三方日志服务
  },
  
  error(message, error = null, meta = {}) {
    if (isDevelopment) {
      console.error(`[ERROR] ${message}`, error, meta);
    }
    // 生产环境可以发送到错误追踪服务
  },
  
  warn(message, meta = {}) {
    if (isDevelopment) {
      console.warn(`[WARN] ${message}`, meta);
    }
  },
};

// 使用示例
export async function createPost(data) {
  try {
    logger.info('Creating post', { title: data.title });
    
    const post = await db.posts.create(data);
    
    logger.info('Post created successfully', { postId: post.id });
    
    return post;
  } catch (error) {
    logger.error('Failed to create post', error, { data });
    throw error;
  }
}
```
