Next.js 16 提供了许多高级特性，包括 API Routes、中间件、图片优化、字体优化以及与 React 19 Server Components 和 Server Actions 的深度集成。本章节将深入介绍这些高级功能。

## 🎯 API Routes

### 基础 API Routes

```js
// app/api/hello/route.js
import { NextResponse } from 'next/server';

export async function GET(request) {
  return NextResponse.json({
    message: 'Hello from Next.js API!',
    timestamp: new Date().toISOString(),
  });
}

export async function POST(request) {
  const body = await request.json();
  
  return NextResponse.json({
    message: 'Data received',
    data: body,
  });
}

// 使用示例：
// GET: fetch('/api/hello')
// POST: fetch('/api/hello', { method: 'POST', body: JSON.stringify({ name: 'Alice' }) })
```

### 动态 API Routes

```js
// app/api/users/[id]/route.js
import { NextResponse } from 'next/server';

async function getUser(id) {
  const res = await fetch(`https://api.example.com/users/${id}`);
  if (!res.ok) {
    throw new Error('User not found');
  }
  return res.json();
}

export async function GET(request, { params }) {
  try {
    const user = await getUser(params.id);
    return NextResponse.json(user);
  } catch (error) {
    return NextResponse.json(
      { error: 'User not found' },
      { status: 404 }
    );
  }
}

export async function PUT(request, { params }) {
  const body = await request.json();
  
  // 更新用户逻辑
  const updatedUser = {
    id: params.id,
    ...body,
    updatedAt: new Date().toISOString(),
  };
  
  return NextResponse.json(updatedUser);
}

export async function DELETE(request, { params }) {
  // 删除用户逻辑
  return NextResponse.json({
    message: `User ${params.id} deleted`,
  });
}
```

## 🎯 中间件

### 基础中间件

```js
// middleware.js
import { NextResponse } from 'next/server';

// 简单的 token 验证函数（实际项目中应该使用更安全的验证方式）
function verifyToken(token) {
  // 这里应该实现真实的 token 验证逻辑
  // 例如：使用 JWT 验证
  if (!token) {
    return false;
  }
  
  // 简单示例：检查 token 是否符合预期格式
  try {
    const parts = token.split('.');
    if (parts.length !== 3) {
      return false;
    }
    
    // 在实际应用中，这里应该验证 JWT 的签名和有效期
    return true;
  } catch (error) {
    return false;
  }
}

export function middleware(request) {
  // 检查认证
  const token = request.cookies.get('token')?.value;
  
  if (!token || !verifyToken(token)) {
    return NextResponse.redirect(new URL('/login', request.url));
  }
  
  return NextResponse.next();
}

export const config = {
  matcher: [
    '/dashboard/:path*',
    '/api/protected/:path*',
  ],
};
```

### 路由保护中间件

```js
// middleware.js
import { NextResponse } from 'next/server';

export function middleware(request) {
  const { pathname } = request.nextUrl;
  
  // 公开路由
  const publicPaths = ['/', '/login', '/register', '/about'];
  
  if (publicPaths.includes(pathname)) {
    return NextResponse.next();
  }
  
  // 检查认证
  const token = request.cookies.get('auth_token');
  
  if (!token) {
    const loginUrl = new URL('/login', request.url);
    loginUrl.searchParams.set('redirect', pathname);
    return NextResponse.redirect(loginUrl);
  }
  
  // 角色检查
  if (pathname.startsWith('/admin')) {
    const userRole = request.cookies.get('user_role')?.value;
    
    if (userRole !== 'admin') {
      return NextResponse.redirect(new URL('/unauthorized', request.url));
    }
  }
  
  return NextResponse.next();
}

export const config = {
  matcher: [
    '/((?!api|_next/static|_next/image|favicon.ico).*)',
  ],
};
```

### 国际化中间件

```js
// middleware.js
import { NextResponse } from 'next/server';

const locales = ['en', 'zh', 'ja'];
const defaultLocale = 'en';

export function middleware(request) {
  const pathname = request.nextUrl.pathname;
  
  // 检查路径中是否已包含语言
  const pathnameIsMissingLocale = locales.every(
    (locale) => !pathname.startsWith(`/${locale}/`) && pathname !== `/${locale}`
  );
  
  if (pathnameIsMissingLocale) {
    const locale = defaultLocale;
    return NextResponse.redirect(
      new URL(`/${locale}${pathname}`, request.url)
    );
  }
  
  return NextResponse.next();
}

export const config = {
  matcher: [
    '/((?!_next|_vercel|.*\\..*).*)',
  ],
};
```

### Server Actions 集成

```js
// app/actions.js
'use server';

import { revalidatePath } from 'next/cache';

export async function createProduct(formData) {
  const name = formData.get('name');
  const price = formData.get('price');
  
  // 模拟数据库操作
  const newProduct = {
    id: Math.random().toString(36).substr(2, 9),
    name,
    price: parseFloat(price),
  };
  
  // 重新验证产品列表页面
  revalidatePath('/products');
  
  return newProduct;
}

export async function deleteProduct(productId) {
  // 删除产品逻辑
  console.log(`Deleting product: ${productId}`);
  
  // 重新验证产品列表页面
  revalidatePath('/products');
  
  return { success: true };
}
```

### 使用 Server Actions

```js
// app/products/page.js
'use client';

import { useState } from 'react';
import { createProduct, deleteProduct } from '../actions';

export default function ProductsPage() {
  const [products, setProducts] = useState([]);
  
  async function handleCreateProduct(formData) {
    const newProduct = await createProduct(formData);
    setProducts([...products, newProduct]);
  }
  
  async function handleDeleteProduct(productId) {
    await deleteProduct(productId);
    setProducts(products.filter(p => p.id !== productId));
  }
  
  return (
    <main>
      <h1>产品管理</h1>
      
      <form action={handleCreateProduct}>
        <input name="name" placeholder="产品名称" required />
        <input name="price" type="number" placeholder="价格" required />
        <button type="submit">添加产品</button>
      </form>
      
      <div>
        {products.map((product) => (
          <div key={product.id}>
            <h3>{product.name}</h3>
            <p>¥{product.price}</p>
            <button onClick={() => handleDeleteProduct(product.id)}>
              删除
            </button>
          </div>
        ))}
      </div>
    </main>
  );
}
```
