# Next.js 基础

> **版本**: v1.0  
> **最后更新**: 2026-04-21

## 📚 概念介绍

Next.js 是一个基于 React 的全栈框架，提供了服务端渲染（SSR）、静态生成（SSG）、增量静态生成（ISR）等渲染策略。Next.js 16 是最新版本，完全支持 React 19 的 Server Components 和 Server Actions。本章节将介绍 Next.js 的核心概念和基础用法。

## 🎯 页面路由与 App Router

### App Router 基础

Next.js 16 使用 App Router 作为默认路由系统，基于文件系统的路由结构：

```js
// app/layout.js - 根布局
export default function RootLayout({ children }) {
  return (
    <html lang="zh-CN">
      <body>

          <a href="/blog">博客</a>
        </nav>
        {children}
      </body>

// app/page.js - 首页
export default function HomePage() {

// app/about/page.js - 关于页面
export default function AboutPage() {
  return (
    <main>
      <h1>关于我们</h1>

// app/blog/[slug]/page.js - 动态路由
```

### 动态路由与参数

```js
// app/products/[id]/page.js
async function getProduct(id) {
  const res = await fetch(`https://api.example.com/products/${id}`);
```

## 🎯 静态生成（SSG）

### 基础静态生成

```js
// app/blog/page.js - 静态生成博客列表
async function getBlogPosts() {
  const res = await fetch('https://api.example.com/posts', {
    cache: 'force-cache', // 强制缓存，用于 SSG
  });
  return res.json();
}

export default async function BlogListPage() {
  const posts = await getBlogPosts();
  
  return (
    <main>
      <h1>博客文章列表</h1>
      <ul>
        {posts.map((post) => (
          <li key={post.id}>
            <a href={`/blog/${post.slug}`}>
              <h2>{post.title}</h2>
              <p>{post.excerpt}</p>
            </a>
          </li>
        ))}
      </ul>
    </main>
  );
}
```

### 生成静态路径

```js
// app/blog/[slug]/page.js - 生成静态路径
import { notFound } from 'next/navigation';

async function getBlogPost(slug) {
  const res = await fetch(`https://api.example.com/posts/${slug}`, {
    cache: 'force-cache',
  });
  
  if (!res.ok) {
    notFound();
  }
  
  return res.json();
}

async function getAllBlogSlugs() {
  const res = await fetch('https://api.example.com/posts', {
    cache: 'force-cache',
  });
  const posts = await res.json();
  return posts.map((post) => post.slug);
}

export async function generateStaticParams() {
  const slugs = await getAllBlogSlugs();
  return slugs.map((slug) => ({
    slug: slug,
  }));
}

export default async function BlogPostPage({ params }) {
  const post = await getBlogPost(params.slug);
  
  return (
    <main>
      <article>
        <h1>{post.title}</h1>
        <p>{post.content}</p>
        <time>{post.publishedAt}</time>
      </article>
    </main>
  );
}
```

## 🎯 服务端渲染（SSR）

### 基础服务端渲染

```js
// app/dashboard/page.js - 服务端渲染
async function getUserData(userId) {
  const res = await fetch(`https://api.example.com/users/${userId}`, {
    cache: 'no-store', // 禁用缓存，用于 SSR
  });
  if (!res.ok) {
    throw new Error('Failed to fetch user data');
  }
  return res.json();
}

export default async function DashboardPage() {
  const userData = await getUserData('user-123');
  
  return (
    <main>
      <h1>用户仪表板</h1>
      <div>
        <h2>欢迎, {userData.name}</h2>
        <p>邮箱: {userData.email}</p>
        <p>角色: {userData.role}</p>
      </div>
    </main>
  );
}
```

### 动态数据服务端渲染

```js
// app/orders/[orderId]/page.js
import { notFound } from 'next/navigation';

async function getOrder(orderId) {
  const res = await fetch(`https://api.example.com/orders/${orderId}`, {
    cache: 'no-store',
    headers: {
      'Authorization': `Bearer ${process.env.API_TOKEN}`,
    },
  });
  
  if (!res.ok) {
    notFound();
  }
  
  return res.json();
}

export default async function OrderPage({ params }) {
  const order = await getOrder(params.orderId);
  
  return (
    <main>
      <h1>订单详情</h1>
      <div>
        <p>订单号: {order.id}</p>
        <p>状态: {order.status}</p>
        <p>总金额: ¥{order.total}</p>
        <ul>
          {order.items.map((item) => (
            <li key={item.id}>
              {item.name} - ¥{item.price} x {item.quantity}
            </li>
          ))}
        </ul>
      </div>
    </main>
  );
}
```

## 🎯 增量静态生成（ISR）

### 基础 ISR

```js
// app/news/page.js - 增量静态生成
async function getNews() {
  const res = await fetch('https://api.example.com/news', {
    next: {
      revalidate: 3600, // 每小时重新验证一次
    },
  });
  return res.json();
}

export default async function NewsPage() {
  const news = await getNews();
  
  return (
    <main>
      <h1>最新新闻</h1>
      <ul>
        {news.map((item) => (
          <li key={item.id}>
            <h2>{item.title}</h2>
            <p>{item.summary}</p>
            <time>{item.publishedAt}</time>
          </li>
        ))}
      </ul>
    </main>
  );
}
```

### 按需重新验证

```js
// app/api/revalidate/route.js - 重新验证 API
import { revalidatePath } from 'next/cache';
import { NextResponse } from 'next/server';

export async function POST(request) {
  const path = request.nextUrl.searchParams.get('path');
  
  if (path) {
    revalidatePath(path);
    return NextResponse.json({ revalidated: true, now: Date.now() });
  }
  
  return NextResponse.json({
    revalidated: false,
    now: Date.now(),
    message: 'Missing path to revalidate',
  });
}

// 使用示例：curl -X POST 'http://localhost:3000/api/revalidate?path=/news'
```

```js
// app/layout.js
export default function RootLayout({ children }) {

            <a href="/blog">博客</a>

export default function HomePage() {

      <a href="/blog">浏览博客文章</a>
    </main>
  );
}

// app/blog/page.js
async function getAllPosts() {
  const res = await fetch('https://api.example.com/posts', {
    next: { revalidate: 3600 },
  });
  return res.json();
}

export default async function BlogListPage() {
  const posts = await getAllPosts();
  
  return (
    <main>
      <h1>博客文章</h1>
      <div>
        {posts.map((post) => (
          <article key={post.id}>
            <h2>
              <a href={`/blog/${post.slug}`}>{post.title}</a>
            </h2>
            <p>{post.excerpt}</p>
            <time>{post.publishedAt}</time>
          </article>
        ))}
      </div>
    </main>
  );
}

// app/blog/[slug]/page.js
async function getPost(slug) {
  const res = await fetch(`https://api.example.com/posts/${slug}`, {
    next: { revalidate: 3600 },
  });
  
  if (!res.ok) {
    notFound();
  }
  
  return res.json();
}

export async function generateStaticParams() {
  const res = await fetch('https://api.example.com/posts');
  const posts = await res.json();
  return posts.map((post) => ({
    slug: post.slug,
  }));
}

export default async function BlogPostPage({ params }) {
  const post = await getPost(params.slug);
  
  return (
    <main>
      <article>
        <h1>{post.title}</h1>
        <div dangerouslySetInnerHTML={{ __html: post.content }} />
        <time>{post.publishedAt}</time>
      </article>
    </main>
  );
}
```
