## 📚 概念介绍

桌面应用开发是现代软件开发的重要组成部分。React 生态提供了多种桌面应用开发解决方案，其中 Electron 和 Tauri 是最流行的跨平台桌面应用开发框架。本节将深入探讨 Electron + React 和 Tauri 框架的集成方式、核心概念和最佳实践。

## 🎯 Electron + React

### Electron 基础

Electron 是由 GitHub 开发的跨平台桌面应用开发框架，允许开发者使用 JavaScript、HTML 和 CSS 构建原生桌面应用。

```js
// 模拟 Electron 核心功能
const Electron = {
  // 主进程模块
  app: {
    on: function(event, callback) {
      console.log(`App event: ${event}`);
      if (event === 'ready') {
        setTimeout(callback, 100);
      }
    },
    quit: function() {
      console.log('App quitting');
    },
    getPath: function(name) {
      const paths = {
        home: '/Users/username',
        appData: '/Users/username/Library/Application Support',
        userData: '/Users/username/Library/Application Support/myapp',
        temp: '/tmp',
        downloads: '/Users/username/Downloads'
      };
      return paths[name] || '';
    }
  },

  // 浏览器窗口模块
  BrowserWindow: function BrowserWindow(options) {
    this.options = options;
    this.id = Date.now();
    
    return {
      id: this.id,
      loadURL: function(url) {
        console.log(`Loading URL: ${url}`);
        return Promise.resolve();
      },
      on: function(event, callback) {
        console.log(`Window event: ${event}`);
        if (event === 'closed') {
          setTimeout(callback, 50);
        }
      },
      close: function() {
        console.log('Window closing');
      },
      webContents: {
        openDevTools: function() {
          console.log('Opening DevTools');
        },
        send: function(channel, ...args) {
          console.log(`Sending to renderer: ${channel}`, args);
        }
      },
      show: function() {
        console.log('Window shown');
      }
    };
  },

  // IPC 通信模块
  ipcMain: {
    on: function(channel, listener) {
      console.log(`IPC main listening on: ${channel}`);
      this.listeners = this.listeners || {};
      this.listeners[channel] = listener;
    },
    handle: function(channel, listener) {
      console.log(`IPC main handling: ${channel}`);
      this.handlers = this.handlers || {};
      this.handlers[channel] = listener;
    }
  },

  // 渲染进程模块
  ipcRenderer: {
    on: function(channel, listener) {
      console.log(`IPC renderer listening on: ${channel}`);
    },
    send: function(channel, ...args) {
      console.log(`Sending to main: ${channel}`, args);
    },
    invoke: async function(channel, ...args) {
      console.log(`Invoking main: ${channel}`, args);
      await new Promise(resolve => setTimeout(resolve, 100));
      return { success: true, data: args };
    }
  },

  // 对话框模块
  dialog: {
    showOpenDialog: async function(options) {
      console.log('Showing open dialog', options);
      await new Promise(resolve => setTimeout(resolve, 200));
      return {
        canceled: false,
        filePaths: ['/path/to/selected/file.txt']
      };
    },
    showSaveDialog: async function(options) {
      console.log('Showing save dialog', options);
      await new Promise(resolve => setTimeout(resolve, 200));
      return {
        canceled: false,
        filePath: '/path/to/save/file.txt'
      };
    },
    showMessageBox: async function(options) {
      console.log('Showing message box', options);
      await new Promise(resolve => setTimeout(resolve, 200));
      return { response: 0 };
    }
  },

  // 菜单模块
  Menu: {
    buildFromTemplate: function(template) {
      console.log('Building menu from template');
      return {
        items: template
      };
    },
    setApplicationMenu: function(menu) {
      console.log('Setting application menu');
    }
  },

  MenuItem: function MenuItem(options) {
    this.options = options;
    return {
      ...options
    };
  }
};

// Electron 主进程代码
const createMainWindow = () => {
  const mainWindow = new Electron.BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,
      preload: '/path/to/preload.js'
    }
  });

  mainWindow.loadURL('http://localhost:3000');
  
  // 开发环境打开 DevTools
  if (process.env.NODE_ENV === 'development') {
    mainWindow.webContents.openDevTools();
  }

  mainWindow.on('closed', () => {
    console.log('Main window closed');
  });

  return mainWindow;
};

// Electron 应用初始化
const initializeElectronApp = () => {
  let mainWindow;

  Electron.app.on('ready', () => {
    console.log('Electron app ready');
    mainWindow = createMainWindow();
  });

  Electron.app.on('window-all-closed', () => {
    if (process.platform !== 'darwin') {
      Electron.app.quit();
    }
  });

  Electron.app.on('activate', () => {
    if (mainWindow === null) {
      mainWindow = createMainWindow();
    }
  });

  console.log('Electron app initialized');
};

// 测试 Electron 基础功能
console.log('=== Testing Electron Basics ===');
initializeElectronApp();

const testWindow = new Electron.BrowserWindow({
  width: 800,
  height: 600
});
console.log('Test window created with ID:', testWindow.id);
```

### 主进程与渲染进程

Electron 应用采用多进程架构，主进程负责应用生命周期和原生功能，渲染进程负责 UI 渲染。

```js
// 主进程代码
const MainProcess = {
  // IPC 处理器
  setupIpcHandlers: function() {
    // 文件操作
    Electron.ipcMain.handle('read-file', async (event, filePath) => {
      console.log('Reading file:', filePath);
      await new Promise(resolve => setTimeout(resolve, 100));
      return {
        success: true,
        content: `File content from ${filePath}`,
        size: 1024
      };
    });

    Electron.ipcMain.handle('write-file', async (event, filePath, content) => {
      console.log('Writing file:', filePath);
      await new Promise(resolve => setTimeout(resolve, 100));
      return {
        success: true,
        bytesWritten: content.length
      };
    });

    // 系统信息
    Electron.ipcMain.handle('get-system-info', async () => {
      return {
        platform: process.platform,
        arch: process.arch,
        version: process.version,
        electronVersion: '25.0.0'
      };
    });

    // 应用路径
    Electron.ipcMain.handle('get-app-path', async (event, pathName) => {
      return Electron.app.getPath(pathName);
    });

    // 对话框
    Electron.ipcMain.handle('show-open-dialog', async (event, options) => {
      return await Electron.dialog.showOpenDialog(options);
    });

    Electron.ipcMain.handle('show-save-dialog', async (event, options) => {
      return await Electron.dialog.showSaveDialog(options);
    });

    console.log('IPC handlers setup complete');
  },

  // 创建应用菜单
  createApplicationMenu: function() {
    const template = [
      {
        label: 'File',
        submenu: [
          new Electron.MenuItem({
            label: 'New',
            accelerator: 'CmdOrCtrl+N',
            click: () => {
              console.log('New file');
            }
          }),
          new Electron.MenuItem({
            label: 'Open',
            accelerator: 'CmdOrCtrl+O',
            click: () => {
              console.log('Open file');
            }
          }),
          { type: 'separator' },
          new Electron.MenuItem({
            label: 'Save',
            accelerator: 'CmdOrCtrl+S',
            click: () => {
              console.log('Save file');
            }
          }),
          { type: 'separator' },
          {
            label: 'Exit',
            accelerator: process.platform === 'darwin' ? 'Cmd+Q' : 'Ctrl+Q',
            click: () => {
              Electron.app.quit();
            }
          }
        ]
      },
      {
        label: 'Edit',
        submenu: [
          { role: 'undo' },
          { role: 'redo' },
          { type: 'separator' },
          { role: 'cut' },
          { role: 'copy' },
          { role: 'paste' }
        ]
      },
      {
        label: 'View',
        submenu: [
          { role: 'reload' },
          { role: 'forceReload' },
          { role: 'toggleDevTools' },
          { type: 'separator' },
          { role: 'resetZoom' },
          { role: 'zoomIn' },
          { role: 'zoomOut' },
          { type: 'separator' },
          { role: 'togglefullscreen' }
        ]
      },
      {
        label: 'Help',
        submenu: [
          {
            label: 'About',
            click: () => {
              console.log('About');
            }
          }
        ]
      }
    ];

    const menu = Electron.Menu.buildFromTemplate(template);
    Electron.Menu.setApplicationMenu(menu);
    console.log('Application menu created');
  }
};

// 渲染进程代码（模拟）
const RendererProcess = {
  // 与主进程通信

  },

    return await Electron.ipcRenderer.invoke('show-open-dialog', options);
  },

    return await Electron.ipcRenderer.invoke('show-save-dialog', options);
  },

  // 监听主进程消息
  listenToMain: function(channel, callback) {
    Electron.ipcRenderer.on(channel, callback);
  }
};

// 测试主进程和渲染进程通信
console.log('\n=== Testing Main and Renderer Process Communication ===');
```

## 🎯 Tauri 框架

### Tauri 基础

Tauri 是一个轻量级的跨平台桌面应用开发框架，使用 Rust 作为后端，Web 技术作为前端。

```js
// 模拟 Tauri 核心功能
const Tauri = {
  // Tauri API
  invoke: async function(cmd, args) {
    console.log(`Tauri invoke: ${cmd}`, args);
    await new Promise(resolve => setTimeout(resolve, 100));
    return {
      success: true,
      data: args
    };
  },

  // 事件系统
  listen: function(event, callback) {
    console.log(`Tauri listening to event: ${event}`);
    return {
      unlisten: function() {
        console.log(`Unlistening from event: ${event}`);
      }
    };
  },

  emit: function(event, payload) {
    console.log(`Tauri emitting event: ${event}`, payload);
  },

  // 窗口管理
  Window: {
    getCurrent: function() {
      return {
        label: 'main',
        setTitle: function(title) {
          console.log('Setting window title:', title);
        },
        maximize: function() {
          console.log('Maximizing window');
        },
    minimize: function() {
          console.log('Minimizing window');
        },
        close: function() {
          console.log('Closing window');
        },
        hide: function() {
          console.log('Hiding window');
        },
        show: function() {
          console.log('Showing window');
        },
        isMaximized: async function() {
          return false;
        },
        isMinimized: async function() {
          return false;
        },
        isVisible: async function() {
          return true;
        }
      };
    }
  },

  // 应用信息
  App: {
    getName: async function() {
      return 'My Tauri App';
    },
    getVersion: async function() {
      return '1.0.0';
    },
    getTauriVersion: async function() {
      return '2.0.0';
    }
  },

  // 对话框
  Dialog: {
    open: async function(options) {
      console.log('Opening dialog with options:', options);
      await new Promise(resolve => setTimeout(resolve, 200));
      return '/path/to/selected/file';
    },
    save: async function(options) {
      console.log('Saving dialog with options:', options);
      await new Promise(resolve => setTimeout(resolve, 200));
      return '/path/to/save/file';
    },
    message: async function(message, title) {
      console.log('Showing message dialog:', message, title);
      await new Promise(resolve => setTimeout(resolve, 200));
      return null;
    },
    ask: async function(message, title) {
      console.log('Showing ask dialog:', message, title);
      await new Promise(resolve => setTimeout(resolve, 200));
      return true;
    },
    confirm: async function(message, title) {
      console.log('Showing confirm dialog:', message, title);
      await new Promise(resolve => setTimeout(resolve, 200));
      return true;
    }
  },

  // 文件系统
  Fs: {
    readTextFile: async function(path) {
      console.log('Reading text file:', path);
      await new Promise(resolve => setTimeout(resolve, 100));
      return 'File content';
    },
    writeTextFile: async function(path, contents) {
      console.log('Writing text file:', path);
      await new Promise(resolve => setTimeout(resolve, 100));
      return null;
    },
    readDir: async function(path) {
      console.log('Reading directory:', path);
      await new Promise(resolve => setTimeout(resolve, 100));
      return [
        { name: 'file1.txt' },
        { name: 'file2.txt' },
        { name: 'subdirectory' }
      ];
    },
    createDir: async function(path) {
      console.log('Creating directory:', path);
      await new Promise(resolve => setTimeout(resolve, 50));
      return null;
    },
    removeDir: async function(path) {
      console.log('Removing directory:', path);
      await new Promise(resolve => setTimeout(resolve, 50));
      return null;
    }
  },

  // 路径
  Path: {
    resolve: function(...paths) {
      return paths.join('/');
    },
    join: function(...paths) {
      return paths.join('/');
    },
    basename: function(path) {
      return path.split('/').pop();
    },
    dirname: function(path) {
      return path.split('/').slice(0, -1).join('/');
    },
    extname: function(path) {
      const ext = path.split('.').pop();
      return ext === path ? '' : `.${ext}`;
    }
  }
};

// 使用 Tauri API
const useTauriAPI = async () => {
  // 调用 Tauri 命令
  const result = await Tauri.invoke('custom_command', { param: 'value' });
  console.log('Tauri invoke result:', result);

  // 监听事件
  const unlisten = Tauri.listen('tauri://update-available', (event) => {
    console.log('Update available:', event);
  });

  // 发送事件
  Tauri.emit('custom-event', { data: 'test' });

  // 窗口操作
  const window = Tauri.Window.getCurrent();
  await window.setTitle('New Title');
  console.log('Window title set');

  // 应用信息
  const appName = await Tauri.App.getName();
  const appVersion = await Tauri.App.getVersion();
  console.log(`App: ${appName} v${appVersion}`);

  // 对话框
  const filePath = await Tauri.Dialog.open({
    multiple: false,
    filters: [
      { name: 'Text', extensions: ['txt'] }
    ]
  });
  console.log('Selected file:', filePath);

  // 文件系统操作
  const content = await Tauri.Fs.readTextFile('/test/file.txt');
  console.log('File content:', content);

  await Tauri.Fs.writeTextFile('/test/output.txt', 'New content');
  console.log('File written');

  const dirContents = await Tauri.Fs.readDir('/test');
  console.log('Directory contents:', dirContents);

  // 路径操作
  const fullPath = Tauri.Path.resolve('/home', 'user', 'documents', 'file.txt');
  console.log('Resolved path:', fullPath);

  const basename = Tauri.Path.basename(fullPath);
  console.log('Basename:', basename);

  const dirname = Tauri.Path.dirname(fullPath);
  console.log('Dirname:', dirname);

  // 清理事件监听
  unlisten.unlisten();
};

// 测试 Tauri 基础功能
console.log('\n=== Testing Tauri Basics ===');
useTauriAPI();
```

### Rust 后端集成

Tauri 使用 Rust 编写后端逻辑，提供高性能和安全性。

```js
        },

        }

    // 数据处理

    encrypt: async function(text, key) {
      console.log('[Rust] Encrypting text');
      // 模拟 Rust 加密
      await new Promise(resolve => setTimeout(resolve, 150));
      return {
        encrypted: true,
        cipher_text: btoa(text + ':' + key),
        algorithm: 'AES-256-GCM'
      };
    },

    decrypt: async function(cipher_text, key) {
      console.log('[Rust] Decrypting text');
      // 模拟 Rust 解密
      await new Promise(resolve => setTimeout(resolve, 150));
      const decoded = atob(cipher_text);
      const [text, originalKey] = decoded.split(':');
      return {
        decrypted: originalKey === key,
        plain_text: originalKey === key ? text : null
      };
    }
  },

  // 调用 Rust 命令
  invoke: async function(commandName, args) {
    const command = this.commands[commandName];
    if (!command) {
      throw new Error(`Unknown command: ${commandName}`);
    }
    return await command.apply(this, Object.values(args || {}));
  }
};
```

```js
// 模拟完整的桌面文本编辑器应用
const DesktopTextEditor = () => {
  // 应用状态
  const state = {
    currentFile: null,
    content: '',
    isModified: false,
    recentFiles: [],
    window: null
  };

  // 初始化应用
  const initializeApp = () => {
    console.log('Initializing text editor...');
    
    // 创建主窗口
    state.window = new Electron.BrowserWindow({
      width: 1200,
      height: 800,
      minWidth: 800,
      minHeight: 600,
      webPreferences: {
        nodeIntegration: false,
        contextIsolation: true,
        preload: '/path/to/preload.js'
      }
    });

    state.window.loadURL('http://localhost:3000');

    // 设置应用菜单
    setupMenus();

    // 设置 IPC 处理器
    setupIpcHandlers();

    console.log('Text editor initialized');
  };

  // 设置菜单
  const setupMenus = () => {
    const template = [
      {
        label: 'File',
        submenu: [
          {
            label: 'New',
            accelerator: 'CmdOrCtrl+N',
            click: () => newFile()
          },
          {
            label: 'Open',
            accelerator: 'CmdOrCtrl+O',
            click: () => openFile()
          },
          { type: 'separator' },
          {
            label: 'Save',
            accelerator: 'CmdOrCtrl+S',
            click: () => saveFile()
          },
          {
            label: 'Save As',
            accelerator: 'CmdOrCtrl+Shift+S',
            click: () => saveFileAs()
          },
          { type: 'separator' },
          {
            label: 'Recent Files',
            submenu: state.recentFiles.map(file => ({
              label: file,
              click: () => openRecentFile(file)
            }))
          },
          { type: 'separator' },
          {
            label: 'Exit',
            accelerator: process.platform === 'darwin' ? 'Cmd+Q' : 'Ctrl+Q',
            click: () => Electron.app.quit()
          }
        ]
      },
      {
        label: 'Edit',
        submenu: [
          { role: 'undo' },
          { role: 'redo' },
          { type: 'separator' },
          { role: 'cut' },
          { role: 'copy' },
          { role: 'paste' },
          { role: 'selectall' }
        ]
      },
      {
        label: 'View',
        submenu: [
          { role: 'reload' },
          { role: 'toggleDevTools' },
          { type: 'separator' },
          { role: 'resetZoom' },
          { role: 'zoomIn' },
          { role: 'zoomOut' },
          { type: 'separator' },
          { role: 'togglefullscreen' }
        ]
      }
    ];

    const menu = Electron.Menu.buildFromTemplate(template);
    Electron.Menu.setApplicationMenu(menu);
  };

  // 设置 IPC 处理器
  const setupIpcHandlers = () => {
    // 文件操作
    Electron.ipcMain.handle('new-file', () => {
      newFile();
      return { success: true };
    });

    Electron.ipcMain.handle('open-file', async () => {
      const result = await openFile();
      return result;
    });

    Electron.ipcMain.handle('save-file', async () => {
      const result = await saveFile();
      return result;
    });

    Electron.ipcMain.handle('save-file-as', async () => {
      const result = await saveFileAs();
      return result;
    });

    // 内容操作
    Electron.ipcMain.handle('get-content', () => {
      return { content: state.content, isModified: state.isModified };
    });

    Electron.ipcMain.handle('set-content', (event, content) => {
      state.content = content;
      state.isModified = true;
      updateWindowTitle();
      return { success: true };
    });

    // 文件信息
    Electron.ipcMain.handle('get-file-info', () => {
      return {
        currentFile: state.currentFile,
        isModified: state.isModified
      };
    });
  };

  // 新建文件
  const newFile = () => {
    if (state.isModified) {
      // 提示保存
      console.log('Prompt to save changes');
    }
    
    state.currentFile = null;
    state.content = '';
    state.isModified = false;
    updateWindowTitle();
    
    // 通知渲染进程
    state.window.webContents.send('file-newed');
    
    console.log('New file created');
  };

  // 打开文件
  const openFile = async () => {
    const result = await Electron.dialog.showOpenDialog({
      properties: ['openFile'],
      filters: [
        { name: 'Text Files', extensions: ['txt', 'md', 'js', 'html', 'css'] },
        { name: 'All Files', extensions: ['*'] }
      ]
    });

    if (!result.canceled && result.filePaths.length > 0) {
      const filePath = result.filePaths[0];
      const fileContent = await NativeFeatures.FileSystem.readFile(filePath);
      
      state.currentFile = filePath;
      state.content = fileContent.content;
      state.isModified = false;
      
      // 添加到最近文件
      addToRecentFiles(filePath);
      
      updateWindowTitle();
      
      // 通知渲染进程
      state.window.webContents.send('file-opened', {
        path: filePath,
        content: state.content
      });
      
      console.log('File opened:', filePath);
    }
    
    return {
      success: !result.canceled,
      filePath: state.currentFile,
      content: state.content
    };
  };

  // 保存文件
  const saveFile = async () => {
    if (!state.currentFile) {
      return await saveFileAs();
    }
    
    await NativeFeatures.FileSystem.writeFile(state.currentFile, state.content);
    state.isModified = false;
    updateWindowTitle();
    
    // 通知渲染进程
    state.window.webContents.send('file-saved', {
      path: state.currentFile
    });
    
    console.log('File saved:', state.currentFile);
    
    return { success: true, filePath: state.currentFile };
  };

  // 另存为
  const saveFileAs = async () => {
    const result = await Electron.dialog.showSaveDialog({
      filters: [
        { name: 'Text Files', extensions: ['txt'] },
        { name: 'All Files', extensions: ['*'] }
      ]
    });

    if (!result.canceled) {
      const filePath = result.filePath;
      await NativeFeatures.FileSystem.writeFile(filePath, state.content);
      
      state.currentFile = filePath;
      state.isModified = false;
      
      addToRecentFiles(filePath);
      updateWindowTitle();
      
      // 通知渲染进程
      state.window.webContents.send('file-saved', {
        path: filePath
      });
      
      console.log('File saved as:', filePath);
    }
    
    return {
      success: !result.canceled,
      filePath: state.currentFile
    };
  };

  // 打开最近文件
  const openRecentFile = async (filePath) => {
    try {
      const fileContent = await NativeFeatures.FileSystem.readFile(filePath);
      
      state.currentFile = filePath;
      state.content = fileContent.content;
      state.isModified = false;
      
      updateWindowTitle();
      
      // 通知渲染进程
      state.window.webContents.send('file-opened', {
        path: filePath,
        content: state.content
      });
      
      console.log('Recent file opened:', filePath);
    } catch (error) {
      console.error('Error opening recent file:', error);
    }
  };

  // 添加到最近文件
  const addToRecentFiles = (filePath) => {
    // 移除重复项
    state.recentFiles = state.recentFiles.filter(f => f !== filePath);
    // 添加到开头
    state.recentFiles.unshift(filePath);
    // 限制数量
    state.recentFiles = state.recentFiles.slice(0, 10);
  };

  // 更新窗口标题
  const updateWindowTitle = () => {
    const title = state.currentFile
      ? `${state.isModified ? '● ' : ''}${state.currentFile.split('/').pop()} - Text Editor`
      : `${state.isModified ? '● ' : ''}Untitled - Text Editor`;
    
    state.window.setTitle(title);
  };

  return {
    initializeApp,
    newFile,
    openFile,
    saveFile,
    saveFileAs,
    getState: () => state
  };
};

})();
```

- **原生模块**: 访问操作系统原生功能的桥梁
- **Rust 后端**: Tauri 使用 Rust 编写高性能后端逻辑
- **安全沙箱**: 隔离应用环境，提高安全性
