## 📚 概念介绍

TypeScript 为 React 应用提供了强大的类型安全，能够在编译时捕获错误，提高代码质量和开发效率。React 19 对 TypeScript 支持进行了重大改进，包括更严格的类型推断、更好的泛型支持以及移除了许多已废弃的类型。

本节将深入探讨如何在 React 19 中有效使用 TypeScript，包括组件类型定义、Hooks 类型安全、事件处理类型以及 React 19 的最新 TypeScript 变更。

## 🎯 TypeScript 基础

### 类型系统

TypeScript 的类型系统是静态的，这意味着类型检查在编译时进行，而不是运行时。这有助于在开发早期发现错误。

```js

  age: 30,
  isActive: true

  if (typeof userData !== 'object' || userData === null) {
    throw new Error('Invalid user data');
  }

    processedAt: new Date().toISOString()
  };
}
```

### Hooks 类型安全

TypeScript 为 React Hooks 提供了完整的类型支持，确保状态和副作用的类型安全。

```js
// 模拟 React Hooks（在实际项目中使用 import { useState, useEffect } from 'react'）
let hookId = 0;
const hooks = new Map();

function useState(initialValue) {
  const id = hookId++;
  
  if (!hooks.has(id)) {
    hooks.set(id, {
      value: typeof initialValue === 'function' ? initialValue() : initialValue
    });
  }
  
  const hook = hooks.get(id);
  
  const setValue = (newValue) => {
    hook.value = typeof newValue === 'function' ? newValue(hook.value) : newValue;
  };
  
  return [hook.value, setValue];
}

function useEffect(callback, deps) {
  const id = hookId++;
  const prevDeps = hooks.get(id)?.deps;
  
  const hasChanged = !prevDeps || !deps || deps.some((dep, i) => dep !== prevDeps[i]);
  
  if (hasChanged) {
    const cleanup = callback();
    hooks.set(id, { deps, cleanup });
  }
}

// useState 类型示例
const Counter = () => {
  const [count, setCount] = useState(0);
  const [user, setUser] = useState(null);

  const increment = () => {
    setCount(prevCount => prevCount + 1);
  };

  const loadUser = () => {
    setUser({
      id: 1,
      name: 'John Doe',
      email: 'john@example.com'
    });
  };

  // 重置 hookId 以便测试
  hookId = 0;
  
  return {
    type: 'div',
    props: {
      children: [
        { type: 'p', props: { children: `Count: ${count}` } },
        { type: 'button', props: { children: 'Increment', onClick: increment } },
        { type: 'button', props: { children: 'Load User', onClick: loadUser } },
        user ? { type: 'div', props: { children: [
          { type: 'p', props: { children: `User: ${user.name}` } },
          { type: 'p', props: { children: `Email: ${user.email}` } }
        ]}} : null
      ].filter(Boolean)
    }
  };
};

// 测试 Counter 组件
```

```js
// 模拟 React 19 的 useActionState Hook
function useActionState(action, initialState, permalink) {
  const [state, setState] = useState(initialState);
  const [isPending, setIsPending] = useState(false);

  const formAction = async (formData) => {
    setIsPending(true);
    try {
      const result = await action(state, formData);
      setState(result);
    } finally {
      setIsPending(false);
    }
  };

  return [state, formAction, isPending];
}
```

```js
const UserManagementApp = () => {
  // 重置 hookId
  hookId = 0;
  
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [editingUser, setEditingUser] = useState(null);
  const [showForm, setShowForm] = useState(false);

  const loadUsers = async () => {
    try {
      setLoading(true);
      const data = await UserAPI.fetchUsers();
      setUsers(data);
    } catch (err) {
      setError('Failed to load users');
    } finally {
      setLoading(false);
    }
  };

  const handleCreateUser = async (userData) => {
    try {
      const newUser = await UserAPI.createUser(userData);
      setUsers(prev => [...prev, newUser]);
      setShowForm(false);
    } catch (err) {
      setError('Failed to create user');
    }
  };

  const handleUpdateUser = async (userId, updates) => {
    try {
      const updatedUser = await UserAPI.updateUser(userId, updates);
      setUsers(prev =>
        prev.map(user => (user.id === userId ? updatedUser : user))
      );
      setEditingUser(null);
    } catch (err) {
      setError('Failed to update user');
    }
  };

  const handleDeleteUser = (userId) => {
    setUsers(prev => prev.filter(user => user.id !== userId));
  };

  const handleEdit = (user) => {
    setEditingUser(user);
    setShowForm(true);
  };

  const handleSubmit = (formData) => {
    if (editingUser) {
      handleUpdateUser(editingUser.id, formData);
    } else {
      handleCreateUser(formData);
    }
  };

  // 初始加载用户数据
  loadUsers();

  return {
    type: 'div',
    props: {
      style: { padding: '20px', maxWidth: '800px', margin: '0 auto' },
      children: [
        { type: 'h1', props: { children: 'User Management' } },
        loading ? { type: 'p', props: { children: 'Loading...' } } : null,
        error ? { type: 'p', props: { style: { color: 'red' }, children: error } } : null,
        !loading && !error ? {
          type: 'div',
          props: {
            children: [
              {
                type: 'button',
                props: {
                  children: 'Add New User',
                  style: { marginBottom: '16px' },
                  onClick: () => {
                    setEditingUser(null);
                    setShowForm(true);
                  }
                }
              },
              showForm ? {
                type: 'div',
                props: {
                  style: { marginBottom: '20px' },
                  children: [
                    {
                      type: 'h2',
                      props: {
                        children: editingUser ? 'Edit User' : 'Create User'
                      }
                    },
                    UserForm({
                      user: editingUser,
                      onSubmit: handleSubmit,
                      onCancel: () => setShowForm(false)
                    })
                  ]
                }
              } : null,
              UserList({
                users: users,
                onEdit: handleEdit,
                onDelete: handleDeleteUser
              })
            ].filter(Boolean)
          }
        } : null
      ].filter(Boolean)
    }
  };
};
```
