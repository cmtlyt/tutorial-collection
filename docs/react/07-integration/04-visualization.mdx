# 可视化与图形

> **版本**: v1.0  
> **最后更新**: 2025-04-21

## 📚 概念介绍

数据可视化和图形渲染是现代 Web 应用的重要组成部分。React 生态提供了丰富的可视化库和图形渲染解决方案，包括基于 D3.js 的 Recharts、功能强大的 ECharts、Canvas 渲染、SVG 图形以及 Three.js 3D 图形。本节将深入探讨这些技术的核心概念、使用方法和最佳实践。

## 🎯 数据可视化库

### Recharts 基础

Recharts 是一个基于 D3.js 和 SVG 的 React 图表库，提供了声明式的 API 和丰富的图表类型。

```js
// 模拟 Recharts 核心功能
const Recharts = {
  // 图表组件
  BarChart: function BarChart({ data, children, width = 500, height = 300 }) {
    this.data = data;
    this.children = children;
    this.width = width;
    this.height = height;
    
    return {
      type: 'BarChart',
      props: { data, width, height },
      render: function() {
        return {
          chartType: 'bar',
          data: this.props.data,
          dimensions: { width: this.props.width, height: this.props.height },
          bars: this.props.data.map(item => ({
            name: item.name,
            value: item.value
          }))
        };
      }
    };
  },

  LineChart: function LineChart({ data, children, width = 500, height = 300 }) {
    this.data = data;
    this.children = children;
    this.width = width;
    this.height = height;
    
    return {
      type: 'LineChart',
      props: { data, width, height },
      render: function() {
        return {
          chartType: 'line',
          data: this.props.data,
          dimensions: { width: this.props.width, height: this.props.height },
          points: this.props.data.map(item => ({
            name: item.name,
            value: item.value
          }))
        };
      }
    };
  },

  PieChart: function PieChart({ data, children, width = 400, height = 400 }) {
    this.data = data;
    this.children = children;
    this.width = width;
    this.height = height;
    
    return {
      type: 'PieChart',
      props: { data, width, height },
      render: function() {
        return {
          chartType: 'pie',
          data: this.props.data,
          dimensions: { width: this.props.width, height: this.props.height },
          slices: this.props.data.map(item => ({
            name: item.name,
            value: item.value
          }))
        };
      }
    };
  },

  // 数据映射组件
  XAxis: function XAxis({ dataKey, label }) {
    return {
      type: 'XAxis',
      props: { dataKey, label }
    };
  },

  YAxis: function YAxis({ label }) {
    return {
      type: 'YAxis',
      props: { label }
    };
  },

  // 数据系列组件
  Bar: function Bar({ dataKey, fill, name }) {
    return {
      type: 'Bar',
      props: { dataKey, fill, name }
    };
  },

  Line: function Line({ dataKey, stroke, name, type = 'monotone' }) {
    return {
      type: 'Line',
      props: { dataKey, stroke, name, type }
    };
  },

  Pie: function Pie({ dataKey, name, fill }) {
    return {
      type: 'Pie',
      props: { dataKey, name, fill }
    };
  },

  Cell: function Cell({ fill }) {
    return {
      type: 'Cell',
      props: { fill }
    };
  },

  // 工具组件
  Tooltip: function Tooltip({ cursor, content }) {
    return {
      type: 'Tooltip',
      props: { cursor, content }
    };
  },

  Legend: function Legend({ verticalAlign, height, align }) {
    return {
      type: 'Legend',
      props: { verticalAlign, height, align }
    };
  },

  CartesianGrid: function CartesianGrid({ strokeDasharray }) {
    return {
      type: 'CartesianGrid',
      props: { strokeDasharray }
    };
  },

  // 响应式容器
  ResponsiveContainer: function ResponsiveContainer({ children, aspect = 2 }) {
    return {
      type: 'ResponsiveContainer',
      props: { aspect },
      children
    };
  }
};
```

## 🎯 Canvas 图形渲染

### Canvas 基础

Canvas 提供了高性能的 2D 图形渲染能力，适合处理大量数据和复杂动画。

```js
// 模拟 Canvas API
const Canvas = {
  // 2D 上下文
  Context2D: function Context2D(canvas) {
    this.canvas = canvas;
    this.fillStyle = '#000000';
    this.strokeStyle = '#000000';
    this.lineWidth = 1;
    this.font = '12px sans-serif';
    this.textAlign = 'start';
    this.textBaseline = 'alphabetic';
    this.shadowColor = 'transparent';
    this.shadowBlur = 0;
    this.shadowOffsetX = 0;
    this.shadowOffsetY = 0;
    this.globalAlpha = 1;
    this.globalCompositeOperation = 'source-over';
    this.transform = { a: 1, b: 0, c: 0, d: 1, e: 0, f: 0 };
    
    return {
      // 绘制矩形
      fillRect: function(x, y, width, height) {
        console.log(`fillRect: ${x}, ${y}, ${width}, ${height}, fill: ${this.fillStyle}`);
      },
      strokeRect: function(x, y, width, height) {
        console.log(`strokeRect: ${x}, ${y}, ${width}, ${height}, stroke: ${this.strokeStyle}`);
      },
      clearRect: function(x, y, width, height) {
        console.log(`clearRect: ${x}, ${y}, ${width}, ${height}`);
      },
      
      // 绘制路径
      beginPath: function() {
        console.log('beginPath');
        this.currentPath = [];
      },
      closePath: function() {
        console.log('closePath');
      },
      moveTo: function(x, y) {
        console.log(`moveTo: ${x}, ${y}`);
        this.currentPath = this.currentPath || [];
        this.currentPath.push({ type: 'move', x, y });
      },
      lineTo: function(x, y) {
        console.log(`lineTo: ${x}, ${y}`);
        this.currentPath = this.currentPath || [];
        this.currentPath.push({ type: 'line', x, y });
      },
      arc: function(x, y, radius, startAngle, endAngle, anticlockwise) {
        console.log(`arc: ${x}, ${y}, ${radius}, ${startAngle}, ${endAngle}`);
        this.currentPath = this.currentPath || [];
        this.currentPath.push({ type: 'arc', x, y, radius, startAngle, endAngle, anticlockwise });
      },
      quadraticCurveTo: function(cpx, cpy, x, y) {
        console.log(`quadraticCurveTo: ${cpx}, ${cpy}, ${x}, ${y}`);
        this.currentPath = this.currentPath || [];
        this.currentPath.push({ type: 'quad', cpx, cpy, x, y });
      },
      bezierCurveTo: function(cp1x, cp1y, cp2x, cp2y, x, y) {
        console.log(`bezierCurveTo: ${cp1x}, ${cp1y}, ${cp2x}, ${cp2y}, ${x}, ${y}`);
        this.currentPath = this.currentPath || [];
        this.currentPath.push({ type: 'bezier', cp1x, cp1y, cp2x, cp2y, x, y });
      },
      fill: function() {
        console.log(`fill with: ${this.fillStyle}`);
      },
      stroke: function() {
        console.log(`stroke with: ${this.strokeStyle}`);
      },
      
      // 绘制文本
      fillText: function(text, x, y) {
        console.log(`fillText: "${text}" at ${x}, ${y}`);
      },
      strokeText: function(text, x, y) {
        console.log(`strokeText: "${text}" at ${x}, ${y}`);
      },
      measureText: function(text) {
        return {
          width: text.length * 10,
          actualBoundingBoxLeft: 0,
          actualBoundingBoxRight: text.length * 10
        };
      },
      
      // 图像处理
      drawImage: function(image, dx, dy) {
        console.log(`drawImage at ${dx}, ${dy}`);
      },
      
      // 变换
      save: function() {
        console.log('save');
        this.savedStates = this.savedStates || [];
        this.savedStates.push({
          fillStyle: this.fillStyle,
          strokeStyle: this.strokeStyle,
          lineWidth: this.lineWidth,
          transform: { ...this.transform },
          globalAlpha: this.globalAlpha
        });
      },
      restore: function() {
        console.log('restore');
        if (this.savedStates && this.savedStates.length > 0) {
          const state = this.savedStates.pop();
          this.fillStyle = state.fillStyle;
          this.strokeStyle = state.strokeStyle;
          this.lineWidth = state.lineWidth;
          this.transform = state.transform;
          this.globalAlpha = state.globalAlpha;
        }
      },
      translate: function(x, y) {
        console.log(`translate: ${x}, ${y}`);
        this.transform.e += x;
        this.transform.f += y;
      },
      rotate: function(angle) {
        console.log(`rotate: ${angle}`);
        const cos = Math.cos(angle);
        const sin = Math.sin(angle);
        const { a, b, c, d } = this.transform;
        this.transform.a = a * cos - b * sin;
        this.transform.b = a * sin + b * cos;
        this.transform.c = c * cos - d * sin;
        this.transform.d = c * sin + d * cos;
      },
      scale: function(x, y) {
        console.log(`scale: ${x}, ${y}`);
        this.transform.a *= x;
        this.transform.b *= y;
        this.transform.c *= x;
        this.transform.d *= y;
      },
      
      // 渐变
      createLinearGradient: function(x0, y0, x1, y1) {
        return {
          addColorStop: function(offset, color) {
            console.log(`addColorStop: ${offset}, ${color}`);
          }
        };
      },
      createRadialGradient: function(x0, y0, r0, x1, y1, r1) {
        return {
          addColorStop: function(offset, color) {
            console.log(`addColorStop: ${offset}, ${color}`);
          }
        };
      }
    };
  },

  // 创建 Canvas 元素
  createElement: function(width, height) {
    return {
      width: width,
      height: height,
      getContext: function(type) {
        if (type === '2d') {
          return new Canvas.Context2D(this);
        }
        return null;
      },
      toDataURL: function(type) {
        return `data:${type || 'image/png'};base64,...`;
      }
    };
  }
};
```

```js
const ReactCanvas = {
  // Canvas 组件
  Canvas: function Canvas({ width, height, onDraw, style }) {
    this.canvasRef = { current: null };
    this.width = width;
    this.height = height;
    this.onDraw = onDraw;
    this.style = style;
    
    return {
      type: 'Canvas',
      props: { width, height, onDraw, style },
      ref: this.canvasRef,
      componentDidMount: function() {
        if (this.props.onDraw && this.canvasRef.current) {
          const ctx = this.canvasRef.current.getContext('2d');
          this.props.onDraw(ctx, this.canvasRef.current);
        }
      },
      componentDidUpdate: function(prevProps) {
        if (this.props.onDraw !== prevProps.onDraw && this.canvasRef.current) {
          const ctx = this.canvasRef.current.getContext('2d');
          this.props.onDraw(ctx, this.canvasRef.current);
        }
      }
    };
  },

  // 动画 Canvas 组件
  AnimatedCanvas: function AnimatedCanvas({ width, height, onDraw, style, fps = 60 }) {
    this.canvasRef = { current: null };
    this.width = width;
    this.height = height;
    this.onDraw = onDraw;
    this.style = style;
    this.fps = fps;
    this.animationId = null;
    
    return {
      type: 'AnimatedCanvas',
      props: { width, height, onDraw, style, fps },
      ref: this.canvasRef,
      componentDidMount: function() {
        this.startAnimation();
      },
      componentWillUnmount: function() {
        this.stopAnimation();
      },
      startAnimation: function() {
        const animate = (timestamp) => {
          if (this.canvasRef.current) {
            const ctx = this.canvasRef.current.getContext('2d');
            this.props.onDraw(ctx, timestamp, this.canvasRef.current);
          }
          this.animationId = requestAnimationFrame(animate);
        };
        this.animationId = requestAnimationFrame(animate);
      },
      stopAnimation: function() {
        if (this.animationId) {
          cancelAnimationFrame(this.animationId);
          this.animationId = null;
        }
      }
    };
  }
};
```

```js
// 测试完整仪表板
const dashboard = DataDashboard();
```

## 🔗 相关概念

- **数据可视化**: 将数据转换为图形表示的技术
- **Canvas**: HTML5 提供的 2D 绘图 API，适合高性能渲染
- **SVG**: 可缩放矢量图形，提供高质量的矢量图形
- **D3.js**: 强大的数据可视化库，提供灵活的数据绑定和转换
- **ECharts**: 功能丰富的可视化库，支持多种图表类型
- **Recharts**: 基于 D3 和 SVG 的 React 图表库
- **Three.js**: 3D 图形库，支持 WebGL 渲染
